# src/riskquant/config/builder.py
"""
Builder module for risk configuration.
Isolates Pandas and DataFrame memory from the runtime config objects.
"""

from __future__ import annotations

import re
from dataclasses import dataclass
from typing import Dict, Final, Tuple

import pandas as pd

from riskquant.config.configs import RiskStaticConfig
from riskquant.config.loader import load_risk_static_data_config_xlsx_into_dict
from riskquant.config.schema_objects import (
    AliasSchema,
    AssetClass,
    AssetSubclass,
    BusinessDayConvention,
    Ccy,
    CompoundingConvention,
    ContainerMeta,
    ContainerProxyMap,
    DayCountConvention,
    ExtrapolationMethod,
    HolidayCalendar,
    InterpolationMethod,
    RiskFactor,
    RiskFactorProxyMap,
    ShiftType,
    SpotRollRule,
    SurfaceYAxisType,
    TenorSchema,
    UnitScale,
    UnitType,
)
from riskquant.config.validate import validate_static_config_dfs
from riskquant.market_data.risk_factors.shift_types import (
    validate_shift_types_dict,
)
from riskquant.utils.printer import print_success_message


@dataclass(frozen=True, slots=True)
class RiskStaticConfigFrames:
    """
    Temporary container that holds pandas DataFrames loaded from Excel.
    Call .compile() to create a runtime RiskStaticConfig that holds no DataFrames.

    The tab names in the Excel workbook must match the field names in this dataclass, which are:
    - container_meta,
    - risk_factors,
    - tenor_schema,
    - alias_schema,
    - spot_roll_rules,
    - asset_classes,
    - asset_subclasses,
    - unit_types,
    - unit_scales,
    - surface_y_axis_schema,
    - ccy,
    - shift_types,
    - interpolation_methods,
    - extrapolation_methods,
    - container_proxies,
    - risk_factor_proxies,
    - holiday_calendars,
    - day_count_conventions,
    - business_day_conventions,
    - compounding_conventions.

    """

    # Core
    container_meta: pd.DataFrame
    risk_factors: pd.DataFrame

    # Schemas
    tenor_schema: pd.DataFrame
    alias_schema: pd.DataFrame
    spot_roll_rules: pd.DataFrame
    surface_y_axis_schema: pd.DataFrame

    # Lookups
    asset_classes: pd.DataFrame
    asset_subclasses: pd.DataFrame
    unit_types: pd.DataFrame
    unit_scales: pd.DataFrame
    ccy: pd.DataFrame
    shift_types: pd.DataFrame
    interpolation_methods: pd.DataFrame
    extrapolation_methods: pd.DataFrame

    # Mappings
    container_proxies: pd.DataFrame
    risk_factor_proxies: pd.DataFrame

    # Calendars / conventions
    holiday_calendars: pd.DataFrame
    day_count_conventions: pd.DataFrame
    business_day_conventions: pd.DataFrame
    compounding_conventions: pd.DataFrame

    def compile(self) -> RiskStaticConfig:
        """
        Compile validated DataFrames into compact Python structures and return runtime config.
        """

        # --- Helper for fast, null-safe DataFrame iteration ---
        def _iter_df(df: pd.DataFrame) -> list[dict]:
            if df is None or df.empty:
                return []
            # Replaces pandas <NA> and np.nan with native Python None for clean extraction
            return df.where(pd.notnull(df), None).to_dict("records")

        # ======================= #
        # --- LOOKUPS & CONVS --- #
        # ======================= #
        asset_classes_dict = {
            str(r["asset_class"]): AssetClass(
                asset_class=str(r["asset_class"]),
                asset_class_id=int(r["asset_class_id"]),
            )
            for r in _iter_df(self.asset_classes)
        }

        asset_subclasses_dict = {
            str(r["asset_subclass"]): AssetSubclass(
                asset_subclass=str(r["asset_subclass"]),
                asset_subclass_id=int(r["asset_subclass_id"]),
                asset_class=str(r["asset_class"]),
            )
            for r in _iter_df(self.asset_subclasses)
        }

        unit_types_dict = {
            str(r["unit_type_name"]): UnitType(
                unit_type_name=str(r["unit_type_name"]),
                unit_type_name_id=int(r["unit_type_name_id"]),
            )
            for r in _iter_df(self.unit_types)
        }

        unit_scales_dict = {
            str(r["unit_scale_name"]): UnitScale(
                unit_scale_name=str(r["unit_scale_name"]),
                unit_scale_name_id=int(r["unit_scale_name_id"]),
                scale_to_base=float(r["scale_to_base"]),
            )
            for r in _iter_df(self.unit_scales)
        }

        day_count_conventions_dict = {
            str(r["day_count_convention"]): DayCountConvention(
                day_count_convention=str(r["day_count_convention"]),
                day_count_convention_id=int(r["day_count_convention_id"]),
            )
            for r in _iter_df(self.day_count_conventions)
        }

        business_day_conventions_dict = {
            str(r["business_day_convention"]): BusinessDayConvention(
                business_day_convention=str(r["business_day_convention"]),
                business_day_convention_id=int(r["business_day_convention_id"]),
            )
            for r in _iter_df(self.business_day_conventions)
        }

        compounding_conventions_dict = {
            str(r["compounding_convention"]): CompoundingConvention(
                compounding_convention=str(r["compounding_convention"]),
                compounding_convention_id=int(r["compounding_convention_id"]),
                compounding_type=str(r["compounding_type"]),
                frequency_per_year=(
                    float(r["frequency_per_year"])
                    if r.get("frequency_per_year") is not None
                    else None
                ),
            )
            for r in _iter_df(self.compounding_conventions)
        }

        interpolation_methods_dict = {
            str(r["interpolation_method"]): InterpolationMethod(
                interpolation_method=str(r["interpolation_method"]),
                interpolation_method_id=int(r["interpolation_method_id"]),
            )
            for r in _iter_df(self.interpolation_methods)
        }

        extrapolation_methods_dict = {
            str(r["extrapolation_method"]): ExtrapolationMethod(
                extrapolation_method=str(r["extrapolation_method"]),
                extrapolation_method_id=int(r["extrapolation_method_id"]),
            )
            for r in _iter_df(self.extrapolation_methods)
        }

        # Fixed: Now mapping using the actual schema objects
        ccy_dict = {
            str(r["ccy"]).upper(): Ccy(
                ccy=str(r["ccy"]).upper(), ccy_id=int(r["ccy_id"])
            )
            for r in _iter_df(self.ccy)
        }

        shift_types_dict = {
            str(r["shift_type"]).upper(): ShiftType(
                shift_type=str(r["shift_type"]).upper(),
                shift_type_id=int(r["shift_type_id"]),
            )
            for r in _iter_df(self.shift_types)
        }
        # Pass a mapped `{str: int}` dict safely into the validator to prevent it from breaking
        validate_shift_types_dict(
            {k: v.shift_type_id for k, v in shift_types_dict.items()}
        )

        # Instantiating the HolidayCalendar schema
        holiday_calendars_dict = {
            str(r["holiday_calendar_name"]): HolidayCalendar(
                holiday_calendar_name=str(r["holiday_calendar_name"]),
                holiday_dates=(
                    set(r["holiday_dates"])
                    if r.get("holiday_dates") is not None
                    else set()
                ),
            )
            for r in _iter_df(self.holiday_calendars)
        }

        # ======================= #
        # --- CORE META & RF  --- #
        # ======================= #
        container_meta_dict: Dict[str, ContainerMeta] = {}
        for r in _iter_df(self.container_meta):
            if not bool(r.get("active", 1)):
                continue

            c_name = str(r["container_name"])
            base_ccy = r.get("base_ccy")
            quote_ccy = str(r["quote_ccy"])
            base_ccy = str(base_ccy) if base_ccy is not None else quote_ccy

            container_meta_dict[c_name] = ContainerMeta(
                container_name=c_name,
                container_name_id=int(r["container_name_id"]),
                base_ccy=base_ccy,
                quote_ccy=quote_ccy,
                unit_type_name=str(r["unit_type_name"]),
                unit_scale_name=str(r["unit_scale_name"]),
                container_dim=int(r["container_dim"]),
                asset_class=str(r["asset_class"]),
                asset_subclass=str(r["asset_subclass"]),
                tenor_type_name=r.get("tenor_type_name"),
                spot_roll_rule_name=r.get("spot_roll_rule_name"),
                alias_type_name=r.get("alias_type_name"),
                surface_y_axis_type_name=r.get("surface_y_axis_type_name"),
                shift_type_name=str(r["shift_type_name"]),
                interp_method_name_x=r.get("interp_method_name_x"),
                interp_method_name_y=r.get("interp_method_name_y"),
                extrap_method_name_x=r.get("extrap_method_name_x"),
                extrap_method_name_y=r.get("extrap_method_name_y"),
            )

        risk_factors_dict: Dict[str, RiskFactor] = {}
        for r in _iter_df(self.risk_factors):
            if not bool(r.get("active", 1)):
                continue

            rf_name = str(r["risk_factor_name"])
            risk_factors_dict[rf_name] = RiskFactor(
                risk_factor_name=rf_name,
                risk_factor_name_id=int(r["risk_factor_name_id"]),
                tenor_token=r.get("tenor_token"),
                container_name=str(r["container_name"]),
            )

        # ======================= #
        # --- SCHEMAS / RULES --- #
        # ======================= #
        tenor_schema_dict = {
            str(r["tenor_type_name"]): TenorSchema(
                tenor_type_name=str(r["tenor_type_name"]),
                tenor_units=str(r["tenor_units"]),
                digits=int(r["digits"]),
                regex=re.compile(str(r["regex"])) if r.get("regex") else None,
            )
            for r in _iter_df(self.tenor_schema)
        }

        alias_schema_dict = {
            str(r["alias_type_name"]): AliasSchema(
                alias_type_name=str(r["alias_type_name"]),
                alias_kind=str(r["alias_kind"]),
                alias_unit=r.get("alias_unit"),
            )
            for r in _iter_df(self.alias_schema)
        }

        surface_y_axis_schema_dict = {
            str(r["y_axis_type_name"]): SurfaceYAxisType(
                y_axis_type_name=str(r["y_axis_type_name"])
            )
            for r in _iter_df(self.surface_y_axis_schema)
        }

        spot_roll_rules_dict: Dict[Tuple[str, str], SpotRollRule] = {}
        for r in _iter_df(self.spot_roll_rules):
            key = (str(r["spot_roll_rule_name"]), str(r["tenor_type_name"]))
            spot_roll_rules_dict[key] = SpotRollRule(
                spot_roll_rule_name=key[0],
                tenor_type_name=key[1],
                spot_roll_method=str(r["spot_roll_method"]),
                offset_in_tenor_units=r.get("offset_in_tenor_units"),
                roll_month=r.get("roll_month"),
                roll_day=r.get("roll_day"),
                fh_bh_ddmm=r.get("fh_bh_ddmm"),
                post_roll_shift=r.get("post_roll_shift"),
            )

        # ======================= #
        # --- PROXIES         --- #
        # ======================= #
        curve_proxies_dict = {
            str(r["container_name"]): ContainerProxyMap(
                container_name=str(r["container_name"]),
                proxy_container_name=str(r["proxy_container_name"]),
                proxy_container_vol_scaling_factor=float(
                    r.get("proxy_container_vol_scaling_factor", 1.0) or 1.0
                ),
            )
            for r in _iter_df(self.container_proxies)
        }

        risk_factor_proxies_dict = {
            str(r["risk_factor_name"]): RiskFactorProxyMap(
                risk_factor_name=str(r["risk_factor_name"]),
                proxy_risk_factor_name=str(r["proxy_risk_factor_name"]),
                proxy_risk_factor_vol_scaling_factor=float(
                    r.get("proxy_risk_factor_vol_scaling_factor", 1.0) or 1.0
                ),
            )
            for r in _iter_df(self.risk_factor_proxies)
        }

        # ======================= #
        # --- FINALIZE CONFIG --- #
        # ======================= #
        # - container_meta,
        # - risk_factors,
        # - tenor_schema,
        # - alias_schema,
        # - spot_roll_rules,
        # - asset_classes,
        # - asset_subclasses,
        # - unit_types,
        # - unit_scales,
        # - surface_y_axis_schema,
        # - ccy,
        # - shift_types,
        # - interpolation_methods,
        # - extrapolation_methods,
        # - container_proxies,
        # - risk_factor_proxies,
        # - holiday_calendars,
        # - day_count_conventions,
        # - business_day_conventions,
        # - compounding_conventions.
        return RiskStaticConfig(
            container_meta_dict=container_meta_dict,
            risk_factors_dict=risk_factors_dict,
            tenor_schema_dict=tenor_schema_dict,
            alias_schema_dict=alias_schema_dict,
            spot_roll_rules_dict=spot_roll_rules_dict,
            surface_y_axis_schema_dict=surface_y_axis_schema_dict,
            asset_classes_dict=asset_classes_dict,
            asset_subclasses_dict=asset_subclasses_dict,
            unit_types_dict=unit_types_dict,

            risk_factor_proxies_dict=risk_factor_proxies_dict,

if __name__ == "__main__":

# src/riskquant/config/compile.py
"""
Main API for compiling configurations from source data (e.g. .xlsx workbooks)
into runtime config objects.
"""

from __future__ import annotations

from pathlib import Path
from typing import Union

from typing_extensions import Final

from riskquant.config.builder import RiskStaticConfigFrames
from riskquant.config.configs import RiskStaticConfig
from riskquant.config.loader import load_risk_static_data_config_xlsx_into_dict
from riskquant.config.resolution import RiskFactorNameAbsToRelMapping
from riskquant.config.validate import validate_static_config_dfs
from riskquant.utils.dates import get_previous_business_date_as_date
from riskquant.utils.printer import (
    print_dict,
    print_success_message,
)


def compile_risk_static_config_from_xlsx(
    path: Union[str, Path],
) -> RiskStaticConfig:
    """
    Load, validate, and compile a risk static config .xlsx workbook into a
    RiskStaticConfig instance.

    Args:
        path: Path to the Excel (.xlsx) workbook containing the static config sheets.

    Returns:
        RiskStaticConfig: the compiled runtime configuration object.

    Raises:
        ValueError: if required sheets are missing, or validation fails on
            uniqueness/referential integrity constraints.
    """

    # 1. Load config dataframes into a dictionary (with optimized dtypes).
    dict_of_dfs = load_risk_static_data_config_xlsx_into_dict(path=path)

    # 2. Vectorized fail-fast validation of raw DataFrames (C-level uniqueness & FK checks).
    validate_static_config_dfs(dict_of_dfs)

    # 3. Create the lightweight frames container (now stripped of heavy validation logic).
    frames_config = RiskStaticConfigFrames(**dict_of_dfs)

    # 4. Compile the DataFrames into our highly efficient, slot-based python structures.
    compiled_config = frames_config.compile()

    # 5. Return the compiled config object ready for use in runtime/model code.
    return compiled_config

        repr=False,
        init=False,

            raise ValueError(
                "container_meta_dict is empty. Check your container_meta sheet and 'active' flags."
            )
        if not self.risk_factors_dict:
            raise ValueError(
                "risk_factors_dict is empty. Check your risk_factors sheet and 'active' flags."
            )

    def summary(self) -> str:
        lines = [
            "=== Risk Static Config Summary ===",
            f"Containers Mapped: {len(self.container_meta_dict)}",
            f"Risk Factors: {len(self.risk_factors_dict)}",
            f"Asset Classes: {len(self.asset_classes_dict)}",
            f"Currencies: {len(self.ccy_dict)}",
            f"Shift Types: {len(self.shift_types_dict)}",
            f"Container Proxies: {len(self.curve_proxies_dict)}",
            f"Risk Factor Proxies: {len(self.risk_factor_proxies_dict)}",
            "==================================",
        ]
        return "\n".join(lines)

    def get_risk_factor_id(self, risk_factor_name: str) -> int:
        return self.risk_factors_dict[risk_factor_name].risk_factor_name_id

    @property
    def rf_name_to_rf_id(self) -> Dict[str, int]:
        """A build-once cache: risk_factor_name -> risk_factor_name_id"""
        if self._rf_name_to_rf_id is None:
            object.__setattr__(
                self,
                "_rf_name_to_rf_id",
                {
                    k: v.risk_factor_name_id
                    for k, v in self.risk_factors_dict.items()
                },
            )
        return self._rf_name_to_rf_id

    @property
    def rf_id_to_proxy_rf_id(self) -> Dict[int, Tuple[int, float]]:
        """A build-once cache of RF ID-level proxy overrides: required_rf_id -> (proxy_rf_id, vol_scale)"""
        if self._rf_id_to_proxy_rf_id is None:
            rf_name_map = self.rf_name_to_rf_id
            out: Dict[int, Tuple[int, float]] = {}
            for p in self.risk_factor_proxies_dict.values():
                rf_id = int(rf_name_map[p.risk_factor_name])
                proxy_rf_id = int(rf_name_map[p.proxy_risk_factor_name])
                scale = float(p.proxy_risk_factor_vol_scaling_factor or 1.0)
                out[rf_id] = (proxy_rf_id, scale)
            object.__setattr__(self, "_rf_id_to_proxy_rf_id", out)
        return self._rf_id_to_proxy_rf_id

    @property
    def rf_id_to_risk_factor(self) -> Dict[int, RiskFactor]:
        """Cached mapping: risk_factor_name_id (int) -> RiskFactor"""
        if self._rf_id_to_risk_factor is None:
            out: Dict[int, RiskFactor] = {}
            for rf in self.risk_factors_dict.values():
                rf_id = int(rf.risk_factor_name_id)
                if rf_id in out:
                    raise ValueError(
                        f"Duplicate risk_factor_name_id detected: {rf_id}"
                    )
                out[rf_id] = rf
            object.__setattr__(self, "_rf_id_to_risk_factor", out)
        return self._rf_id_to_risk_factor

    @property
    def rf_id_to_shift_type_id(self) -> Dict[int, int]:
        """Cached mapping: risk_factor_name_id (int) -> shift_type_id (int)"""
        if self._rf_id_to_shift_type_id is None:
            out: Dict[int, int] = {}
            for rf in self.risk_factors_dict.values():
                rf_id = int(rf.risk_factor_name_id)
                cm = self.container_meta_dict.get(rf.container_name)
                if cm is None:
                    raise KeyError(
                        f"curve_name={rf.container_name!r} not found in container_meta_dict."
                    )

                st_obj = self.shift_types_dict.get(cm.shift_type_name)
                if st_obj is None:
                    raise KeyError(
                        f"shift_type_name={cm.shift_type_name!r} not found in shift_types_dict."
                    )

                out[rf_id] = int(st_obj.shift_type_id)
            object.__setattr__(self, "_rf_id_to_shift_type_id", out)
        return self._rf_id_to_shift_type_id

    @property
    def curve_tenor_index(
        self,
    ) -> Dict[str, Tuple[np.ndarray, np.ndarray, Dict[str, int]]]:
        """Cached per-curve tenor lookup used by curve-level proxy resolution."""
        if self._curve_tenor_index is not None:
            return self._curve_tenor_index

        offsets: Dict[str, list[int]] = {}
        offset_ids: Dict[str, list[int]] = {}
        labels: Dict[str, Dict[str, int]] = {}

        for rf_id, rf in self.rf_id_to_risk_factor.items():
            rf_curve_name = rf.container_name
            rf_tenor_token = rf.tenor_token

            if rf_tenor_token is None:
                labels.setdefault(rf_curve_name, {})[""] = int(rf_id)
                continue

            rf_unit, rf_offset = parse_bracket_token(rf_tenor_token)

            if rf_offset is None:
                labels.setdefault(rf_curve_name, {})[rf_unit] = int(rf_id)
            else:
                offsets.setdefault(rf_curve_name, []).append(int(rf_offset))
                offset_ids.setdefault(rf_curve_name, []).append(int(rf_id))

        out: Dict[str, Tuple[np.ndarray, np.ndarray, Dict[str, int]]] = {}

        for rf_curve_name, rf_offset_list in offsets.items():
            rf_ids_list = offset_ids[rf_curve_name]
            rf_offsets_arr = np.asarray(rf_offset_list, dtype=np.int32)
            rf_ids_arr = np.asarray(rf_ids_list, dtype=np.int32)

            rf_offset_idx = np.argsort(rf_offsets_arr, kind="mergesort")

            out[rf_curve_name] = (
                rf_offsets_arr[rf_offset_idx],
                rf_ids_arr[rf_offset_idx],
                labels.get(rf_curve_name, {}),
            )

        for rf_curve_name, rf_curve_label_to_rf_id in labels.items():
            if rf_curve_name not in out:
                out[rf_curve_name] = (
                    np.empty(0, dtype=np.int32),
                    np.empty(0, dtype=np.int32),
                    rf_curve_label_to_rf_id,
                )

        object.__setattr__(self, "_curve_tenor_index", out)
        return out

def write_config_dict_to_llm_txt(
    *,
    data: Dict[str, pd.DataFrame],
    output_path: Union[str, Path],
) -> None:
    """
    Serialize config DataFrames into an LLM-friendly text format.

    Design goals:
        - Deterministic
        - Token efficient
        - Easily parsable by LLMs
        - No dataframe truncation
        - Explicit structure boundaries
    """
    output_path = Path(output_path)

    with output_path.open("w", encoding="utf-8", newline="\n") as f:

        for sheet_name, df in data.items():

            # Ensure stable column order
            df = df.copy(deep=False)

            f.write(f"=== SHEET: {sheet_name} ===\n")
            f.write(f"ROWS: {len(df)}\n")
            f.write("COLUMNS: " + ", ".join(map(str, df.columns)) + "\n\n")

            # CSV is extremely LLM-friendly.
            # Use na_rep="" instead of .fillna("") to safely handle strict dtypes like Int64.
            df.to_csv(
                f,
                index=False,
                lineterminator="\n",
                na_rep="",
            )

            f.write("\n=== END SHEET ===\n\n")


def load_risk_static_data_config_xlsx_into_dict(
    *,
    path: Union[str, Path],
    sheet_names: Optional[Iterable[str]] = None,
) -> Dict[str, pd.DataFrame]:
    """
    Load selected sheets from a config workbook.

    Notebook-safe:
      - Ensures the Excel file handle is closed so the workbook can be saved/edited after load.
    Memory/CPU efficient:
      - Reads selected sheets in one call where possible.
      - Avoids unnecessary copies.
    """
    path_str = str(path)

    # Use a context manager to guarantee the file handle is closed
    with pd.ExcelFile(path_str, engine="openpyxl") as xls:
        available = xls.sheet_names

        if sheet_names is None:
            sheets = available  # reuse list
        else:
            # materialize once; also ensures list-like -> dict return
            sheets = tuple(sheet_names)

        missing = tuple(s for s in sheets if s not in available)
        if missing:
            raise ValueError(
                f"Missing required sheet(s) in '{path_str}': {missing}"
            )

        # One read call for all requested sheets
        dict_of_dfs = pd.read_excel(xls, sheet_name=sheets)

    # Validate that dfs is a mapping of sheet_name -> DataFrame
    if not isinstance(dict_of_dfs, Mapping):
        raise ValueError(
            "Expected multiple sheets to return a Mapping[str, DataFrame]."
        )

    # Clean the DataFrames to ensure consistent dtypes and memory layouts.
    clean_dict_of_dfs = {}
    for sheet_name, df in dict_of_dfs.items():
        # Convert dtypes: floats with NaNs become nullable Int64, objects become strings.
        # This prevents precision leaks and enforces tight memory layouts.
        clean_dict_of_dfs[sheet_name] = df.convert_dtypes()

    # Return a regular dict to avoid potential issues with pandas' internal dict subclass
    return clean_dict_of_dfs

# src/riskquant/config/resolution.py
from __future__ import annotations

from bisect import bisect_left
from dataclasses import dataclass
from datetime import date
from typing import (
    Dict,
    Mapping,
    Sequence,
    Tuple,
    overload,

import numpy as np
import pandas as pd

# Ensure these utility functions are available in riskquant.utils.tenors
from riskquant.config.configs import RiskStaticConfig
from riskquant.utils.tenors import (
    add_months_to_date_and_return_yyyymm01,
    fmt_date_as_str_yyyymm,
    get_spot_year,
    parse_bracket_token,
)


def build_risk_factor_name_mapping_for_as_of_date(
    *,
    config: RiskStaticConfig,
    as_of_date: date,
) -> RiskFactorNameAbsToRelMapping:
    """
    Resolve relative risk factor names into absolute names for a given as_of_date.

    Design:
        - Interrogates `config.risk_factors_dict` to define the universe of valid factors.
        - Translates each relative factor to its absolute name based on spot roll rules.
    """
    abs_to_rf_name_and_id: Dict[str, Tuple[str, int]] = {}

    for rf in config.risk_factors_dict.values():
        rf_rel_name = rf.risk_factor_name
        rf_rel_name_id = rf.risk_factor_name_id
        rf_container_name = rf.container_name
        rf_tenor_token = rf.tenor_token

        # Get container meta
        rf_cm = config.container_meta_dict.get(rf_container_name)
        if rf_cm is None:
            raise ValueError(
                "risk_factors_dict references unknown container_name.\n"
                f"container_name={rf_container_name!r}\n"
                f"relative={rf_rel_name!r}"
            )

        # ======================================================== #
        # === HANDLE SCALAR CONTAINERS (FX SPOT, EQUITIES, ETC) == #
        # ======================================================== #
        if rf_cm.alias_type_name is None:
            if rf_tenor_token is not None:
                raise ValueError(
                    f"Risk factor ID tenor_token ({rf_tenor_token}) must be None when alias_type_name is None.\n"
                    f"relative={rf_rel_name!r}\n"
                    "container_name={rf_container_name!r}"
                )
            # Scalar factors just map to themselves (no term structure to roll)
            abs_to_rf_name_and_id[rf_rel_name] = (rf_rel_name, rf_rel_name_id)
            continue

        # Get alias schema
        rf_alias = config.alias_schema_dict.get(rf_cm.alias_type_name)
        if rf_alias is None:
            raise ValueError(
                "container_meta_dict references unknown alias_type_name.\n"
                f"container_name={rf_cm.container_name!r}\n"
                "alias_type_name={rf_cm.alias_type_name!r}"
            )

        # ================================ #
        # === RESOLVE TO ABSOLUTE NAME === #
        # ================================ #

        if rf_alias.alias_kind == "NONE":
            if rf_tenor_token is not None:
                raise ValueError(
                    f"Risk factor ID tenor_token ({rf_tenor_token}) must be None for alias_kind = 'NONE'.\n"
                    f"relative={rf_rel_name!r}\ncurve_name={rf_container_name!r}"
                )
            abs_to_rf_name_and_id[rf_rel_name] = (rf_rel_name, rf_rel_name_id)
            continue

        elif rf_alias.alias_kind == "LABEL":
            if rf_tenor_token is None:
                raise ValueError(
                    "Risk factor missing tenor_token for alias_kind = 'LABEL'.\n"
                    f"relative={rf_rel_name!r}\ncurve_name={rf_container_name!r}"
                )
            label, signed_offset = parse_bracket_token(rf_tenor_token)
            if signed_offset is not None:
                raise ValueError(
                    f"alias_kind = 'LABEL' tenor_token ('{rf_tenor_token}') must not have +/- offset.\n"
                    f"relative={rf_rel_name!r}\ncurve_name={rf_container_name!r}"
                )
            rf_abs_name = f"{rf_container_name}_{label}"
            abs_to_rf_name_and_id[rf_abs_name] = (rf_rel_name, rf_rel_name_id)
            continue

        elif rf_alias.alias_kind == "OFFSET":
            if rf_tenor_token is None:
                raise ValueError(
                    "alias_kind = OFFSET risk factor missing tenor_token.\n"
                    f"relative={rf_rel_name!r}\ncurve_name={rf_container_name!r}"
                )

            if (
                not rf_cm.spot_roll_rule_name
                or rf_cm.spot_roll_rule_name == "None"
            ):
                raise ValueError(
                    "alias_kind = OFFSET curve missing spot_roll_rule_name in container_meta.\n"
                    f"relative={rf_rel_name!r}\ncurve_name={rf_container_name!r}"
                )

            tenor_token_unit, tenor_token_signed_offset = parse_bracket_token(
                rf_tenor_token
            )
            if tenor_token_signed_offset is None:
                raise ValueError(
                    "alias_kind = OFFSET tenor_token must have +/- integer.\n"
                    f"relative={rf_rel_name!r}\ncurve_name={rf_container_name!r}\ntenor_token={rf_tenor_token!r}\n"
                )

            # Retrieve rule
            rf_srr = config.spot_roll_rules_dict.get(
                (rf_cm.spot_roll_rule_name, rf_cm.tenor_type_name)
            )
            if rf_srr is None:
                raise ValueError(
                    "Missing spot roll rule in spot_roll_rules_dict!\n"
                    f"spot_roll_rule_name={rf_cm.spot_roll_rule_name!r}\n"
                    f"tenor_type_name={rf_cm.tenor_type_name!r}\n"
                    f"relative={rf_rel_name!r}\ncurve_name={rf_container_name!r}"
                )

            rf_spot_roll_method = rf_srr.spot_roll_method
            rf_tenor_type_name = rf_cm.tenor_type_name

            rf_offset = int(rf_srr.offset_in_tenor_units or 0)
            rf_roll_month = int(rf_srr.roll_month or 0)
            rf_roll_day = int(rf_srr.roll_day or 0)
            rf_post_shift = int(rf_srr.post_roll_shift or 0)

            # Resolution Logic
            if (
                rf_tenor_type_name == "TENOR_YM"
                and rf_spot_roll_method == "CALENDAR_PERIOD"
            ):
                spot_month_yyyymm01 = add_months_to_date_and_return_yyyymm01(
                    d=date(as_of_date.year, as_of_date.month, 1),
                    months_to_add=rf_offset,
                )
                rf_abs_date_yyyymm01 = add_months_to_date_and_return_yyyymm01(
                    d=spot_month_yyyymm01,
                    months_to_add=int(tenor_token_signed_offset),
                )
                rf_abs_name = f"{rf_container_name}_{fmt_date_as_str_yyyymm(rf_abs_date_yyyymm01)}"
                abs_to_rf_name_and_id[rf_abs_name] = (
                    rf_rel_name,
                    rf_rel_name_id,
                )
                continue

            elif (
                rf_tenor_type_name == "TENOR_Y"
                and rf_spot_roll_method == "ROLL_ON_MMDD"
            ):
                spot_year = get_spot_year(
                    as_of_date=as_of_date,
                    roll_month=rf_roll_month,
                    roll_day=rf_roll_day,
                    post_roll_shift=rf_post_shift,
                )
                year = spot_year + int(tenor_token_signed_offset)
                rf_abs_name = f"{rf_container_name}_{year:04d}"
                abs_to_rf_name_and_id[rf_abs_name] = (
                    rf_rel_name,
                    rf_rel_name_id,
                )
                continue

            elif (
                rf_tenor_type_name == "TENOR_YMD"
                and rf_spot_roll_method == "HALF_YEAR_BY_DATE"
            ):
                fh_bh_ddmm = rf_srr.fh_bh_ddmm
                if not fh_bh_ddmm:
                    raise ValueError(
                        "spot_roll_method = 'HALF_YEAR_BY_DATE' missing fh_bh_ddmm!\n"
                        f"spot_roll_rule_name={rf_cm.spot_roll_rule_name!r}\n"
                        f"relative={rf_rel_name!r}\ncurve_name={rf_container_name!r}"
                    )

                fh_mmdd, bh_mmdd = fh_bh_ddmm.split("|")
                if len(fh_mmdd) != 4 or len(bh_mmdd) != 4:
                    raise ValueError(
                        f"fh_bh_ddmm ('{fh_bh_ddmm}') must be of form 'ddmm|ddmm'!"
                    )

                as_of_year = as_of_date.year
                look_years = max(3, abs(int(tenor_token_signed_offset)) + 1)

                boundaries: list[date] = []
                for y in range(
                    as_of_year - look_years, as_of_year + look_years + 1
                ):
                    boundaries.append(
                        date(y, int(fh_mmdd[:2]), int(fh_mmdd[2:]))
                    )
                    boundaries.append(
                        date(y, int(bh_mmdd[:2]), int(bh_mmdd[2:]))
                    )
                boundaries.sort()

                idx_hy0 = bisect_left(boundaries, as_of_date)
                idx_target = idx_hy0 + int(tenor_token_signed_offset)

                if idx_target < 0 or idx_target >= len(boundaries):
                    raise ValueError(
                        "HALF_YEAR_BY_DATE offset out of range for constructed boundary grid.\n"
                        f"curve_name={rf_container_name!r}\nrelative={rf_rel_name!r}\n"
                    )

                abs_dt = boundaries[idx_target]
                yyyymmdd_str = (
                    f"{abs_dt.year:04d}{abs_dt.month:02d}{abs_dt.day:02d}"
                )
                rf_abs_name = f"{rf_container_name}_{yyyymmdd_str}"

                abs_to_rf_name_and_id[rf_abs_name] = (
                    rf_rel_name,
                    rf_rel_name_id,
                )
                continue

            else:
                raise ValueError(
                    "Unsupported OFFSET resolution combination: "
                    f"(tenor_type_name, rf_spot_roll_method) = ('{rf_tenor_type_name}', '{rf_spot_roll_method}')\n"
                    f"curve_name={rf_container_name!r}\nrelative={rf_rel_name!r}\n"
                )

    return RiskFactorNameAbsToRelMapping(
        as_of_date=as_of_date,
        abs_to_rf_name_and_id=abs_to_rf_name_and_id,
    )


@dataclass(frozen=True, slots=True)
class RiskFactorNameAbsToRelMapping:
    as_of_date: date
    abs_to_rf_name_and_id: Mapping[str, Tuple[str, int]]  # abs -> (rel, id)

    def __post_init__(self) -> None:
        seen: dict[int, str] = {}

        for abs_name, (rel_name, rf_id) in self.abs_to_rf_name_and_id.items():
            if not isinstance(rel_name, str):
                raise ValueError(
                    f"rel_name for '{abs_name}' must be str, got {type(rel_name).__name__}"
                )
            if not isinstance(rf_id, (int, np.integer)):
                raise ValueError(
                    f"id for '{abs_name}' must be an integer, got {type(rf_id).__name__}={rf_id!r}"
                )

            iv = int(rf_id)
            prev = seen.get(iv)
            if prev is not None and prev != abs_name:
                raise ValueError(
                    "Mapping contains duplicate ID assignment:\n"
                    f"  '{prev}' and '{abs_name}' both map to ID={iv}\n"
                    "Each absolute risk factor must map to a unique risk_factor_name_id."
                )
            seen[iv] = abs_name

    @staticmethod
    def from_config(
        config: RiskStaticConfig,
        as_of_date: date,
    ) -> "RiskFactorNameAbsToRelMapping":
        return build_risk_factor_name_mapping_for_as_of_date(
            config=config,
            as_of_date=as_of_date,
        )

    @overload
    def get(self, abs_name: str) -> tuple[str, int]: ...

    @overload
    def get(self, abs_name: Sequence[str]) -> list[tuple[str, int]]: ...

    def get(
        self,
        abs_name: str | Sequence[str],
    ) -> tuple[str, int] | list[tuple[str, int]]:
        m = self.abs_to_rf_name_and_id
        if isinstance(abs_name, str):
            return m[abs_name]
        return [m[a] for a in abs_name]

    @overload
    def get_rf_name_for_abs_name(self, abs_name: str) -> str: ...

    @overload
    def get_rf_name_for_abs_name(
        self, abs_name: Sequence[str]
    ) -> list[str]: ...

    def get_rf_name_for_abs_name(
        self,
        abs_name: str | Sequence[str],
    ) -> str | list[str]:
        m = self.abs_to_rf_name_and_id
        if isinstance(abs_name, str):
            return m[abs_name][0]
        return [m[a][0] for a in abs_name]

    @overload
    def get_rf_id_for_abs_name(self, abs_name: str) -> int: ...

    @overload
    def get_rf_id_for_abs_name(self, abs_name: Sequence[str]) -> list[int]: ...

    def get_rf_id_for_abs_name(
        self,
        abs_name: str | Sequence[str],
    ) -> int | list[int]:
        m = self.abs_to_rf_name_and_id
        if isinstance(abs_name, str):
            return m[abs_name][1]
        return [m[a][1] for a in abs_name]

    def to_df(self) -> pd.DataFrame:
        """
        Return a DataFrame with columns:
          [risk_factor_name, risk_factor_id]
        Rows are deterministically ordered by absolute_name (the index).
        """
        items = sorted(self.abs_to_rf_name_and_id.items(), key=lambda x: x[0])
        abs_names = [k for k, _ in items]
        rel_names = [v[0] for _, v in items]
        ids = np.fromiter(
            (int(v[1]) for _, v in items), dtype=np.int32, count=len(items)
        )

        df = pd.DataFrame(
            {
                "risk_factor_name": rel_names,
                "risk_factor_id": ids,
            },
            index=pd.Index(abs_names, name="absolute_risk_factor_name"),
        )
        return df

    @property
    def df(self) -> pd.DataFrame:
        """Alias for to_dataframe(), for convenient access."""
        return self.to_df()

    @property
    def abs_to_rf_name(self) -> Dict[str, str]:

"""
High-performance vectorized validation for risk static config data.

Design goals:
  - Fail-fast before memory allocation of runtime objects.
  - Zero native Python loops over dataframe rows.
  - Utilize NumPy/Pandas C-level vectorization for constraints (Unique, FK).
"""

from typing import Dict, Final

import numpy as np
import pandas as pd

from riskquant.config.loader import load_risk_static_data_config_xlsx_into_dict
from riskquant.utils.printer import (
    print_success_message,
)


def validate_static_config_dfs(dict_of_dfs: Dict[str, pd.DataFrame]) -> None:
    """
    Executes vectorized schema checks (uniqueness and foreign keys) across the
    raw configuration dataframes.

    Args:
        dict_of_dfs: The raw loaded dataframes mapped by sheet name.

    Raises:
        ValueError: On the first violation of schema constraints.
    """

    # --------------------------------------------------------- #
    # 1. UNIQUENESS CONSTRAINTS (Primary Keys / Alt Keys)
    # --------------------------------------------------------- #
    # Structure: sheet_name -> tuple of column names that must be unique
    unique_constraints = {
        "container_meta": ("container_name", "container_name_id"),
        "risk_factors": ("risk_factor_name", "risk_factor_name_id"),
        "asset_classes": ("asset_class", "asset_class_id"),
        "asset_subclasses": ("asset_subclass", "asset_subclass_id"),
        "unit_types": ("unit_type_name", "unit_type_name_id"),
        "unit_scales": ("unit_scale_name", "unit_scale_name_id"),
        "ccy": ("ccy", "ccy_id"),
        "shift_types": ("shift_type", "shift_type_id"),
        "container_proxies": ("container_name",),
        "risk_factor_proxies": ("risk_factor_name",),
    }

    for sheet_name, cols in unique_constraints.items():
        df = dict_of_dfs.get(sheet_name)
        if df is None or df.empty:
            continue

        for col in cols:
            if col not in df.columns:
                continue  # Missing columns handled separately if strictly required

            # .is_unique is an O(N) C-level check.
            if not df[col].is_unique:
                # Only pay the cost of finding the exact duplicates if we fail
                duplicates = df[col][df[col].duplicated()].unique()
                raise ValueError(
                    f"Validation failed: '{sheet_name}.{col}' must be unique. "
                    f"Duplicates found: {duplicates.tolist()}"
                )

    # --------------------------------------------------------- #
    # 2. FOREIGN KEY CONSTRAINTS (Referential Integrity)
    # --------------------------------------------------------- #
    # Structure: tuple(child_sheet, child_col) -> tuple(parent_sheet, parent_col)
    fk_constraints = {
        ("risk_factors", "container_name"): (
            "container_meta",
            "container_name",
        ),
        ("asset_subclasses", "asset_class"): ("asset_classes", "asset_class"),
    }

    for (c_sheet, c_col), (p_sheet, p_col) in fk_constraints.items():
        c_df = dict_of_dfs.get(c_sheet)
        p_df = dict_of_dfs.get(p_sheet)

        if (
            c_df is None
            or p_df is None
            or c_col not in c_df.columns
            or p_col not in p_df.columns
        ):
            continue

        # Extract unique arrays directly. Dropping NA ensures we don't flag optional/blank FKs.
        child_vals = c_df[c_col].dropna().unique()
        parent_vals = p_df[p_col].dropna().unique()

        # np.setdiff1d is a highly optimized C-level array operation.
        # It is significantly faster and uses less memory than Python set(A) - set(B).
        missing_fks = np.setdiff1d(child_vals, parent_vals, assume_unique=True)

        if missing_fks.size > 0:
            raise ValueError(
                f"Validation failed: '{c_sheet}.{c_col}' contains values not present in "
                f"'{p_sheet}.{p_col}'. Missing: {missing_fks.tolist()}"
            )

def validate_interpolation_methods_dict(
    interpolation_methods_dict: Mapping[str, int],
) -> None:
    expected = {
        name: int(member)
        for name, member in InterpolationMethod.__members__.items()
    }
    got = {str(k): int(v) for k, v in interpolation_methods_dict.items()}

    if got == expected:
        return

    missing = {k: expected[k] for k in expected.keys() - got.keys()}
    extra = {k: got[k] for k in got.keys() - expected.keys()}
    different = {
        k: {"expected": expected[k], "got": got[k]}
        for k in (expected.keys() & got.keys())
        if expected[k] != got[k]
    }

    raise ValueError(
        "interpolation_methods sheet does not match riskquant InterpolationMethod enum.\n"
        f"missing={missing}\n"
        f"extra={extra}\n"
        f"different={different}\n"
        "Fix the Excel interpolation_methods table or update riskquant.market_data.risk_factors.interpolation_methods.InterpolationMethod."

from __future__ import annotations

from pathlib import Path
from typing import Mapping, Optional, Union

import numpy as np
import pandas as pd
from pandas.api.types import is_datetime64_any_dtype

from riskquant.market_data.risk_factors.builder_utils import (
    _require_mapping_key,
    _validate_mapping_dict,
)
from riskquant.market_data.risk_factors.matrix import RiskFactorMatrix
from riskquant.utils.dates import (
    coerce_as_of_date_to_pd_timestamp,
    datetime64D_to_ordinal32,
)

# --- BUILD A RISK FACTOR MATRIX FROM A DATAFRAME


def build_risk_factor_matrix_from_df(
    *,
    df: pd.DataFrame,
    name_to_id_map: Optional[Mapping[str, int]] = None,
    use_int32_dates: bool = True,
    float_dtype: type[np.floating] = np.float32,
) -> RiskFactorMatrix:
    """Construct a RiskFactorMatrix from a wide pandas DataFrame.

    Parameters
    ----------
    df : pd.DataFrame
        Wide DataFrame with dates on the index and one column per risk factor.
    name_to_id_map : Optional[Mapping[str, int]]
        Optional mapping from column name (string) to integer factor id. If
        omitted, columns are parsed as integers directly.
    use_int32_dates : bool
        If True convert dates to int32 ordinal representation; otherwise
        keep datetime64[D] dtype.
    float_dtype : numpy floating type
        dtype used for the `values` array (default: np.float32).

    Returns
    -------
    RiskFactorMatrix
        Canonicalized matrix with validated shapes, dtypes and contiguity.

    Raises
    ------
    ValueError
        If the index is not date-like, not monotonic increasing, or contains
        duplicates; or if factor id mapping contains errors.

    Notes
    -----
    The function intentionally does not auto-sort the DataFrame rows
    (dates). If the provided index is unsorted we raise an error because
    auto-sorting would require an expensive O(T*N) copy of the values.
    """
    if not isinstance(df, pd.DataFrame):
        raise ValueError("Expected a DataFrame!")

    # Accept a wider set of date-like index types; coerce if needed.
    if not is_datetime64_any_dtype(df.index):
        # Convert in-place where possible; this preserves row order and
        # avoids unnecessary copying of the data matrix.
        df = df.copy(deep=False)
        df.index = pd.to_datetime(df.index, errors="raise")

    # Enforce invariants early with clear errors (and avoid needless allocations)
    if df.index.has_duplicates:
        raise ValueError("DataFrame index (dates) must be unique!")
    if not df.index.is_monotonic_increasing:
        raise ValueError("DataFrame index (dates) must be sorted ascending!")

    # Dates: obtain a datetime64[D] view and optionally convert to int32 ordinal
    dts_64D = df.index.to_numpy(dtype="datetime64[D]", copy=False)
    dates = datetime64D_to_ordinal32(dts_64D) if use_int32_dates else dts_64D

    # Columns -> factor_ids: either direct integer parsing or mapping lookup.
    cols = df.columns
    n = cols.size

    if name_to_id_map is None:
        # Parse column labels as integers; will raise on non-integer labels.
        factor_ids = np.fromiter(
            (int(c) for c in cols),
            dtype=np.int32,
            count=n,
        )
    else:
        # Validate mapping once (cheap for small config dicts) and then map.
        _validate_mapping_dict(name_to_id_map)
        factor_ids = np.fromiter(
            (_require_mapping_key(name_to_id_map, str(c)) for c in cols),
            dtype=np.int32,
            count=n,
        )

    # Canonicalize factor axis BEFORE allocating `values` (avoids extra copies).
    if n > 1 and np.any(factor_ids[1:] < factor_ids[:-1]):
        perm = np.argsort(factor_ids, kind="mergesort")
        factor_ids = factor_ids[perm]

        if np.any(factor_ids[1:] == factor_ids[:-1]):
            raise ValueError("factor_ids must be unique!")

        # Reorder the DataFrame columns to match the canonical factor order.
        df = df.iloc[:, perm]
    else:
        if n > 1 and np.any(factor_ids[1:] == factor_ids[:-1]):
            raise ValueError("factor_ids must be unique!")

    # Materialize the numeric matrix as a C-contiguous array with requested dtype.
    values = np.ascontiguousarray(
        df.to_numpy(dtype=float_dtype, copy=True, na_value=np.nan)
    )

    return RiskFactorMatrix(dates=dates, factor_ids=factor_ids, values=values)


# --- BUILD A RISK FACTOR MATRIX DIRECTLY FROM AN XLSX FILE


def build_risk_factor_matrix_from_xlsx(
    *,
    path: Union[str, Path],
    sheet_name: Union[str, int] = 0,
    name_to_id_map: Optional[Mapping[str, int]] = None,
    use_int32_dates: bool = True,
    float_dtype: type[np.floating] = np.float32,
    as_of_date: Optional[pd.Timestamp] = None,
) -> "RiskFactorMatrix":
    """Load a RiskFactorMatrix from an Excel workbook/sheet.

    Parameters
    ----------
    path : str | Path
        Path to the Excel file.
    sheet_name : str | int
        Sheet name or index to read (forwarded to pandas.read_excel).
    name_to_id_map : Optional[Mapping[str, int]]
        Optional mapping for column name -> factor id (same semantics as
        `from_df`).
    use_int32_dates : bool
        If True convert dates to int32 ordinal representation.
    float_dtype : numpy floating type
        dtype used for the `values` array.
    as_of_date : Optional[pd.Timestamp]
        If provided, rows with date > as_of_date are filtered out.

    Returns
    -------
    RiskFactorMatrix
        Canonicalized matrix constructed from the Excel sheet.
    """
    # Read sheet with the first column treated as the index (dates expected).
    df = pd.read_excel(
        str(path),
        sheet_name=sheet_name,
        index_col=0,
    )

    # If empty.
    if df.empty:
        raise ValueError("dataframe is empty - check: {path}")

    # Convert index to datetime if needed (common for Excel imports).
    if not is_datetime64_any_dtype(df.index):
        df.index = pd.to_datetime(df.index, errors="raise")

    # Optional filter by as_of_date; preserves monotonicity if present.
    if as_of_date is not None:
        cutoff = coerce_as_of_date_to_pd_timestamp(as_of_date)
        df = df.loc[df.index <= cutoff]

    # Enforce same sorted/unique contract as `from_df`.

    # Duplicated dates.
    if df.index.has_duplicates:
        duplicated = df.index[df.index.duplicated(keep=False)]
        unique_dupes = duplicated.unique()
        sample = unique_dupes[:10]
        raise ValueError(
            f"Excel sheet index (dates) must be unique.\n"
            f"Found {len(unique_dupes)} duplicated date(s).\n"
            f"Sample duplicates: {list(sample)}"
        )

    # Non-monotonic dates
    if not df.index.is_monotonic_increasing:
        raise ValueError("Excel sheet index (dates) must be sorted ascending!")

    # Reuse the robust `from_df` constructor for the remaining work.
    return build_risk_factor_matrix_from_df(
        df=df,
        name_to_id_map=name_to_id_map,
        use_int32_dates=use_int32_dates,
        float_dtype=float_dtype,
    )

from __future__ import annotations

from datetime import date
from pathlib import Path
from typing import Dict, Mapping, Optional, Union

import numpy as np
import pandas as pd

from riskquant.market_data.risk_factors.snapshot import RiskFactorLevelSnapshot

from riskquant.market_data.risk_factors.builder_utils import (
    _require_mapping_key,
    _validate_mapping_dict,
)


def build_risk_factor_level_snapshot_from_dict(
    *,
    as_of_date: date,
    levels_dict: Mapping[str, float],
    name_to_id_map: Optional[Mapping[str, int]] = None,
    float_dtype: type[np.floating] = np.float32,
) -> RiskFactorLevelSnapshot:
    """
    Build a snapshot from a dictionary mapping risk factor names to their levels.
    """
    if not levels_dict:
        raise ValueError("levels_dict cannot be empty.")

    n = len(levels_dict)

    if name_to_id_map is None:
        # Assume the keys in the dictionary are already integer IDs formatted as strings.
        ids = np.fromiter(
            (int(k) for k in levels_dict.keys()), dtype=np.int32, count=n
        )
    else:
        # Resolve names to IDs
        _validate_mapping_dict(name_to_id_map)
        ids = np.fromiter(
            (
                _require_mapping_key(name_to_id_map, str(k))
                for k in levels_dict.keys()
            ),
            dtype=np.int32,
            count=n,
        )

    # Extract levels into a contiguous array
    levels = np.fromiter(levels_dict.values(), dtype=float_dtype, count=n)

    return RiskFactorLevelSnapshot(
        as_of_date=as_of_date,
        factor_ids=ids,
        factor_levels=levels,
    )


def build_risk_factor_level_snapshot_from_df(
    *,
    as_of_date: date,
    df: pd.DataFrame,
    factor_col: str = "risk_factor_name",
    level_col: str = "level",
    name_to_id_map: Optional[Mapping[str, int]] = None,

from typing import Any, Mapping

import numpy as np


def _validate_mapping_dict(m: Mapping[str, Any]) -> None:
    """Validate a small static mapping of risk factor name -> integer id.

    Checks performed:
      - keys must be non-empty strings
      - values must be integer-like (reject bool, float, str)
      - no two keys map to the same integer id

    This helper is optimized for configuration-time validation (small dicts),
    not for per-row lookups in tight loops.
    """
    # Empty/blank key detection (single pass)
    for k in m.keys():
        if not isinstance(k, str) or not k.strip():
            raise ValueError(f"Mapping contains invalid/blank key: {k!r}")

    # Ensure values are integer-like and detect duplicates by tracking seen ids.
    seen: dict[int, str] = {}
    for k, v in m.items():
        # `bool` is a subclass of int in Python; explicitly reject it here because
        # booleans are not valid factor ids.
        if isinstance(v, (bool,)):
            raise ValueError(
                f"Mapping value for '{k}' must be an int id, got bool={v!r}"
            )

        # Accept Python ints and NumPy integer scalars; reject other types.
        if not isinstance(v, (int, np.integer)):
            raise ValueError(
                f"Mapping value for '{k}' must be an int id, got {type(v).__name__}={v!r}"
            )

        iv = int(v)
        prev = seen.get(iv)
        if prev is not None:
            # Clear, actionable error message when multiple names map to same id.
            raise ValueError(
                "Mapping contains duplicate ID assignment:\n"
                f"  '{prev}' and '{k}' both map to ID={iv}\n"
                "Each risk factor must have a unique risk_factor_name_id."
            )
        seen[iv] = k


def _require_mapping_key(

    key: str,

    """Lookup `key` in mapping `m` and raise a detailed error if missing.

    showing a short preview of known keys and a suggested fix. Designed to be
    inexpensive on the success path (single hash lookup); the preview list is
    materialized only on error.

    Parameters
    ----------
    m : Mapping[str, Any]
        Mapping from risk factor name to id.
    key : str
        Key to lookup (column label converted to string prior to lookup).
    preview_n : int
        Maximum number of known keys to show in the error message.
    """

        return m[key]
    except KeyError:
        # Materialize a small preview of keys only when diagnosing a failure --
        # prevents huge error messages for very large config maps.
        keys = list(m.keys())
        n = len(keys)
        preview = keys[:preview_n]
        preview_str = ", ".join(preview) + (" ..." if n > preview_n else "")

            "Fix:\n"
            "  Add this risk factor to the STATIC config table that defines risk_factor_name_id.\n"
            "  Example row to add:\n"
            f"    risk_factor_name = {key}\n"
            "    risk_factor_name_id = <new unique integer>\n\n"
            f"Known keys preview ({min(preview_n, n)} of {n}):\n"
            f"  {preview_str}\n"
        ) from None

    """
    Canonical invariants (enforced):
      - dates: 1D, sorted ascending, unique
      - factor_ids: 1D int32, sorted ascending, unique
      - values: 2D floating, shape (T, N), C-contiguous
      - shapes aligned: len(dates)=T, len(factor_ids)=N

    Canonicalization:
      - factor axis is automatically sorted (and values columns permuted) if needed.
      - dates are validated as sorted+unique (not auto-sorted) to avoid an O(T*N) row copy.
    """

    dates: np.ndarray  # datetime64[D] or int32 ordinal
    factor_ids: np.ndarray  # int32, shape (N,)
    values: np.ndarray  # float32/float64, shape (T, N), contiguous

    def __post_init__(self) -> None:
        """Validate and canonicalize a newly-created RiskFactorMatrix.

        Responsibilities:
          - verify simple shape/dtype/contiguity invariants cheaply (no heavy copies)
          - enforce that date encoding is either int32 ordinal or datetime64[D]
          - ensure `factor_ids` are int32 and unique; *automatically* sort + permute columns
            if the provided factor_ids are not sorted (keeps values aligned)

        The method raises clear, early exceptions to avoid subtle bugs later in
        numerical code that assumes the canonical invariants described on the
        class docstring.
        """
        # ---- Basic dimensionality checks (cheap) ----
        if self.dates.ndim != 1:
            raise ValueError("dates must be 1D!")
        if self.factor_ids.ndim != 1:
            raise ValueError("factor_ids must be 1D!")
        if self.values.ndim != 2:
            raise ValueError("values must be 2D!")

        # ---- dtype invariants ----
        # `values` must be a floating-point array (float32/float64 allowed).
        if not np.issubdtype(self.values.dtype, np.floating):
            raise TypeError(
                f"values must be floating dtype; got {self.values.dtype}"
            )

        # Dates may be encoded as int32 ordinals or as datetime64[D].
        # If integer-encoded, require int32 to avoid accidental upcasting.
        if np.issubdtype(self.dates.dtype, np.integer):
            if self.dates.dtype != np.int32:
                raise TypeError(
                    f"int date encoding must be int32 days since DATE_EPOCH; got {self.dates.dtype}"
                )
        else:
            if self.dates.dtype != np.dtype("datetime64[D]"):
                raise TypeError(
                    f"datetime date encoding must be datetime64[D]; got {self.dates.dtype}"
                )

        # Factor IDs must be 32-bit integers for compact storage and stable hashing.
        if self.factor_ids.dtype != np.int32:
            raise TypeError(
                f"factor_ids must be int32; got {self.factor_ids.dtype}"
            )

        # ---- shape invariants ----
        T, N = self.values.shape
        if self.dates.size != T:
            raise ValueError(
                f"dates length ({self.dates.size}) must match values.shape[0] ({T})"
            )
        if self.factor_ids.size != N:
            raise ValueError(
                f"factor_ids length ({self.factor_ids.size}) must match values.shape[1] ({N})"
            )

        # ---- contiguity invariant for values ----
        # Many numerical libraries expect C-contiguous 2D arrays; detect accidental
        # non-contiguous views early and ask callers to construct canonical arrays.
        if not self.values.flags["C_CONTIGUOUS"]:
            raise ValueError(
                "values must be C-contiguous (use np.ascontiguousarray at construction). If you sliced, "
                "call np.ascontiguousarray(...) or use the provided slice_* helpers."
            )

        # ---- dates: require sorted + unique (cheap, alloc-free checks) ----
        # We *do not* auto-sort dates because that would require an O(T*N) row copy.
        if T > 1:
            if np.any(self.dates[1:] < self.dates[:-1]):
                raise ValueError("dates must be sorted ascending.")
            if np.any(self.dates[1:] == self.dates[:-1]):
                raise ValueError("dates must be unique!")

        # ---- factor_ids: canonicalize to sorted + unique with minimal copying ----
        if N > 1:
            # Fast path: if already non-decreasing, just check uniqueness and return.
            if not np.any(self.factor_ids[1:] < self.factor_ids[:-1]):
                if np.any(self.factor_ids[1:] == self.factor_ids[:-1]):
                    raise ValueError("factor_ids must be unique!")
                return

            # Slow path: factor_ids are unsorted -> build a stable permutation and
            # reorder both `factor_ids` and the corresponding columns of `values`.
            # We use `mergesort` for determinism; `np.argsort` here allocates O(N).
            perm = np.argsort(self.factor_ids, kind="mergesort")
            sorted_ids = self.factor_ids[perm]

            # Check uniqueness after sorting (adjacent duplicates are easy to detect).
            if np.any(sorted_ids[1:] == sorted_ids[:-1]):
                raise ValueError("factor_ids must be unique!")

            # Permuting columns via fancy indexing produces a new array; ensure it's
            # C-contiguous so downstream code's contiguity assumptions hold.
            sorted_vals = np.ascontiguousarray(self.values[:, perm])

            # Bypass dataclass immutability to install canonicalized arrays.
            object.__setattr__(self, "factor_ids", sorted_ids)
            object.__setattr__(self, "values", sorted_vals)

    Rationale:
      - Code is the contract; Excel is data.
      - Kernels rely on stable integer codes.
    """
    # Expected mapping from enum
    expected: dict[str, int] = {
        name: int(member) for name, member in ShiftType.__members__.items()
    }

    got: dict[str, int] = {str(k): int(v) for k, v in shift_types_dict.items()}

    extra = {k: got[k] for k in got.keys() - expected.keys()}
    different = {
        k: {"expected": expected[k], "got": got[k]}
        for k in (expected.keys() & got.keys())
        if expected[k] != got[k]
    }

from __future__ import annotations

from dataclasses import dataclass
from datetime import date

import numpy as np
import pandas as pd

from riskquant.utils.dates import ordinal32_to_datetime64D


@dataclass(frozen=True, slots=True)
class RiskFactorLevelSnapshot:
    """Immutable snapshot of risk factor levels at a single as-of date.

    Attributes
    ----------
    as_of_date : datetime.date
        The calendar date for which the snapshot applies.
    factor_ids : np.ndarray[int32]
        1-D array of unique risk-factor identifiers (dtype int32, C-contiguous).
        Must be sorted ascending; if not sorted, the constructor will re-order
        the ids and corresponding `factor_levels`.
    factor_levels : np.ndarray[float32|float64]
        1-D numeric array of factor levels aligned with `factor_ids`.
        Must be the same length as `factor_ids` and C-contiguous.

    Invariants (enforced on construction)
    -------------------------------------
    - `factor_ids` is 1D, dtype int32, C-contiguous, sorted ascending and unique
    - `factor_levels` is 1D, dtype float32 or float64, C-contiguous
    - lengths of `factor_ids` and `factor_levels` match

    The class is intentionally lightweight and validates eagerly so downstream
    numerical kernels can assume compact, contiguous, and deterministic
    memory layouts (fast gathers, stable hashing, etc.).
    """

    as_of_date: date
    factor_ids: np.ndarray  # (N0,) int32
    factor_levels: np.ndarray  # (N0,) float32/float64

    def __post_init__(self) -> None:
        """Validate and canonicalize snapshot arrays.

        The method performs cheap, early checks and performs the minimal amount
        of work required to restore canonical invariants (sorting + contiguity).
        It raises clear exceptions on invalid input rather than attempting to
        silently coerce data into a valid form.
        """
        # Local aliases to make subsequent checks concise and explicit.
        ids = self.factor_ids
        lv = self.factor_levels

        # ----- shape checks -----
        # Ensure both arrays are 1-D vectors.
        if ids.ndim != 1:
            raise ValueError(f"factor_ids must be 1D; got shape={ids.shape}")
        if lv.ndim != 1:
            raise ValueError(f"factor_levels must be 1D; got shape={lv.shape}")

        # Ensure arrays have the same length (alignment requirement).
        if ids.shape[0] != lv.shape[0]:
            raise ValueError(
                f"factor_ids and factor_levels length mismatch: "
                f"{ids.shape[0]} vs {lv.shape[0]}"
            )

        # ----- dtype checks -----
        # factor_ids must be 32-bit integers for compactness and predictable
        # behavior when used as indices or keys in mappings.
        if ids.dtype != np.int32:
            raise TypeError(f"factor_ids must be int32; got {ids.dtype}")

        # factor_levels must be floating point (single or double precision).
        if lv.dtype not in (np.float32, np.float64):
            raise TypeError(
                f"factor_levels must be float32 or float64; got {lv.dtype}"
            )

        # ----- memory layout checks -----
        # Many numerical routines assume C-contiguous 1-D arrays; check early so
        # callers get an explicit error instead of subtle performance bugs.
        if not ids.flags["C_CONTIGUOUS"]:
            raise ValueError("factor_ids must be C-contiguous")
        if not lv.flags["C_CONTIGUOUS"]:
            raise ValueError("factor_levels must be C-contiguous")

        # Fast path: zero or one element requires no further sorting/uniqueness checks.
        n = ids.size
        if n <= 1:
            return

        # ----- sorted + unique enforcement -----
        # If ids are already non-decreasing, perform an adjacent-duplicate check
        # which is allocation-free. If not sorted, build a stable permutation and
        # reorder both arrays so the object is canonical afterwards.
        if not np.any(ids[1:] < ids[:-1]):
            # ids are non-decreasing — ensure strict uniqueness.
            if np.any(ids[1:] == ids[:-1]):
                raise ValueError("factor_ids must be unique")
            return

        # ids are unsorted: compute a stable sort permutation (mergesort).
        perm = np.argsort(ids, kind="mergesort")

        # Apply the permutation to ids (produces a new array view/copy).
        ids_sorted = ids[perm]

        # Check for duplicates after sorting (adjacent duplicates are easy to detect).
        if np.any(ids_sorted[1:] == ids_sorted[:-1]):
            raise ValueError("factor_ids must be unique")

        # Reorder factor levels to keep alignment with the now-sorted ids.
        lv_sorted = lv[perm]

        # Fancy indexing typically returns a contiguous array for 1-D, but
        # explicitly enforce C-contiguity to preserve the class invariant.
        if not lv_sorted.flags["C_CONTIGUOUS"]:
            lv_sorted = np.ascontiguousarray(lv_sorted)

        # Bypass dataclass immutability to install the canonicalized arrays.
        object.__setattr__(self, "factor_ids", ids_sorted)
        object.__setattr__(self, "factor_levels", lv_sorted)

    def to_df(self) -> pd.DataFrame:
        """Return the snapshot as a 1-row wide DataFrame."""
        # Reshape the 1D levels array into a 2D row vector (1, N)
        data_2d = self.factor_levels[np.newaxis, :]

        df = pd.DataFrame(
            data=data_2d,
            index=pd.DatetimeIndex([self.as_of_date], name="date"),
            columns=pd.Index(self.factor_ids, name="factor_id"),
        )
        return df

    def to_series(self) -> pd.Series:
        """Return the snapshot as a pandas Series."""
        return pd.Series(
            data=self.factor_levels,
            index=pd.Index(self.factor_ids, name="factor_id"),
            name=self.as_of_date,
        )

    @property
    def df(self) -> pd.DataFrame:
        """Alias for to_df()."""
        return self.to_df()

    @property

from __future__ import annotations

import datetime as dt
from typing import Final, Iterable, List, Tuple, Union

import numpy as np
import pandas as pd

DATE_EPOCH: Final[dt.date] = dt.date(1970, 1, 1)
DATE_EPOCH_AS_DATETIME64D: Final[np.datetime64] = np.datetime64(
    DATE_EPOCH.isoformat(),
    "D",
)


def coerce_as_of_date_to_pd_timestamp(
    as_of_date: pd.Timestamp | dt.date | pd.Timestamp | str,
) -> pd.Timestamp:
    """
    Accepts: pd.Timestamp | datetime.datetime | datetime.date | str
    Returns: pd.Timestamp (tz-naive)
    """
    # If str, validate format.
    if isinstance(as_of_date, str):
        assert_date_str_is_iso_format(as_of_date)

    # As pd.Timestamp
    ts = pd.Timestamp(as_of_date)

    # If caller passed a date (no time), treat it as end-of-day inclusive
    # so 2026-01-02 includes all times on 2026-01-02.
    if isinstance(as_of_date, dt.date) and not isinstance(
        as_of_date, dt.datetime
    ):
        ts = ts + pd.Timedelta(days=1) - pd.Timedelta(nanoseconds=1)

    # Normalize tz handling (if any). Adjust policy as you like.
    if ts.tzinfo is not None:
        ts = ts.tz_convert(None)

    return ts


def is_date_str_valid(date_str: str) -> bool:
    """Checks if date_str which must have form %Y-%m-%d is a valid calendar date."""
    if not isinstance(date_str, str):
        return False
    if len(date_str) != 10 or date_str[4] != "-" or date_str[7] != "-":
        return False
    if not date_str.replace("-", "").isdigit():
        return False

    try:
        pd.to_datetime(date_str, format="%Y-%m-%d", errors="raise")
    except ValueError:
        return False

    return True


def assert_date_str_is_iso_format(date_str: str) -> None:
    """Assert that date_str is in ISO format 'YYYY-MM-DD'."""
    if not is_date_str_valid(date_str):
        raise ValueError(
            f"Date string '{date_str}' is not in 'YYYY-MM-DD' format."
        )


def date_str_to_date(date_str: str) -> dt.date:
    """Convert a date string in 'YYYY-MM-DD' format to a datetime.date object."""
    assert_date_str_is_iso_format(date_str)
    return pd.to_datetime(date_str, format="%Y-%m-%d", errors="raise").date()


def date_to_ordinal32(d: dt.date) -> np.int32:
    return np.int32((d - DATE_EPOCH).days)


def ordinal32_to_date(days: int) -> dt.date:
    return DATE_EPOCH + pd.Timedelta(days=int(days))  # or datetime.timedelta


def datetime64D_to_ordinal32(dts: np.ndarray) -> np.ndarray:
    """
    Convert numpy datetime64[D] array -> int32 days since DATE_EPOCH.
    Returns a view/copy depending on input; no pandas involvement.
    """
    dts = np.asarray(dts, dtype="datetime64[D]")
    return (dts - DATE_EPOCH_AS_DATETIME64D).astype(np.int32, copy=False)


def ordinal32_to_datetime64D(days: np.ndarray) -> np.ndarray:
    """
    Convert int32 days since DATE_EPOCH -> numpy datetime64[D] array.
    """
    days = np.asarray(days, dtype=np.int32)
    # Cast to timedelta64[D] then add epoch
    return DATE_EPOCH_AS_DATETIME64D + days.astype("timedelta64[D]", copy=False)


def ym_str_to_int_tuple(ym: str) -> tuple[int, int]:
    """Convert 'YYYYMM' -> (YYYY, MM)."""
    if len(ym) != 6 or not ym.isdigit():
        raise ValueError(f"Invalid YM string: {ym}")
    y, m = int(ym[:4]), int(ym[4:6])
    if not 1 <= m <= 12:
        raise ValueError(f"Invalid YM string: {ym}")
    return y, m


def ymd_str_to_date(ymd: str) -> dt.date:
    """Convert 'YYYYMMDD' -> datetime.date."""
    if len(ymd) != 8 or not ymd.isdigit():
        raise ValueError(f"Invalid YMD string: {ymd}")
    return dt.date(int(ymd[:4]), int(ymd[4:6]), int(ymd[6:8]))


def month_diff(
    *,
    anchor_ym: tuple[int, int],
    target_ym: tuple[int, int],
) -> int:
    """Difference in months: target - anchor."""
    ay, am = anchor_ym
    ty, tm = target_ym
    return (ty - ay) * 12 + (tm - am)


def next_half_year_boundary(
    t0: dt.date,
    ym_boundaries: Iterable[str] = ("0630", "1231"),
) -> dt.date:
    """
    Return the first boundary date >= t0 among {YYYY0630, YYYY1231} (or configured boundaries).
    If none remain in the year, return the first boundary in the next year.
    """
    ym_bounds = list(ym_boundaries)
    if not ym_bounds:
        raise ValueError("No half-year boundaries provided.")

    year = t0.year
    candidates = []
    for b in ym_bounds:
        if len(b) != 4 or not b.isdigit():
            raise ValueError(f"Invalid year-month boundary 'MMDD': {b}")
        m = int(b[:2])
        d = int(b[2:])
        candidates.append(dt.date(year, m, d))

    for d in sorted(candidates):
        if d >= t0:
            return d

    # none left this year -> first boundary next year (sorted)
    year += 1
    b0 = sorted(ym_bounds)[0]
    return dt.date(year, int(b0[:2]), int(b0[2:]))


def half_year_index(d: dt.date) -> tuple[int, int]:
    """
    Map boundary date to a half-year index:
      YYYY-06-30 -> (YYYY, 0)
      YYYY-12-31 -> (YYYY, 1)
    """
    if (d.month, d.day) == (6, 30):
        return d.year, 0
    if (d.month, d.day) == (12, 31):
        return d.year, 1
    raise ValueError(f"Not a recognized half-year boundary: {d.isoformat()}")


def half_year_diff(
    *,
    anchor: dt.date,
    target: dt.date,
) -> int:
    """Difference in half-years: target - anchor, where each boundary step counts as 1."""
    ay, ah = half_year_index(anchor)
    ty, th = half_year_index(target)
    return (ty - ay) * 2 + (th - ah)


def get_previous_business_date_as_iso_str(
    *,
    lag_in_bus_days: int = 0,
    holiday_dates: Union[List[str], Tuple[str, ...]] = None,
    time_str_to_append_to_date: str = None,
) -> str:
    # Ensure holidays is a list of strings.
    if holiday_dates is None:
        holiday_dates = []
    if isinstance(holiday_dates, tuple):
        holiday_dates = list(holiday_dates)
    if not isinstance(holiday_dates, list):
        raise TypeError("holiday_dates must be a list!")

    # Create a custom business day with the holidays.
    custom_business_day = pd.offsets.CustomBusinessDay(holidays=holiday_dates)

    # Start from today and apply the lag and one additional business day.
    start_date = dt.datetime.today().date()
    adjusted_date = start_date - custom_business_day * (lag_in_bus_days + 1)
    date_as_str = adjusted_date.strftime("%Y-%m-%d")

    # Append the time string if provided, and return.
    if time_str_to_append_to_date is not None:
        return date_as_str + " {}".format(time_str_to_append_to_date)
    return date_as_str


def get_previous_business_date_as_date(
    *,
    lag_in_bus_days: int = 0,
    holiday_dates: Union[List[str], Tuple[str, ...]] = None,
    time_str_to_append_to_date: str = None,
) -> dt.date:
    # Get as string.
    date_string = get_previous_business_date_as_iso_str(
        lag_in_bus_days=lag_in_bus_days,
        holiday_dates=holiday_dates,
        time_str_to_append_to_date=time_str_to_append_to_date,
    )

    # To date.
    return dt.date.fromisoformat(date_string)

def as_list_str(x: Iterable) -> str:

def print_wrapped_text_box(
    text: str,
    *,
    width: int = 70,
    padding: int = 1,
    corner_char: str = "+",
    top_border: str = "-",
    bottom_border: str = "-",
    horiz_border: str = "|",
    add_space_before: bool = True,
    add_space_after: bool = True,
) -> None:
    """Word‑wrap `text` to `width` characters and display it in an ASCII box.

    The box will have a single‑character border and the given number of spaces
    between the text and the vertical edges.

    Example:

        +-------------------------------+
        | This is a long sentence that  |
        | will be wrapped automatically |
        | and displayed in a neat box.  |
        +-------------------------------+

    Args:
        text: string to render.
        width: max characters per line inside the box.
        padding: spaces between text and left/right borders.
    """
    import textwrap  # local import keeps top‑level imports unchanged

    lines = textwrap.wrap(text, width=width)
    if not lines:
        lines = [""]
    max_len = max(len(line) for line in lines)

    inner_width = max_len + padding * 2
    horiz = corner_char + top_border * inner_width + corner_char

    if add_space_before:
        print()
    print(horiz)
    for line in lines:
        print(
            horiz_border
            + " " * padding
            + line.ljust(max_len)
            + " " * padding
            + horiz_border
        )
    print(corner_char + bottom_border * inner_width + corner_char)
    if add_space_after:
        print()


def print_success_message() -> None:
    print_wrapped_text_box(
        "SUCCESS! CODE RAN SUCCESSFULLY!",
        width=60,
        padding=2,
        corner_char="*",
        top_border="=",
        bottom_border="=",
    )


def print_dict(
    d: Mapping[str, Any],
    *,
    indent: int = 0,
    indent_step: int = 4,
    sort_keys: bool = False,
) -> None:
    """
    Pretty-print a dictionary with aligned keys and nested support.

    Designed for low memory overhead:
        - No large intermediate strings
        - Streams output directly via print
        - Minimal recursion footprint

    Args:
        d: Dictionary (or mapping) to print.
        indent: Initial indentation level.
        indent_step: Spaces added per nesting level.
        sort_keys: If True, keys are printed alphabetically.
    """
    if not isinstance(d, Mapping):
        raise TypeError("print_dict_pretty expects a mapping type.")

    items = sorted(d.items()) if sort_keys else d.items()

    # Compute key width once for alignment
    try:
        max_key_len = max(len(str(k)) for k in d.keys())
    except ValueError:  # empty dict
        max_key_len = 0

    pad = " " * indent

    print(pad + "{")
    for key, value in items:
        key_str = str(key)
        key_pad = key_str.ljust(max_key_len)

        if isinstance(value, Mapping):
            print(pad + " " * indent_step + f"{key_pad}:")
            print_dict(
                value,
                indent=indent + indent_step * 2,
                indent_step=indent_step,
                sort_keys=sort_keys,
            )
        else:
            print(pad + " " * indent_step + f"{key_pad}: {value}")

    print(pad + "}")


class _FloatFormatter:
    __slots__ = ("_fmt", "_zero_tol")

    def __init__(self, spec: str, zero_tol: float = 0.0) -> None:
        # spec like ".2f", "_.2f", ",.3f"
        self._fmt = "{:" + spec + "}"
        self._zero_tol = float(zero_tol)

    def __call__(self, x: float) -> str:
        # pandas will only call this for float entries
        if x is None or (isinstance(x, float) and math.isnan(x)):
            return "nan"
        if self._zero_tol and abs(x) < self._zero_tol:
            x = 0.0
        return self._fmt.format(x)


def print_df(

# riskquant/utils/tenors.py
from __future__ import annotations

from datetime import date
from typing import Optional, Tuple


def parse_bracket_token(tenor_token: str) -> Tuple[str, Optional[int]]:
    """
    "[Y-2]" -> ("Y", -2)
    "[M+0]" -> ("M", 0)
    "[D+24]" -> ("D", 24)
    "[HY+1]" -> ("HY", 1)
    "[ON]" -> ("ON", None)   (LABEL)
    """
    # Error.
    if tenor_token is None:
        raise ValueError("tenor_token cannot be None!")

    # Strip white space.
    s = tenor_token.strip()

    # Must start with '[' and end with ']'.
    if s.startswith("[") and s.endswith("]"):
        s = s[1:-1].strip()

    # Find the plus or minus sign.
    plus = s.rfind("+")
    minus = s.rfind("-")
    sign_pos = max(plus, minus)

    # If no sign.
    if sign_pos <= 0:
        return s, None

    # Unit (Y, M, D, HY, etc.)
    unit = s[:sign_pos]

    # Signed offset.
    offset = int(s[sign_pos:])

    # Return.
    return unit, offset


def get_spot_year(
    *,
    as_of_date: date,
    roll_month: int,
    roll_day: int,
    post_roll_shift: int,
) -> int:
    """
    Defines which delivery year corresponds to Y+0 under ROLL_ON_MMDD.
    """

    # Get the roll date,
    roll_date = date(as_of_date.year, roll_month, roll_day)

    # Compare to as-of-date.
    # As-of-date is on or after the roll date.
    if as_of_date >= roll_date:
        return (
            as_of_date.year + post_roll_shift
        )  # Current year plus the post_roll_shift amount (in years).

    # As-of-date is before the roll date.
    # Previous year plus the post_roll_shift amount (in years).
    return (as_of_date.year - 1) + post_roll_shift


def add_months_to_date_and_return_yyyymm01(
    *,
    d: date,
    months_to_add: int,
) -> date:
    """Add months to a month-start date (day=1)."""
    y = d.year + (d.month - 1 + months_to_add) // 12
    m = (d.month - 1 + months_to_add) % 12 + 1
    return date(y, m, 1)
