# riskquant/portfolio/payloads/base.py
from dataclasses import dataclass


@dataclass(frozen=True, slots=True)
class Payload:
    """
    Marker base class for instrument payloads.
    No methods; payloads are compiled into blocks.
    """

    def __repr__(self) -> str:
        return self.__str__()


# ===== FILE: portfolio\payloads\forward.py =====

"""
Payload definition for forward contract trades.

This module defines a lightweight, immutable container for forward positions.
The payload stores only numeric identifiers, primitive fields, and date ordinals
(int32 days-since-epoch). External pricers and valuation engines resolve the
identifiers to market data and perform pricing, discounting, and FX conversions.

Key design principle:
  Exposure timing (via `price_rf_id` and `price_source_type`) is intentionally
  decoupled from settlement timing (`settlement_date_ordinal32`). This enables
  support for forwards on long-dated production with arbitrary payment schedules,
  power and commodity contracts, and custom discounting logic.
"""

from dataclasses import dataclass

import numpy as np

# Import price source type indicators and date/FX sentinels for defaults
from riskquant.portfolio.constants import (
    PRICE_SOURCE_NODE,
    PRICE_SOURCE_CURVE,
    NO_DELIVERY_DATE_ORDINAL32,
    NO_SETTLEMENT_DATE_ORDINAL32,
    NO_FX_RF_REQUIRED,
)

# Import base Payload interface for type polymorphism
from riskquant.portfolio.payloads.base import Payload


@dataclass(frozen=True, slots=True)
class ForwardPayload(Payload):
    """
    Immutable payload for a forward contract position.

    Encodes the minimal economic and market-linkage information required
    by pricers and valuation engines. All fields use compact NumPy dtypes
    to support efficient array-based processing and serialization.

    The payload intentionally separates:
      - Exposure timing (price_rf_id + price_source_type): defines which
        market curve/node is shocked in scenarios.
      - Settlement timing (delivery_date_ordinal32 + settlement_date_ordinal32):
        used for payment and discounting logic.

    This design enables:
      - Forwards on long-dated underlying with custom payment schedules.
      - Decoupled scenario shock and valuation logic.
      - Flexible discounting strategies (can be disabled via sentinel values).
    """

    # ========== UNDERLYING LINKAGES ==========

    # Unique identifier for the forward price/curve node used by the pricer.
    # Resolves to a time series or market surface in scenario shocks.
    price_rf_id: np.int32

    # ========== ECONOMIC TERMS ==========

    # Signed quantity/notional (long > 0, short < 0). dtype: np.float64
    quantity: np.float64

    # Strike/forward level K in trade currency units. dtype: np.float64
    strike: np.float64

    # ========== PRICE SOURCE TYPE ==========

    # Indicator for how to interpret price_rf_id (0=NODE, 1=CURVE).
    # Helps pricers select the correct market data interpretation.
    price_source_type: np.int8 = PRICE_SOURCE_NODE

    # ========== DELIVERY & SETTLEMENT DATES (ORDINALS) ==========

    # Physical delivery date encoded as int32 ordinal (days since DATE_EPOCH).
    # Default is sentinel NO_DELIVERY_DATE_ORDINAL32 if not applicable.
    delivery_date_ordinal32: np.int32 = NO_DELIVERY_DATE_ORDINAL32

    # Settlement/payment date encoded as int32 ordinal (days since DATE_EPOCH).
    # Used exclusively for discounting. Default sentinel indicates no discounting.
    settlement_date_ordinal32: np.int32 = NO_SETTLEMENT_DATE_ORDINAL32

    # ========== FX LINKAGES ==========

    # FX risk factor ID mapping underlying_ccy -> settlement_ccy.
    # Sentinel NO_FX_RF_REQUIRED (-1) indicates no FX conversion needed.
    fx_under_to_settle_rf_id: np.int32 = NO_FX_RF_REQUIRED

    # FX risk factor ID mapping strike_ccy -> settlement_ccy.
    # Sentinel NO_FX_RF_REQUIRED (-1) indicates no FX conversion needed.
    fx_strike_to_settle_rf_id: np.int32 = NO_FX_RF_REQUIRED

    def __str__(self) -> str:
        """Return a compact, single-line string representation for logging and debugging."""
        return (
            f"ForwardPayload("
            f"quantity={self.quantity}, "
            f"strike={self.strike}, "
            f"delivery_date_ordinal32={self.delivery_date_ordinal32}, "
            f"settlement_date_ordinal32={self.settlement_date_ordinal32}, "
            f"price_rf_id={self.price_rf_id}, "
            f"price_source_type={self.price_source_type}, "
            f"fx_under_to_settle_rf_id={self.fx_under_to_settle_rf_id}, "
            f"fx_strike_to_settle_rf_id={self.fx_strike_to_settle_rf_id}"
            f")"
        )


if __name__ == "__main__":
    # Example: Create and print a ForwardPayload representing a 1000-unit
    # position on a forward contract with strike 1.2345, using a curve-based
    # price source, settling on ordinal day 18993 with no FX conversions needed.
    example = ForwardPayload(
        price_rf_id=np.int32(101),
        price_source_type=PRICE_SOURCE_CURVE,
        quantity=np.float64(1_000.0),
        strike=np.float64(1.2345),
        settlement_date_ordinal32=np.int32(18993),
        fx_under_to_settle_rf_id=NO_FX_RF_REQUIRED,
        fx_strike_to_settle_rf_id=NO_FX_RF_REQUIRED,
    )
    print(example)


# ===== FILE: portfolio\payloads\spot.py =====

"""
Payload for spot positions.

This lightweight, immutable container holds the minimal numeric
information required to price and risk a spot position. It stores only
primitive numeric types (NumPy dtypes) so it can be stored in compact
arrays and passed efficiently to vectorized valuation kernels.

Fields
------
quantity
    Signed position size (float64). Positive values represent long
    exposure; negative values represent short exposure.
price_rf_id
    Integer identifier (np.int32) referring to the price/forward risk
    factor that provides market data for this spot position.
fx_rf_id
    Integer identifier (np.int32) for FX conversion underlying_ccy ->
    settlement_ccy. Uses the sentinel `NO_FX_RF_REQUIRED` when no FX
    conversion is needed.
"""

from dataclasses import dataclass

import numpy as np

from riskquant.portfolio.constants import NO_FX_RF_REQUIRED
from riskquant.portfolio.payloads.base import Payload


@dataclass(frozen=True, slots=True)
class SpotPayload(Payload):
    """Immutable payload for a single spot trade.

    The class is frozen to guarantee immutability and uses slots to
    reduce per-instance memory overhead. All fields are NumPy dtypes
    to remain compatible with array-based processing.
    """

    # ========== ECONOMIC TERMS ==========

    # Signed quantity / notional for the spot position. dtype: np.float64
    quantity: np.float64

    # ========== UNDERLYING LINKAGES ==========

    # Price risk factor id used to look up market data for this position. dtype: np.int32
    price_rf_id: np.int32

    # ========== FX LINKAGES ==========

    # FX risk factor id for underlying->settlement conversion, or sentinel when not required.
    fx_rf_id: np.int32 = NO_FX_RF_REQUIRED

    def __str__(self) -> str:
        # Single-line, human-readable representation for logging and debugging
        return (
            f"SpotPayload("
            f"quantity={self.quantity}, "
            f"price_rf_id={self.price_rf_id}, "
            f"fx_rf_id={self.fx_rf_id}"
            f")"
        )


if __name__ == "__main__":
    # Small usage example showing construction and printing of a SpotPayload.
    # Example values chosen for illustration only.
    example = SpotPayload(
        price_rf_id=np.int32(101),
        quantity=np.float64(1_000.0),
        fx_rf_id=NO_FX_RF_REQUIRED,
    )
    print(example)


# ===== FILE: portfolio\payloads\vanilla_european_option.py =====

"""
Payload definition for vanilla European option trades.

This module defines a compact, immutable container that encodes the
minimal economic and market-linkage information required by option pricers
and valuation engines. The payload deliberately carries identifiers
for risk factors (price, vol, FX) and a date encoded as an ordinal32
(days since DATE_EPOCH), rather than storing strings or full market
objects. This keeps memory footprint small and makes the payload
well-suited for large portfolios and high-throughput simulation.

Notes on expiry encoding:
  - The payload uses `expiry_date_ordinal32` (int32 days-since-epoch)
    to represent the option expiry date, aligning with how settlement
    dates are encoded in ForwardPayload (`settlement_date_ordinal32`).
  - The sentinel value `NO_EXPIRY_DATE` indicates a missing expiry.
"""

from dataclasses import dataclass

import numpy as np

from riskquant.portfolio.constants import (
    NO_FX_RF_REQUIRED,
    NO_EXPIRY_DATE_ORDINAL32,
    NO_SETTLEMENT_DATE_ORDINAL32,
    PRICE_SOURCE_NODE,
    PRICE_SOURCE_CURVE,
)
from riskquant.portfolio.payloads.base import Payload


@dataclass(frozen=True, slots=True)
class VanillaEuropeanOptionPayload(Payload):
    """
    Immutable payload for a single vanilla European option position.

    Lightweight container that exposes only numeric identifiers and
    primitive fields. External pricers resolve the IDs to time series
    or market objects and perform pricing, discounting and FX logic.
    """

    # ID referencing the underlying price/forward risk factor used by the pricer
    underlying_price_rf_id: np.int32

    # Signed position size (contracts / notional)
    quantity: np.float64

    # Strike price K expressed in strike currency units
    strike: np.float64

    # 1 => call, 0 => put; stored as int8 for compactness
    is_call: np.int8

    # --- VOL LINKAGE ---
    # Implied volatility risk factor id (volatility series used by pricer)
    vol_rf_id: np.int32

    # Indicates how to interpret `underlying_price_rf_id`: 0=NODE, 1=CURVE. dtype: np.int8
    # 0=NODE, 1=CURVE
    underlying_price_source_type: np.int8 = PRICE_SOURCE_NODE

    # --- EXPIRY DATE ---
    # Expiry encoded as int32 ordinal (days since DATE_EPOCH). Matches ForwardPayload.
    expiry_date_ordinal32: np.int32 = NO_EXPIRY_DATE_ORDINAL32

    # --- SETTLEMENT DATE ---
    # Optional settlement date (days-since-epoch). Used for discounting only. dtype: np.int32
    payoff_settlement_date_ordinal32: np.int32 = NO_SETTLEMENT_DATE_ORDINAL32

    # --- FX LINKAGE ---
    # FX id for underlying_ccy -> settlement_ccy, or FX_NONE if no conversion
    fx_under_to_settle_rf_id: np.int32 = NO_FX_RF_REQUIRED

    # FX id for strike_ccy -> settlement_ccy, or FX_NONE if no conversion
    fx_strike_to_settle_rf_id: np.int32 = NO_FX_RF_REQUIRED

    def __str__(self) -> str:
        """Return a compact single-line representation for logging/debugging."""
        return (
            f"VanillaEuropeanOptionPayload("
            f"underlying_price_rf_id={self.underlying_price_rf_id}, "
            f"underlying_price_source_type={self.underlying_price_source_type}, "
            f"quantity={self.quantity}, "
            f"strike={self.strike}, "
            f"is_call={self.is_call}, "
            f"vol_rf_id={self.vol_rf_id}, "
            f"expiry_date_ordinal32={self.expiry_date_ordinal32}, "
            f"settlement_date_ordinal32={self.payoff_settlement_date_ordinal32}, "
            f"fx_under_to_settle_rf_id={self.fx_under_to_settle_rf_id}, "
            f"fx_strike_to_settle_rf_id={self.fx_strike_to_settle_rf_id}"
            f")"
        )


if __name__ == "__main__":
    # Small usage example showing construction and printing of a VanillaEuropeanOptionPayload.
    # Example values chosen for illustration only.
    example = VanillaEuropeanOptionPayload(
        underlying_price_rf_id=np.int32(101),
        underlying_price_source_type=PRICE_SOURCE_CURVE,
        quantity=np.float64(1_000.0),
        strike=np.float64(1.2345),
        # 1 => call, 0 => put; using int8 for compact storage
        is_call=np.int8(1),
        vol_rf_id=np.int32(202),
        # Use a hypothetical expiry ordinal (days since 1970-01-01)
        expiry_date_ordinal32=np.int32(18993),
        # No settlement date specified in this example, using the sentinel
        payoff_settlement_date_ordinal32=NO_SETTLEMENT_DATE_ORDINAL32,
        # No FX conversion required in this example, using the sentinel
        fx_under_to_settle_rf_id=NO_FX_RF_REQUIRED,
        fx_strike_to_settle_rf_id=NO_FX_RF_REQUIRED,
    )
    print(example)
