## C. Additional observations (Observed, sandbox)

1. P5: the per-UID aggregation stage (`rows[row_order]` gather plus `np.add.reduceat`, api._value_positions) costs 20 to 31 ns per cell against 1.1 to 1.3 ns per cell for SPOT valuation (ratio 19 to 25). Consequence for the design: on a SPOT book the gather and the conversion temporaries dominate the per-window cost, not valuation; a fused convert-and-fold kernel and a gather-free aggregation for single-row UIDs are the first performance items.

### extract_tree.py

```python
# extract_tree.py: rebuild a source tree from context_riskquant_*.txt profiles
import glob, hashlib, os, re, sys
ROOT = sys.argv[1]
SOURCES = sys.argv[2:]
MARK = re.compile(r"^=== FILE: (.+?) ===\s*$", re.M)
FENCE_OPEN = re.compile(r"^```[A-Za-z0-9_+-]*[ \t]*\n")
written = {}
per_profile = {}
for src in SOURCES:
    for p in sorted(glob.glob(src)):
        text = open(p, encoding="utf-8", errors="replace").read().replace("\r\n", "\n")
        parts = MARK.split(text)
        n = 0
        for i in range(1, len(parts), 2):
            path, body = parts[i].strip(), parts[i + 1]
            body = body.lstrip("\n")
            m = FENCE_OPEN.match(body)
            if m:
                body = body[m.end():]
                end = body.rfind("\n```")
                if end >= 0:
                    body = body[:end + 1]
            if path in written and written[path] != body:
                # keep the longer representation; report the conflict
                print(f"CONFLICT {path}: {os.path.basename(p)} differs ({len(body)} vs {len(written[path])} bytes)")
                if len(body) <= len(written[path]):
                    continue
            written[path] = body
            n += 1
        per_profile[os.path.basename(p)] = n
for path, body in written.items():
    full = os.path.join(ROOT, path)
    os.makedirs(os.path.dirname(full), exist_ok=True)
    with open(full, "w", encoding="utf-8") as fh:
        fh.write(body)
for k, v in sorted(per_profile.items()):
    print(f"{k}: {v} file blocks")
print(f"TOTAL unique files written: {len(written)}")
```

def bitwise_equal(a: np.ndarray, b: np.ndarray) -> bool:
    a = np.asarray(a)
    b = np.asarray(b)
    if a.shape != b.shape or a.dtype != b.dtype:
        return False
    return bool(np.array_equal(a.view(np.uint8), b.view(np.uint8)))


def first_diff(a: np.ndarray, b: np.ndarray) -> str:
    a = np.asarray(a)
    b = np.asarray(b)
    if a.shape != b.shape:
        return f"shape {a.shape} vs {b.shape}"
    if a.dtype != b.dtype:
        return f"dtype {a.dtype} vs {b.dtype}"
    if a.dtype.kind == "f":
        neq = np.argwhere(a.view(np.uint64 if a.dtype == np.float64 else np.uint32) != b.view(
            np.uint64 if a.dtype == np.float64 else np.uint32
        ))
    else:
        neq = np.argwhere(a != b)
    if neq.size == 0:
        return "no difference"
    idx = tuple(int(i) for i in neq[0])
    va, vb = a[idx], b[idx]
    ha = float(va).hex() if a.dtype.kind == "f" else repr(va)
    hb = float(vb).hex() if b.dtype.kind == "f" else repr(vb)
    return f"first diff at {idx}: {ha} vs {hb} ({neq.shape[0]} cells differ)"


def verdict(label: str, ok: bool, detail: str = "") -> bool:
    print(f"[{'EQUAL' if ok else 'DIFF '}] {label}" + (f": {detail}" if detail else ""))
    return ok


def hexs(x: float) -> str:

# Shocked levels: the api.py stage (proxy compile -> lower -> bind -> stream)

def shocked_levels_by_window(
    *,
    static_config,
    as_of_ord32: int,
    axis_factor_ids: np.ndarray,
    axis_shift_type_ids: np.ndarray,
    base_levels: np.ndarray,
    proxy_terms,
    scenario_config,
    data,
    batch_size: int,
):
    """Yield (offset, (K, w) float64 C) windows, as run_scenario_var_es builds them."""
    from riskquant.config.proxy_model.compile import compile_proxy_model
    from riskquant.simlab.shocks.axis_operator import lower_axis_operator
    from riskquant.simlab.shocks.linear_term_apply import apply_linear_terms_to_base_levels
    from riskquant.solutions.scenario_models.api import build_scenario_generator
    from riskquant.solutions.scenario_models.world_columns import bind_operator_to_world
    from riskquant.utils.business_days import ord_to_date

    model = compile_proxy_model(
        authored_terms=proxy_terms,
        static_config=static_config,
        as_of_date=ord_to_date(int(as_of_ord32)),
        run_axis_rf_ids=np.ascontiguousarray(axis_factor_ids, dtype=np.int32),
    )
    operator = lower_axis_operator(
        model=model,
        axis_rf_ids=np.ascontiguousarray(axis_factor_ids, dtype=np.int32),
        axis_shift_type_ids=np.ascontiguousarray(axis_shift_type_ids, dtype=np.int8),
    )
    binding = bind_operator_to_world(operator, scenario_config, data)
    generator = build_scenario_generator(scenario_config, data=data)
    k = int(base_levels.shape[0])
    for batch in generator.iter_batches(
        driver_rf_ids=binding.request_driver_rf_ids,
        chart_ids=binding.request_chart_ids,
        batch_size=batch_size,
    ):
        out = np.empty((k, int(batch.shocks.shape[0])), dtype=np.float64, order="C")
        apply_linear_terms_to_base_levels(
            base_levels=np.ascontiguousarray(base_levels, dtype=np.float64),
            row_term_offsets=operator.row_term_offsets,
            term_column_uix=binding.term_column_uix,
            term_loadings=operator.term_loadings,
            target_shift_types=operator.target_shift_type_ids,
            shocks=batch.shocks,
            out=out,
        )
        yield int(batch.scenario_offset), out


def shocked_levels_full(**kw) -> np.ndarray:
    return np.ascontiguousarray(np.hstack([w for _, w in shocked_levels_by_window(**kw)]))

    env = dict(os.environ)
    env["NUMBA_NUM_THREADS"] = str(threads)
    env["PROBE_CHILD"] = "1"
    env.update(extra_env or {})
    cmd = [sys.executable, script, *(args or [])]
    t0 = time.perf_counter()
    res = subprocess.run(cmd, env=env, capture_output=True, text=True)
    dt = time.perf_counter() - t0
    print(f"--- child NUMBA_NUM_THREADS={threads} rc={res.returncode} wall={dt:.1f}s ---")
    print(res.stdout.rstrip())
    if res.returncode != 0:
        print(res.stderr[-4000:])
        raise SystemExit(f"child failed: {' '.join(cmd)}")
    return res.stdout

"""

Books: SPOT (_certfix, M positions) and VANILLA_FORWARD_NODE with re-tagged

    from riskquant.portfolio.engine import build_uix_mapper, compile_portfolio_from_ingest_batch
    from riskquant.solutions.scenario_models.api import parse_scenario_model_config

    world = fw.build_world()
    batch = world.context.ingest_batch
    uids = np.asarray(batch.uids, dtype=np.int64).copy()
    # Re-tag N_PAIRS pairs of rows with the same settlement currency to one UID
    # (D055 multi-row trade identity; same technique as
    # tests/valuation/oracle/test_multi_row_trade_identity.py).
    settle = np.asarray(batch.fx_u2s_ids)
    pairs = []
    used = set()
    order = np.arange(batch.size)
    for i in order:
        if len(pairs) == N_PAIRS:
            break
        if i in used:
            continue
        for j in order[i + 1 :]:
            if j in used or settle[j] != settle[i]:
                continue
            pairs.append((int(i), int(j)))
            used.update((int(i), int(j)))
            break

        uids[j] = uids[i]
    retagged = dataclasses.replace(batch, uids=np.ascontiguousarray(uids))
    mapper = build_uix_mapper([retagged], as_of_ord32=fw.AS_OF_ORD32)
    portfolio = compile_portfolio_from_ingest_batch(ingest_batch=retagged, uix_mapper=mapper)
    umap = build_universe_mapping(
        static_config=world.rsc, uix_mapper=mapper, discounting_mode=fw.DISCOUNTING
    )
    axis = world.axis
    print(
        f"forward book: rows={retagged.size} unique UIDs={np.unique(uids).shape[0]} "
        f"multi-row UIDs={len(pairs)} mapper fingerprint equal to world mapper: "
        f"{mapper.fingerprint == world.context.uix_mapper.fingerprint}"

def compile_tiles(book, tile_size):
    from riskquant.portfolio.engine import compile_portfolio_from_ingest_batch

    batch = book["batch"]
    row_uids = np.asarray(batch.uids, dtype=np.int64)
    sorted_uids = np.unique(row_uids)
    tiles = []
    for tile_uids in pc.uid_tiles(sorted_uids, tile_size):
        mask = np.isin(row_uids, tile_uids)
        sub = pc.sub_batch(batch, mask)
        tiles.append((tile_uids, mask, compile_portfolio_from_ingest_batch(ingest_batch=sub, uix_mapper=book["mapper"])))
    return tiles


def run_book(book, results: dict) -> bool:
    name = book["name"]
    ok_all = True
    n_cmp = {"V1": 0, "V3": 0}
    m = int(np.unique(np.asarray(book["batch"].uids)).shape[0])
    full = book["portfolio"]
    full_uids = np.asarray(full.trade_uids, dtype=np.int64)
    tile_cache = {}
    for ts in TILE_SIZES:
        size = m - 1 if ts == "M-1" else ts
        tile_cache[ts] = compile_tiles(book, size)
        # V2: within-UID row order equals the full compiled order (per tile)
        for tile_uids, mask, tp in tile_cache[ts]:
            tp_uids = np.asarray(tp.trade_uids, dtype=np.int64)
            if not np.array_equal(full_uids[np.isin(full_uids, tile_uids)], tp_uids):
                ok_all = pc.verdict(f"{name} V2 row order tile={ts}", False, f"{full_uids[np.isin(full_uids, tile_uids)][:8]} vs {tp_uids[:8]}") and ok_all
    pc.verdict(f"{name} V2 compiled row order equal in every tile of every tile size", ok_all)
    windows = list(
        pc.shocked_levels_by_window(
            static_config=book["static_config"],
            as_of_ord32=book["as_of"],
            axis_factor_ids=book["axis_ids"],
            axis_shift_type_ids=book["axis_shift"],
            base_levels=book["base"],
            proxy_terms=(),
            scenario_config=book["scenario_config"],
            data=book["data"],
            batch_size=4096,
        )
    )
    levels = np.ascontiguousarray(np.hstack([w for _, w in windows]))[: book["k_val"]]
    s_total = levels.shape[1]
    for width in WIDTHS:
        stop = s_total if width >= s_total else SMALL_SCENARIOS
        rows_full = np.empty((full.total_trades, stop), dtype=np.float64)
        for s0 in range(0, stop, width):
            s1 = min(s0 + width, stop)
            state = pc.build_state(
                mapper=book["mapper"], universe_map=book["universe_map"],
                flat_levels=np.ascontiguousarray(levels[:, s0:s1]),
                as_of_ord32=book["as_of"], discounting_mode=book["discounting"],
            )
            rows_full[:, s0:s1] = pc.value_rows(full, state)
            for ts in TILE_SIZES:
                for tile_uids, mask, tp in tile_cache[ts]:
                    rows_tile = pc.value_rows(tp, state)
                    n_cmp["V1"] += 1
                    sel = np.isin(full_uids, tile_uids)
                    if not pc.bitwise_equal(rows_full[sel, s0:s1], rows_tile):
                        ok_all = pc.verdict(f"{name} V1 rows width={width} tile={ts} window=[{s0},{s1})", False, pc.first_diff(rows_full[sel, s0:s1], rows_tile)) and ok_all
        pc.verdict(f"{name} V1 valuation rows bitwise equal, full vs every tile, width={width}, scenarios={stop}, tile comparisons so far={n_cmp['V1']}", ok_all)
        # V3: per-UID reduceat sums
        uids_full, agg_full = pc.aggregate_by_uid(full, rows_full)
        for ts in TILE_SIZES:
            for tile_uids, mask, tp in tile_cache[ts]:
                sel_rows = np.isin(full_uids, tile_uids)
                uids_t, agg_t = pc.aggregate_by_uid(tp, rows_full[sel_rows])
                n_cmp["V3"] += 1
                ref = agg_full[np.isin(uids_full, tile_uids)]
                if not pc.bitwise_equal(ref, agg_t):
                    ok_all = pc.verdict(f"{name} V3 per-UID sums width={width} tile={ts}", False, pc.first_diff(ref, agg_t)) and ok_all
        pc.verdict(f"{name} V3 per-UID reduceat sums bitwise equal, full vs every tile, width={width}, tile comparisons so far={n_cmp['V3']}", ok_all)
        results[f"{name}_rows_w{width}"] = rows_full
        results[f"{name}_agg_w{width}"] = agg_full
    return ok_all

    print(f"P1 verdict: {'ALL EQUAL' if ok else 'DIFFERENCES FOUND'}")


if __name__ == "__main__":

def row_slices(model):
    """{target rf_id: (drivers, charts, loadings)} in canonical term order."""
    off = np.asarray(model.row_term_offsets)
    out = {}
    for r, rf in enumerate(np.asarray(model.row_target_rf_ids).tolist()):
        a, b = int(off[r]), int(off[r + 1])
        out[int(rf)] = (
            np.asarray(model.term_driver_rf_ids[a:b]),
            np.asarray(model.term_chart_ids[a:b]),
            np.asarray(model.term_loadings[a:b]),
        )
    return out


def compare(label, full_model, sub_model) -> bool:
    f, s = row_slices(full_model), row_slices(sub_model)
    common = sorted(set(f) & set(s))
    ok = True
    n_terms = 0
    for rf in common:
        for i, (x, y) in enumerate(zip(f[rf], s[rf], strict=True)):
            n_terms += x.shape[0] if i == 0 else 0
            if not pc.bitwise_equal(x, y):
                ok = pc.verdict(f"{label} target rf_id={rf} array {i}", False, pc.first_diff(x, y)) and ok
    only_sub = sorted(set(s) - set(f))
    return pc.verdict(
        f"{label}: {len(common)} common targets, {n_terms} terms compared bitwise; "
        f"targets only on sub-axis={len(only_sub)}; rows(full)={len(f)} rows(sub)={len(s)}",
        ok and not only_sub,
    )

# probes/p4_fold_metrics.py
"""
P4: fold determinism and per-row VaR/ES.

Converted reporting-currency rows of the SPOT book (production D081 conversion
through api._reporting_route / api._conversion_plan /
convert_scenario_pnl_to_reporting_currency) are folded into node vectors in
ascending UID order:
  (a) a NumPy sequential fold through tiles of 1, 7, 64 and M UIDs;
  (b) a probe-local Numba fold with prange over scenario-column strips and
      node-major CSR membership (the DP-12 layout), at 1 and 4 threads.
compute_var_es is called once on (N, S) and once per row; results are compared
bitwise, at 1 and 4 threads. NumPy's converted.sum(axis=0) is compared with the
sequential fold for information only (D041 F-17 context).
"""

from __future__ import annotations

import os
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parent))
import probe_common as pc  # noqa: E402

N_NODES = 50
STRIP = 64


def converted_rows():
    """(M, S) converted rows of the SPOT book, ascending UID, via production functions."""
    import _certfix as cf
    from riskquant.reporting.fx_conversion import convert_scenario_pnl_to_reporting_currency
    from riskquant.solutions.risk_run import api
    from riskquant.solutions.risk_run.schemas import ValuationContext
    from riskquant.solutions.scenario_models.api import parse_scenario_model_config

    context = ValuationContext(static_config=cf.RSC, ingest_batch=cf._BATCH, uix_mapper=cf.MAPPER, discounting_mode=cf.NONE)
    universe = cf.UIDS
    agg = api._row_aggregation(portfolio=cf.PORTFOLIO, universe=universe)
    positions = api._positions(agg.economic_uids, universe)
    route = api._reporting_route(context=context, universe_map=cf.UNIVERSE_MAP, economic_uids=agg.economic_uids, reporting_ccy="USD")
    plan, fx_rows, _ = api._conversion_plan(route=route, universe_positions=positions, static_config=cf.RSC, reporting_ccy="USD")
    cfg = parse_scenario_model_config(dict(cf.CFGS["sbb"]))
    levels = pc.shocked_levels_full(
        static_config=cf.RSC, as_of_ord32=cf.AS_OF, axis_factor_ids=route.axis.factor_ids,
        axis_shift_type_ids=route.axis.shift_type_ids, base_levels=np.ascontiguousarray(
            np.concatenate([cf.BASE, cf.VALS[-1][route.axis.factor_ids[cf.K:] - 1]])), proxy_terms=(),
        scenario_config=cfg, data=cf.matrix(), batch_size=512,
    )
    k_val = route.axis.num_valuation_factors
    base_full = np.ascontiguousarray(np.concatenate([cf.BASE, cf.VALS[-1][route.axis.factor_ids[cf.K:] - 1]]))
    st0 = pc.build_state(mapper=cf.MAPPER, universe_map=cf.UNIVERSE_MAP, flat_levels=base_full[:k_val, None], as_of_ord32=cf.AS_OF, discounting_mode=cf.NONE)
    _, v0 = pc.aggregate_by_uid(cf.PORTFOLIO, pc.value_rows(cf.PORTFOLIO, st0))
    st = pc.build_state(mapper=cf.MAPPER, universe_map=cf.UNIVERSE_MAP, flat_levels=levels[:k_val], as_of_ord32=cf.AS_OF, discounting_mode=cf.NONE)
    _, vs = pc.aggregate_by_uid(cf.PORTFOLIO, pc.value_rows(cf.PORTFOLIO, st))
    dv32 = (vs - v0[:, [0]]).astype(np.float32)  # PnlGrid f32 quantisation, as _write_full_reval_block
    settlement = np.ascontiguousarray(dv32.astype(np.float64))
    conv = convert_scenario_pnl_to_reporting_currency(
        settlement_pnl=settlement[positions], fx_levels=np.ascontiguousarray(levels[fx_rows]), plan=plan,
        base_settlement_values=np.ascontiguousarray(v0[positions, 0]), base_fx_levels=np.ascontiguousarray(base_full[fx_rows]),
    )
    return np.ascontiguousarray(conv.converted_pnl), np.ascontiguousarray(conv.portfolio_pnl)


def numpy_fold(rows: np.ndarray, tile: int) -> np.ndarray:
    acc = np.zeros(rows.shape[1], dtype=np.float64)
    for t0 in range(0, rows.shape[0], tile):
        for r in range(t0, min(t0 + tile, rows.shape[0])):
            acc += rows[r]
    return acc


def strip_fold_kernel():
    import numba as nb

    @nb.njit(parallel=True, cache=False, fastmath=False)
    def fold(rows, node_offsets, node_rows, node_pnl, s0):
        n_nodes = node_offsets.shape[0] - 1
        w = rows.shape[1]
        n_strips = (w + STRIP - 1) // STRIP
        for strip in nb.prange(n_strips):
            a = strip * STRIP
            b = min(a + STRIP, w)
            for n in range(n_nodes):
                for k in range(node_offsets[n], node_offsets[n + 1]):
                    r = node_rows[k]
                    for s in range(a, b):
                        node_pnl[n, s0 + s] += rows[r, s]

    return fold


def membership(m: int, rng: np.random.Generator):
    """node -> ascending member row positions (CSR), N_NODES disjoint leaves + one root."""
    leaf = rng.integers(0, N_NODES - 1, size=m)
    offsets = [0]
    members = []
    for n in range(N_NODES - 1):
        rows_n = np.flatnonzero(leaf == n)
        members.append(rows_n)
        offsets.append(offsets[-1] + rows_n.shape[0])
    members.append(np.arange(m))  # root
    offsets.append(offsets[-1] + m)
    return np.asarray(offsets, dtype=np.int32), np.ascontiguousarray(np.concatenate(members).astype(np.int32)), leaf


def child(tag: str) -> None:
    pc.banner(f"P4 child {tag}")
    from riskquant.scenario_metrics.api.var_es import compute_var_es

    rows, portfolio_np = converted_rows()
    m, s = rows.shape
    print(f"converted rows: {rows.shape} float64 C={rows.flags['C_CONTIGUOUS']}")
    ok = True
    # (a) sequential fold through tiles
    ref = numpy_fold(rows, m)
    for tile in (1, 7, 64, m):
        f = numpy_fold(rows, tile)
        ok = pc.verdict(f"NumPy sequential fold tile={tile} vs tile=M", pc.bitwise_equal(f, ref), "" if pc.bitwise_equal(f, ref) else pc.first_diff(f, ref)) and ok
    eq_np = pc.bitwise_equal(portfolio_np, ref)
    print(f"[INFO ] production converted.sum(axis=0) vs sequential fold: {'EQUAL' if eq_np else 'DIFFERENT'}; {pc.first_diff(portfolio_np, ref)}")
    # (b) strip-parallel CSR fold, windows of 512 columns, tiles of UIDs
    rng = np.random.default_rng(11)
    offsets, members, leaf = membership(m, rng)
    fold = strip_fold_kernel()
    results = {}
    for tile in (1, 7, 64, m):
        node_pnl = np.zeros((N_NODES, s), dtype=np.float64)
        for s0 in range(0, s, 512):
            s1 = min(s0 + 512, s)
            for t0 in range(0, m, tile):
                t1 = min(t0 + tile, m)
                # tile-local CSR: members of each node within [t0, t1), rebased to tile rows
                t_off = [0]
                t_rows = []
                for n in range(N_NODES):
                    mem = members[offsets[n] : offsets[n + 1]]
                    sel = mem[(mem >= t0) & (mem < t1)] - t0
                    t_rows.append(sel)
                    t_off.append(t_off[-1] + sel.shape[0])
                fold(np.ascontiguousarray(rows[t0:t1, s0:s1]), np.asarray(t_off, dtype=np.int32), np.ascontiguousarray(np.concatenate(t_rows).astype(np.int32)), node_pnl, s0)
        results[f"strip_fold_tile{tile}"] = node_pnl
    ref_nodes = results[f"strip_fold_tile{m}"]
    for tile in (1, 7, 64):
        eq = pc.bitwise_equal(results[f"strip_fold_tile{tile}"], ref_nodes)
        ok = pc.verdict(f"strip-parallel CSR fold tile={tile} vs tile=M (window 512)", eq, "" if eq else pc.first_diff(results[f"strip_fold_tile{tile}"], ref_nodes)) and ok
    # root row equals the sequential fold; leaf rows equal their own sequential folds
    ok = pc.verdict("strip fold root row == NumPy sequential fold of all rows", pc.bitwise_equal(ref_nodes[-1], ref), pc.first_diff(ref_nodes[-1], ref)) and ok
    leaf_ok = True
    for n in range(N_NODES - 1):
        sel = np.flatnonzero(leaf == n)
        if sel.size and not pc.bitwise_equal(ref_nodes[n], numpy_fold(rows[sel], sel.size)):
            leaf_ok = False
    ok = pc.verdict("strip fold leaf rows == sequential folds of their members", leaf_ok) and ok
    # metrics: once on (N, S) vs per row
    once = compute_var_es(np.ascontiguousarray(ref_nodes), 0.99)
    per_var = np.empty(N_NODES); per_es = np.empty(N_NODES)
    for n in range(N_NODES):
        r = compute_var_es(np.ascontiguousarray(ref_nodes[n : n + 1]), 0.99)
        per_var[n], per_es[n] = r.var[0], r.es[0]
    ok = pc.verdict("compute_var_es once (N,S) vs per row: VaR", pc.bitwise_equal(once.var, per_var), pc.first_diff(once.var, per_var)) and ok
    ok = pc.verdict("compute_var_es once (N,S) vs per row: ES", pc.bitwise_equal(once.es, per_es), pc.first_diff(once.es, per_es)) and ok
    results["var"] = np.asarray(once.var); results["es"] = np.asarray(once.es); results["seq_fold"] = ref

### p5_cost_per_cell.py

```python
# probes/p5_cost_per_cell.py
"""
P5: per-stage cost per (position, scenario) cell. Indicative only.

SPOT book of M positions (RQ_CERT_POSITIONS) on a window of w scenarios
(RQ_CERT_SCENARIOS = w, SBB). Stages timed with time.perf_counter, best of 3
after one warm-up (Numba JIT excluded):
  state   build_market_data_matrix_from_flat_levels + assemble_valuation_state
  value   run_valuation into (R, w)
  agg     rows[row_order] + np.add.reduceat (api._value_positions arithmetic)
  vp      api._value_positions as one call (state + value + agg)
  conv    convert_scenario_pnl_to_reporting_currency (D081, NumPy temporaries)
  fold    NumPy sequential fold of converted rows into one (w,) vector
  strip   probe-local Numba strip fold into N_NODES node rows (DP-12 layout)
Cells per M x w: (M, w) = 2000 x 1024, 2000 x 4096, 18000 x 1024. The sandbox
has 3.9 GiB RAM, so 20000 x 4096 (five (M, w) float64 temporaries, about
3.3 GiB) is not run; the large cell is 18000 x 1024. M = 20000 is not
reachable with _certfix: its fixed-seed quantity draw rounds one row to 0.00
and ingest refuses the dust quantity (also at M = 19000).
"""

from __future__ import annotations

import gc
import os
import sys
import time
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parent))
import probe_common as pc  # noqa: E402

CELLS = ((2000, 1024), (2000, 4096), (18000, 1024))
N_NODES = 50


def best_of(fn, n=3):
    fn()  # warm-up (JIT, page faults)
    ts = []
    for _ in range(n):
        gc.collect()
        t0 = time.perf_counter()
        fn()
        ts.append(time.perf_counter() - t0)
    return min(ts)


def child() -> None:
    import _certfix as cf
    from riskquant.reporting.fx_conversion import convert_scenario_pnl_to_reporting_currency
    from riskquant.solutions.risk_run import api
    from riskquant.solutions.risk_run.schemas import ValuationContext
    from riskquant.solutions.scenario_models.api import parse_scenario_model_config
    from riskquant.valuation.engine import run_valuation

    sys.path.insert(0, str(Path(__file__).resolve().parent))
    from p4_fold_metrics import membership, numpy_fold, strip_fold_kernel

    m_req, w = int(os.environ["RQ_CERT_POSITIONS"]), int(os.environ["RQ_CERT_SCENARIOS"])
    pc.banner(f"P5 child M={m_req} w={w}")
    context = ValuationContext(static_config=cf.RSC, ingest_batch=cf._BATCH, uix_mapper=cf.MAPPER, discounting_mode=cf.NONE)
    agg = api._row_aggregation(portfolio=cf.PORTFOLIO, universe=cf.UIDS)
    positions = api._positions(agg.economic_uids, cf.UIDS)
    route = api._reporting_route(context=context, universe_map=cf.UNIVERSE_MAP, economic_uids=agg.economic_uids, reporting_ccy="USD")
    plan, fx_rows, _ = api._conversion_plan(route=route, universe_positions=positions, static_config=cf.RSC, reporting_ccy="USD")
    base_full = np.ascontiguousarray(np.concatenate([cf.BASE, cf.VALS[-1][route.axis.factor_ids[cf.K:] - 1]]))
    k_val = route.axis.num_valuation_factors
    cfg = parse_scenario_model_config(dict(cf.CFGS["sbb"]))
    levels = pc.shocked_levels_full(
        static_config=cf.RSC, as_of_ord32=cf.AS_OF, axis_factor_ids=route.axis.factor_ids,
        axis_shift_type_ids=route.axis.shift_type_ids, base_levels=base_full, proxy_terms=(),
        scenario_config=cfg, data=cf.matrix(), batch_size=w,
    )
    assert levels.shape[1] == w
    m = int(cf.UIDS.shape[0]); r_rows = int(cf.PORTFOLIO.total_trades)
    e = cf.MAPPER
    e_eff = dict(spot=int(e.spot_effective_pairs.shape[0]), edf=int(e.edf_effective_triples.shape[0]), efp=int(e.efp_effective_quads.shape[0]), fx=int(e.fx_effective_pairs.shape[0]))
    print(f"M={m} R={r_rows} K={route.axis.factor_ids.shape[0]} K_val={k_val} E_eff={e_eff} w={w} cells={m * w}")
    vroute = api._ValuationRoute(portfolio=cf.PORTFOLIO, uix_mapper=cf.MAPPER, universe_map=cf.UNIVERSE_MAP)
    lv = np.ascontiguousarray(levels[:k_val])
    st0 = pc.build_state(mapper=cf.MAPPER, universe_map=cf.UNIVERSE_MAP, flat_levels=base_full[:k_val, None], as_of_ord32=cf.AS_OF, discounting_mode=cf.NONE)
    v0 = pc.aggregate_by_uid(cf.PORTFOLIO, pc.value_rows(cf.PORTFOLIO, st0))[1][:, 0]

    times = {}
    state_holder = {}
    def f_state():
        state_holder["s"] = pc.build_state(mapper=cf.MAPPER, universe_map=cf.UNIVERSE_MAP, flat_levels=lv, as_of_ord32=cf.AS_OF, discounting_mode=cf.NONE)
    times["state"] = best_of(f_state)
    rows = np.empty((r_rows, w), dtype=np.float64)
    def f_value():
        run_valuation(portfolio=cf.PORTFOLIO, valuation_state=state_holder["s"], out=rows)
    times["value"] = best_of(f_value)
    agg_holder = {}
    def f_agg():
        agg_holder["a"] = np.add.reduceat(rows[agg.row_order], agg.segment_starts, axis=0)
    times["agg"] = best_of(f_agg)
    def f_vp():
        api._value_positions(route=vroute, aggregation=agg, flat_levels=lv)
    times["vp"] = best_of(f_vp)
    settlement = np.ascontiguousarray((agg_holder["a"] - v0[:, None]).astype(np.float32).astype(np.float64))
    del rows; gc.collect()
    conv_holder = {}
    def f_conv():
        conv_holder["c"] = convert_scenario_pnl_to_reporting_currency(
            settlement_pnl=settlement, fx_levels=np.ascontiguousarray(levels[fx_rows]), plan=plan,
            base_settlement_values=np.ascontiguousarray(v0), base_fx_levels=np.ascontiguousarray(base_full[fx_rows]))
    times["conv"] = best_of(f_conv)
    conv = np.ascontiguousarray(conv_holder["c"].converted_pnl)
    del conv_holder, settlement, agg_holder, state_holder; gc.collect()
    times["fold"] = best_of(lambda: numpy_fold(conv, m))
    offsets, members, _ = membership(m, np.random.default_rng(5))
    fold = strip_fold_kernel()
    node_pnl = np.zeros((N_NODES, w), dtype=np.float64)
    def f_strip():
        node_pnl.fill(0.0)
        fold(conv, offsets, members, node_pnl, 0)
    times["strip"] = best_of(f_strip)
    cells = m * w
    print("stage      seconds    ns/cell   share_of(state+value+agg+conv+fold)")
    tot = times["state"] + times["value"] + times["agg"] + times["conv"] + times["fold"]
    for k in ("state", "value", "agg", "vp", "conv", "fold", "strip"):
        print(f"{k:8s} {times[k]:10.4f} {1e9 * times[k] / cells:9.1f}   {100 * times[k] / tot:6.1f}%" + ("  (vp = state+value+agg in one call; not in the total)" if k == "vp" else "") + ("  (root fold of depth-2 tree, 50 nodes; not in the total)" if k == "strip" else ""))
    print(f"ratios: state/value={times['state'] / times['value']:.2f}  conv/value={times['conv'] / times['value']:.2f}  fold/value={times['fold'] / times['value']:.2f}  strip/value={times['strip'] / times['value']:.2f}  agg/value={times['agg'] / times['value']:.2f}")


def parent() -> None:
    pc.banner("P5 parent")
    here = str(Path(__file__).resolve())
    for m, w in CELLS:
        pc.run_child(here, 1, extra_env={"RQ_CERT_POSITIONS": str(m), "RQ_CERT_SCENARIOS": str(w), "PROBE_TAG": f"M{m}_w{w}"})
    print("P5 done: timings are indicative (1 CPU sandbox); ratios between stages are the usable output.")


if __name__ == "__main__":
    if os.environ.get("PROBE_CHILD"):
        child()
    else:
        parent()

```python
# probes/p6_window_memory.py
"""
P6: memory of one scenario window's market-data build and state assembly.

tracemalloc peak over build_market_data_matrix_from_flat_levels +
assemble_valuation_state at widths 256, 1024 and 4096, on the SPOT book
(_certfix) and the forward book (_fwdcertfix). Reported per width: peak bytes,
bytes retained by the returned state, K, K_v, each effective-table row count,
and the fitted c_E in peak = 8 * w * (K + K_v + c_E * E_eff) with
E_eff = spot + edf + efp + fk rows. tracemalloc traces NumPy allocations
through PyDataMem hooks; it does not trace Numba-internal allocations.
"""

from __future__ import annotations

import gc
import sys
import tracemalloc
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parent))
import probe_common as pc  # noqa: E402

WIDTHS = (256, 1024, 4096)


def measure(book) -> None:
    from riskquant.market_data.assemble import build_market_data_matrix_from_flat_levels
    from riskquant.market_data.compile import assemble_valuation_state
    from riskquant.utils.business_days import ord_to_date

    mp = book["mapper"]
    eff = dict(
        spot=int(mp.spot_effective_pairs.shape[0]),
        edf=int(mp.edf_effective_triples.shape[0]),
        efp=int(mp.efp_effective_quads.shape[0]),
        fk=int(mp.fk_effective_pairs.shape[0]),
        vol=int(np.asarray(mp.vol_effective_uix).shape[0]),
        fx=int(mp.fx_effective_pairs.shape[0]),
    )
    e_eff = eff["spot"] + eff["edf"] + eff["efp"] + eff["fk"]
    k = int(book["axis_ids"].shape[0])
    k_v = int(book["k_val"])
    print(f"{book['name']}: M={book['m']} K={k} K_v={k_v} effective rows={eff} E_eff={e_eff}")
    print("width   peak_bytes  retained_bytes   peak/(8w)   c_E_fit   peak_per_scenario_bytes")
    rng = np.random.default_rng(3)
    for w in WIDTHS:
        lv = np.ascontiguousarray(book["base"][:k_v, None] * np.exp(0.01 * rng.standard_normal((k_v, w))))
        gc.collect()
        tracemalloc.start()
        tracemalloc.reset_peak()
        base0, _ = tracemalloc.get_traced_memory()
        matrix = build_market_data_matrix_from_flat_levels(
            as_of_date=ord_to_date(int(book["as_of"])), uix_mapper=mp, universe_map=book["universe_map"],
            flat_levels=lv, num_scenarios=w, discounting_mode=book["discounting"],
        )
        state = assemble_valuation_state(matrix=matrix, uix_mapper=mp)
        cur, peak = tracemalloc.get_traced_memory()
        tracemalloc.stop()
        peak -= base0
        cur -= base0
        c_e = ((peak / (8.0 * w)) - k - k_v) / e_eff if e_eff else float("nan")
        print(f"{w:5d} {peak:12d} {cur:15d} {peak / (8.0 * w):11.1f} {c_e:9.2f} {peak / w:12.0f}")
        del matrix, state
        gc.collect()


def main() -> None:
    pc.banner("P6")
    import _certfix as cf
    import _fwdcertfix as fw
    from riskquant.market_data.assemble import build_universe_mapping

    measure(dict(name="SPOT", m=int(cf.UIDS.shape[0]), mapper=cf.MAPPER, universe_map=cf.UNIVERSE_MAP, axis_ids=cf.FIDS, k_val=int(cf.FIDS.shape[0]), base=cf.BASE, as_of=cf.AS_OF, discounting=cf.NONE))
    world = fw.build_world()
    umap = build_universe_mapping(static_config=world.rsc, uix_mapper=world.context.uix_mapper, discounting_mode=fw.DISCOUNTING)
    measure(dict(name="FORWARD", m=int(world.uids.shape[0]), mapper=world.context.uix_mapper, universe_map=umap, axis_ids=world.axis.factor_ids, k_val=int(world.axis.num_valuation_factors), base=world.base, as_of=fw.AS_OF_ORD32, discounting=fw.DISCOUNTING))
    print("P6 done (measurement; no bitwise verdict).")


if __name__ == "__main__":
    main()

# probes/p7_taylor_subset.py
"""
P7: Taylor engine over a UID subset (V6).

The certified TAYLOR/DERIVED calculator of the SPOT book (_certfix.calculator
over every UID, SECOND_ORDER_FULL) prices the SBB world (K_v, S) once. Subsets
of 1, 7 and 64 positions (low end and high end of the UID axis) are priced by
  (a) a producer-recompiled calculator (_certfix.calculator(uids=subset));
  (b) a CSR row slice of the FULL compiled book (dataclasses.replace, offsets
      rebased, active-factor axis unchanged) bound to SensitivityPnlEngine
      through the production make_engine / stream_scenarios pipeline.
Per-position float32 rows are compared bitwise with the full-book rows, at
NUMBA_NUM_THREADS in {1, 4}.
"""

from __future__ import annotations

import dataclasses
import os
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parent))
import probe_common as pc  # noqa: E402

SIZES = (1, 7, 64)


def csr_row_slice(book, keep_rows: np.ndarray):
    """Row slice of a CompiledSensitivityBook (probe-local; no production code)."""
    keep_rows = np.asarray(keep_rows, dtype=np.intp)
    po, co = np.asarray(book.position_offsets), np.asarray(book.cross_offsets)
    d_idx = np.concatenate([np.arange(po[r], po[r + 1]) for r in keep_rows]) if keep_rows.size else np.empty(0, np.intp)
    x_idx = np.concatenate([np.arange(co[r], co[r + 1]) for r in keep_rows]) if keep_rows.size else np.empty(0, np.intp)
    d_counts = (po[keep_rows + 1] - po[keep_rows]).astype(np.int64)
    x_counts = (co[keep_rows + 1] - co[keep_rows]).astype(np.int64)
    new_po = np.ascontiguousarray(np.concatenate([[0], np.cumsum(d_counts)]).astype(np.int32))
    new_co = np.ascontiguousarray(np.concatenate([[0], np.cumsum(x_counts)]).astype(np.int32))
    return dataclasses.replace(
        book,
        position_uids=np.ascontiguousarray(book.position_uids[keep_rows]),
        position_offsets=new_po,
        factor_uix=np.ascontiguousarray(book.factor_uix[d_idx]),
        first_order=np.ascontiguousarray(book.first_order[d_idx]),
        second_order=np.ascontiguousarray(book.second_order[d_idx]),
        cross_offsets=new_co,
        cross_factor_uix_i=np.ascontiguousarray(book.cross_factor_uix_i[x_idx]),
        cross_factor_uix_j=np.ascontiguousarray(book.cross_factor_uix_j[x_idx]),
        cross_second_order=np.ascontiguousarray(book.cross_second_order[x_idx]),
    )


def child(tag: str) -> None:
    pc.banner(f"P7 child {tag}")
    import _certfix as cf
    from riskquant.solutions.sensitivity_pnl.pipeline import make_engine, resolve_batch_size, stream_scenarios
    from riskquant.solutions.scenario_models.api import parse_scenario_model_config

    full_calc = cf.calculator(uids=cf.UIDS)
    print(f"full book: positions={full_calc.num_positions} active factors={full_calc.num_active_factors} nnz={full_calc.num_first_order_entries} cross={full_calc.num_cross_entries} order={full_calc.taylor_declaration.order.name}")
    cfg = parse_scenario_model_config(dict(cf.CFGS["sbb"]))
    levels = pc.shocked_levels_full(
        static_config=cf.RSC, as_of_ord32=cf.AS_OF, axis_factor_ids=cf.FIDS,
        axis_shift_type_ids=np.ascontiguousarray(cf.UNIVERSE_MAP.required_shift_type_ids, np.int8),
        base_levels=cf.BASE, proxy_terms=(), scenario_config=cfg, data=cf.matrix(), batch_size=512,
    )
    market = full_calc.axis.levels(base_levels=cf.BASE, shocked_levels=levels)
    full = full_calc.run(market)
    full_rows = np.asarray(full.pnl)
    full_uids = np.asarray(full.position_uids)
    print(f"full rows: {full_rows.shape} {full_rows.dtype}; engine batch_size={full.provenance.batch_size}")
    ok = True
    results = {"full": full_rows}
    for end, pick in (("low", lambda n: cf.UIDS[:n]), ("high", lambda n: cf.UIDS[-n:])):
        for n in SIZES:
            subset = np.asarray(pick(n), dtype=np.int64)
            sel = np.isin(full_uids, subset)
            ref = full_rows[sel]
            # (a) producer-recompiled calculator over the subset
            sub_calc = cf.calculator(uids=subset)
            sub_market = sub_calc.axis.levels(base_levels=cf.BASE, shocked_levels=levels)
            a_rows = np.asarray(sub_calc.run(sub_market).pnl)
            eq = pc.bitwise_equal(ref, a_rows)
            ok = pc.verdict(f"(a) recompiled calculator {end} n={n}: rows {a_rows.shape} vs full", eq, "" if eq else pc.first_diff(ref, a_rows)) and ok
            # (b) CSR row slice of the full compiled book + production engine
            keep_rows = np.flatnonzero(sel)
            sliced = csr_row_slice(full_calc.book, keep_rows)
            out = np.empty((sliced.num_positions, levels.shape[1]), dtype=np.float32, order="C")
            bs = resolve_batch_size(num_active_factors=sliced.num_active_factors, num_scenarios=levels.shape[1])
            engine = make_engine(book=sliced, market=market, out=out, batch_size=bs)
            stream_scenarios(market=market, engine=engine, num_scenarios=levels.shape[1])
            eq = pc.bitwise_equal(ref, out)
            ok = pc.verdict(f"(b) CSR row slice + engine {end} n={n}: rows {out.shape} vs full (nnz={sliced.nnz}, cross={sliced.nnz_cross}, batch={bs})", eq, "" if eq else pc.first_diff(ref, out)) and ok
            results[f"a_{end}_{n}"] = a_rows
            results[f"b_{end}_{n}"] = out
    # (c) whole book through the sliced path with a non-trivial batch size: engine identity
    sliced_all = csr_row_slice(full_calc.book, np.arange(full_rows.shape[0]))
    out = np.empty(full_rows.shape, dtype=np.float32, order="C")
    engine = make_engine(book=sliced_all, market=market, out=out, batch_size=7)
    stream_scenarios(market=market, engine=engine, num_scenarios=levels.shape[1])
    eq = pc.bitwise_equal(full_rows, out)
    ok = pc.verdict("(c) full book re-sliced, engine batch_size=7 vs calculator run", eq, "" if eq else pc.first_diff(full_rows, out)) and ok
    np.savez(pc.OUT / f"p7_{tag}.npz", **results)
    print(f"P7 child verdict: {'ALL EQUAL' if ok else 'DIFFERENCES FOUND'}")
