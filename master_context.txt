# src/riskquant/simlab/apply/shock_kernel.py
from __future__ import annotations

import numpy as np
import numba as nb
from typing import Optional

from riskquant.simlab.scenarios.base import ScenarioSetBatch
from riskquant.market_data.risk_factors.shift_types import (
    SHIFT_DIFFERENCE,
    SHIFT_LOG_RETURN,
    SHIFT_SIMPLE_RETURN,
)

# ============================================================
# ZERO-ALLOCATION NUMBA APPLY KERNEL
# ============================================================


@nb.njit(parallel=True, fastmath=True, cache=True)
def _apply_shocks_numba_core(
    base_levels: np.ndarray,  # (N,) float64
    shocks: np.ndarray,  # (B, K) float64 (Fortran-contiguous)
    req_to_uix: np.ndarray,  # (N,) int32
    shock_scale: np.ndarray,  # (N,) float64
    shift_types: np.ndarray,  # (N,) int8
    out: np.ndarray,  # (N, B) float64 (C-contiguous)
):
    """
    The inner JIT-compiled loop.
    Parallelized across the N required exposures.
    """
    N = base_levels.shape[0]
    B = shocks.shape[0]

    # Parallelize over the portfolio exposures (N)
    for n in nb.prange(N):
        # 1. Fetch factor-specific metadata once per row
        base = base_levels[n]
        k = req_to_uix[n]
        scale = shock_scale[n]
        st = shift_types[n]

        # Pull the shock column once per row. This is the only way to avoid
        # redundant memory reads of the huge shocks array inside the B-loop.
        shock_col = shocks[:, k]

        # 2. Branching is hoisted OUTSIDE the B-loop.
        # The CPU executes the inner loop with zero branch prediction misses.
        if st == SHIFT_LOG_RETURN:
            for b in range(B):
                # out is (N, B) C-contiguous.
                # Looping over b writes sequentially into memory. Cache-perfect.
                out[n, b] = base * np.exp(shock_col[b] * scale)

        elif st == SHIFT_SIMPLE_RETURN:
            for b in range(B):
                out[n, b] = base * (1.0 + (shock_col[b] * scale))

        elif st == SHIFT_DIFFERENCE:
            for b in range(B):
                out[n, b] = base + (shock_col[b] * scale)


# ============================================================
# PYTHON DISPATCH WRAPPER
# ============================================================


def apply_shocks_to_base_levels(
    *,
    base_levels: np.ndarray,
    batch: ScenarioSetBatch,
    out: Optional[np.ndarray] = None,
) -> np.ndarray:
    """
    Unpacks the Python ScenarioSetBatch dataclass and feeds the raw
    NumPy arrays into the Numba kernel.
    """
    N = base_levels.shape[0]
    B = batch.shocks.shape[0]

    # Force shocks to be Fortran-contiguous for optimal access patterns in the Numba kernel.
    if not batch.shocks.flags["F_CONTIGUOUS"]:
        shocks = np.asfortranarray(batch.shocks)
    else:
        shocks = batch.shocks

    # Pre-allocate if not provided. In the hot VaR loop,
    # you should always be passing a slice of the global array here.
    if out is None:
        out = np.empty((N, B), dtype=np.float64, order="C")

    _apply_shocks_numba_core(
        base_levels=base_levels,
        shocks=shocks,
        req_to_uix=batch.required_to_shock_uix,
        shock_scale=batch.shock_scale,
        shift_types=batch.required_shift_type_id,
        out=out,
    )

    return out

# riskquant/simlab/resolve/proxy.py
from __future__ import annotations

from dataclasses import dataclass
from typing import Dict, Final, Tuple

import numpy as np

from riskquant.config.static_config.static_config import RiskStaticConfig
from riskquant.exceptions import PrettyKeyError
from riskquant.utils.tenors import parse_bracket_token

# =============================== #
# --- SOURCE SELECTION CODES ---- #
# =============================== #

# --- RESOLUTION CODES --- #
RES_NOT_SIMULATED: Final[np.int8] = np.int8(-1)
RES_DIRECT: Final[np.int8] = np.int8(0)
RES_TENOR_FALLBACK: Final[np.int8] = np.int8(1)
RES_ID_PROXY: Final[np.int8] = np.int8(2)
RES_CONTAINER_PROXY: Final[np.int8] = np.int8(3)

# ======================================================= #
# --- COMPILED OUTPUT: REQUIRED -> SOURCE (PROXY) MAP --- #
# ======================================================= #


@dataclass(frozen=True, slots=True, kw_only=True)
class ProxyResolution:
    """
    Compiled proxy mapping between portfolio-required risk factors and
    available market data risk factors.

    This object defines, for each required risk factor ID referenced by
    instruments in the compiled portfolio:

      - which historical time series should be used to generate shocks
      - how those shocks should be scaled
      - how to efficiently reuse shared source series across multiple exposures

    It is a pure mapping layer:
      - no current levels
      - no shift types
      - no currency logic
      - no valuation logic

    Attributes
    ----------
    required_factor_ids : (N,) int32
        Ordered list of risk factor IDs required by portfolio exposures.
        This is the domain being resolved (portfolio universe A).

    source_factor_ids : (N,) int32
        For each required_factor_id, the resolved risk factor ID whose
        historical time series will be used as the shock source.
        Each entry is guaranteed to exist in RiskFactorMatrix.factor_ids
        (market data universe B).

    required_to_source_index : (N,) int32
        For each required_factor_id, an integer index into unique_source_factor_ids.
        Enables:
          - computing returns/shocks once per unique source series (K series),
          - then broadcasting those shocks to all required IDs that map to it.

    shock_scale : (N,) float32
        Volatility scaling factor applied to the shock derived from the source
        factor for this required exposure. Defaults to 1.0 when no scaling is
        configured.

        IMPORTANT: scaling applies to the shock (return/diff), not to levels.

    unique_source_factor_ids : (K,) int32
        Deduplicated list of source factor IDs required from the RiskFactorMatrix.
        K <= N. The ordering is deterministic: first-seen order during resolution.

    resolution_code : (N,) int8
        Per-required-id resolution type code:

          - RES_NOT_SIMULATED     (-1): required_id is not simulated (e.g. shift type = "none")
          - RES_DIRECT            (0): required_id exists directly in market data universe B
          - RES_TENOR_FALLBACK    (1): same-container nearest-tenor fallback within B applied
          - RES_ID_PROXY          (2): explicit ID-level proxy (risk_factor_proxies) applied
          - RES_CONTAINER_PROXY   (3): explicit container-level proxy (container_proxies) applied
    """

    # Attribute list
    required_factor_ids: np.ndarray
    source_factor_ids: np.ndarray
    required_to_source_index: np.ndarray
    shock_scale: np.ndarray
    unique_source_factor_ids: np.ndarray
    resolution_code: np.ndarray


# ========================================== #
# --- SMALL HOT HELPERS (NO ALLOCATIONS) --- #
# ========================================== #


def _nearest_offset_match(
    *,
    offsets_sorted: np.ndarray,
    rf_ids_sorted: np.ndarray,
    target: int,
) -> int:
    """
    Nearest neighbor selection on signed OFFSET tenors.

    - offsets_sorted: sorted int32 offsets for proxy container, aligned with rf_ids_sorted
    - rf_ids_sorted:  rf_id aligned with offsets_sorted
    - target:         required signed offset to match

    Deterministic tie-break:
      - if equally close to two offsets, choose the lower offset (j-1).
    """
    j = int(np.searchsorted(offsets_sorted, np.int32(target), side="left"))
    if j <= 0:
        return int(rf_ids_sorted[0])
    if j >= offsets_sorted.shape[0]:
        return int(rf_ids_sorted[-1])

    lo = int(offsets_sorted[j - 1])
    hi = int(offsets_sorted[j])
    if (target - lo) <= (hi - target):
        return int(rf_ids_sorted[j - 1])
    return int(rf_ids_sorted[j])


def _build_available_container_tenor_index(
    *,
    cfg: RiskStaticConfig,
    market_data_factor_ids: np.ndarray,  # (Nb,) int32 (B)
) -> Dict[str, Tuple[np.ndarray, np.ndarray, Dict[str, int]]]:
    """
    Build a per-container tenor lookup restricted to the IDs that actually exist in market data (B).

    Output:
        container_name -> (offsets_sorted, rf_ids_sorted, label_to_rf_id)

    This is used for the default "same-container extrapolation" behavior:
      - if required_id is missing in B but the container exists in B, match to nearest tenor on the same container.

    Notes (CPU/RAM):
      - Allocates small per-container Python lists/dicts, proportional to Nb (market data size), not config size.
      - Uses cfg.rf_id_to_risk_factor for O(1) id -> (container, tenor_token).
      - Sorting is per-container and typically small.
    """
    # --- COLLECT OFFSETS + LABELS FOR IDS PRESENT IN MARKET DATA --- #
    offsets: Dict[str, list[int]] = {}
    offset_ids: Dict[str, list[int]] = {}
    labels: Dict[str, Dict[str, int]] = {}

    rf_by_id = cfg.rf_id_to_risk_factor  # local binding

    for mid in market_data_factor_ids:
        rid = int(mid)
        rf = rf_by_id.get(rid)
        if rf is None:
            # Market data contains an id unknown to config: fail-fast, because
            # tenor/container matching is undefined without config metadata.
            raise PrettyKeyError(
                f"market_data_factor_id not found in config: {rid}"
            )

        container = rf.container_name
        tok = rf.tenor_token

        if tok is None:
            # NONE tenor => empty label key
            labels.setdefault(container, {})[""] = rid
            continue

        unit, off = parse_bracket_token(tok)
        if off is None:
            # LABEL tenor => exact label match
            labels.setdefault(container, {})[unit] = rid
        else:
            # OFFSET tenor => nearest neighbor match
            offsets.setdefault(container, []).append(int(off))
            offset_ids.setdefault(container, []).append(rid)

    # --- MATERIALIZE SORTED ARRAYS PER CURVE --- #
    out: Dict[str, Tuple[np.ndarray, np.ndarray, Dict[str, int]]] = {}

    for container, olist in offsets.items():
        ids_list = offset_ids[container]
        o = np.asarray(olist, dtype=np.int32)
        r = np.asarray(ids_list, dtype=np.int32)
        idx = np.argsort(o, kind="mergesort")  # deterministic ties
        out[container] = (o[idx], r[idx], labels.get(container, {}))

    for container, d in labels.items():
        if container not in out:
            out[container] = (
                np.empty(0, dtype=np.int32),
                np.empty(0, dtype=np.int32),
                d,
            )

    return out


# ============================================================= #
# --- MAIN RESOLVER: REQUIRED IDS -> MARKET DATA SOURCE IDS --- #
# ============================================================= #


def resolve_required_factor_ids_to_market_data_factor_ids(
    *,
    cfg: RiskStaticConfig,
    required_factor_ids: np.ndarray,  # (N,) int32 (A)
    market_data_factor_ids: np.ndarray,  # (Nb,) int32 (B)
) -> ProxyResolution:
    """
    Resolve each required risk factor ID onto a source ID present in market data.

    Input universes:
      - required_factor_ids: portfolio-required factor IDs (A)
      - market_data_factor_ids: factor IDs present in RiskFactorMatrix.factor_ids (B)

    Resolution priority (first match wins):
      (1) Direct: required_id exists in B

      (2) ID-level proxy (explicit, highest override):
            - cfg.rf_id_to_proxy_rf_id[required_id] -> (proxy_id, scale)

      (3) Container-level proxy (explicit, second override):
            - map required container -> proxy container via cfg.container_proxies_dict
            - match tenor on proxy container (LABEL exact match; OFFSET nearest match)
            - apply container-level scaling factor

      (4) Same-container tenor fallback (default extrapolation, lowest priority):
            - if required_id missing but the container exists in B, match tenor on the *same* container
              (LABEL exact match; OFFSET nearest match, including extrapolation to end points)
            - scale = 1.0

    IMPORTANT DESIGN RULE:
      - This resolver selects *data sources only*.
      - Any metadata (ccy / shift type / risk factor group) must come from the
        *required* container later, not from the proxy container.
    """

    # --- SMOKE TESTS --- #
    if required_factor_ids.dtype != np.int32:
        raise TypeError(
            f"required_factor_ids must be int32; got {required_factor_ids.dtype}"
        )
    if market_data_factor_ids.dtype != np.int32:
        raise TypeError(
            f"market_data_factor_ids must be int32; got {market_data_factor_ids.dtype}"
        )

    # --- FAST MEMBERSHIP STRUCTURE FOR AVAILABLE MARKET DATA IDS --- #
    #
    # avail_get(rf_id) is not None <=> rf_id exists in market data universe B
    avail: Dict[int, int] = {}
    for mid in market_data_factor_ids:
        avail[int(mid)] = 1
    avail_get = avail.get

    # --- CONFIG LOOKUPS (LOCAL BINDINGS) --- #
    #
    # Local bindings reduce attribute resolution overhead inside the per-id loop.
    id_proxy_get = cfg.rf_id_to_proxy_rf_id.get
    rf_by_id = cfg.rf_id_to_risk_factor
    container_proxy_get = cfg.container_proxies_dict.get

    # --- BUILD MARKET-DATA-RESTRICTED CONTAINER INDEX (FOR CONTAINER PROXY + TENOR FALLBACK) --- #
    #
    # Critical design rule:
    #   Any chosen source_id must exist in market data universe B.
    # Therefore:
    #   - same-container fallback matches tenors using only IDs that exist in B
    #   - container-level proxy also matches tenors using only IDs that exist in B
    #
    # This prevents "config-only" proxy IDs that are absent from market data.
    avail_container_index = _build_available_container_tenor_index(
        cfg=cfg,
        market_data_factor_ids=market_data_factor_ids,
    )
    avail_container_index_get = avail_container_index.get

    # --- OUTPUT ARRAYS --- #
    num_required_ids = int(required_factor_ids.shape[0])
    source_factor_ids = np.empty(num_required_ids, dtype=np.int32)
    shock_scale = np.empty(num_required_ids, dtype=np.float32)
    resolution_code = np.empty(num_required_ids, dtype=np.int8)

    # --- UNIQUE SOURCE UNIVERSE (FIRST-SEEN ORDER) --- #
    #
    # We build:
    #   src_to_uix: source_id -> universe index (0..K-1)
    #   unique_src: list of unique source ids in first-seen order
    src_to_uix: Dict[int, int] = {}
    unique_src: list[np.int32] = []
    src_to_uix_get = src_to_uix.get
    src_to_uix_set = src_to_uix.__setitem__
    unique_append = unique_src.append

    # --- RESOLVE EACH REQUIRED ID --- #
    for i in range(num_required_ids):
        required_id = int(required_factor_ids[i])

        # ============================================================ #
        # --- (0) REQUIRED METADATA LOOKUP (CURVE + TENOR TOKEN) --- #
        # ============================================================ #
        #
        # We need container_name and tenor_token for:
        #   - container-level proxy tenor matching
        #   - same-container tenor fallback matching
        rf = rf_by_id.get(required_id)
        if rf is None:
            raise PrettyKeyError(
                f"Required factor_id not found in config: {required_id}"
            )

        required_container = rf.container_name
        tok = rf.tenor_token
        if tok is None:
            target_unit = ""
            target_off = None
        else:
            target_unit, target_off = parse_bracket_token(tok)

        # ================================= #
        # --- (1) DIRECT (NO PROXYING) --- #
        # ================================= #
        if avail_get(required_id) is not None:
            source_id = required_id
            sc = 1.0
            rc = RES_DIRECT

        else:
            # =================================================== #
            # --- (2) ID-LEVEL PROXY (EXPLICIT OVERRIDE) --- #
            # =================================================== #
            #
            # Highest priority override: if the user explicitly proxied the point,
            # we use it even if other fallbacks would be possible.
            pr = id_proxy_get(required_id)
            if pr is not None:
                source_id = int(pr[0])
                sc = float(pr[1])
                rc = RES_ID_PROXY

                # Must exist in market data universe B.
                if avail_get(source_id) is None:
                    raise PrettyKeyError(
                        "ID-level proxy resolved to missing market data factor_id:\n"
                        f"- required_id={required_id}\n"
                        f"- required_container={required_container!r}\n"
                        f"- proxy_id={source_id}\n"
                    )

            else:
                # ===================================================== #
                # --- (3) CONTAINER-LEVEL PROXY (EXPLICIT OVERRIDE) --- #
                # ===================================================== #
                #
                # Second priority override: if the container has been configured to proxy
                # onto another container, use that proxy container's data + tenor matching.
                cp = container_proxy_get(required_container)
                if cp is not None:
                    proxy_container = cp.proxy_container_name
                    container_scale = float(
                        cp.proxy_container_vol_scaling_factor or 1.0
                    )

                    # --- MARKET-DATA-RESTRICTED TENOR MATCH ON PROXY CONTAINER --- #
                    #
                    # We match on proxy_container using only IDs that exist in B, otherwise
                    # we'd select an ID not present in the matrix (the original bug).
                    ci_px = avail_container_index_get(proxy_container)
                    if ci_px is None:
                        raise PrettyKeyError(
                            "Container-level proxy container has no market data coverage:\n"
                            f"- required_id={required_id}\n"
                            f"- required_container={required_container!r}\n"
                            f"- proxy_container={proxy_container!r}\n"
                            f"- tenor_token={tok!r}\n"
                        )

                    offsets_sorted_px, rf_ids_sorted_px, label_map_px = ci_px

                    # LABEL/NONE tenor: exact label match on proxy container (within B)
                    if target_off is None:
                        sid2 = label_map_px.get(target_unit)
                        if sid2 is None:
                            raise PrettyKeyError(
                                "Cannot proxy LABEL/NONE tenor onto proxy container (no matching label in market data):\n"
                                f"- required_id={required_id}\n"
                                f"- required_container={required_container!r}\n"
                                f"- proxy_container={proxy_container!r}\n"
                                f"- tenor_token={tok!r}\n"
                                f"- label={target_unit!r}\n"
                            )
                        source_id = int(sid2)

                    # OFFSET tenor: nearest signed offset match on proxy container (within B)
                    else:
                        if offsets_sorted_px.shape[0] == 0:
                            raise PrettyKeyError(
                                "Proxy container has no OFFSET tenors in market data to match required tenor:\n"
                                f"- required_id={required_id}\n"
                                f"- required_container={required_container!r}\n"
                                f"- proxy_container={proxy_container!r}\n"
                                f"- tenor_token={tok!r}\n"
                            )
                        source_id = _nearest_offset_match(
                            offsets_sorted=offsets_sorted_px,
                            rf_ids_sorted=rf_ids_sorted_px,
                            target=int(target_off),
                        )

                    # Defensive: should always be in B by construction.
                    if avail_get(source_id) is None:
                        raise PrettyKeyError(
                            "Internal error: container-level proxy produced id not in market data:\n"
                            f"- required_id={required_id}\n"
                            f"- required_container={required_container!r}\n"
                            f"- proxy_container={proxy_container!r}\n"
                            f"- proxy_id={source_id}\n"
                        )

                    sc = container_scale
                    rc = RES_CONTAINER_PROXY

                else:
                    # =============================================================== #
                    # --- (4) SAME-C CONTAINER TENOR FALLBACK (DEFAULT EXTRAPOLATION) --- #
                    # =============================================================== #
                    #
                    # Lowest priority behavior: if we have the container in market data but
                    # not the exact tenor point, map to the nearest available tenor on
                    # the *same container within B* (end-point extrapolation by selection).
                    ci_av = avail_container_index_get(required_container)
                    if ci_av is None:
                        raise PrettyKeyError(
                            "Missing history for required_id and no proxy configured:\n"
                            f"- required_id={required_id}\n"
                            f"- required_container={required_container!r}\n"
                            f"- tenor_token={tok!r}\n"
                            "No ID proxy, no container proxy, and container not present in market data universe.\n"
                        )

                    offsets_sorted, rf_ids_sorted, label_map = ci_av

                    # LABEL/NONE tenor: exact label match on same container (within B)
                    if target_off is None:
                        sid2 = label_map.get(target_unit)
                        if sid2 is None:
                            raise PrettyKeyError(
                                "Same-container fallback failed for LABEL/NONE tenor:\n"
                                f"- required_id={required_id}\n"
                                f"- required_container={required_container!r}\n"
                                f"- tenor_token={tok!r}\n"
                                f"- label={target_unit!r}\n"
                                "Container exists in market data, but no matching label exists.\n"
                            )
                        source_id = int(sid2)

                    # OFFSET tenor: nearest signed offset match on same container (within B)
                    else:
                        if offsets_sorted.shape[0] == 0:
                            raise PrettyKeyError(
                                "Same-container fallback failed: container has no OFFSET tenors in market data:\n"
                                f"- required_id={required_id}\n"
                                f"- required_container={required_container!r}\n"
                                f"- tenor_token={tok!r}\n"
                            )
                        source_id = _nearest_offset_match(
                            offsets_sorted=offsets_sorted,
                            rf_ids_sorted=rf_ids_sorted,
                            target=int(target_off),
                        )

                    # Defensive: should always be in B by construction.
                    if avail_get(source_id) is None:
                        raise PrettyKeyError(
                            "Internal error: same-container fallback produced id not in market data:\n"
                            f"- required_id={required_id}\n"
                            f"- required_container={required_container!r}\n"
                            f"- resolved_source_id={source_id}\n"
                        )

                    sc = 1.0
                    rc = RES_TENOR_FALLBACK

        # --- WRITE PER-REQUIRED OUTPUTS --- #
        source_factor_ids[i] = np.int32(source_id)
        shock_scale[i] = np.float32(sc)
        resolution_code[i] = np.int8(rc)

        # --- MAINTAIN UNIQUE SOURCE UNIVERSE --- #
        uix = src_to_uix_get(source_id)
        if uix is None:
            uix = len(unique_src)
            src_to_uix_set(source_id, uix)
            unique_append(np.int32(source_id))

    unique_source_factor_ids = np.asarray(unique_src, dtype=np.int32)

    # --- BUILD REQUIRED -> UNIQUE SOURCE INDEX VECTOR --- #
    required_to_source_index = np.empty(num_required_ids, dtype=np.int32)
    for i in range(num_required_ids):
        required_to_source_index[i] = np.int32(
            src_to_uix[int(source_factor_ids[i])]
        )

    # build and return
    return ProxyResolution(
        required_factor_ids=required_factor_ids,
        source_factor_ids=source_factor_ids,
        required_to_source_index=required_to_source_index,
        shock_scale=shock_scale,
        unique_source_factor_ids=unique_source_factor_ids,
        resolution_code=resolution_code,
    )

from __future__ import annotations

import json
import logging

from dataclasses import dataclass, field

import numpy as np
import pandas as pd

from riskquant.config.static_config.static_config import RiskStaticConfig
from riskquant.exceptions import PrettyKeyError
from riskquant.rq_logging import LogContext, get_logger, with_context
from riskquant.simlab.resolve.proxy import (

def _build_audit_df(
    *,
    cfg: RiskStaticConfig,
    proxy_resolution: ProxyResolution,
) -> pd.DataFrame:
    """
    Internal implementation. Intentionally materializes Python strings / DataFrame.
    Audit-only; not kernel-friendly.
    """
    _smoke_test_proxy_resolution(proxy_resolution)

    rf_by_id = cfg.rf_id_to_risk_factor
    num_required_factors = int(proxy_resolution.required_factor_ids.shape[0])

    # Avoid copies where possible.
    req_id = proxy_resolution.required_factor_ids.astype(np.int32, copy=False)
    src_id = proxy_resolution.source_factor_ids.astype(np.int32, copy=False)
    code = proxy_resolution.resolution_code

    required_curve = [""] * num_required_factors
    required_tok = [""] * num_required_factors
    required_name = [""] * num_required_factors

    source_curve = [""] * num_required_factors
    source_tok = [""] * num_required_factors
    source_name = [""] * num_required_factors

    source_code = [""] * num_required_factors

    id_proxy_target = [""] * num_required_factors
    curve_proxy_target = [""] * num_required_factors

    for i in range(num_required_factors):
        rid = int(req_id[i])
        sid = int(src_id[i])

        rrf = rf_by_id.get(rid)
        srf = rf_by_id.get(sid)

        if rrf is None:
            raise PrettyKeyError(
                f"required_factor_id not found in config: {rid}"
            )
        if srf is None:
            raise PrettyKeyError(f"source_factor_id not found in config: {sid}")

        required_curve[i] = rrf.container_name
        required_tok[i] = "" if rrf.tenor_token is None else rrf.tenor_token
        required_name[i] = rrf.risk_factor_name

        source_curve[i] = srf.container_name
        source_tok[i] = "" if srf.tenor_token is None else srf.tenor_token
        source_name[i] = srf.risk_factor_name

        c = int(code[i])
        source_code[i] = _SOURCE_CODE_TO_LABEL.get(c, f"UNKNOWN_{c}")

        if c == int(RES_ID_PROXY):
            # ID proxy resolves directly to the proxy rf (sid), so source_name is the target rf name.
            id_proxy_target[i] = source_name[i]
        elif c == int(RES_CONTAINER_PROXY):
            # Curve proxy: show the proxy curve name if present.
            cp = cfg.curve_proxies_dict.get(required_curve[i])
            curve_proxy_target[i] = "" if cp is None else cp.proxy_curve_name

    return pd.DataFrame(
        {
            "required_factor_id": req_id,
            "required_risk_factor_name": required_name,
            "required_curve_name": required_curve,
            "required_tenor_token": required_tok,
            "source_factor_id": src_id,
            "source_risk_factor_name": source_name,
            "source_curve_name": source_curve,
            "source_tenor_token": source_tok,
            "source_code": source_code,
            "shock_scale": proxy_resolution.shock_scale,
            "id_proxy_target_risk_factor_name": id_proxy_target,
            "curve_proxy_target_curve_name": curve_proxy_target,
        }
    )


def _jsonable(v: Any) -> Any:
    """
    Convert non-JSON-native scalars to JSON-friendly Python types.
    """
    # numpy scalar -> python scalar
    if isinstance(v, np.generic):
        return v.item()
    # handle NaN -> None (safer for downstream ingestion)
    if isinstance(v, float) and np.isnan(v):
        return None
    return v


@dataclass(slots=True)
class ProxyResolutionAudit:
    """
    Handle for producing and logging an audit of a ProxyResolution.

    - Lazy: DataFrame is built only when needed.
    - Cached: repeated calls reuse the same DataFrame.
    - Logging: supports summary logging and NDJSON row logging.
    """

    cfg: RiskStaticConfig
    proxy_resolution: ProxyResolution
    name: str = "proxy_resolution_audit"
    component: str = "simlab.resolve.proxy_audit"

    # Tie all row-lines to one emission/run for easy filtering/grouping.
    run_id: str = field(default_factory=lambda: uuid.uuid4().hex)

    # Cached df - build once.
    _df: Optional[pd.DataFrame] = field(default=None, init=False, repr=False)

    @property
    def df(self) -> pd.DataFrame:
        return self.to_frame()

    def to_frame(self) -> pd.DataFrame:
        if self._df is None:
            self._df = _build_audit_df(
                cfg=self.cfg, proxy_resolution=self.proxy_resolution
            )
        return self._df

    def to_csv(self, path: str, *, index: bool = False, **kwargs) -> str:
        self.to_frame().to_csv(path, index=index, **kwargs)
        return path

    def _logger(self, logger: Optional[LoggerLike]) -> logging.LoggerAdapter:
        # Normalize to a base Logger, regardless of whether caller passed a LoggerAdapter.
        base: Optional[logging.Logger]
        if isinstance(logger, logging.LoggerAdapter):
            base = logger.logger
        else:
            base = logger

        if base is None:
            base = get_logger(__name__)

        ctx = LogContext(
            component=self.component,
            object_name=self.name,
            object_id=self.run_id,
        )
        return with_context(base, ctx)

    def log_summary(
        self,
        *,
        logger: Optional[LoggerLike] = None,
        level: int = logging.INFO,
    ) -> None:
        """
        Human-friendly headline summary: row count + source_code counts.
        """
        log = self._logger(logger)
        df = self.to_frame()

        counts = df["source_code"].value_counts(dropna=False)
        counts_str = ", ".join(f"{k}={int(v)}" for k, v in counts.items())

        log.log(
            level,
            "audit_summary rows=%d source_code_counts=[%s]",
            int(df.shape[0]),
            counts_str,
        )

    def log_rows_ndjson(
        self,
        *,
        logger: Optional[LoggerLike] = None,
        level: int = logging.INFO,
        event: str = "proxy_resolution_audit_row",
        columns: Optional[Iterable[str]] = None,
    ) -> None:
        """
        Emit the entire audit table as NDJSON: one log line per row.

        This is designed to be:
          - easy to search (grep/ripgrep)
          - easy to reconstruct (pandas.read_json(..., lines=True)) provided the log line is JSON.

        NOTE:
        - For best ingestion, write these lines to a dedicated handler/file using a formatter of '%(message)s'
          so each line is pure JSON (no timestamp/level prefix).
        """
        log = self._logger(logger)
        df = self.to_frame()

        if columns is not None:
            cols = list(columns)
            missing = [c for c in cols if c not in df.columns]
            if missing:
                raise KeyError(
                    f"Requested audit columns not in DataFrame: {missing}"
                )
            df = df.loc[:, cols]

        col_names = list(df.columns)
        audit_name = self.name
        run_id = self.run_id

        # Iterate without materializing a big list of dicts.
        for i, row in enumerate(df.itertuples(index=False, name=None)):
            row_dict = {
                col_names[j]: _jsonable(row[j]) for j in range(len(col_names))
            }
            payload = {
                "event": event,
                "audit_name": audit_name,
                "run_id": run_id,
                "row_idx": i,
                **row_dict,
            }
            msg = json.dumps(payload, separators=(",", ":"), ensure_ascii=False)
            log.log(level, msg)

from typing import Dict
import numpy as np

from riskquant.simlab.shock_plan.plan import (

@dataclass(frozen=True, slots=True, kw_only=True)
class ShockUniverse:
    """
    Canonical definition of the shock space.

    This object is:
      - deterministic
      - reusable
      - zero-allocation at runtime
    """

    # --- UNIQUE SHOCK KEYS --- #
    shock_source_ids: np.ndarray  # (K,) int32
    shock_shift_type_ids: np.ndarray  # (K,) int8

    # --- REQUIRED → SHOCK MAPPING --- #
    required_to_shock_uix: np.ndarray  # (Nreq,) int32

    # --- COMPUTE PLAN --- #
    compute_plan: ShockComputePlan

    # --- OPTIONAL: RAW KEYS (DEBUG / HASHING) --- #
    shock_keys: np.ndarray  # (K,) int64


def build_shock_universe(
    *,
    required_shift_type_id: np.ndarray,  # (Nreq,) int8
    source_factor_ids: np.ndarray,  # (Nreq,) int32
    id_to_col: Dict[int, int],
) -> ShockUniverse:
    """
    Build a canonical ShockUniverse from proxy resolution outputs.
    """
    # Assertions to enforce preconditions for bit-packing and type assumptions
    assert required_shift_type_id.ndim == 1
    assert (
        required_shift_type_id.max() < 256 and required_shift_type_id.min() >= 0
    )
    assert required_shift_type_id.dtype == np.int8
    assert source_factor_ids.ndim == 1
    assert source_factor_ids.dtype == np.int32
    assert source_factor_ids.shape == required_shift_type_id.shape

    n = int(source_factor_ids.shape[0])

    # --- PACK KEYS --- #
    keys = (source_factor_ids.astype(np.int64) << 8) | (
        required_shift_type_id.astype(np.int64) & 0xFF
    )

    # --- DEDUPLICATION --- #
    key_to_uix: Dict[int, int] = {}
    unique_src: list[np.int32] = []
    unique_st: list[np.int8] = []
    unique_keys: list[np.int64] = []

    req_to_uix = np.empty(n, dtype=np.int32)

    get = key_to_uix.get
    set_ = key_to_uix.__setitem__

    for i in range(n):
        k = int(keys[i])

        uix = get(k)
        if uix is None:
            uix = len(unique_src)
            set_(k, uix)

            unique_src.append(source_factor_ids[i])
            unique_st.append(required_shift_type_id[i])
            unique_keys.append(np.int64(k))

        req_to_uix[i] = np.int32(uix)

    shock_source_ids = np.asarray(unique_src, dtype=np.int32)
    shock_shift_type_ids = np.asarray(unique_st, dtype=np.int8)
    shock_keys = np.asarray(unique_keys, dtype=np.int64)

    # --- BUILD COMPUTE PLAN --- #
    compute_plan = build_shock_compute_plan(
        shock_source_ids=shock_source_ids,
        shock_shift_type_ids=shock_shift_type_ids,
        id_to_col=id_to_col,
    )

    return ShockUniverse(
        shock_source_ids=shock_source_ids,
        shock_shift_type_ids=shock_shift_type_ids,
        required_to_shock_uix=req_to_uix,
        compute_plan=compute_plan,
        shock_keys=shock_keys,
    )

# =========================================================================
# ZERO-ALLOCATION PATH-AGGREGATION KERNEL - SB BOOTSTRAP SCENARIO GENERATOR
# =========================================================================


@nb.njit(parallel=True, fastmath=True, cache=True)
def _kernel_fused_sbb_h_day_shock(
    *,
    values: np.ndarray,  # Fortran-contiguous (T x K_total)
    cols: np.ndarray,  # K_req, column indices into values
    shift_types: np.ndarray,  # K_req
    inv_perm: np.ndarray,  # K_req, output column positions
    random_t_indices: np.ndarray,  # (B, h) matrix of synthetic daily paths
    h: int,  # Effective horizon in days
    B: int,  # Number of scenarios in the batch
    out: np.ndarray,  # Fortran-contiguous (B x K_req)
) -> None:
    """
    Computes effective h-day shocks by aggregating a synthetic path of 1-day returns.
    Parallelized across requested columns for optimal Fortran cache-line usage.
    """
    K = cols.shape[0]

    for k in nb.prange(K):
        col = cols[k]
        st = shift_types[k]
        out_col = inv_perm[k]

        # Bind column for cache efficiency
        col_vals = values[:, col]
        out_col_vals = out[:, out_col]

        if st == SHIFT_LOG_RETURN:
            for b in range(B):
                # Product of ratios over the h-day path
                cum_ratio = 1.0
                for step in range(h):
                    t1 = random_t_indices[b, step]
                    t0 = t1 - 1
                    v1 = col_vals[t1]
                    v0 = col_vals[t0]
                    cum_ratio *= v1 / v0
                # Single log call per scenario!
                out_col_vals[b] = np.log(cum_ratio)

        elif st == SHIFT_SIMPLE_RETURN:
            for b in range(B):
                # Compound product of (1 + r) over the h-day path
                cum_ret = 1.0
                for step in range(h):
                    t1 = random_t_indices[b, step]
                    t0 = t1 - 1
                    v1 = col_vals[t1]
                    v0 = col_vals[t0]
                    cum_ret *= v1 / v0
                out_col_vals[b] = cum_ret - 1.0

        elif st == SHIFT_DIFFERENCE:
            for b in range(B):
                # Sum of absolute differences over the h-day path
                cum_diff = 0.0
                for step in range(h):
                    t1 = random_t_indices[b, step]
                    t0 = t1 - 1
                    v1 = col_vals[t1]
                    v0 = col_vals[t0]
                    cum_diff += v1 - v0
                out_col_vals[b] = cum_diff


# =========================================================================
# STATIONARY BLOCK BOOTSTRAP GENERATOR
# =========================================================================


@dataclass(frozen=True, slots=True, kw_only=True)
class BootstrapMCScenarioGenerator(ScenarioGenerator):
    """
    Generates scenarios via Stationary Block Bootstrap (SBB).

    Constructs an h-day synthetic path for each scenario by stitching together
    blocks of contiguous 1-day historical returns. Block lengths are drawn from
    a Geometric distribution (expected_block_size).

    References
    ----------
    Politis, D. N., & Romano, J. P. (1994). The Stationary Bootstrap. Journal of the American Statistical Association, 89(428), 1303–1313. https://doi.org/10.2307/2290993

    Politis, D. N. (2003). The impact of bootstrap methods on time series analysis. Statistical Science, 18. https://doi.org/10.1214/ss/1063994977

    """

    matrix: RiskFactorMatrix
    total_scenarios: int
    expected_block_size: float = 1.0
    horizon: int = 1
    seed: int | None = None
    shock_dtype: type[np.float32] | type[np.float64] = np.float32

    def __post_init__(self) -> None:
        if self.total_scenarios <= 0:
            raise ValueError("total_scenarios must be positive")
        if self.expected_block_size < 1.0:
            raise ValueError("expected_block_size must be >= 1.0")
        if self.horizon < 1:
            raise ValueError("horizon must be >= 1")

        T = self.matrix.values.shape[0]
        if T < 2:
            raise ValueError(
                f"Matrix history (T={T}) is too short to compute 1-day returns."
            )

    def iter_batches(
        self,
        *,
        required_shift_type_id: np.ndarray,
        source_factor_ids: np.ndarray,
        shock_scale: np.ndarray,
        batch_size: int,
    ) -> Iterator[ScenarioSetBatch]:

        # Build factor id -> column map
        id_to_col = build_factor_id_to_col_index(self.matrix.factor_ids)

        # Build shock universe and compute plan
        shock_universe = build_shock_universe(
            required_shift_type_id=required_shift_type_id,
            source_factor_ids=source_factor_ids,
            id_to_col=id_to_col,
        )

        T = self.matrix.values.shape[0]
        h = int(self.horizon)
        K_req = shock_universe.compute_plan.cols_sorted.shape[0]

        # --- 1. GENERATE ALL RANDOM PATHS UPFRONT (OUTSIDE THE LOOP) ---
        # This guarantees batch-size invariance and maximizes throughput
        # by only calling the RNG C-API once for the entire simulation.
        rng = np.random.default_rng(self.seed)
        master_t_indices = generate_sbb_path_indices(
            T=T,
            h=h,
            B=self.total_scenarios,  # <-- Generates all paths at once
            expected_block_size=self.expected_block_size,
            rng=rng,
        )

        scenario_offset = 0

        # --- 2. STREAM SCENARIO BATCHES ---
        while scenario_offset < self.total_scenarios:
            B = min(batch_size, self.total_scenarios - scenario_offset)

            # Slice the master index array for this specific batch
            batch_t_indices = master_t_indices[
                scenario_offset : scenario_offset + B
            ]

            shocks = np.empty((B, K_req), dtype=self.shock_dtype, order="F")

            _kernel_fused_sbb_h_day_shock(
                values=self.matrix.values,
                cols=shock_universe.compute_plan.cols_sorted,
                shift_types=shock_universe.compute_plan.shift_types_sorted,
                inv_perm=shock_universe.compute_plan.inverse_permutation,
                random_t_indices=batch_t_indices,
                B=B,
                h=h,
                out=shocks,
            )

            yield ScenarioSetBatch(
                shocks=shocks,
                required_to_shock_uix=shock_universe.required_to_shock_uix,
                shock_scale=shock_scale,
                required_shift_type_id=required_shift_type_id,
                scenario_offset=scenario_offset,
                scenario_count_total=self.total_scenarios,
            )

            scenario_offset += B

    out: np.ndarray,  # Fortran-contiguous (B x K_req)
) -> None:
    """
    Computes effective h-day shocks by aggregating a synthetic path of 1-day returns.
    Parallelized across requested columns for optimal Fortran cache-line usage.
    """
    K = cols.shape[0]

    for k in nb.prange(K):
        col = cols[k]
        st = shift_types[k]
        out_col = inv_perm[k]

        # Bind column for cache efficiency
        col_vals = values[:, col]
        col_vols = vol_ratios[:, k]  # <-- Reads from K_req aligned array
        out_col_vals = out[:, out_col]

        if st == SHIFT_LOG_RETURN:
            for b in range(B):
                # Product of ratios over the h-day path
                cum_shock = 0.0
                for step in range(h):
                    t1 = random_t_indices[b, step]
                    t0 = t1 - 1
                    v1 = col_vals[t1]
                    v0 = col_vals[t0]
                    r_log = np.log(v1 / v0)
                    cum_shock += r_log * col_vols[t1]
                out_col_vals[b] = cum_shock

        elif st == SHIFT_SIMPLE_RETURN:
            for b in range(B):
                # Compound product of (1 + r) over the h-day path
                cum_ret = 1.0
                for step in range(h):
                    t1 = random_t_indices[b, step]
                    t0 = t1 - 1
                    v1 = col_vals[t1]
                    v0 = col_vals[t0]
                    r_simple = (v1 / v0) - 1.0
                    cum_ret *= 1.0 + (r_simple * col_vols[t1])
                out_col_vals[b] = cum_ret - 1.0

        elif st == SHIFT_DIFFERENCE:
            for b in range(B):
                # Sum of absolute differences over the h-day path
                cum_diff = 0.0
                for step in range(h):
                    t1 = random_t_indices[b, step]
                    t0 = t1 - 1
                    v1 = col_vals[t1]
                    v0 = col_vals[t0]
                    cum_diff += (v1 - v0) * col_vols[t1]
                out_col_vals[b] = cum_diff

# src/riskquant/simlab/scenarios/core_math/ewma_volatility.py


from riskquant.market_data.risk_factors.shift_types import (
    SHIFT_DIFFERENCE,
    SHIFT_LOG_RETURN,
    SHIFT_SIMPLE_RETURN,
)


import numba as nb
import numpy as np


# =========================================================================
# EWMA VOLATILITY PRE-COMPILER (DYNAMIC TO SHIFT TYPES)
# =========================================================================


@nb.njit(parallel=True, fastmath=True, cache=True)
def kernel_ewma_vol_ratios(
    values: np.ndarray,  # Fortran-contiguous (T x K_total)
    cols: np.ndarray,  # K_req, column indices into values
    shift_types: np.ndarray,  # K_req
    decay_factor: float,  # lambda (e.g., 0.94)
    out_ratios: np.ndarray,  # Fortran-contiguous (T x K_req)
) -> None:
    """
    Computes the FHS Z-score matrix: Z_t = (Sigma_today / Sigma_t)
    dynamically respecting the specific shift type requested for each column.
    """
    T = values.shape[0]
    K = cols.shape[0]

    for k in nb.prange(K):
        col = cols[k]
        st = shift_types[k]

        col_vals = values[:, col]
        col_out_ratios = out_ratios[:, k]

        variances = np.empty(T, dtype=np.float64)

        # 1. Initialize first return based on strict shift type
        if st == SHIFT_LOG_RETURN:
            r1 = np.log(col_vals[1] / col_vals[0])
        elif st == SHIFT_SIMPLE_RETURN:
            r1 = (col_vals[1] / col_vals[0]) - 1.0
        elif st == SHIFT_DIFFERENCE:
            r1 = col_vals[1] - col_vals[0]
        else:
            r1 = 0.0

        var_t = r1 * r1
        variances[0] = var_t
        variances[1] = var_t

        # 2. EWMA loop matching strict shift type
        if st == SHIFT_LOG_RETURN:
            for t in range(2, T):
                rt = np.log(col_vals[t] / col_vals[t - 1])
                var_t = (1.0 - decay_factor) * (rt * rt) + decay_factor * var_t
                variances[t] = var_t

        elif st == SHIFT_SIMPLE_RETURN:
            for t in range(2, T):
                rt = (col_vals[t] / col_vals[t - 1]) - 1.0
                var_t = (1.0 - decay_factor) * (rt * rt) + decay_factor * var_t
                variances[t] = var_t

        elif st == SHIFT_DIFFERENCE:
            for t in range(2, T):
                rt = col_vals[t] - col_vals[t - 1]
                var_t = (1.0 - decay_factor) * (rt * rt) + decay_factor * var_t
                variances[t] = var_t

        # 3. Today's variance is the very last element
        var_today = variances[T - 1]

        # 4. Calculate Z-ratios (Sigma_today / Sigma_t)
        for t in range(T):
            if variances[t] < 1e-12:
                col_out_ratios[t] = 1.0
            else:
                col_out_ratios[t] = np.sqrt(var_today / variances[t])

# src/riskquant/simlab/scenarios/core_math/sbb_block_generator.py


import numpy as np


def generate_sbb_path_indices(
    *,
    T: int,
    h: int,
    B: int,
    expected_block_size: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Vectorized generator for Stationary Block Bootstrap (SBB) path indices.

    Returns
    -------
    random_t_indices : np.ndarray
        A (B, h) matrix of int32 indices bounded strictly between [1, T-1].
        These represent the valid t1 indices to calculate 1-day historical returns.
    """
    # We need 1-day returns, so valid t1 indices are [1, T-1]
    max_valid_length = T - 1
    p = 1.0 / float(expected_block_size)

    # --- VECTORIZED RANDOM GENERATION ---
    # (B * h) is the absolute maximum number of blocks we could ever need
    max_possible_blocks = B * h

    if p >= 1.0:
        block_lengths = np.ones(max_possible_blocks, dtype=np.int32)
    else:
        block_lengths = rng.geometric(p, size=max_possible_blocks).astype(
            np.int32
        )

    # Draw unconstrained valid starting indices (1 to T-1)
    start_indices = rng.integers(
        low=1, high=T, size=max_possible_blocks, dtype=np.int32
    )

    # --- PATH STITCHING ---
    random_t_indices = np.empty((B, h), dtype=np.int32)

    block_idx = 0
    for b in range(B):
        step = 0
        while step < h:
            b_len = min(block_lengths[block_idx], h - step)
            start_t = start_indices[block_idx]

            # Generate raw sequence
            raw_indices = np.arange(start_t, start_t + b_len, dtype=np.int32)

            # Circular Wrap-Around: keeps indices bounded strictly between [1, T-1]
            wrapped = 1 + ((raw_indices - 1) % max_valid_length)

            random_t_indices[b, step : step + b_len] = wrapped

            step += b_len
            block_idx += 1

    return random_t_indices

@nb.njit(parallel=True, fastmath=True, cache=True)
def _kernel_fused_h_day_shock(
    *,
    values: np.ndarray,  # Fortran-contiguous (T x K)
    cols: np.ndarray,  # K, column indices into values
    shift_types: np.ndarray,  # K
    inv_perm: np.ndarray,  # K, output column positions
    t_start: int,
    h: int,  # Effective horizon in days
    B: int,  # Number of scenarios in the batch
    out: np.ndarray,  # Fortran-contiguous (B x K)
):
    """
    Computes h-day shock from 1-day returns by directly indexing into the values array at the start and end of the h-day period.
    Parallelized across factors (columns) for optimal Fortran cache-line usage.
    """
    K = cols.shape[0]

    # Parallel over factors (columns)
    for k in nb.prange(K):
        col = cols[k]
        st = shift_types[k]
        out_col = inv_perm[k]

        # Precompute base indices for the h-day period to avoid redundant arithmetic in the B-loop
        base_t0 = t_start - h

        # Bind column for cache efficiency
        col_vals = values[:, col]
        out_col_vals = out[:, out_col]

        # Bind t1 and t0
        t1 = t_start
        t0 = base_t0

        # Hoist branch out of B-loop
        if st == SHIFT_LOG_RETURN:
            for b in range(B):
                v1 = col_vals[t1]
                v0 = col_vals[t0]
                out_col_vals[b] = np.log(v1 / v0)
                t1 += 1
                t0 += 1

        elif st == SHIFT_SIMPLE_RETURN:
            for b in range(B):
                v1 = col_vals[t1]
                v0 = col_vals[t0]
                out_col_vals[b] = v1 / v0 - 1.0
                t1 += 1
                t0 += 1

        elif st == SHIFT_DIFFERENCE:
            for b in range(B):
                v1 = col_vals[t1]
                v0 = col_vals[t0]
                out_col_vals[b] = v1 - v0
                t1 += 1
                t0 += 1

                out=shocks,

    def _apply_distribution_scaling(
        self,
        *,
        rng: np.random.Generator,
        shocks: np.ndarray,
        gmm_sliced_multipliers: float | np.ndarray | None,
    ) -> None:
        if self.distribution_kind == "standard_normal":
            return

        elif self.distribution_kind == "student_t":
            assert self.student_t_df is not None
            chi_square = rng.chisquare(
                df=self.student_t_df, size=shocks.shape[0]
            )
            row_scale = np.sqrt((self.student_t_df - 2.0) / chi_square)
            np.multiply(shocks, row_scale[:, None], out=shocks)

        elif self.distribution_kind == "laplace":
            exponential_mix = rng.exponential(scale=1.0, size=shocks.shape[0])
            row_scale = np.sqrt(exponential_mix)
            np.multiply(shocks, row_scale[:, None], out=shocks)

        elif self.distribution_kind == "variance_gamma":
            assert self.vg_nu is not None
            gamma_mix = rng.gamma(
                shape=1.0 / self.vg_nu, scale=self.vg_nu, size=shocks.shape[0]
            )
            row_scale = np.sqrt(gamma_mix)
            np.multiply(shocks, row_scale[:, None], out=shocks)

        elif self.distribution_kind == "regime_switching_gmm":
            assert self.prob_crisis is not None
            assert gmm_sliced_multipliers is not None
            crisis_mask = rng.random(shocks.shape[0]) < self.prob_crisis
            if np.any(crisis_mask):
                shocks[crisis_mask, :] *= gmm_sliced_multipliers

    def iter_batches(
        self,
        *,
        required_shift_type_id: np.ndarray,
        source_factor_ids: np.ndarray,
        shock_scale: np.ndarray,
        batch_size: int,
    ) -> Iterator[ScenarioSetBatch]:

        if batch_size <= 0:
            raise ValueError("batch_size must be positive")

        shock_source_ids, _, req_to_uix = build_shock_key_plan(
            required_shift_type_id=required_shift_type_id,
            source_factor_ids=source_factor_ids,
        )

        K_total = int(self.cholesky_matrix.shape[0])
        K_req = int(shock_source_ids.shape[0])

        id_to_col = {int(fid): i for i, fid in enumerate(self.factor_ids)}
        col_indices = np.empty(K_req, dtype=np.int32)

        for i in range(K_req):
            fid = int(shock_source_ids[i])
            idx = id_to_col.get(fid)
            if idx is None:
                raise KeyError(
                    f"Required factor_id {fid} not found in parametric universe."
                )
            col_indices[i] = idx

        # --- Pre-slice the mathematical parameters ---

        # Slice cholesky
        L_T_sliced = np.ascontiguousarray(
            self._cholesky_transpose[:, col_indices]
        )

        # Slice vols
        vols_sliced = np.ascontiguousarray(self._volatilities64[col_indices])

        # Slice GMM multipliers
        gmm_mults_sliced = None
        if self.distribution_kind == "regime_switching_gmm":
            assert self.crisis_vol_multipliers is not None

            # Type-checker safe: Explicitly isolate the ndarray for slicing
            if isinstance(self.crisis_vol_multipliers, np.ndarray):
                gmm_mults_sliced = np.ascontiguousarray(
                    self.crisis_vol_multipliers[col_indices]
                )
            else:
                # Both floats and ints fall safely here
                gmm_mults_sliced = float(self.crisis_vol_multipliers)

        rng = np.random.default_rng(self.seed)

        # Raw normals MUST remain K_total wide to ensure correct cross-asset correlation
        work = np.empty((batch_size, K_total), dtype=np.float64, order="C")

        # Correlated output is now directly allocated at K_req width
        correlated = np.empty((batch_size, K_req), dtype=np.float64, order="C")

        scenario_offset = 0

        while scenario_offset < self.total_scenarios:
            scenario_count = min(
                batch_size, self.total_scenarios - scenario_offset
            )

            work_view = work[:scenario_count, :]
            correlated_view = correlated[:scenario_count, :]

            # 1. Draw raw (B, K_total)
            rng.standard_normal(size=work_view.shape, out=work_view)

            # 2. Optimized Matmul: (B, K_total) @ (K_total, K_req) -> (B, K_req)
            np.matmul(work_view, L_T_sliced, out=correlated_view)

            # 3. Apply distribution scaling and base volatilities to the smaller matrix
            self._apply_distribution_scaling(
                rng=rng,
                shocks=correlated_view,
                gmm_sliced_multipliers=gmm_mults_sliced,
            )
            np.multiply(correlated_view, vols_sliced, out=correlated_view)

            # 4. Final copy to Fortran contiguous target
            shocks = np.empty(
                (scenario_count, K_req), dtype=self.shock_dtype, order="F"
            )
            np.copyto(shocks, correlated_view, casting="same_kind")

            yield ScenarioSetBatch(
                shocks=shocks,
                required_to_shock_uix=req_to_uix,
                shock_scale=shock_scale,
                required_shift_type_id=required_shift_type_id,
                scenario_offset=scenario_offset,
                scenario_count_total=self.total_scenarios,
            )

            scenario_offset += scenario_count

# src/riskquant/simlab/shocks/plan.py

from dataclasses import dataclass
import numpy as np
from typing import Dict


@dataclass(slots=True)
class ShockComputePlan:
    cols_sorted: np.ndarray
    shift_types_sorted: np.ndarray
    inverse_permutation: np.ndarray


def build_shock_compute_plan(
    *,
    shock_source_ids: np.ndarray,
    shock_shift_type_ids: np.ndarray,
    id_to_col: Dict[int, int],
) -> ShockComputePlan:

    cols = np.array(
        [id_to_col[int(sid)] for sid in shock_source_ids],
        dtype=np.int32,
    )

    shift_types = shock_shift_type_ids.astype(np.int8, copy=False)

    order = np.argsort(cols)

    cols_sorted = cols[order]
    shift_sorted = shift_types[order]

    inv = np.empty_like(order)
    inv[order] = np.arange(order.shape[0])

    return ShockComputePlan(
        cols_sorted=cols_sorted,
        shift_types_sorted=shift_sorted,
        inverse_permutation=inv,
    )
