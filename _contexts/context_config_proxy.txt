# src/riskquant/config/enums/discount_basis.py
"""
Discount basis taxonomy for IR discount curves.

A single currency carries multiple discount curves: one per OIS / IBOR
tenor / CSA collateral choice. The basis enum is the per-(currency,
basis) discriminator the ingest layer uses to resolve a trade's
discount-curve linkage when it specifies a non-default basis.

Sentinel:
  NOT_A_DISCOUNT_CURVE - explicitly marks a curve as non-discount. All
  curve_meta rows carry a basis; non-discount curves carry this value.
  The validator rejects any attempt to use a non-discount curve as a
  discount-curve linkage.

Standard convention: the firm-wide default discount basis is OIS. The
static-config validator enforces exactly one is_default_discount=True
curve per currency that has any discount curve, and that the default's
basis is OIS.
"""

from __future__ import annotations

from enum import IntEnum


class DiscountBasis(IntEnum):
    NOT_A_DISCOUNT_CURVE = 0

    OIS = 1

    IBOR_1M = 11
    IBOR_3M = 13
    IBOR_6M = 16

    CSA_USD = 30
    CSA_EUR = 31
    CSA_GBP = 32
    CSA_CHF = 33
    CSA_JPY = 34

    INTERNAL_FVA = 90


# Firm-wide standard. The default discount curve per currency must use this basis.
FIRM_DEFAULT_DISCOUNT_BASIS: DiscountBasis = DiscountBasis.OIS


VALID_DISCOUNT_BASES: frozenset[str] = frozenset(b.name for b in DiscountBasis)

DISCOUNT_BASES_EXCLUDING_SENTINEL: frozenset[str] = frozenset(
    b.name for b in DiscountBasis if b is not DiscountBasis.NOT_A_DISCOUNT_CURVE
)

"""
Roll method taxonomy for spot roll rules.

Replaces the prior two ad-hoc declarations: the string-keyed
`_VALID_SPOT_ROLL_METHODS` frozenset in validate.py and the
`_SpotRollMethod` constant class in rf_resolution.py.

A roll method specifies how a relative tenor token (e.g. [M+1], [Y+0],
[HY+0]) resolves to an absolute risk-factor name for a given as-of date.
The (XAxisKind, RollMethod) pair determines both the token grammar and
the absolute name format; allowed pairs are enumerated explicitly in
ALLOWED_X_AXIS_KIND_ROLL_METHOD_PAIRS so the validator and the
resolution dispatch cannot drift.

Absolute name formats by combination:
  (OFFSET_M,  CALENDAR_PERIOD)     -> "<curve>_YYYYMM"
  (OFFSET_M,  HALF_YEAR_BY_DATE)   -> "<curve>_YYYYMMDD"
  (OFFSET_Y,  CALENDAR_PERIOD)     -> "<curve>_YYYY"
  (OFFSET_Y,  ROLL_ON_MMDD)        -> "<curve>_YYYY"
  (OFFSET_HY, HALF_YEAR_BY_DATE)   -> "<curve>_YYYYMMDD"
  (OFFSET_HY, HALF_YEAR_BY_LABEL)  -> "<curve>_YYYY_FH" or "<curve>_YYYY_BH"
"""

from __future__ import annotations

from enum import IntEnum

from riskquant.config.enums.x_axis_kinds import XAxisKind


class RollMethod(IntEnum):
    CALENDAR_PERIOD = 0
    ROLL_ON_MMDD = 1
    HALF_YEAR_BY_DATE = 2
    HALF_YEAR_BY_LABEL = 3


VALID_ROLL_METHODS: frozenset[str] = frozenset(m.name for m in RollMethod)


ALLOWED_X_AXIS_KIND_ROLL_METHOD_PAIRS: frozenset[tuple[XAxisKind, RollMethod]] = frozenset(

        (XAxisKind.OFFSET_Y, RollMethod.ROLL_ON_MMDD),
        (XAxisKind.OFFSET_HY, RollMethod.HALF_YEAR_BY_DATE),
        (XAxisKind.OFFSET_HY, RollMethod.HALF_YEAR_BY_LABEL),

# src/riskquant/config/enums/x_axis_kinds.py
"""
X-axis kind taxonomy for curves and surfaces.

Replaces the legacy (TenorType, AliasType, AliasKind) triple. A single
XAxisKind tells the resolution layer exactly how a curve / surface point
addresses its position on the time axis:

  LABEL      - fixed, as-of-independent string labels (e.g. [ON], [1W], [1M]).
               Used by IR conventional curves.
  OFFSET_M   - monthly offsets from a roll-determined spot month (e.g. [M+0],
               [M-1], [M+12]). Used by monthly commodity contracts.
  OFFSET_Y   - annual offsets from a roll-determined spot year (e.g. [Y+0],
               [Y-2]). Used by annual commodity contracts.
  OFFSET_HY  - half-year offsets from a roll-determined spot half (e.g.
               [HY+0], [HY+3]). Used by half-year power contracts.

The X-axis kind defines token grammar only. How an offset resolves to an
absolute date label (calendar period, roll-on-MMDD, half-year boundary)
is independent and lives on the spot roll rule via RollMethod. Allowed
(XAxisKind, RollMethod) pairs are enumerated explicitly in
ALLOWED_X_AXIS_KIND_ROLL_METHOD_PAIRS.
"""

from __future__ import annotations

import re
from enum import IntEnum
from re import Pattern


class XAxisKind(IntEnum):
    LABEL = 0
    OFFSET_M = 1
    OFFSET_Y = 2
    OFFSET_HY = 3


# Token grammars by kind. LABEL matches the standard IR conventions plus the
# generic [\d+(D|W|M|Y)] form. OFFSET kinds match a strictly signed integer
# inside brackets prefixed by the unit (M, Y, HY).
X_AXIS_TOKEN_REGEX: dict[XAxisKind, Pattern[str]] = {
    XAxisKind.LABEL: re.compile(r"^\[(?:ON|SN|TN|\d+D|\d+W|\d+M|\d+Y)\]$"),
    XAxisKind.OFFSET_M: re.compile(r"^\[M[+-]\d+\]$"),
    XAxisKind.OFFSET_Y: re.compile(r"^\[Y[+-]\d+\]$"),
    XAxisKind.OFFSET_HY: re.compile(r"^\[HY[+-]\d+\]$"),
}

X_AXIS_TOKEN_EXAMPLE: dict[XAxisKind, str] = {
    XAxisKind.LABEL: "[3M]",
    XAxisKind.OFFSET_M: "[M+1]",
    XAxisKind.OFFSET_Y: "[Y-2]",
    XAxisKind.OFFSET_HY: "[HY+0]",
}

VALID_X_AXIS_KINDS: frozenset[str] = frozenset(k.name for k in XAxisKind)

# OFFSET kinds require a spot roll rule. LABEL does not (and must not).
X_AXIS_KINDS_REQUIRING_ROLL: frozenset[str] = frozenset(
    k.name for k in (XAxisKind.OFFSET_M, XAxisKind.OFFSET_Y, XAxisKind.OFFSET_HY)
)

# src/riskquant/config/proxy_config/audit.py
"""
Tabular audit of a compiled ProxyTopology.

Imports Pandas deliberately — auditing is an edge-only diagnostic, not a
runtime path. Never imported by valuation, simlab, or any kernel.

This module also hosts the consolidated name-resolved DataFrame view
(`to_df`) that used to live as a method on ProxyTopology. Keeping it
here ensures the runtime container itself stays Pandas-free.
"""

from __future__ import annotations

import numpy as np
import pandas as pd

from riskquant.config.proxy_config.proxy_topology import ProxyTopology
from riskquant.config.static_config.static_config import RiskStaticConfig

_SECTION = "=" * 72
_SUB = "-" * 72


def _header(title: str) -> None:
    print(_SECTION)
    print(title)
    print(_SECTION)


def _sub(title: str) -> None:
    print(f"\n{_SUB}")
    print(title)
    print(_SUB)


# ============================================================
# Name-resolved DataFrame projection
# ============================================================


def to_df(topology: ProxyTopology, static_config: RiskStaticConfig) -> pd.DataFrame:
    """
    Consolidate container and rf proxies into a single name+id DataFrame.

    Mirrors the legacy `ProxyTopology.to_df()` output. Lives here rather
    than on the runtime container so the runtime stays Pandas-free.

    Columns
    -------
        proxying_level, required_name, required_id,
        proxy_name, proxy_id, vol_scaling_factor
    """
    name_to_cid = static_config.container_name_to_id
    name_to_rfid = static_config.rel_name_to_rf_id

    records: list[dict[str, object]] = []

    for pm in topology.container_proxies.values():
        records.append(
            {
                "proxying_level": "CONTAINER",
                "required_name": pm.required_name,
                "required_id": int(name_to_cid[pm.required_name]),
                "proxy_name": pm.proxy_name,
                "proxy_id": int(name_to_cid[pm.proxy_name]),
                "vol_scaling_factor": float(pm.vol_scaling_factor),
            }
        )

    for pm in topology.rf_proxies.values():
        records.append(
            {
                "proxying_level": "RISK_FACTOR",
                "required_name": pm.required_name,
                "required_id": int(name_to_rfid[pm.required_name]),
                "proxy_name": pm.proxy_name,
                "proxy_id": int(name_to_rfid[pm.proxy_name]),
                "vol_scaling_factor": float(pm.vol_scaling_factor),
            }
        )

    return pd.DataFrame(
        records,
        columns=[
            "proxying_level",
            "required_name",
            "required_id",
            "proxy_name",
            "proxy_id",
            "vol_scaling_factor",
        ],
    )


# ============================================================
# Section renderers
# ============================================================


def _audit_summary(topology: ProxyTopology) -> None:
    _header("ProxyTopology Audit")
    print(f"  Container proxies   : {topology.num_container_proxies}")
    print(f"  RF proxies          : {topology.num_rf_proxies}")
    print(f"  Total               : {topology.num_total_proxies}")
    print(f"  c_proxy array size  : {topology.container_id_to_proxy_id.size}")
    print(f"  rf_proxy array size : {topology.rf_id_to_proxy_rf_id.size}")
    arr = topology.container_id_to_proxy_id
    n_set_c = int((arr >= 0).sum())
    arr = topology.rf_id_to_proxy_rf_id
    n_set_rf = int((arr >= 0).sum())
    print(f"  c_proxy slots set   : {n_set_c}")
    print(f"  rf_proxy slots set  : {n_set_rf}")


def _audit_proxy_table(
    topology: ProxyTopology, static_config: RiskStaticConfig, n: int
) -> None:
    _sub("1. Name-resolved proxy table")
    df = to_df(topology, static_config)
    if df.empty:
        print("(no proxies registered)")
        return
    print(df.head(n).to_string(index=False))


def _audit_vol_scale_distribution(topology: ProxyTopology, n: int) -> None:
    _sub("2. vol_scaling_factor distribution")
    scales = [pm.vol_scaling_factor for pm in topology.container_proxies.values()] + [
        pm.vol_scaling_factor for pm in topology.rf_proxies.values()
    ]
    if not scales:
        print("(no proxies)")
        return
    arr = np.asarray(scales, dtype=np.float64)
    print(f"  count = {arr.size}")
    print(f"  min   = {arr.min():.4f}")
    print(f"  max   = {arr.max():.4f}")
    print(f"  mean  = {arr.mean():.4f}")
    print(f"  != 1  = {int((arr != 1.0).sum())} entries differ from identity")


def _audit_proxied_containers_by_kind(
    topology: ProxyTopology, static_config: RiskStaticConfig, n: int
) -> None:
    _sub("3. Container proxies — by container kind (curve / surface)")
    if not topology.container_proxies:
        print("(none)")
        return

"""
Single entry-point for building a ProxyTopology from source data.

Pipeline:
    xlsx  ->  load  ->  validate_df  ->  partition  ->  validate_maps  ->  build  ->  ProxyTopology

The compile step is the only place the raw DataFrame is consumed. After
this layer no caller reads strings, levels, or vol-scaling factors as
named columns: the runtime container exposes only dense int arrays.
"""

from __future__ import annotations

from pathlib import Path

import numpy as np
import pandas as pd

from riskquant.config.proxy_config.loader import load_proxy_config_xlsx_into_df
from riskquant.config.proxy_config.proxy_topology import ProxyTopology
from riskquant.config.proxy_config.schema_objects import ProxyMap
from riskquant.config.proxy_config.validate import (
    validate_proxy_config_df,
    validate_proxy_maps,
)
from riskquant.config.static_config.static_config import RiskStaticConfig

# ============================================================
# Internal helpers
# ============================================================


def _build_proxy_map_dict(level_df: pd.DataFrame) -> dict[str, ProxyMap]:
    """
    Vectorized: extract typed columns from the level-partitioned df and
    materialize one ProxyMap per row. No iterrows.

    `level_df` has been validated upstream — required_name uniqueness
    per level is guaranteed, so the dict keyed by required_name has no
    collisions.
    """
    if level_df.empty:
        return {}

    required_names = level_df["required_name"].astype(str).str.strip().to_numpy()
    proxy_names = level_df["proxy_name"].astype(str).str.strip().to_numpy()
    vol_scales = level_df["vol_scaling_factor"].astype(np.float64).to_numpy()

    return {
        str(req): ProxyMap(
            required_name=str(req),
            proxy_name=str(prx),
            vol_scaling_factor=float(scl),
        )
        for req, prx, scl in zip(required_names, proxy_names, vol_scales, strict=True)
    }


# ============================================================
# Public API
# ============================================================


def compile_proxy_topology_from_df(
    df: pd.DataFrame,
    static_config: RiskStaticConfig,
) -> ProxyTopology:
    """
    Validate and compile an active-filtered proxy DataFrame into a
    ProxyTopology.

    Parameters
    ----------
    df
        Active-filtered proxy DataFrame from the loader. Must have
        columns: proxying_level, required_name, proxy_name,
        vol_scaling_factor.
    static_config
        Compiled static config used to resolve names to ids and to
        validate cross-level conflicts via the static config's
        parent-kind / parent-id arrays.

    Returns
    -------
    ProxyTopology
        Fully compiled, Pandas-free runtime topology with dense int32
        lookup arrays already materialized.

    Raises
    ------
    ValueError
        On any validation failure — Stage 1 (DataFrame schema/type/range),
        Stage 2 (FK / chain / cross-level conflict).
    """
    # ------------------------------------------------------------------ #
    # Stage 1: DataFrame-tier validation                                 #
    # ------------------------------------------------------------------ #
    validate_proxy_config_df(df)

    if df.empty:
        # Build a no-op topology: all lookups pass through unchanged.
        return ProxyTopology.build(
            container_proxies={},
            rf_proxies={},
            static_config=static_config,
        )

    # ------------------------------------------------------------------ #
    # Partition by level (vectorized; no row loop)                       #
    # ------------------------------------------------------------------ #
    level_norm = df["proxying_level"].astype(str).str.upper()
    container_df = df.loc[level_norm == "CONTAINER"]
    rf_df = df.loc[level_norm == "RISK_FACTOR"]

    container_proxies = _build_proxy_map_dict(container_df)
    rf_proxies = _build_proxy_map_dict(rf_df)

    # ------------------------------------------------------------------ #
    # Stage 2: cross-table validation                                    #
    # ------------------------------------------------------------------ #
    validate_proxy_maps(
        container_proxies=container_proxies,
        rf_proxies=rf_proxies,
        static_config=static_config,
    )

    # ------------------------------------------------------------------ #
    # Assemble: build dense arrays inside ProxyTopology.build            #
    # ------------------------------------------------------------------ #
    return ProxyTopology.build(
        container_proxies=container_proxies,
        rf_proxies=rf_proxies,
        static_config=static_config,
    )

# src/riskquant/config/proxy_config/loader.py
"""
Loads proxy configuration from an Excel workbook into a clean DataFrame.

The workbook MUST carry an `active` column. Rows with active=True are
returned; the `active` column is dropped from the result. If `active`
is absent the loader raises — silently treating an absent column as
"all rows active" is a real production foot-gun (drafts and historical
entries get compiled).

This loader only handles the human-facing xlsx convention. The in-memory
canonical fixture is post-filtered (no active column) and goes through
compile_proxy_topology_from_df directly.
"""

from __future__ import annotations

from pathlib import Path

import numpy as np
import pandas as pd

_REQUIRED_COLS: frozenset[str] = frozenset(
    {
        "active",
        "proxying_level",
        "required_name",
        "proxy_name",
        "vol_scaling_factor",
    }
)

# Columns that survive the active-filter and are returned to compile.py.
_POST_FILTER_COLS: tuple[str, ...] = (
    "proxying_level",
    "required_name",
    "proxy_name",
    "vol_scaling_factor",
)


def load_proxy_config_xlsx_into_df(
    *,
    path: str | Path,
    sheet_name: str | None = None,
) -> pd.DataFrame:
    """
    Load the proxy-config sheet and return active rows only.

    Parameters
    ----------
    path
        Path to the Excel workbook.
    sheet_name
        Sheet to read. Defaults to the first sheet if None.

    Returns
    -------
    pd.DataFrame
        Columns: proxying_level, required_name, proxy_name,
        vol_scaling_factor. `active` is consumed and dropped.

    Raises
    ------
    ValueError
        If a required column is missing, or if the sheet name does not
        exist in the workbook.
    """
    path_str = str(path)

    with pd.ExcelFile(path_str, engine="openpyxl") as xls:
        available = list(xls.sheet_names)
        if sheet_name is not None and sheet_name not in available:
            raise ValueError(
                f"Missing required sheet {sheet_name!r} in {path_str!r}. "
                f"Available sheets: {available}"
            )
        _sheet: str = str(sheet_name) if sheet_name is not None else str(available[0])
        df: pd.DataFrame = pd.read_excel(xls, sheet_name=_sheet)

    # --- required columns: explicit (no silent active fallback) ---
    missing = _REQUIRED_COLS - set(df.columns)
    if missing:
        raise ValueError(
            f"Missing required column(s) in sheet {_sheet!r} of {path_str!r}: "
            f"{sorted(missing)}.\n"
            f"Required: {sorted(_REQUIRED_COLS)}"
        )

    # --- active filter ---
    mask = df["active"].fillna(False).astype(bool)
    df = df.loc[mask, list(_POST_FILTER_COLS)].reset_index(drop=True)

    # --- type coercions ---
    # String columns: strip whitespace, leave nulls as nulls (validator
    # rejects both null and blank; loader does NOT fillna to "").
    for col in ("proxying_level", "required_name", "proxy_name"):
        df[col] = df[col].astype("string").str.strip()

    # vol_scaling_factor: float64 with missing -> 1.0 (identity).
    df["vol_scaling_factor"] = df["vol_scaling_factor"].fillna(1.0).astype(np.float64)

    return df

        print_calling_file,
        print_success_message,
    )

    print_calling_file()

    DEV_DATA_DIR: str = "C:/Users/mharvey/Documents/code/riskquant/data/dev"
    DEV_DATA_FILE_NAME: str = "dev_risk_base_proxy_config_v01.xlsx"

    print(f"\nLoaded proxy config DataFrame (shape: {df.shape}):")
    print(df.to_string(index=False))

# src/riskquant/config/proxy_config/proxy_topology.py
"""
Runtime proxy-topology container.

Pandas-free. Hot-path lookups are dense int32 / float64 NumPy arrays
indexed by required-id; sentinel -1 means "no proxy registered for this
id, pass through unchanged."

Built once at compile time. Immutable thereafter; the `with_overrides`
factory returns a brand-new instance.

Resolution semantics (single-hop, enforced by the validator)
------------------------------------------------------------
For any required_id `r`:
    proxy_id    = container_id_to_proxy_id[r]            (or rf_id_to_proxy_rf_id[r])
    proxy_scale = container_id_to_proxy_vol_scale[r]     (or rf_id_to_proxy_vol_scale[r])

If proxy_id < 0, no proxy is registered — pass through.
If proxy_id >= 0, the proxy must NOT itself be proxied (the validator
forbids chains). The kernel performs a single gather, no iteration.

The validator also forbids container-level proxies whose children are
rf-level proxied (cross-level conflict) — they are mutually exclusive
policies and coexisting them silently mis-applies the user's intent.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import TYPE_CHECKING

import numpy as np
import numpy.typing as npt

from riskquant.config.proxy_config.schema_objects import ProxyMap

if TYPE_CHECKING:
    from riskquant.config.static_config.static_config import RiskStaticConfig


# Sentinel for "no proxy registered for this id" in the dense arrays.
# Scoped to ProxyTopology — not in riskquant.constants because no
# consumer outside this module compares against it directly (the kernel
# uses `proxy_id >= 0` branchless masking instead).
_NO_PROXY_ID: int = -1


# Default vol scale for ids with no proxy. 1.0 is the identity
# multiplier; using NaN here would propagate into downstream math.
_DEFAULT_VOL_SCALE: float = 1.0


@dataclass(frozen=True, slots=True)
class ProxyTopology:
    """
    Compiled proxy topology.

    Init fields are supplied by `ProxyTopology.build` (or by
    `with_overrides`, which re-routes through `build`); they are not
    intended to be constructed directly by user code.

    Hot-path readers consume only the four dense arrays:
        container_id_to_proxy_id
        container_id_to_proxy_vol_scale
        rf_id_to_proxy_rf_id
        rf_id_to_proxy_vol_scale

    The two dicts (container_proxies, rf_proxies) are retained for
    audit and override purposes and are never touched on the hot path.
    """

    # ------------------------------------------------------------------ #
    # Human-readable layer (audit + override path)                       #
    # ------------------------------------------------------------------ #

    container_proxies: dict[str, ProxyMap] = field(repr=False)
    rf_proxies: dict[str, ProxyMap] = field(repr=False)

    # ------------------------------------------------------------------ #
    # Hot-path dense arrays                                              #
    # ------------------------------------------------------------------ #
    # Length = static_config.max_container_id + 1.
    # Indexed by required_container_id.
    container_id_to_proxy_id: npt.NDArray[np.int32] = field(repr=False)
    container_id_to_proxy_vol_scale: npt.NDArray[np.float64] = field(repr=False)

    # Length = static_config.max_rf_id + 1. Indexed by required_rf_id.
    rf_id_to_proxy_rf_id: npt.NDArray[np.int32] = field(repr=False)
    rf_id_to_proxy_vol_scale: npt.NDArray[np.float64] = field(repr=False)

    # ------------------------------------------------------------------ #
    # Factories                                                          #
    # ------------------------------------------------------------------ #

    @classmethod
    def build(
        cls,
        *,
        container_proxies: dict[str, ProxyMap],
        rf_proxies: dict[str, ProxyMap],
        static_config: RiskStaticConfig,
    ) -> ProxyTopology:
        """
        Build a ProxyTopology by resolving names against the static config
        and emitting the dense lookup arrays.

        Parameters
        ----------
        container_proxies, rf_proxies
            Already validated by `validate_proxy_maps`. Required name and
            proxy name are guaranteed to resolve in static_config; no
            chains, no cycles, no cross-level conflicts.
        static_config
            Source of truth for name -> id resolution and id-space sizes.
        """
        n_container = int(static_config.max_container_id) + 1
        n_rf = int(static_config.max_rf_id) + 1

        c_proxy = np.full(n_container, _NO_PROXY_ID, dtype=np.int32)
        c_scale = np.full(n_container, _DEFAULT_VOL_SCALE, dtype=np.float64)
        r_proxy = np.full(n_rf, _NO_PROXY_ID, dtype=np.int32)
        r_scale = np.full(n_rf, _DEFAULT_VOL_SCALE, dtype=np.float64)

        name_to_cid = static_config.container_name_to_id
        for pm in container_proxies.values():
            req_id = int(name_to_cid[pm.required_name])
            proxy_id = int(name_to_cid[pm.proxy_name])
            c_proxy[req_id] = np.int32(proxy_id)
            c_scale[req_id] = float(pm.vol_scaling_factor)

        name_to_rfid = static_config.rel_name_to_rf_id
        for pm in rf_proxies.values():
            req_id = int(name_to_rfid[pm.required_name])
            proxy_id = int(name_to_rfid[pm.proxy_name])
            r_proxy[req_id] = np.int32(proxy_id)
            r_scale[req_id] = float(pm.vol_scaling_factor)

        return cls(
            container_proxies=container_proxies,
            rf_proxies=rf_proxies,
            container_id_to_proxy_id=c_proxy,
            container_id_to_proxy_vol_scale=c_scale,
            rf_id_to_proxy_rf_id=r_proxy,
            rf_id_to_proxy_vol_scale=r_scale,
        )

    def with_overrides(
        self,
        *,
        static_config: RiskStaticConfig,
        container_overrides: dict[str, ProxyMap] | None = None,
        rf_overrides: dict[str, ProxyMap] | None = None,
    ) -> ProxyTopology:
        """
        Return a new topology with overrides layered on top of self.

        Overrides are full-replace, keyed by required_name: any
        required_name appearing in the overrides supersedes the base
        topology's mapping for that name (or adds a new one).

        Re-runs the full proxy-map validation against the supplied
        static_config so override-induced chains, cycles, or cross-level
        conflicts surface immediately.
        """
        # Local import to avoid a cycle: validate imports schema_objects,
        # which proxy_topology already imports; validate also imports the
        # module that ends up being this one transitively via compile.
        from riskquant.config.proxy_config.validate import validate_proxy_maps

        merged_containers = dict(self.container_proxies)
        if container_overrides:
            merged_containers.update(container_overrides)

        merged_rfs = dict(self.rf_proxies)
        if rf_overrides:
            merged_rfs.update(rf_overrides)

        validate_proxy_maps(
            container_proxies=merged_containers,
            rf_proxies=merged_rfs,
            static_config=static_config,
        )

        return ProxyTopology.build(
            container_proxies=merged_containers,
            rf_proxies=merged_rfs,
            static_config=static_config,
        )

    # ------------------------------------------------------------------ #
    # Single-id helpers (debug / unit-test only — not for hot path)      #
    # ------------------------------------------------------------------ #

    def resolve_container(self, required_container_id: int) -> tuple[int, float]:
        """
        Return (proxy_container_id, vol_scale) for a single required id.

        If no proxy is registered, returns (required_container_id, 1.0)
        — the identity mapping the kernel applies via branchless masking.
        Hot paths must use the dense arrays directly; this is a
        convenience for tests and notebook sessions.
        """
        i = int(required_container_id)
        proxy_id = int(self.container_id_to_proxy_id[i])
        if proxy_id < 0:
            return (i, _DEFAULT_VOL_SCALE)
        return (proxy_id, float(self.container_id_to_proxy_vol_scale[i]))

    def resolve_rf(self, required_rf_id: int) -> tuple[int, float]:
        """Same as resolve_container, but for the rf-id space."""
        i = int(required_rf_id)
        proxy_id = int(self.rf_id_to_proxy_rf_id[i])
        if proxy_id < 0:
            return (i, _DEFAULT_VOL_SCALE)
        return (proxy_id, float(self.rf_id_to_proxy_vol_scale[i]))

    # ------------------------------------------------------------------ #
    # Scalar accessors                                                   #
    # ------------------------------------------------------------------ #

    @property
    def num_container_proxies(self) -> int:
        return len(self.container_proxies)

    @property
    def num_rf_proxies(self) -> int:
        return len(self.rf_proxies)

    @property
    def num_total_proxies(self) -> int:
        return self.num_container_proxies + self.num_rf_proxies

    def summary(self) -> None:
        """Print a console-friendly overview of the compiled topology."""
        lines = [
            "================================",
            "===   ProxyTopology Summary  ===",
            "================================",
            f"  Container proxies   : {self.num_container_proxies}",
            f"  RF proxies          : {self.num_rf_proxies}",
            f"  Total               : {self.num_total_proxies}",
            f"  c_proxy array size  : {self.container_id_to_proxy_id.size}",
            f"  rf_proxy array size : {self.rf_id_to_proxy_rf_id.size}",
            "================================",
        ]
        print("\n".join(lines))

# src/riskquant/config/proxy_config/schema_objects.py
"""
Immutable value object for one proxy mapping.

The proxying-level discriminator (container vs risk-factor) is carried by
the partition the ProxyMap lives in (ProxyTopology.container_proxies or
.rf_proxies), not by a field on the dataclass. Strings die at the edge;
the runtime never reads a level string.
"""

from __future__ import annotations

from dataclasses import dataclass


@dataclass(frozen=True, slots=True, kw_only=True)
class ProxyMap:
    """
    A single proxy mapping entry.

    The required name's historical data is unavailable or insufficient
    for risk estimation; the proxy name's data is used in its place,
    optionally scaled by `vol_scaling_factor` to account for known
    relative-vol differences between the two series.

    Attributes
    ----------
    required_name
        The name whose data is being substituted (the "missing" one).
        A container name when used in container_proxies, a relative
        risk-factor name when used in rf_proxies.
    proxy_name
        The name supplying the substitute data. Must be a peer of
        required_name (same level — container or rf — same namespace).
    vol_scaling_factor
        Strictly positive multiplier applied to the proxy's vol/shocks
        before substitution. Loader defaults missing cells to 1.0

# src/riskquant/config/proxy_config/validate.py
"""
Two-stage validation for proxy configuration.

Stage 1: `validate_proxy_config_df`
    Runs against the active-filtered DataFrame. Vectorized — no row
    iteration. Catches schema, type, null, blank, and domain-range
    issues before any name resolution is attempted.

Stage 2: `validate_proxy_maps`
    Runs against the typed ProxyMap dicts (after compile partitions them
    by level) and the compiled RiskStaticConfig. Catches:
      - foreign-key drift against the static config,
      - self-references,
      - chained proxies (A->B->C is forbidden — multi-hop is not allowed),
      - cross-level conflicts (a container is proxied AND one of its
        child risk factors is also proxied).

The override path on ProxyTopology calls Stage 2 directly without
roundtripping through a DataFrame.
"""

from __future__ import annotations

from typing import TYPE_CHECKING

import numpy as np
import pandas as pd

from riskquant.config.enums.container_topology import ContainerTopologyType
from riskquant.config.proxy_config.schema_objects import ProxyMap

if TYPE_CHECKING:
    from riskquant.config.static_config.static_config import RiskStaticConfig


# ============================================================
# Constants
# ============================================================


VALID_PROXYING_LEVELS: frozenset[str] = frozenset({"CONTAINER", "RISK_FACTOR"})

# vol_scaling_factor sanity bounds. The lower bound is strict (a zero
# scale silently zeros the proxy's vol, which is a class-A modelling bug);
# the upper bound is conservative but catches typos like missing decimal
# point ("15" intended as "1.5").
_VOL_SCALE_MIN_EXCLUSIVE: float = 0.0
_VOL_SCALE_MAX_INCLUSIVE: float = 10.0


REQUIRED_COLUMNS: frozenset[str] = frozenset(
    {"proxying_level", "required_name", "proxy_name", "vol_scaling_factor"}
)


# ============================================================
# Generic helpers
# ============================================================


def _check_no_blank_strings(df: pd.DataFrame, col: str) -> None:
    if col not in df.columns:
        return
    blanks = df[col].astype(str).str.strip() == ""
    if blanks.any():
        idx = df.index[blanks].tolist()[:10]
        raise ValueError(
            f"proxy_config.{col}: blank string at rows {idx}. "
            "Loader fills null cells with the empty string; the validator "
            "treats both as invalid."
        )


# ============================================================
# Stage 1 — DataFrame-tier validation
# ============================================================


def validate_proxy_config_df(df: pd.DataFrame) -> None:
    """
    Validate an active-filtered proxy DataFrame.

    Tiers:
      1. Required columns present.
      2. No null values in required columns.
      3. No blank strings in string columns.
      4. proxying_level is in VALID_PROXYING_LEVELS.
      5. vol_scaling_factor in (0, 10].
      6. (proxying_level, required_name) is unique — a required name
         can only be proxied once per level.
      7. required_name != proxy_name (self-reference).

    Cross-table checks (FK against static_config, chains, cycles,
    cross-level conflicts) are deferred to `validate_proxy_maps`,
    which runs after the DataFrame has been partitioned by level.

    Raises
    ------
    ValueError
        On the first violation encountered, with a descriptive message.
    """
    if df.empty:
        return  # An empty proxy map is valid: no proxies applied.

    # --- 1. required columns ---
    missing = REQUIRED_COLUMNS - set(df.columns)
    if missing:
        raise ValueError(
            f"proxy_config missing required column(s): {sorted(missing)}.\n"
            f"Found: {list(df.columns)}"
        )

    # --- 2. no nulls in required columns ---
    null_mask = df[list(REQUIRED_COLUMNS)].isnull().any()
    if null_mask.any():
        null_cols = null_mask[null_mask].index.tolist()
        raise ValueError(
            f"proxy_config has null values in required column(s): {null_cols}."
        )

    # --- 3. no blank strings (loader fills nulls with "" which would slip past .isnull) ---
    for col in ("proxying_level", "required_name", "proxy_name"):
        _check_no_blank_strings(df, col)

    # --- 4. proxying_level domain ---
    # Coerce to uppercase before set membership (loader leaves case as-is).
    level_norm = df["proxying_level"].astype(str).str.upper()
    invalid = ~level_norm.isin(VALID_PROXYING_LEVELS)
    if invalid.any():
        bad = sorted(level_norm[invalid].unique().tolist())
        raise ValueError(
            f"proxy_config has invalid proxying_level value(s): {bad}.\n"
            f"Must be one of: {sorted(VALID_PROXYING_LEVELS)} (case-insensitive)."
        )

    # --- 5. vol_scaling_factor range ---
    scale = df["vol_scaling_factor"].astype(np.float64)
    bad_scale = (
        ~np.isfinite(scale.to_numpy())
        | (scale <= _VOL_SCALE_MIN_EXCLUSIVE)
        | (scale > _VOL_SCALE_MAX_INCLUSIVE)
    )
    if bad_scale.any():
        offending = df.loc[
            bad_scale, ["required_name", "proxy_name", "vol_scaling_factor"]
        ]
        raise ValueError(
            f"proxy_config has vol_scaling_factor outside "
            f"({_VOL_SCALE_MIN_EXCLUSIVE}, {_VOL_SCALE_MAX_INCLUSIVE}]:\n"
            f"{offending.to_string(index=False)}"
        )

    # --- 6. (proxying_level, required_name) composite uniqueness ---
    work = df.assign(
        _level=level_norm,
        _required=df["required_name"].astype(str).str.strip(),
    )
    dupes_mask = work.duplicated(subset=["_level", "_required"], keep=False)
    if dupes_mask.any():
        offending = df.loc[
            dupes_mask, ["proxying_level", "required_name", "proxy_name"]
        ]
        raise ValueError(
            "proxy_config has duplicate (proxying_level, required_name) rows. "
            "A required_name can be proxied at most once per level.\n"
            f"Conflicting rows:\n{offending.to_string(index=False)}"
        )

    # --- 7. self-reference ---
    self_ref = (
        df["required_name"].astype(str).str.strip()
        == df["proxy_name"].astype(str).str.strip()
    )
    if self_ref.any():
        offending = df.loc[self_ref, ["proxying_level", "required_name"]]
        raise ValueError(
            "proxy_config has self-referencing entries "
            "(required_name == proxy_name):\n"
            f"{offending.to_string(index=False)}"
        )


# ============================================================
# Stage 2 — ProxyMap-tier validation (FK + chain + cross-level)
# ============================================================


def _check_chain(proxies: dict[str, ProxyMap], level_label: str) -> None:
    """
    Forbid multi-hop proxies within a single level.

    A chain is any (A -> B) where B is itself a required_name in another
    (B -> C) entry. Multi-hop is banned: every proxy must point at a
    terminal node. Cycles are a special case of chains where some B == A,
    so this single check covers cycles too.
    """
    required_set: set[str] = set(proxies.keys())
    proxy_targets: list[tuple[str, str]] = [
        (pm.required_name, pm.proxy_name) for pm in proxies.values()
    ]

    chained: list[tuple[str, str]] = [
        (req, prx) for req, prx in proxy_targets if prx in required_set
    ]
    if chained:
        raise ValueError(
            f"Chained {level_label} proxies are not allowed (multi-hop is "
            f"banned — every proxy must point at a terminal name).\n"
            f"The proxy targets below are themselves required_names in other "
            f"rows:\n"
            + "\n".join(
                f"  {req!r} -> {prx!r}  (but {prx!r} is also proxied)"
                for req, prx in chained
            )
            + f"\nResolve by pointing both {level_label} mappings at a "
            "common terminal name."
        )


def _check_fk(
    proxies: dict[str, ProxyMap],
    name_to_id: dict[str, int] | dict[str, np.int32],
    namespace_label: str,
) -> None:
    """
    Every required_name and proxy_name must resolve in name_to_id.

    Bulk set-difference: one error message lists every unknown name,
    not one PrettyKeyError per row.
    """
    referenced: set[str] = set()
    for pm in proxies.values():
        referenced.add(pm.required_name)
        referenced.add(pm.proxy_name)

    unknown = referenced - set(name_to_id)
    if unknown:
        raise ValueError(
            f"proxy_config references {len(unknown)} unknown {namespace_label} "
            f"name(s): {sorted(unknown)[:25]}.\n"
            f"Every required_name and proxy_name must exist in the static "
            f"config's {namespace_label} namespace."
        )


def _check_cross_level_conflict(
    container_proxies: dict[str, ProxyMap],
    rf_proxies: dict[str, ProxyMap],
    static_config: RiskStaticConfig,
) -> None:
    """
    Forbid a container being proxied while any of its child risk factors
    is ALSO proxied.

    Rationale: container-level "use the whole proxy container's curve"
    and rf-level "use this specific other point" are competing policies;
    coexisting them silently mis-applies the user's intent.
    """
    if not container_proxies or not rf_proxies:
        return

    # Resolve each proxied container to its container_id.
    name_to_cid = static_config.container_name_to_id
    container_id_to_topology_type = (
        static_config.container_id_to_topology_type
    )  # 0=node, 1=curve, 2=surface

    proxied_curve_ids: set[int] = set()
    proxied_surface_ids: set[int] = set()
    for pm in container_proxies.values():
        cid = int(name_to_cid[pm.required_name])
        topology_type = int(container_id_to_topology_type[cid])
        if topology_type == ContainerTopologyType.CURVE.value:
            proxied_curve_ids.add(cid)
        elif topology_type == ContainerTopologyType.SURFACE.value:
            proxied_surface_ids.add(cid)

    if not (proxied_curve_ids or proxied_surface_ids):
        # FK passed but no proxied container resolved to curve or surface.
        # This should be unreachable; if we get here, container_id_to_topology_type
        # has a bug or container_proxies contains a node name (forbidden).
        raise ValueError(
            "internal: cross-level conflict check found proxied containers "
            "with topology_type not in {curve, surface}."
        )

    # Walk every rf-level proxy. The required_name is a relative
    # risk-factor name; its parent_id is on the dense rf_id_to_parent_id
    # array. If parent_id is in any proxied container set, conflict.
    name_to_rfid = static_config.rel_name_to_rf_id
    parent_topology_type_arr = static_config.rf_id_to_parent_topology_type
    parent_id_arr = static_config.rf_id_to_parent_id

    conflicts: list[tuple[str, str]] = []
    for pm in rf_proxies.values():
        rfid = int(name_to_rfid[pm.required_name])
        topology_type = int(parent_topology_type_arr[rfid])
        if topology_type <= ContainerTopologyType.NODE.value:
            continue  # Topology type is NODE (0) or unoccupied (-1) - skip.
        pid = int(parent_id_arr[rfid])
        if (
            topology_type == ContainerTopologyType.CURVE.value
            and pid in proxied_curve_ids
        ):
            conflicts.append((pm.required_name, _curve_name(pid, static_config)))
        elif (
            topology_type == ContainerTopologyType.SURFACE.value
            and pid in proxied_surface_ids
        ):
            conflicts.append((pm.required_name, _surface_name(pid, static_config)))

    if conflicts:
        lines = [
            f"  {rf_name!r} (parent={parent_name!r})"
            for rf_name, parent_name in conflicts
        ]
        raise ValueError(
            "proxy_config has cross-level conflicts: the rf-level proxies "
            "below belong to containers that are ALSO container-level "
            "proxied.\n"
            "Container-level proxy and rf-level proxy are competing "
            "policies; choose ONE per affected container.\n" + "\n".join(lines)
        )


def _curve_name(curve_id: int, sc: RiskStaticConfig) -> str:
    cm = sc.curve_meta_dict.get(curve_id)
    return cm.curve_name if cm is not None else f"<curve_id={curve_id}>"


def _surface_name(surface_id: int, sc: RiskStaticConfig) -> str:
    sm = sc.surface_meta_dict.get(surface_id)
    return sm.surface_name if sm is not None else f"<surface_id={surface_id}>"


def validate_proxy_maps(
    *,
    container_proxies: dict[str, ProxyMap],
    rf_proxies: dict[str, ProxyMap],
    static_config: RiskStaticConfig,
) -> None:
    """
    Validate partitioned proxy maps against the static config.

    Runs after compile.py has split the DataFrame by level (or after
    overrides have been merged in `ProxyTopology.with_overrides`).

    Checks
    ------
      A. Foreign-key: every name in container_proxies resolves in
         `static_config.container_name_to_id`; every name in rf_proxies
         resolves in `static_config.rel_name_to_rf_id`.
      B. Chain ban: no proxy target is itself a required name (per level).
         Implicitly catches cycles, since a cycle requires the target to
         be a required name.
      C. Cross-level: no container is both container-proxied and has any
         of its child risk factors rf-proxied.

    Raises
    ------
    ValueError
        On the first violation, with a descriptive message naming every
        offending pair / name found.
    """
    _check_fk(container_proxies, static_config.container_name_to_id, "container")
    _check_fk(rf_proxies, static_config.rel_name_to_rf_id, "risk-factor")
    _check_chain(container_proxies, "container")
    _check_chain(rf_proxies, "risk-factor")
    _check_cross_level_conflict(container_proxies, rf_proxies, static_config)

    LINEAR = 2

    "LINEAR": ExtrapMethod.LINEAR,
    "FLAT": ExtrapMethod.FLAT,

def validate_extrapolation_methods_dict(
    extrapolation_methods_dict: dict[str, int],
) -> None:
    expected = {
        name: int(member) for name, member in ExtrapMethod.__members__.items()
    }
    got = {str(k): int(v) for k, v in extrapolation_methods_dict.items()}

    if got == expected:
        return

    missing = {k: expected[k] for k in expected.keys() - got.keys()}
    extra = {k: got[k] for k in got.keys() - expected.keys()}
    different = {
        k: {"expected": expected[k], "got": got[k]}
        for k in (expected.keys() & got.keys())
        if expected[k] != got[k]
    }

    raise ValueError(
        "extrapolation_methods sheet does not match riskquant ExtrapMethod enum.\n"
        f"missing={missing}\n"
        f"extra={extra}\n"
        f"different={different}\n"
        "Fix the Excel extrapolation_methods table or update riskquant.market_data.curves.extrap_methods.ExtrapMethod."

from __future__ import annotations

import math

import numba as nb
import numpy as np

# ==============================================================================
# --- LINEAR INTERPOLATION WITH FLAT EXTRAPOLATION ---
# ==============================================================================


def interp_linear_flat_extrap_3d(
    tenors: np.ndarray,
    values: np.ndarray,
    targets: np.ndarray,
) -> np.ndarray:
    """
    Vectorized 3D Linear Interpolation with Flat Extrapolation.
    Shapes: tenors (c_curves, m_pillars), values (c_curves, m_pillars, s_scenarios), targets (d_targets,)
    Returns: (c_curves, d_targets, s_scenarios)
    """
    c_curves, m_pillars = tenors.shape
    idx1 = np.clip(
        np.sum(tenors[:, None, :] <= targets[None, :, None], axis=2),
        1,
        m_pillars - 1,
    )
    idx0 = idx1 - 1

    c_idx = np.arange(c_curves)[:, None]
    t0, t1 = tenors[c_idx, idx0], tenors[c_idx, idx1]
    y0, y1 = values[c_idx, idx0, :], values[c_idx, idx1, :]

    dt = np.where((t1 - t0) == 0, 1.0, t1 - t0)
    w = (targets[None, :] - t0) / dt

    # FLAT EXTRAPOLATION: Clamp weights to [0.0, 1.0]
    w = np.clip(w, 0.0, 1.0)

    return y0 + w[:, :, None] * (y1 - y0)


@nb.njit(
    parallel=True,
    fastmath={"contract", "reassoc", "nsz"},
    nogil=True,
    boundscheck=False,
    error_model="numpy",
    cache=True,
)
def interp_linear_flat_extrap_3d_numba_kernel(
    tenors: np.ndarray,
    values: np.ndarray,
    targets: np.ndarray,
    out: np.ndarray,
) -> None:
    """Zero-allocation 3D Linear Interpolator with Flat Extrapolation."""
    c_curves, m_pillars = tenors.shape
    d_target_dates = targets.shape[0]
    s_scenarios = values.shape[2]

    for c in nb.prange(c_curves):
        for d in range(d_target_dates):
            target = targets[d]

            idx = np.searchsorted(tenors[c], target, side="right")
            idx1 = min(max(idx, 1), m_pillars - 1)
            idx0 = idx1 - 1

            t0 = tenors[c, idx0]
            t1 = tenors[c, idx1]

            dt = t1 - t0
            if dt == 0.0:
                dt = 1.0

            w = (target - t0) / dt
            w = min(max(w, 0.0), 1.0)

            for s in range(s_scenarios):
                y0 = values[c, idx0, s]
                y1 = values[c, idx1, s]
                out[c, d, s] = y0 + w * (y1 - y0)

# --- LOG-LINEAR INTERPOLATION WITH FLAT EXTRAPOLATION ---

def interp_log_linear_flat_extrap_3d(

    values: np.ndarray,

    """

    idx1 = np.sum(mask, axis=2)

    y1 = values[c_idx, idx1, :]

    dt = np.where(dt == 0, 1.0, dt)
    w = (targets[None, :] - t0) / dt

    ln_interp = y0 + w[:, :, None] * (y1 - y0)
    return np.exp(ln_interp)

def interp_strike_linear_flat_extrap(
    pillars: np.ndarray,  # Shape: (N_trades, K_strikes)
    values: np.ndarray,  # Shape: (N_trades, K_strikes, S_scenarios)
    targets: np.ndarray,  # Shape: (N_trades, S_scenarios)
) -> np.ndarray:
    """
    Vectorized 1D linear interpolation along the strike/Y-axis with flat extrapolation.
    Used dynamically inside Option Pricers to resolve exact strike vol across scenarios.
    """
    N, K_dim = pillars.shape
    S = targets.shape[1]

    mask = pillars[:, None, :] <= targets[:, :, None]
    idx1 = np.sum(mask, axis=2)
    idx1 = np.clip(idx1, 1, K_dim - 1)
    idx0 = idx1 - 1

    n_idx = np.arange(N)[:, None]
    s_idx = np.arange(S)[None, :]

    x0 = pillars[n_idx, idx0]
    x1 = pillars[n_idx, idx1]

    y0 = values[n_idx, idx0, s_idx]
    y1 = values[n_idx, idx1, s_idx]

    dx = np.where((x1 - x0) == 0, 1.0, x1 - x0)
    w = (targets - x0) / dx

    w = np.clip(w, 0.0, 1.0)
    return y0 + w * (y1 - y0)

    interpolation_methods_dict: Mapping[str, int],
) -> None:

    got = {str(k): int(v) for k, v in interpolation_methods_dict.items()}

    if got == expected:
        return

    different = {
        k: {"expected": expected[k], "got": got[k]}
        for k in (expected.keys() & got.keys())
        if expected[k] != got[k]
    }

    raise ValueError(
        "interpolation_methods sheet does not match riskquant InterpolationMethod enum.\n"
        f"missing={missing}\n"
        f"extra={extra}\n"
        f"different={different}\n"
        "Fix the Excel interpolation_methods table or update riskquant.market_data.curves.interp_methods.InterpMethod."

# riskquant/utils/tenors.py
from __future__ import annotations

from datetime import date
from typing import Optional, Tuple


def parse_bracket_token(tenor_token: str) -> Tuple[str, Optional[int]]:
    """
    "[Y-2]" -> ("Y", -2)
    "[M+0]" -> ("M", 0)
    "[D+24]" -> ("D", 24)
    "[HY+1]" -> ("HY", 1)
    "[ON]" -> ("ON", None)   (LABEL)
    """
    # Error.
    if tenor_token is None:
        raise ValueError("tenor_token cannot be None!")

    # Strip white space.
    s = tenor_token.strip()

    # Must start with '[' and end with ']'.
    if s.startswith("[") and s.endswith("]"):
        s = s[1:-1].strip()

    # Find the plus or minus sign.
    plus = s.rfind("+")
    minus = s.rfind("-")
    sign_pos = max(plus, minus)

    # If no sign.
    if sign_pos <= 0:
        return s, None

    # Unit (Y, M, D, HY, etc.)
    unit = s[:sign_pos]

    # Signed offset.
    offset = int(s[sign_pos:])

    # Return.
    return unit, offset


def get_spot_year(
    *,
    as_of_date: date,
    roll_month: int,
    roll_day: int,
    post_roll_shift: int,
) -> int:
    """
    Defines which delivery year corresponds to Y+0 under ROLL_ON_MMDD.
    """

    # Get the roll date,
    roll_date = date(as_of_date.year, roll_month, roll_day)

    # Compare to as-of-date.
    # As-of-date is on or after the roll date.
    if as_of_date >= roll_date:
        return (
            as_of_date.year + post_roll_shift
        )  # Current year plus the post_roll_shift amount (in years).

    # As-of-date is before the roll date.
    # Previous year plus the post_roll_shift amount (in years).
    return (as_of_date.year - 1) + post_roll_shift


def add_months_to_date_and_return_yyyymm01(
    *,
    d: date,
    months_to_add: int,
) -> date:
    """Add months to a month-start date (day=1)."""
    y = d.year + (d.month - 1 + months_to_add) // 12
    m = (d.month - 1 + months_to_add) % 12 + 1
    return date(y, m, 1)


def fmt_date_as_str_yyyy(d: date) -> str:
    return f"{d.year:04d}"


def fmt_date_as_str_yyyymm(d: date) -> str:
    return f"{d.year:04d}{d.month:02d}"


def fmt_date_as_str_yyyymmdd(d: date) -> str:
    return f"{d.year:04d}{d.month:02d}{d.day:02d}"
