# src/riskquant/ingest_fixtures/config/risk_static_config/_parts/curve_point_meta.py

from __future__ import annotations

import pandas as pd

from riskquant.ingest_fixtures.schemas import RiskStaticConfigCurvePointMeta
from riskquant.ingest_fixtures.utils import validate_column_lengths

# rf_id allocation per curve. Each curve gets a contiguous block; the
# first rf_id of the block is `_rf_id_base`, and points are emitted in
# the order their tenor_tokens appear in `_tenor_tokens`.
#
# rf_id partition (must not collide with node_meta or surface_point_meta):
#   nodes          : 0 ... 99_999
#   curve_points   : 100_000 ... 199_999
#   surface_points : 200_000 ... 299_999
#
# Tenor labels follow the IR conventional schedule: [ON]..[10Y].
# Monthly commodities use OFFSET_M tokens [M+k]; annual carbon uses
# OFFSET_Y tokens [Y+k]; half-year power uses OFFSET_HY tokens [HY+k].

_LABEL_TENORS: tuple[str, ...] = (
    "[ON]",
    "[1W]",
    "[1M]",
    "[3M]",
    "[6M]",
    "[1Y]",
    "[2Y]",
    "[5Y]",
    "[10Y]",
)

_MONTHLY_OFFSETS: tuple[str, ...] = (
    "[M+0]",
    "[M+1]",
    "[M+2]",
    "[M+3]",
    "[M+6]",
    "[M+12]",
)

_ANNUAL_OFFSETS: tuple[str, ...] = (
    "[Y+0]",
    "[Y+1]",
    "[Y+2]",
    "[Y+3]",
)

_HALF_YEAR_OFFSETS: tuple[str, ...] = (
    "[HY+0]",
    "[HY+1]",
    "[HY+2]",
    "[HY+3]",
)


# (curve_id, rf_id_base, tenor_tokens)
_CURVE_BLOCKS: tuple[tuple[int, int, tuple[str, ...]], ...] = (
    # --- USD discount curves ---
    (100_000_000, 1000, _LABEL_TENORS),  # USD_SOFR_OIS
    (100_000_001, 1010, _LABEL_TENORS),  # USD_LIBOR_3M
    (100_000_002, 1020, _LABEL_TENORS),  # USD_CSA_EUR
    # --- EUR discount curves ---
    (100_000_003, 1100, _LABEL_TENORS),  # EUR_ESTR_OIS
    (100_000_004, 1110, _LABEL_TENORS),  # EUR_EURIBOR_6M
    # --- GBP discount curves ---
    (100_000_005, 1200, _LABEL_TENORS),  # GBP_SONIA_OIS
    # --- CHF discount curves ---
    (100_000_006, 1300, _LABEL_TENORS),  # CHF_SARON_OIS
    # --- AUD discount curves ---
    (100_000_007, 1400, _LABEL_TENORS),  # AUD_AONIA_OIS
    # --- Commodity forward curves ---
    (100_000_008, 2000, _MONTHLY_OFFSETS),  # BRENT_CRUDE_FWD
    (100_000_009, 2010, _MONTHLY_OFFSETS),  # WTI_CRUDE_FWD
    (100_000_010, 2020, _MONTHLY_OFFSETS),  # HENRY_HUB_GAS_FWD
    (100_000_011, 2100, _ANNUAL_OFFSETS),  # CO2_EUA_ANNUAL_FWD
    (100_000_012, 2200, _HALF_YEAR_OFFSETS),  # POWER_CH_BASELOAD_HY
)


def _get_data() -> RiskStaticConfigCurvePointMeta:
    rf_ids: list[int] = []
    curve_ids: list[int] = []
    tenor_tokens: list[str] = []

    for curve_id, rf_id_base, tokens in _CURVE_BLOCKS:
        for offset, token in enumerate(tokens):
            rf_ids.append(rf_id_base + offset)
            curve_ids.append(curve_id)
            tenor_tokens.append(token)

    return {
        "rf_id": rf_ids,
        "curve_id": curve_ids,
        "tenor_token": tenor_tokens,
    }


CANONICAL_CURVE_POINT_META_DATA: RiskStaticConfigCurvePointMeta = _get_data()

validate_column_lengths(CANONICAL_CURVE_POINT_META_DATA)

_CANONICAL_CURVE_POINT_META_DF = pd.DataFrame(CANONICAL_CURVE_POINT_META_DATA)

        print_success_message,
    )

    print_calling_file()

    print("As DataFrame:")
    print_df(_CANONICAL_CURVE_POINT_META_DF, index=False)

    print_success_message(add_space_before=False)


if __name__ == "__main__":
    _debug_print()

Builds a synthetic `(T, N)` historical risk-factor matrix whose columns are

referential integrity with the static metadata.

Math
----
- Equity / FX / commodity spot-like factors follow geometric Brownian motion
  with drift `-0.5 * sigma**2 * dt` so simulated levels remain unbiased in

- Rate factors follow arithmetic Brownian motion (random walk).

Memory layout
-------------
The output matrix is allocated F-contiguous so a downstream caller doing
column-major time-series access reads cache-aligned. `np.cumsum(..., out=...)`
and `np.exp(..., out=...)` mutate the buffer in place; no intermediates.

Output dtype is `float32` to match `build_risk_factor_matrix_from_df`'s
default — generating in float64 would just be downcast immediately.

TODO: split this file into `_parts/{nodes,curves,surfaces}.py` once the
universe count grows past comfort. The dynamic-build pattern makes the split
mechanical.

# Path generation (in-place, F-contiguous, batched RNG draw)
# ==============================================================================

_rng: np.random.Generator = get_canonical_rng()
_dt: float = 1.0 / BUSINESS_DAYS_PER_YEAR
_sqrt_dt: float = float(np.sqrt(_dt))

# Pre-allocate F-contiguous so per-column traversal is contiguous in memory.
_historical_values: npt.NDArray[np.floating] = np.empty(
    (CANONICAL_HISTORY_BUSINESS_DAYS, CANONICAL_HISTORICAL_FACTOR_COUNT),
    dtype=HISTORY_DTYPE,
    order="F",
)

# Single batched draw — avoids per-column RNG state thrash.
_Z: npt.NDArray[np.floating] = _rng.standard_normal(
    (CANONICAL_HISTORY_BUSINESS_DAYS, CANONICAL_HISTORICAL_FACTOR_COUNT),
    dtype=HISTORY_DTYPE,
)

for _i, (_, _params) in enumerate(_FACTOR_SPECS):
    _col = _historical_values[:, _i]
    _z_col = _Z[:, _i]
    _sigma = _params.annualized_vol

    if _params.is_rate:
        # Arithmetic Brownian motion: level_t = s0 + cumsum(sigma * sqrt(dt) * Z)
        np.cumsum(_sigma * _sqrt_dt * _z_col, out=_col)
        _col += _params.s0
    else:
        # Geometric Brownian motion in log-space, then exponentiate in place.
        _drift = -0.5 * (_sigma * _sigma) * _dt
        _log_returns = _drift + (_sigma * _sqrt_dt * _z_col)
        np.cumsum(_log_returns, out=_col)
        np.exp(_col, out=_col)
        _col *= _params.s0

# src/riskquant/market_data/builder.py
"""
Market-data matrix builder.

Sits on the producer side of the valuation contract. Takes raw
scenario-shocked levels plus the `UixMapper` describing the run's
global universe, and emits a frozen `MarketDataMatrix` whose every
axis is aligned with the per-trade `_uix` coordinates carried by the
compiled blocks.

Key responsibilities
--------------------
1. FX effective matrix. Reads `UixMapper.fx_effective_pairs` to learn
   which `(fx_uix, invert)` pairs are needed across the portfolio,
   then materialises `fx_spots_effective: (K, S)` from the raw
   `(F, S)` direct-quote scenario matrix. Inverted rows are resolved
   via vectorised `np.reciprocal(out=...)`; direct rows are a single
   block copy. The pricers see zero inversion math.

2. Discount-factor grid. Log-linear interpolation in time, flat
   extrapolation. Routed through the zero-allocation Numba kernel
   with the pre-allocated output buffer.

3. Price-curve grid. Same shape as discount, but interpolation
   method varies per curve and is dispatched via `CURVE_MATH_REGISTRY`
   on `(interp_method, extrap_method)` group masks.

4. Time-to-expiry grid. Act/365.25 day count from `as_of_date` to
   each unique portfolio date, floored at 1e-9 to keep `sqrt(tte)`
   well-defined for same-day expiries.

5. Vol surfaces. Time-slice interpolation onto the unique-date
   universe. Output layout is `[surface, expiry, scenario, pillar]`
   so the strike-axis interpolation in the option-curve kernel can
   walk a contiguous pillar run per (trade, scenario).
"""

from __future__ import annotations

from datetime import date
from typing import TYPE_CHECKING

import numpy as np
import numpy.typing as npt

from riskquant.market_data.curves.interp_defs import (
    interp_log_linear_flat_extrap_3d_numba_kernel,
)
from riskquant.market_data.curves.interp_dispatch import CURVE_MATH_REGISTRY
from riskquant.market_data.state import MarketDataMatrix
from riskquant.market_data.surfaces.y_axes_types import (
    SURFACE_Y_AXIS_TYPE_DELTA_CALL,
)
from riskquant.utils.dates import date_to_ord32

if TYPE_CHECKING:
    from riskquant.portfolio.engine import UixMapper

_ACT_365_25: float = 365.25
_TTE_FLOOR: float = 1e-9


# ==============================================================================
# FX EFFECTIVE
# ==============================================================================


def _build_fx_spots_effective(
    *,
    raw_fx_spots: npt.NDArray[np.float64],
    fx_effective_pairs: npt.NDArray[np.int32],
    num_scenarios: int,
) -> npt.NDArray[np.float64]:
    """
    Materialise the `(K, S)` effective FX matrix from the dedup pair table.

    `fx_effective_pairs[k] == (fx_uix, invert)` — `fx_uix` is a row index
    into `raw_fx_spots` (or `< 0` for an identity-row sentinel), `invert`
    is `0` for direct and `1` for inverted (i.e. `1 / rate`).
    """
    k_pairs = fx_effective_pairs.shape[0]
    out = np.empty((k_pairs, num_scenarios), dtype=np.float64)

    if k_pairs == 0:
        return out

    if fx_effective_pairs.shape[1] != 2:
        raise ValueError(
            f"fx_effective_pairs must have shape (K, 2); got {fx_effective_pairs.shape}"
        )

    fx_uixs = fx_effective_pairs[:, 0]
    invs = fx_effective_pairs[:, 1]

    sentinel_mask = fx_uixs < 0
    direct_mask = (~sentinel_mask) & (invs == 0)
    invert_mask = (~sentinel_mask) & (invs == 1)

    if sentinel_mask.any():
        out[sentinel_mask, :] = 1.0

    if direct_mask.any():
        np.copyto(out[direct_mask, :], raw_fx_spots[fx_uixs[direct_mask]])

    if invert_mask.any():
        np.reciprocal(
            raw_fx_spots[fx_uixs[invert_mask]],
            out=out[invert_mask, :],
        )

    return out


# ==============================================================================
# CURVE GRIDS
# ==============================================================================


def _build_price_curve_grid(
    *,
    raw_tenors: npt.NDArray[np.float64],
    raw_values: npt.NDArray[np.float64],
    target_dates: npt.NDArray[np.int32],
    interp_methods: npt.NDArray[np.int8],
    extrap_methods: npt.NDArray[np.int8],
    num_scenarios: int,
) -> npt.NDArray[np.float64]:
    """
    Group-dispatched 3-D curve interpolation onto the unique-date grid.

    Curves sharing the same `(interp_method, extrap_method)` pair are
    gathered into a contiguous scratch buffer, fed to the dispatched
    kernel, then scatter-assigned back into `out_grid`.

    Scatter contract
    ----------------
    `out_grid[idx]` on the LHS of an assignment is NumPy fancy-index
    setitem and is in-place. `out_grid[idx]` on the RHS (the kernel's
    original signature) would have been a fresh COPY — the kernel
    would have written into the copy, the copy would have been
    discarded, and `out_grid` would have been left as `np.empty()`
    garbage. The scratch + scatter pattern is the correct fix.
    """
    c_curves = raw_values.shape[0]
    d_dates = target_dates.shape[0]

    out_grid = np.empty((c_curves, d_dates, num_scenarios), dtype=np.float64)

    if d_dates == 0 or c_curves == 0:
        return out_grid

    for (interp_id, extrap_id), kernel in CURVE_MATH_REGISTRY.items():
        mask = (interp_methods == interp_id) & (extrap_methods == extrap_id)
        if not mask.any():
            continue

        idx = np.where(mask)[0]
        n_in_group = idx.shape[0]

        # Gather to a contiguous scratch buffer the kernel can write to.
        scratch = np.empty(
            (n_in_group, d_dates, num_scenarios),
            dtype=np.float64,
        )
        kernel(
            np.ascontiguousarray(raw_tenors[idx]),
            np.ascontiguousarray(raw_values[idx]),
            target_dates,
            scratch,
        )
        # Scatter back. Fancy-indexed ASSIGNMENT is in-place.
        out_grid[idx] = scratch

    return out_grid


def _build_discount_factor_grid(
    *,
    raw_tenors: npt.NDArray[np.float64],
    raw_ln_dfs: npt.NDArray[np.float64],
    target_dates: npt.NDArray[np.int32],
    num_scenarios: int,
) -> npt.NDArray[np.float64]:
    """
    Discount-factor grid: log-linear interpolation in time, flat extrap.

    Domain-convention: every discount curve in the system is quoted as
    log(DF) at pillar dates and interpolated linearly on log(DF), which
    is equivalent to linear interpolation on zero rates between pillars.
    The flat extrapolation prevents pathological exponentials at the long
    end and matches the front-office convention.
    """
    c_curves = raw_ln_dfs.shape[0]
    d_dates = target_dates.shape[0]

    out_grid = np.empty((c_curves, d_dates, num_scenarios), dtype=np.float64)

    if d_dates == 0 or c_curves == 0:
        return out_grid

    interp_log_linear_flat_extrap_3d_numba_kernel(
        raw_tenors,
        raw_ln_dfs,
        target_dates,
        out_grid,
    )
    return out_grid


# ==============================================================================
# VOL SURFACES
# ==============================================================================


def _build_vol_surfaces(
    *,
    raw_vol_surface_tenors: npt.NDArray[np.float64],
    raw_vol_surface_strikes: npt.NDArray[np.float64],
    raw_vol_surface_values: npt.NDArray[np.float64],
    surface_interp_x_methods: npt.NDArray[np.int8],
    surface_extrap_x_methods: npt.NDArray[np.int8],
    surface_y_axis_type_ids: npt.NDArray[np.int8],
    target_dates: npt.NDArray[np.int32],
    num_scenarios: int,
) -> npt.NDArray[np.float64]:
    """
    Time-slice every vol surface onto the portfolio's unique-date axis
    and emit the canonical kernel-facing layout.

    Input  : `raw_vol_surface_values` is `(C_surf, M_tenors, K_strikes, S_scn)`.
    Output : `(C_surf, D_dates, S_scn, K_strikes)` — pillar in trailing axis.
    """
    if (surface_y_axis_type_ids == SURFACE_Y_AXIS_TYPE_DELTA_CALL).any():
        raise ValueError(
            "FATAL: Delta-quoted Volatility Surfaces detected in runtime builder. "
            "DELTA_CALL surfaces must be pre-processed and converted to STRIKE or "
            "MONEYNESS by the Calibration module before being passed to the "
            "MarketDataMatrix."
        )

    c_surf = raw_vol_surface_tenors.shape[0] if raw_vol_surface_tenors.ndim > 0 else 0
    k_strikes = (
        raw_vol_surface_strikes.shape[1] if raw_vol_surface_strikes.ndim > 1 else 0
    )
    d_dates = target_dates.shape[0]

    if d_dates == 0 or raw_vol_surface_values.size == 0 or c_surf == 0:
        return np.empty((c_surf, d_dates, num_scenarios, k_strikes), dtype=np.float64)

    c_in, m_tenors, k_in, s_in = raw_vol_surface_values.shape
    if c_in != c_surf or k_in != k_strikes or s_in != num_scenarios:
        raise ValueError(
            f"raw_vol_surface_values shape {raw_vol_surface_values.shape} is "
            f"inconsistent with tenors/strikes/scenarios "
            f"({c_surf}, {m_tenors}, {k_strikes}, {num_scenarios})."
        )

    # Step 1. Fold (pillar, scenario) into one trailing axis and feed
    # the time-interp kernels as a 3-D (c, m_tenors, k * s) array.
    flat_in = raw_vol_surface_values.reshape(
        c_surf, m_tenors, k_strikes * num_scenarios
    )
    time_sliced_flat = _build_price_curve_grid(
        raw_tenors=raw_vol_surface_tenors,
        raw_values=np.ascontiguousarray(flat_in),
        target_dates=target_dates,
        interp_methods=surface_interp_x_methods,
        extrap_methods=surface_extrap_x_methods,
        num_scenarios=k_strikes * num_scenarios,
    )

    # Step 2. Reshape back to (c, d, k, s) and swap to land in the kernel-
    # facing (c, d, s, k) layout. Force C-contiguous because the kernel
    # signature pins `[:, :, :, ::1]`.
    canonical = time_sliced_flat.reshape(
        c_surf, d_dates, k_strikes, num_scenarios
    ).swapaxes(2, 3)
    return np.ascontiguousarray(canonical)


# ==============================================================================
# TOP-LEVEL ENTRY POINT
# ==============================================================================


def build_market_data_matrix(
    *,
    as_of_date: date,
    uix_mapper: UixMapper,
    num_scenarios: int,
    # ---- Raw scalar matrices (F/P/Fwd/Vol, scenarios) ----
    raw_fx_spots: npt.NDArray[np.float64],
    raw_prices: npt.NDArray[np.float64],
    raw_forward_nodes: npt.NDArray[np.float64],
    raw_vol_nodes: npt.NDArray[np.float64],
    # ---- Curve configs ----
    raw_discount_tenors: npt.NDArray[np.float64],
    raw_discount_ln_dfs: npt.NDArray[np.float64],
    raw_price_curve_tenors: npt.NDArray[np.float64],
    raw_price_curve_values: npt.NDArray[np.float64],
    curve_interp_methods: npt.NDArray[np.int8],
    curve_extrap_methods: npt.NDArray[np.int8],
    # ---- Surface configs ----
    raw_vol_surface_tenors: npt.NDArray[np.float64],
    raw_vol_surface_strikes: npt.NDArray[np.float64],
    raw_vol_surface_values: npt.NDArray[np.float64],
    surface_interp_x_methods: npt.NDArray[np.int8],
    surface_extrap_x_methods: npt.NDArray[np.int8],
    surface_y_axis_type_ids: npt.NDArray[np.int8],
) -> MarketDataMatrix:
    """
    Assemble a frozen `MarketDataMatrix` aligned to the run's UIX universes.

    Parameters
    ----------
    uix_mapper
        Source of truth for the FX-effective pair table and the unique-
        date axis. Built once per run by `build_uix_mapper`.

    Returns
    -------
    MarketDataMatrix
        Valuation-ready container; pricers consume directly.
    """
    unique_dates_ord32 = uix_mapper.date_ord32_universe
    d_dates = unique_dates_ord32.shape[0]

    # ---- 1. FX effective matrix ------------------------------------------
    fx_spots_effective = _build_fx_spots_effective(
        raw_fx_spots=raw_fx_spots,
        fx_effective_pairs=uix_mapper.fx_effective_pairs,
        num_scenarios=num_scenarios,
    )

    # ---- 2. Discount factor grid -----------------------------------------
    discount_factor_grid = _build_discount_factor_grid(
        raw_tenors=raw_discount_tenors,
        raw_ln_dfs=raw_discount_ln_dfs,
        target_dates=unique_dates_ord32,
        num_scenarios=num_scenarios,
    )

    # ---- 3. Price curve grid ---------------------------------------------
    price_curve_grid = _build_price_curve_grid(
        raw_tenors=raw_price_curve_tenors,
        raw_values=raw_price_curve_values,
        target_dates=unique_dates_ord32,
        interp_methods=curve_interp_methods,
        extrap_methods=curve_extrap_methods,
        num_scenarios=num_scenarios,
    )

    # ---- 4. TTE grid (act/365.25) ----------------------------------------
    as_of_ord32 = date_to_ord32(as_of_date)
    if d_dates == 0:
        tte_grid = np.empty(0, dtype=np.float64)
    else:
        tte_grid = np.maximum(
            (unique_dates_ord32.astype(np.float64) - float(as_of_ord32)) / _ACT_365_25,
            _TTE_FLOOR,
        )

    # ---- 5. Vol surfaces (canonical [c, d, s, k] layout) -----------------
    vol_surfaces = _build_vol_surfaces(
        raw_vol_surface_tenors=raw_vol_surface_tenors,
        raw_vol_surface_strikes=raw_vol_surface_strikes,
        raw_vol_surface_values=raw_vol_surface_values,
        surface_interp_x_methods=surface_interp_x_methods,
        surface_extrap_x_methods=surface_extrap_x_methods,
        surface_y_axis_type_ids=surface_y_axis_type_ids,
        target_dates=unique_dates_ord32,
        num_scenarios=num_scenarios,
    )

    return MarketDataMatrix(
        num_scenarios=num_scenarios,
        fx_spots_effective=fx_spots_effective,
        prices=raw_prices,
        forward_nodes=raw_forward_nodes,
        vol_nodes=raw_vol_nodes,
        discount_factor_grid=discount_factor_grid,
        price_curve_grid=price_curve_grid,
        tte_grid=tte_grid,
        vol_surface_strikes=raw_vol_surface_strikes,
        vol_surfaces=vol_surfaces,
    )

from __future__ import annotations

from enum import IntEnum

import numpy as np


class ExtrapMethod(IntEnum):
    FLAT = 1
    LINEAR = 2


# compiler-friendly int8 constants
EXTRAP_FLAT: np.int8 = np.int8(ExtrapMethod.FLAT)
EXTRAP_LINEAR: np.int8 = np.int8(ExtrapMethod.LINEAR)


# Extrap methods
EXTRAP_METHOD_NAMES: tuple[str, ...] = tuple(
    method.name.upper() for method in ExtrapMethod
)

# Central registry
EXTRAP_METHOD_REGISTRY: dict[str, ExtrapMethod] = {
    "LINEAR": ExtrapMethod.LINEAR,
    "FLAT": ExtrapMethod.FLAT,
}


def validate_extrapolation_methods_dict(
    extrapolation_methods_dict: dict[str, int],
) -> None:
    expected = {
        name: int(member) for name, member in ExtrapMethod.__members__.items()
    }
    got = {str(k): int(v) for k, v in extrapolation_methods_dict.items()}

    if got == expected:
        return

    missing = {k: expected[k] for k in expected.keys() - got.keys()}
    extra = {k: got[k] for k in got.keys() - expected.keys()}
    different = {
        k: {"expected": expected[k], "got": got[k]}
        for k in (expected.keys() & got.keys())
        if expected[k] != got[k]
    }

    raise ValueError(
        "extrapolation_methods sheet does not match riskquant ExtrapMethod enum.\n"
        f"missing={missing}\n"
        f"extra={extra}\n"
        f"different={different}\n"
        "Fix the Excel extrapolation_methods table or update riskquant.market_data.curves.extrap_methods.ExtrapMethod."
    )

# src/riskquant/market_data/curves/interp_defs.py
from __future__ import annotations

import math

import numba as nb
import numpy as np

# ==============================================================================
# --- LINEAR INTERPOLATION WITH FLAT EXTRAPOLATION ---
# ==============================================================================


def interp_linear_flat_extrap_3d(
    tenors: np.ndarray,
    values: np.ndarray,
    targets: np.ndarray,
) -> np.ndarray:
    """
    Vectorized 3D Linear Interpolation with Flat Extrapolation.
    Shapes: tenors (c_curves, m_pillars), values (c_curves, m_pillars, s_scenarios), targets (d_targets,)
    Returns: (c_curves, d_targets, s_scenarios)
    """
    c_curves, m_pillars = tenors.shape
    idx1 = np.clip(
        np.sum(tenors[:, None, :] <= targets[None, :, None], axis=2),
        1,
        m_pillars - 1,
    )
    idx0 = idx1 - 1

    c_idx = np.arange(c_curves)[:, None]
    t0, t1 = tenors[c_idx, idx0], tenors[c_idx, idx1]
    y0, y1 = values[c_idx, idx0, :], values[c_idx, idx1, :]

    dt = np.where((t1 - t0) == 0, 1.0, t1 - t0)
    w = (targets[None, :] - t0) / dt

    # FLAT EXTRAPOLATION: Clamp weights to [0.0, 1.0]
    w = np.clip(w, 0.0, 1.0)

    return y0 + w[:, :, None] * (y1 - y0)


@nb.njit(
    parallel=True,
    fastmath={"contract", "reassoc", "nsz"},
    nogil=True,
    boundscheck=False,
    error_model="numpy",
    cache=True,
)
def interp_linear_flat_extrap_3d_numba_kernel(
    tenors: np.ndarray,
    values: np.ndarray,
    targets: np.ndarray,
    out: np.ndarray,
) -> None:
    """Zero-allocation 3D Linear Interpolator with Flat Extrapolation."""
    c_curves, m_pillars = tenors.shape
    d_target_dates = targets.shape[0]
    s_scenarios = values.shape[2]

    for c in nb.prange(c_curves):
        for d in range(d_target_dates):
            target = targets[d]

            idx = np.searchsorted(tenors[c], target, side="right")
            idx1 = min(max(idx, 1), m_pillars - 1)
            idx0 = idx1 - 1

            t0 = tenors[c, idx0]
            t1 = tenors[c, idx1]

            dt = t1 - t0
            if dt == 0.0:
                dt = 1.0

            w = (target - t0) / dt
            w = min(max(w, 0.0), 1.0)

            for s in range(s_scenarios):
                y0 = values[c, idx0, s]
                y1 = values[c, idx1, s]
                out[c, d, s] = y0 + w * (y1 - y0)


# ==============================================================================
# --- LINEAR INTERPOLATION WITH LINEAR EXTRAPOLATION ---
# ==============================================================================


def interp_linear_linear_extrap_3d(
    tenors: np.ndarray,
    values: np.ndarray,
    targets: np.ndarray,
) -> np.ndarray:
    """
    Vectorized 3D Linear Interpolation with Linear Extrapolation.
    Shapes: tenors (c_curves, m_pillars), values (c_curves, m_pillars, s_scenarios), targets (d_targets,)
    Returns: (c_curves, d_targets, s_scenarios)
    """
    c_curves, m_pillars = tenors.shape
    idx1 = np.clip(
        np.sum(tenors[:, None, :] <= targets[None, :, None], axis=2),
        1,
        m_pillars - 1,
    )
    idx0 = idx1 - 1

    c_idx = np.arange(c_curves)[:, None]
    t0, t1 = tenors[c_idx, idx0], tenors[c_idx, idx1]
    y0, y1 = values[c_idx, idx0, :], values[c_idx, idx1, :]

    dt = np.where((t1 - t0) == 0, 1.0, t1 - t0)
    w = (targets[None, :] - t0) / dt

    # LINEAR EXTRAPOLATION: Do not clamp weights
    return y0 + w[:, :, None] * (y1 - y0)


@nb.njit(
    parallel=True,
    fastmath={"contract", "reassoc", "nsz"},
    nogil=True,
    boundscheck=False,
    error_model="numpy",
    cache=True,
)
def interp_linear_linear_extrap_3d_numba_kernel(
    tenors: np.ndarray,
    values: np.ndarray,
    targets: np.ndarray,
    out: np.ndarray,
) -> None:
    """Zero-allocation 3D Linear Interpolator with Linear Extrapolation."""
    c_curves, m_pillars = tenors.shape
    d_target_dates = targets.shape[0]
    s_scenarios = values.shape[2]

    for c in nb.prange(c_curves):
        for d in range(d_target_dates):
            target = targets[d]

            idx = np.searchsorted(tenors[c], target, side="right")
            idx1 = min(max(idx, 1), m_pillars - 1)

    values: np.ndarray,
    targets: np.ndarray,
) -> np.ndarray:

    Vectorized 3D Log-Linear Interpolation with Constant Forward Extrapolation.
    """
    c_curves, m_pillars = tenors.shape
    mask = tenors[:, None, :] <= targets[None, :, None]
    idx1 = np.sum(mask, axis=2)
    idx1 = np.clip(idx1, 1, m_pillars - 1)
    idx0 = idx1 - 1

    c_idx = np.arange(c_curves)[:, None]
    t0 = tenors[c_idx, idx0]
    t1 = tenors[c_idx, idx1]

    y0 = values[c_idx, idx0, :]
    y1 = values[c_idx, idx1, :]

    dt = t1 - t0
    dt = np.where(dt == 0, 1.0, dt)
    w = (targets[None, :] - t0) / dt

    ln_interp = y0 + w[:, :, None] * (y1 - y0)
    return np.exp(ln_interp)

# --- 2D SURFACE STRIKE INTERPOLATOR ---
# ==============================================================================


def interp_strike_linear_flat_extrap(
    pillars: np.ndarray,  # Shape: (N_trades, K_strikes)
    values: np.ndarray,  # Shape: (N_trades, K_strikes, S_scenarios)
    targets: np.ndarray,  # Shape: (N_trades, S_scenarios)
) -> np.ndarray:
    """
    Vectorized 1D linear interpolation along the strike/Y-axis with flat extrapolation.
    Used dynamically inside Option Pricers to resolve exact strike vol across scenarios.
    """
    N, K_dim = pillars.shape
    S = targets.shape[1]

    mask = pillars[:, None, :] <= targets[:, :, None]
    idx1 = np.sum(mask, axis=2)
    idx1 = np.clip(idx1, 1, K_dim - 1)
    idx0 = idx1 - 1

    n_idx = np.arange(N)[:, None]
    s_idx = np.arange(S)[None, :]

    x0 = pillars[n_idx, idx0]
    x1 = pillars[n_idx, idx1]

    y0 = values[n_idx, idx0, s_idx]
    y1 = values[n_idx, idx1, s_idx]

    dx = np.where((x1 - x0) == 0, 1.0, x1 - x0)
    w = (targets - x0) / dx

    w = np.clip(w, 0.0, 1.0)
    return y0 + w * (y1 - y0)

    Pre-allocates an (N_trades, S_scenarios) array and resolves exactly.

    s_scenarios = targets.shape[1]

    for n in nb.prange(n_trades):

            y0 = values[n, idx0, s]
            y1 = values[n, idx1, s]

    interpolation_methods_dict: Mapping[str, int],
) -> None:
    expected = {name: int(member) for name, member in InterpMethod.__members__.items()}
    got = {str(k): int(v) for k, v in interpolation_methods_dict.items()}

    if got == expected:
        return

    missing = {k: expected[k] for k in expected.keys() - got.keys()}
    extra = {k: got[k] for k in got.keys() - expected.keys()}
    different = {
        k: {"expected": expected[k], "got": got[k]}
        for k in (expected.keys() & got.keys())
        if expected[k] != got[k]
    }

    raise ValueError(
        "interpolation_methods sheet does not match riskquant InterpolationMethod enum.\n"
        f"missing={missing}\n"
        f"extra={extra}\n"
        f"different={different}\n"
        "Fix the Excel interpolation_methods table or update riskquant.market_data.curves.interp_methods.InterpMethod."

from __future__ import annotations

from collections.abc import Mapping
from pathlib import Path

import numpy as np
import pandas as pd
from pandas.api.types import is_datetime64_any_dtype

from riskquant.market_data.risk_factors.builder_utils import (
    _require_mapping_key,
    _validate_mapping_dict,
)
from riskquant.market_data.risk_factors.matrix import RiskFactorMatrix
from riskquant.utils.dates import (

    datetime64D_to_ord32,
)

# --- BUILD A RISK FACTOR MATRIX FROM A DATAFRAME


def build_risk_factor_matrix_from_df(
    *,
    df: pd.DataFrame,
    name_to_id_map: Mapping[str, int] | None = None,
    use_int32_dates: bool = True,
    float_dtype: type[np.floating] = np.float32,
) -> RiskFactorMatrix:
    """Construct a RiskFactorMatrix from a wide pandas DataFrame.

    Parameters
    ----------
    df : pd.DataFrame
        Wide DataFrame with dates on the index and one column per risk factor.
    name_to_id_map : Optional[Mapping[str, int]]
        Optional mapping from column name (string) to integer factor id. If
        omitted, columns are parsed as integers directly.
    use_int32_dates : bool
        If True convert dates to int32 ordinal representation; otherwise
        keep datetime64[D] dtype.
    float_dtype : numpy floating type
        dtype used for the `values` array (default: np.float32).

    Returns
    -------
    RiskFactorMatrix
        Canonicalized matrix with validated shapes, dtypes and contiguity.

    Raises
    ------
    ValueError
        If the index is not date-like, not monotonic increasing, or contains
        duplicates; or if factor id mapping contains errors.

    Notes
    -----
    The function intentionally does not auto-sort the DataFrame rows
    (dates). If the provided index is unsorted we raise an error because
    auto-sorting would require an expensive O(T*N) copy of the values.
    """
    if not isinstance(df, pd.DataFrame):
        raise ValueError("Expected a DataFrame!")

    # Accept a wider set of date-like index types; coerce if needed.
    if not is_datetime64_any_dtype(df.index):
        # Convert in-place where possible; this preserves row order and
        # avoids unnecessary copying of the data matrix.
        df = df.copy(deep=False)
        df.index = pd.to_datetime(df.index, errors="raise")

    # Enforce invariants early with clear errors (and avoid needless allocations)
    if df.index.has_duplicates:
        raise ValueError("DataFrame index (dates) must be unique!")
    if not df.index.is_monotonic_increasing:
        raise ValueError("DataFrame index (dates) must be sorted ascending!")

    # Check for duplicated column labels (factor names); this would cause silent data corruption
    if df.columns.has_duplicates:
        duplicated = df.columns[df.columns.duplicated(keep=False)]
        unique_dupes = duplicated.unique()
        sample = unique_dupes[:10]
        raise ValueError(
            f"DataFrame columns (factor names) must be unique.\n"
            f"Found {len(unique_dupes)} duplicated column(s).\n"
            f"Sample duplicates: {list(sample)}"
        )

    # Dates: obtain a datetime64[D] view and optionally convert to int32 ordinal
    dts_64D = df.index.to_numpy(dtype="datetime64[D]", copy=False)
    dates = datetime64D_to_ord32(dts_64D) if use_int32_dates else dts_64D

    # Columns -> factor_ids: either direct integer parsing or mapping lookup.
    cols = df.columns
    n = cols.size

    if name_to_id_map is None:
        # Parse column labels as integers; will raise on non-integer labels.
        factor_ids = np.fromiter(
            (int(c) for c in cols),
            dtype=np.int32,
            count=n,
        )
    else:
        # Validate mapping once (cheap for small config dicts) and then map.
        _validate_mapping_dict(name_to_id_map)
        factor_ids = np.fromiter(
            (_require_mapping_key(name_to_id_map, str(c)) for c in cols),
            dtype=np.int32,
            count=n,
        )

    # Canonicalize factor axis BEFORE allocating `values` (avoids extra copies).
    if n > 1 and np.any(factor_ids[1:] < factor_ids[:-1]):
        perm = np.argsort(factor_ids, kind="mergesort")
        factor_ids = factor_ids[perm]

        if np.any(factor_ids[1:] == factor_ids[:-1]):
            raise ValueError("factor_ids must be unique!")

        # Reorder the DataFrame columns to match the canonical factor order.
        df = df.iloc[:, perm]
    else:
        if n > 1 and np.any(factor_ids[1:] == factor_ids[:-1]):
            raise ValueError("factor_ids must be unique!")

    # Materialize the numeric matrix as a C-contiguous array with requested dtype.
    values = np.ascontiguousarray(
        df.to_numpy(dtype=float_dtype, copy=True, na_value=np.nan)
    )

    return RiskFactorMatrix(dates=dates, factor_ids=factor_ids, values=values)

def build_risk_factor_matrix_from_xlsx(
    *,
    path: str | Path,
    sheet_name: str | int = 0,
    name_to_id_map: Mapping[str, int] | None = None,
    use_int32_dates: bool = True,
    float_dtype: type[np.floating] = np.float32,
    as_of_date: pd.Timestamp | None = None,
) -> RiskFactorMatrix:
    """Load a RiskFactorMatrix from an Excel workbook/sheet.

    Parameters
    ----------
    path : str | Path
        Path to the Excel file.
    sheet_name : str | int
        Sheet name or index to read (forwarded to pandas.read_excel).
    name_to_id_map : Optional[Mapping[str, int]]
        Optional mapping for column name -> factor id (same semantics as
        `from_df`).
    use_int32_dates : bool
        If True convert dates to int32 ordinal representation.
    float_dtype : numpy floating type
        dtype used for the `values` array.
    as_of_date : Optional[pd.Timestamp]
        If provided, rows with date > as_of_date are filtered out.

    Returns
    -------
    RiskFactorMatrix
        Canonicalized matrix constructed from the Excel sheet.
    """
    # Read sheet with the first column treated as the index (dates expected).
    df = pd.read_excel(
        str(path),
        sheet_name=sheet_name,
        index_col=0,
    )

    # If empty.
    if df.empty:
        raise ValueError("dataframe is empty - check: {path}")

    # Convert index to datetime if needed (common for Excel imports).
    if not is_datetime64_any_dtype(df.index):
        df.index = pd.to_datetime(df.index, errors="raise")

    # Optional filter by as_of_date; preserves monotonicity if present.
    if as_of_date is not None:
        cutoff = coerce_as_of_date_to_pd_timestamp(as_of_date)
        df = df.loc[df.index <= cutoff]

    # Enforce same sorted/unique contract as `from_df`.

    # Duplicated dates.
    if df.index.has_duplicates:
        duplicated = df.index[df.index.duplicated(keep=False)]
        unique_dupes = duplicated.unique()
        sample = unique_dupes[:10]
        raise ValueError(
            f"Excel sheet index (dates) must be unique.\n"
            f"Found {len(unique_dupes)} duplicated date(s).\n"
            f"Sample duplicates: {list(sample)}"

    # Non-monotonic dates
    if not df.index.is_monotonic_increasing:
        raise ValueError("Excel sheet index (dates) must be sorted ascending!")

    # Reuse the robust `from_df` constructor for the remaining work.
    return build_risk_factor_matrix_from_df(
        df=df,
        name_to_id_map=name_to_id_map,
        use_int32_dates=use_int32_dates,
        float_dtype=float_dtype,
    )

# src\riskquant\market_data\risk_factors\builder_utils.py
from __future__ import annotations

from collections.abc import Mapping
from typing import Any

import numpy as np


def _validate_mapping_dict(m: Mapping[str, Any]) -> None:
    """Validate a small static mapping of risk factor name -> integer id.

    Checks performed:
      - keys must be non-empty strings
      - values must be integer-like (reject bool, float, str)
      - no two keys map to the same integer id

    This helper is optimized for configuration-time validation (small dicts),
    not for per-row lookups in tight loops.
    """
    # Empty/blank key detection (single pass)
    for k in m.keys():
        if not isinstance(k, str) or not k.strip():
            raise ValueError(f"Mapping contains invalid/blank key: {k!r}")

    # Ensure values are integer-like and detect duplicates by tracking seen ids.
    seen: dict[int, str] = {}
    for k, v in m.items():
        # `bool` is a subclass of int in Python; explicitly reject it here because
        # booleans are not valid factor ids.
        if isinstance(v, (bool,)):
            raise ValueError(
                f"Mapping value for '{k}' must be an int id, got bool={v!r}"
            )

        # Accept Python ints and NumPy integer scalars; reject other types.
        if not isinstance(v, (int, np.integer)):
            raise ValueError(
                f"Mapping value for '{k}' must be an int id, got {type(v).__name__}={v!r}"
            )

        iv = int(v)
        prev = seen.get(iv)
        if prev is not None:
            # Clear, actionable error message when multiple names map to same id.
            raise ValueError(
                "Mapping contains duplicate ID assignment:\n"
                f"  '{prev}' and '{k}' both map to ID={iv}\n"
                "Each risk factor must have a unique risk_factor_name_id."
            )
        seen[iv] = k


def _require_mapping_key(
    m: Mapping[str, Any],
    key: str,
    *,
    preview_n: int = 25,
) -> Any:
    """Lookup `key` in mapping `m` and raise a detailed error if missing.

    This helper provides user-friendly diagnostics on configuration errors by
    showing a short preview of known keys and a suggested fix. Designed to be
    inexpensive on the success path (single hash lookup); the preview list is
    materialized only on error.

    Parameters
    ----------
    m : Mapping[str, Any]
        Mapping from risk factor name to id.
    key : str
        Key to lookup (column label converted to string prior to lookup).
    preview_n : int
        Maximum number of known keys to show in the error message.
    """
    try:
        return m[key]
    except KeyError:
        # Materialize a small preview of keys only when diagnosing a failure --
        # prevents huge error messages for very large config maps.
        keys = list(m.keys())
        n = len(keys)
        preview = keys[:preview_n]
        preview_str = ", ".join(preview) + (" ..." if n > preview_n else "")

        raise ValueError(
            "Risk factor ID mapping lookup failed.\n"
            f"Missing key: '{key}'\n\n"
            "Fix:\n"
            "  Add this risk factor to the STATIC config table that defines risk_factor_name_id.\n"
            "  Example row to add:\n"
            f"    risk_factor_name = {key}\n"
            "    risk_factor_name_id = <new unique integer>\n\n"
            f"Known keys preview ({min(preview_n, n)} of {n}):\n"
            f"  {preview_str}\n"
        ) from None

# src/riskquant/market_data/risk_factors/compile_snapshot.py
"""
Compile a flat `(rel_name, level)` levels dictionary into a typed
`RiskFactorLevelSnapshot`, with unit-scale conversion applied.

Public API mirror — analogous to `compile_risk_static_config_from_dict_of_dfs`:

  compile_risk_static_config_from_dict_of_dfs(dict_of_dfs) -> RiskStaticConfig
  compile_risk_factor_level_snapshot_from_levels_dict(
      *, as_of_date, levels_dict, static_config
  ) -> RiskFactorLevelSnapshot

Contract
--------
- Every key in `levels_dict` is a relative risk-factor name (e.g.
  "USD_SOFR_OIS[1Y]", "BRENT_CRUDE_VOL_SURF[M+0][50.00]") that
  `static_config.rel_name_to_rf_id` resolves.
- Each value is the level *in the unit_scale declared by the static
  config for that rf*. This compiler applies the per-rf
  `unit_scale_factor` so the emitted `factor_levels` are always in
  base DECIMAL — e.g. PERCENT inputs (5.30) become decimal (0.0530),
  and BP inputs (50.0) become decimal (0.0050).
- The resolved `factor_ids` array is sorted strictly ascending and
  C-contiguous; `RiskFactorLevelSnapshot.__post_init__` enforces this
  invariant downstream.

Why a separate compiler rather than reusing `build_*_from_dict`?
----------------------------------------------------------------
The existing `build_risk_factor_level_snapshot_from_dict` is layout-
agnostic and unit-agnostic — fine for already-canonical numeric inputs.
The compiler here adds two pieces the canonical fixture needs:
  1. Name resolution via the static config (rel_name -> rf_id).
  2. Unit-scale promotion to base DECIMAL.

Both belong to the ingest perimeter, not the hot-path snapshot
container, so they live here in a dedicated compile step.
"""

from __future__ import annotations

from collections.abc import Mapping
from datetime import date

import numpy as np

from riskquant.config.static_config.static_config import RiskStaticConfig
from riskquant.market_data.risk_factors.snapshot import RiskFactorLevelSnapshot


def compile_risk_factor_level_snapshot_from_levels_dict(
    *,
    as_of_date: date,
    levels_dict: Mapping[str, float],
    static_config: RiskStaticConfig,
    float_dtype: type[np.floating] = np.float64,
) -> RiskFactorLevelSnapshot:
    """
    Resolve relative-name -> rf_id, apply unit_scale, and return a snapshot.

    Parameters
    ----------
    as_of_date
        Calendar date the snapshot applies to. Pure metadata on the
        returned object — no arithmetic uses it.
    levels_dict
        Flat mapping of relative-name -> level. Levels are in each rf's
        declared `unit_scale` (PERCENT for IR, DECIMAL for prices /
        FX / vols, BP if anything declares it).
    static_config
        Compiled static config. Source of truth for the rel-name -> rf_id
        index and the per-rf unit_scale_factor.
    float_dtype
        Output dtype for `factor_levels`. Default `float64` to preserve
        precision through the unit-scale multiplication; downstream
        scenario buffers narrow to `float32` at their own perimeter.

    Returns
    -------
    RiskFactorLevelSnapshot
        `factor_ids` int32 sorted-ascending unique; `factor_levels`
        in base DECIMAL, parallel to `factor_ids`.

    Raises
    ------
    ValueError
        Empty `levels_dict`.
    KeyError
        A relative name in `levels_dict` does not resolve to any rf_id
        in the static config. The error names the offending key.
    """
    if not levels_dict:
        raise ValueError(
            "levels_dict cannot be empty; the snapshot must reference at "
            "least one rf_id."
        )

    n = len(levels_dict)
    rel_to_rf_id = static_config.rel_name_to_rf_id
    rf_id_to_unit_scale = static_config.rf_id_to_unit_scale_factor

    # Single-pass resolution: name -> rf_id + per-rf scale -> base level.
    # Pre-allocated int32 / float64 buffers; no Python list intermediates
    # past the dict iteration itself.
    ids_buf = np.empty(n, dtype=np.int32)
    levels_buf = np.empty(n, dtype=float_dtype)

    for i, (rel_name, raw_level) in enumerate(levels_dict.items()):
        rf_id = rel_to_rf_id.get(rel_name)
        if rf_id is None:
            raise KeyError(
                f"rel_name {rel_name!r} did not resolve through "
                f"static_config.rel_name_to_rf_id. The snapshot fixture is "
                f"out of sync with the static config."
            )
        ids_buf[i] = np.int32(rf_id)
        # rf_id_to_unit_scale_factor is keyed by rf_id (numpy array indexed
        # by id). float()-wrap because the result will be cast to
        # float_dtype anyway and we want native multiplication.
        levels_buf[i] = float_dtype(raw_level) * float_dtype(
            rf_id_to_unit_scale[int(rf_id)]
        )

    # The snapshot constructor enforces sorted-unique + contiguous; pass
    # the buffers in directly. If callers fed us duplicate rel-names that
    # map to the same rf_id, the snapshot constructor will raise.
    return RiskFactorLevelSnapshot(
        as_of_date=as_of_date,
        factor_ids=ids_buf,
        factor_levels=levels_buf,
    )

# riskquant/market_data/risk_factors/matrix.py
from __future__ import annotations

from dataclasses import dataclass
from datetime import date

import numpy as np
import pandas as pd

from riskquant.market_data.risk_factors.snapshot import RiskFactorLevelSnapshot
from riskquant.utils.dates import ordinal32_to_datetime64D


@dataclass(frozen=True, slots=True)
class RiskFactorMatrix:
    """
    Canonical RiskFactorMatrix with God-Tier memory layout for time-series operations.

    Invariants:
      - dates: 1D, sorted ascending, unique (int32 ordinal or datetime64[D])
      - factor_ids: 1D int32, sorted ascending, unique
      - values: 2D float32/64, shape (T, N), F-contiguous by default for column-major access
      - len(dates)=T, len(factor_ids)=N
    """

    # datetime64[D] or int32 ordinal
    dates: np.ndarray

    # int32
    factor_ids: np.ndarray

    # float32/64, shape (T, N) --> (observations, factors)
    values: np.ndarray

    # Default memory layout (F-contiguous for time-series)
    fortran_order: bool = True

    def __post_init__(self) -> None:
        # ---- Basic checks ----
        t_obs, n_factors = self.values.shape
        if self.dates.ndim != 1:
            raise ValueError("dates must be 1D")
        if self.factor_ids.ndim != 1:
            raise ValueError("factor_ids must be 1D")
        if self.values.ndim != 2:
            raise ValueError("values must be 2D")

        # ---- dtype checks ----
        if not np.issubdtype(self.values.dtype, np.floating):
            raise TypeError(f"values must be float dtype, got {self.values.dtype}")

        if np.issubdtype(self.dates.dtype, np.integer):
            if self.dates.dtype != np.int32:
                raise TypeError(f"int dates must be int32, got {self.dates.dtype}")
        elif self.dates.dtype != np.dtype("datetime64[D]"):
            raise TypeError(
                f"datetime dates must be datetime64[D], got {self.dates.dtype}"
            )

        if self.factor_ids.dtype != np.int32:
            raise TypeError(f"factor_ids must be int32, got {self.factor_ids.dtype}")

        # ---- shape checks ----
        if self.dates.size != t_obs:
            raise ValueError(
                f"len(dates) ({self.dates.size}) != values.shape[0] ({t_obs})"
            )
        if self.factor_ids.size != n_factors:
            raise ValueError(
                f"len(factor_ids) ({self.factor_ids.size}) != values.shape[1] ({n_factors})"
            )

        # ---- memory layout ----
        if self.fortran_order:
            if not self.values.flags["F_CONTIGUOUS"]:
                vals = np.asfortranarray(self.values)
                object.__setattr__(self, "values", vals)
        else:
            if not self.values.flags["C_CONTIGUOUS"]:
                vals = np.ascontiguousarray(self.values)
                object.__setattr__(self, "values", vals)

        # ---- dates sorted + unique ----
        if t_obs > 1:
            if np.any(self.dates[1:] < self.dates[:-1]):
                raise ValueError("dates must be sorted ascending")
            if np.any(self.dates[1:] == self.dates[:-1]):
                raise ValueError("dates must be unique")

        # ---- factor_ids sorted + unique ----
        if n_factors > 1 and np.any(self.factor_ids[1:] < self.factor_ids[:-1]):
            perm = np.argsort(self.factor_ids, kind="mergesort")
            sorted_ids = self.factor_ids[perm]
            if np.any(sorted_ids[1:] == sorted_ids[:-1]):
                raise ValueError("factor_ids must be unique")
            sorted_vals = (
                np.asfortranarray(self.values[:, perm])
                if self.fortran_order
                else np.ascontiguousarray(self.values[:, perm])
            )
            object.__setattr__(self, "factor_ids", sorted_ids)
            object.__setattr__(self, "values", sorted_vals)

    # ---------------------------- Convenience Methods ----------------------------
    def to_df(self) -> pd.DataFrame:
        index = pd.DatetimeIndex(
            ordinal32_to_datetime64D(self.dates)
            if np.issubdtype(self.dates.dtype, np.integer)
            else self.dates
        )
        df = pd.DataFrame(data=self.values, index=index, columns=self.factor_ids)
        df.index.name = "date"
        df.columns.name = "factor_id"
        return df

    def head(self, n: int = 5) -> pd.DataFrame:
        return self.to_df().head(n=n)

    def tail(self, n: int = 5) -> pd.DataFrame:
        return self.to_df().tail(n=n)

    def plot(self, **kwargs):
        return self.to_df().plot(**kwargs)

    # ---------------------------- Properties ----------------------------
    @property
    def index_dates(self) -> pd.DatetimeIndex:
        return pd.DatetimeIndex(
            ordinal32_to_datetime64D(self.dates)
            if np.issubdtype(self.dates.dtype, np.integer)
            else self.dates
        )

    @property
    def index_factors(self) -> pd.Index:
        return pd.Index(self.factor_ids, name="factor_id")

    @property
    def df(self) -> pd.DataFrame:
        return self.to_df()

    @property
    def shape(self) -> tuple[int, ...]:
        return self.values.shape

    @property
    def num_dates(self) -> int:
        return self.values.shape[0]

    @property
    def num_factors(self) -> int:
        return self.values.shape[1]

    # ---------------------------- Slicing Methods ----------------------------
    def slice_dates(
        self, start: int | None = None, end: int | None = None
    ) -> RiskFactorMatrix:
        start = 0 if start is None else start
        end = self.num_dates if end is None else end
        vals = self.values[start:end, :]
        if self.fortran_order and not vals.flags["F_CONTIGUOUS"]:
            vals = np.asfortranarray(vals)
        elif not self.fortran_order and not vals.flags["C_CONTIGUOUS"]:
            vals = np.ascontiguousarray(vals)
        return RiskFactorMatrix(
            dates=self.dates[start:end],
            factor_ids=self.factor_ids,
            values=vals,
            fortran_order=self.fortran_order,
        )

    def slice_factors(self, factor_indices: np.ndarray) -> RiskFactorMatrix:
        vals = self.values[:, factor_indices]
        if self.fortran_order and not vals.flags["F_CONTIGUOUS"]:
            vals = np.asfortranarray(vals)
        elif not self.fortran_order and not vals.flags["C_CONTIGUOUS"]:
            vals = np.ascontiguousarray(vals)
        return RiskFactorMatrix(
            dates=self.dates,
            factor_ids=self.factor_ids[factor_indices],
            values=vals,
            fortran_order=self.fortran_order,
        )

    # ---------------------------- Time-Series Optimization ----------------------------
    def to_time_series_optimized_array(self) -> np.ndarray:
        """Return F-contiguous array for column-major operations (Numba/Cython)."""
        return np.asfortranarray(self.values) if not self.fortran_order else self.values

    def get_factor_time_series(self, factor_id: int) -> np.ndarray:
        idx = np.searchsorted(self.factor_ids, factor_id)
        if idx >= self.factor_ids.size or self.factor_ids[idx] != factor_id:
            raise KeyError(f"factor_id {factor_id} not found")
        return np.ascontiguousarray(self.values[:, idx])

    # ---------------------------- Snapshot ----------------------------
    def get_snapshot(self, target_date: date) -> RiskFactorLevelSnapshot:
        from riskquant.utils.dates import date_to_ord32

        target_val = (
            date_to_ord32(target_date)
            if np.issubdtype(self.dates.dtype, np.integer)
            else np.datetime64(target_date, "D")
        )
        idx = np.searchsorted(self.dates, target_val)
        if idx >= self.dates.size or self.dates[idx] != target_val:
            raise KeyError(f"Date {target_date} not found in matrix")

        levels = self.values[idx, :]
        valid_mask = ~np.isnan(levels)
        return RiskFactorLevelSnapshot(
            as_of_date=target_date,
            factor_ids=self.factor_ids[valid_mask],
            factor_levels=levels[valid_mask],
        )

    @classmethod
    def from_snapshot(
        cls,
        snapshot: RiskFactorLevelSnapshot,
        *,
        fortran_order: bool = True,
    ) -> RiskFactorMatrix:
        """
        Elevate this snapshot into a 1-row RiskFactorMatrix.

        S-Tier mechanics:
        - factor_ids: Passed by reference (zero-copy).
        - values: Reshaped via np.newaxis (zero-copy view).
        - dates: Minimal 1-element array allocation aligned to datetime64[D].
        """

        # 1. Align dates invariant: 1D array of datetime64[D]
        dates_arr = np.array([snapshot.as_of_date], dtype="datetime64[D]")

        # 2. Align values invariant: 2D array (T, N) -> (1, N)
        # Using np.newaxis creates a fast, zero-copy view of the 1D C-contiguous array.
        values_2d = snapshot.factor_levels[np.newaxis, :]

        return RiskFactorMatrix(
            dates=dates_arr,
            factor_ids=snapshot.factor_ids,  # Already validated, sorted, and unique
            values=values_2d,
            fortran_order=fortran_order,
        )

# Registry
SHIFT_TYPE_NAME_TO_ID_REGISTRY: dict[str, np.int8] = {
    "NO_SHIFT": SHIFT_NO_SHIFT_ID,
    "LOG_RETURN": SHIFT_LOG_RETURN_ID,
    "SIMPLE_RETURN": SHIFT_SIMPLE_RETURN_ID,
    "DIFFERENCE": SHIFT_DIFFERENCE_ID,
}


def validate_shift_types_dict(shift_types_dict: Mapping[str, int]) -> None:
    """
    Validate that the Excel-provided mapping exactly matches the code enum.

    Rationale:
      - Code is the contract; Excel is data.
      - Kernels rely on stable integer codes.
    """
    # Expected mapping from enum
    expected: dict[str, int] = {
        name: int(member) for name, member in ShiftType.__members__.items()
    }

    # Normalize "got" mapping to plain ints for stable comparison
    got: dict[str, int] = {str(k): int(v) for k, v in shift_types_dict.items()}

    if got == expected:
        return

    missing = {k: expected[k] for k in expected.keys() - got.keys()}
    extra = {k: got[k] for k in got.keys() - expected.keys()}
    different = {
        k: {"expected": expected[k], "got": got[k]}
        for k in (expected.keys() & got.keys())
        if expected[k] != got[k]
    }

    raise ValueError(
        "shift_types sheet does not match riskquant ShiftType enum.\n"
        f"missing={missing}\n"
        f"extra={extra}\n"
        f"different={different}\n"
        "Fix the Excel shift_types table or update riskquant.market_data.risk_factors.shift_types.ShiftType."
    )

# riskquant/market_data/risk_factors/snapshot.py

from __future__ import annotations

from dataclasses import dataclass
from datetime import date

import numpy as np
import pandas as pd


@dataclass(frozen=True, slots=True)
class RiskFactorLevelSnapshot:
    """Immutable snapshot of risk factor levels at a single as-of date.

    Attributes
    ----------
    as_of_date : datetime.date
        The calendar date for which the snapshot applies.
    factor_ids : np.ndarray[int32]
        1-D array of unique risk-factor identifiers (dtype int32, C-contiguous).
        Must be sorted ascending; if not sorted, the constructor will re-order
        the ids and corresponding `factor_levels`.
    factor_levels : np.ndarray[float32|float64]
        1-D numeric array of factor levels aligned with `factor_ids`.
        Must be the same length as `factor_ids` and C-contiguous.

    Invariants (enforced on construction)
    -------------------------------------
    - `factor_ids` is 1D, dtype int32, C-contiguous, sorted ascending and unique
    - `factor_levels` is 1D, dtype float32 or float64, C-contiguous
    - lengths of `factor_ids` and `factor_levels` match

    The class is intentionally lightweight and validates eagerly so downstream
    numerical kernels can assume compact, contiguous, and deterministic
    memory layouts (fast gathers, stable hashing, etc.).
    """

    as_of_date: date
    factor_ids: np.ndarray  # (N0,) int32
    factor_levels: np.ndarray  # (N0,) float32/float64

    def __post_init__(self) -> None:
        """Validate and canonicalize snapshot arrays.

        The method performs cheap, early checks and performs the minimal amount
        of work required to restore canonical invariants (sorting + contiguity).
        It raises clear exceptions on invalid input rather than attempting to
        silently coerce data into a valid form.
        """
        # Local aliases to make subsequent checks concise and explicit.
        ids = self.factor_ids
        lv = self.factor_levels

        # ----- shape checks -----
        # Ensure both arrays are 1-D vectors.
        if ids.ndim != 1:
            raise ValueError(f"factor_ids must be 1D; got shape={ids.shape}")
        if lv.ndim != 1:
            raise ValueError(f"factor_levels must be 1D; got shape={lv.shape}")

        # Ensure arrays have the same length (alignment requirement).
        if ids.shape[0] != lv.shape[0]:
            raise ValueError(
                f"factor_ids and factor_levels length mismatch: "
                f"{ids.shape[0]} vs {lv.shape[0]}"
            )

        # ----- dtype checks -----
        # factor_ids must be 32-bit integers for compactness and predictable
        # behavior when used as indices or keys in mappings.
        if ids.dtype != np.int32:
            raise TypeError(f"factor_ids must be int32; got {ids.dtype}")

        # factor_levels must be floating point (single or double precision).
        if lv.dtype not in (np.float32, np.float64):
            raise TypeError(f"factor_levels must be float32 or float64; got {lv.dtype}")

        # ----- memory layout checks -----
        # Many numerical routines assume C-contiguous 1-D arrays; check early so
        # callers get an explicit error instead of subtle performance bugs.
        if not ids.flags["C_CONTIGUOUS"]:
            raise ValueError("factor_ids must be C-contiguous")
        if not lv.flags["C_CONTIGUOUS"]:
            raise ValueError("factor_levels must be C-contiguous")

        # Fast path: zero or one element requires no further sorting/uniqueness checks.
        n = ids.size
        if n <= 1:
            return

        # ----- sorted + unique enforcement -----
        # If ids are already non-decreasing, perform an adjacent-duplicate check
        # which is allocation-free. If not sorted, build a stable permutation and
        # reorder both arrays so the object is canonical afterwards.
        if not np.any(ids[1:] < ids[:-1]):
            # ids are non-decreasing — ensure strict uniqueness.
            if np.any(ids[1:] == ids[:-1]):
                raise ValueError("factor_ids must be unique")
            return

        # ids are unsorted: compute a stable sort permutation (mergesort).
        perm = np.argsort(ids, kind="mergesort")

        # Apply the permutation to ids (produces a new array view/copy).
        ids_sorted = ids[perm]

        # Check for duplicates after sorting (adjacent duplicates are easy to detect).
        if np.any(ids_sorted[1:] == ids_sorted[:-1]):
            raise ValueError("factor_ids must be unique")

        # Reorder factor levels to keep alignment with the now-sorted ids.
        lv_sorted = lv[perm]

        # Fancy indexing typically returns a contiguous array for 1-D, but
        # explicitly enforce C-contiguity to preserve the class invariant.
        if not lv_sorted.flags["C_CONTIGUOUS"]:
            lv_sorted = np.ascontiguousarray(lv_sorted)

        # Bypass dataclass immutability to install the canonicalized arrays.
        object.__setattr__(self, "factor_ids", ids_sorted)
        object.__setattr__(self, "factor_levels", lv_sorted)

    def to_df(self) -> pd.DataFrame:
        """Return the snapshot as a 1-row wide DataFrame."""
        # Reshape the 1D levels array into a 2D row vector (1, N)
        data_2d = self.factor_levels[np.newaxis, :]

        df = pd.DataFrame(
            data=data_2d,
            index=pd.DatetimeIndex([self.as_of_date], name="date"),
            columns=pd.Index(self.factor_ids, name="factor_id"),
        )
        return df

    def to_series(self) -> pd.Series:
        """Return the snapshot as a pandas Series."""
        return pd.Series(
            data=self.factor_levels,
            index=pd.Index(self.factor_ids, name="factor_id"),
            name=self.as_of_date,
        )

    @property
    def df(self) -> pd.DataFrame:
        """Alias for to_df()."""
        return self.to_df()

    @property
    def sr(self) -> pd.Series:
        """Alias for to_series()."""
        return self.to_series()

    @property
    def num_factors(self) -> int:
        return len(self.factor_ids)

    def summary(self) -> None:
        """Print a console-friendly overview of the snapshot."""
        lines = [
            "=======================================",
            "=== RiskFactorLevelSnapshot Summary ===",
            "=======================================",
            f"  As of date          : {self.as_of_date}",
            f"  Number of factors   : {self.num_factors}",
            f"  Factor ID range     : {self.factor_ids.min()} to {self.factor_ids.max()}",
            f"  Factor level range  : {self.factor_levels.min()} to {self.factor_levels.max()}",
            "=======================================",
        ]
        print("\n".join(lines))

# src/riskquant/market_data/state.py
"""
Vectorized market data container for the valuation engine.

`MarketDataMatrix` is the frozen, valuation-ready SoA representation of all
market data required to price a `CompiledPortfolio` across `num_scenarios`
parallel scenarios.

Axis conventions
----------------
* Dimension 0 always corresponds to a `_uix` (price / fx_effective / vol /
  discount universe).
* For 2-D scalar grids the final dimension is the scenario index.
* 3-D grids (`discount_factor_grid`, `price_curve_grid`) carry a middle
  date-uix axis aligned with `CompiledPortfolio.unique_dates_ord32`, and
  scenario in the trailing axis.
* `vol_surfaces` is 4-D and arranged as `[surface, expiry, scenario, pillar]`
  so that the strike-axis pillars for one trade-scenario are a single
  contiguous run of memory — see Vol-surface layout below.

FX semantics
------------
`fx_spots_effective` holds the deduplicated, fully-resolved FX transform
matrix. Row `k` is the spot transform identified by row `k` of the
portfolio's `UixMapper.fx_effective_pairs`. Trades that need no FX carry
`INVALID_UIX` (< 0) in their per-trade `fx_*_effective_uix` field and skip
the FX gather entirely via a hoisted branch in the pricer.

Pricer contract for FX:
    out[i, s] *= fx_spots_effective[fx_effective_uix[i], s]

No inversion math, no `1.0 / rate`, no scenario-level branches. The raw
`fx_spots` matrix is producer-side only and never reaches the pricers.

Vol-surface layout (CRITICAL)
-----------------------------
`vol_surfaces` is `(num_vol_surfaces, num_unique_dates, num_scenarios,
max_vol_strikes)` — pillar in the trailing axis. This guarantees that
the strike pillars for one (surface, expiry, scenario) tuple form one
C-contiguous run, which is exactly what the strike-axis interpolation
in the option-curve kernel needs for SIMD-friendly reads.

The market-data builder is responsible for emitting this layout; the
kernels rely on it for `v_plane[s]` to be a contiguous 1-D pillar row.
"""

from __future__ import annotations

from dataclasses import dataclass, field

import numpy as np
import numpy.typing as npt


@dataclass(frozen=True, slots=True, kw_only=True)
class MarketDataMatrix:
    """Frozen, validated SoA container of market data for vectorized valuation."""

    num_scenarios: int

    # ========== 1. SCALARS / NODES ==========

    # (num_fx_effective_pairs, num_scenarios) — deduplicated, pre-resolved FX
    # transforms. Pricers gather one row per leg per trade and multiply.
    fx_spots_effective: npt.NDArray[np.float64]

    # (num_price_factors, num_scenarios)
    prices: npt.NDArray[np.float64]

    # (num_fwd_nodes, num_scenarios)
    forward_nodes: npt.NDArray[np.float64] = field(
        default_factory=lambda: np.empty((0, 0), dtype=np.float64),
    )

    # (num_vol_nodes, num_scenarios)
    vol_nodes: npt.NDArray[np.float64] = field(
        default_factory=lambda: np.empty((0, 0), dtype=np.float64),
    )

    # ========== 2. DISCOUNT GRID ==========
    # (num_discount_curves, num_unique_dates, num_scenarios)
    discount_factor_grid: npt.NDArray[np.float64] = field(
        default_factory=lambda: np.empty((0, 0, 0), dtype=np.float64),
    )

    # ========== 3. CURVE-BASED DATA ==========
    # (num_price_curves, num_unique_dates, num_scenarios)
    price_curve_grid: npt.NDArray[np.float64] = field(
        default_factory=lambda: np.empty((0, 0, 0), dtype=np.float64),
    )

    # ========== 4. TIME GRIDS ==========
    # (num_unique_dates,) — year fractions from as_of to each unique date.
    tte_grid: npt.NDArray[np.float64] = field(
        default_factory=lambda: np.empty(0, dtype=np.float64),
    )

    sqrt_tte_grid: npt.NDArray[np.float64] = field(init=False)

    half_tte_grid: npt.NDArray[np.float64] = field(init=False)

    # ========== 5. SURFACES ==========
    # (num_vol_surfaces, max_vol_strikes) — strike pillars per surface.
    vol_surface_strikes: npt.NDArray[np.float64] = field(
        default_factory=lambda: np.empty((0, 0), dtype=np.float64),
    )

    # (num_vol_surfaces, num_unique_dates, num_scenarios, max_vol_strikes)
    # Pillar-in-trailing-axis layout — see module docstring.
    vol_surfaces: npt.NDArray[np.float64] = field(
        default_factory=lambda: np.empty((0, 0, 0, 0), dtype=np.float64),
    )

    def __post_init__(self):

        object.__setattr__(
            self,
            "sqrt_tte_grid",
            np.sqrt(self.tte_grid),
        )

        object.__setattr__(
            self,
            "half_tte_grid",
            0.5 * self.tte_grid,
        )
