# src/riskquant/ingest_fixtures/config/risk_static_config/_parts/curve_point_meta.py

from __future__ import annotations

import pandas as pd

from riskquant.ingest_fixtures.schemas import RiskStaticConfigCurvePointMeta
from riskquant.ingest_fixtures.utils import validate_column_lengths

# rf_id allocation per curve. Each curve gets a contiguous block; the
# first rf_id of the block is `_rf_id_base`, and points are emitted in
# the order their tenor_tokens appear in `_tenor_tokens`.
#
# rf_id partition (must not collide with node_meta or surface_point_meta):
#   nodes          : 0 ... 99_999
#   curve_points   : 100_000 ... 199_999
#   surface_points : 200_000 ... 299_999
#
# Tenor labels follow the IR conventional schedule: [ON]..[10Y].
# Monthly commodities use OFFSET_M tokens [M+k]; annual carbon uses
# OFFSET_Y tokens [Y+k]; half-year power uses OFFSET_HY tokens [HY+k].

_LABEL_TENORS: tuple[str, ...] = (
    "[ON]",
    "[1W]",
    "[1M]",
    "[3M]",
    "[6M]",
    "[1Y]",
    "[2Y]",
    "[5Y]",
    "[10Y]",
)

_MONTHLY_OFFSETS: tuple[str, ...] = (
    "[M+0]",
    "[M+1]",
    "[M+2]",
    "[M+3]",
    "[M+6]",
    "[M+12]",
)

_ANNUAL_OFFSETS: tuple[str, ...] = (
    "[Y+0]",
    "[Y+1]",
    "[Y+2]",
    "[Y+3]",
)

_HALF_YEAR_OFFSETS: tuple[str, ...] = (
    "[HY+0]",
    "[HY+1]",
    "[HY+2]",
    "[HY+3]",
)


# (curve_id, rf_id_base, tenor_tokens)
_CURVE_BLOCKS: tuple[tuple[int, int, tuple[str, ...]], ...] = (
    # --- USD discount curves ---
    (100_000_000, 1000, _LABEL_TENORS),  # USD_SOFR_OIS
    (100_000_001, 1010, _LABEL_TENORS),  # USD_LIBOR_3M
    (100_000_002, 1020, _LABEL_TENORS),  # USD_CSA_EUR
    # --- EUR discount curves ---
    (100_000_003, 1100, _LABEL_TENORS),  # EUR_ESTR_OIS
    (100_000_004, 1110, _LABEL_TENORS),  # EUR_EURIBOR_6M
    # --- GBP discount curves ---
    (100_000_005, 1200, _LABEL_TENORS),  # GBP_SONIA_OIS
    # --- CHF discount curves ---
    (100_000_006, 1300, _LABEL_TENORS),  # CHF_SARON_OIS
    # --- AUD discount curves ---
    (100_000_007, 1400, _LABEL_TENORS),  # AUD_AONIA_OIS
    # --- Commodity forward curves ---
    (100_000_008, 2000, _MONTHLY_OFFSETS),  # BRENT_CRUDE_FWD
    (100_000_009, 2010, _MONTHLY_OFFSETS),  # WTI_CRUDE_FWD
    (100_000_010, 2020, _MONTHLY_OFFSETS),  # HENRY_HUB_GAS_FWD
    (100_000_011, 2100, _ANNUAL_OFFSETS),  # CO2_EUA_ANNUAL_FWD
    (100_000_012, 2200, _HALF_YEAR_OFFSETS),  # POWER_CH_BASELOAD_HY
)


def _get_data() -> RiskStaticConfigCurvePointMeta:
    rf_ids: list[int] = []
    curve_ids: list[int] = []
    tenor_tokens: list[str] = []

    for curve_id, rf_id_base, tokens in _CURVE_BLOCKS:
        for offset, token in enumerate(tokens):
            rf_ids.append(rf_id_base + offset)
            curve_ids.append(curve_id)
            tenor_tokens.append(token)

    return {
        "rf_id": rf_ids,
        "curve_id": curve_ids,
        "tenor_token": tenor_tokens,
    }


CANONICAL_CURVE_POINT_META_DATA: RiskStaticConfigCurvePointMeta = _get_data()

validate_column_lengths(CANONICAL_CURVE_POINT_META_DATA)

_CANONICAL_CURVE_POINT_META_DF = pd.DataFrame(CANONICAL_CURVE_POINT_META_DATA)

        print_success_message,
    )

    print_calling_file()

    print_dict(CANONICAL_CURVE_POINT_META_DATA)

    print("As DataFrame:")
    print_df(_CANONICAL_CURVE_POINT_META_DF, index=False)

    print_success_message(add_space_before=False)


if __name__ == "__main__":
    _debug_print()

# src/riskquant/ingest_fixtures/credit_factors/minimal_factor_universe.py
"""
Canonical credit-factor REGISTRY for tests and demos.

This file is structural metadata, NOT historical data. It defines:
- the set of factor *names* that obligor JTD ingest cases may reference,
- a name -> int32 UIX mapping (`factor_name_to_uix`) that
  `compile_jtd_portfolio_from_df` uses to dereference factor names into
  `CreditFactorMatrix` row indices,
- which UIX positions carry a chi-square mixing variable (t-copula), and
  the degrees of freedom for each one.

The credit code path is currently parametric-simulation only — the
copula generators draw fresh latent factors every run from
`np.random.default_rng`. No part of the credit layer consumes a
historical time series of credit-factor levels.

If/when historical or FHS scenario generation lands for credit, a
parallel `ingest_cases/credit_factors/historical/` would supply
`(T, F_credit)` factor observations through the existing
`market_data/risk_factors/matrix.py::RiskFactorMatrix` container —
F-contiguous, int32 ordinal dates, int32 factor ids — which is already
the right shape for credit history too. Until then, only the registry
metadata below is required.

Chi/nu alignment (mathematical invariant)
-----------------------------------------
Each chi-square factor in the registry has its own dof
(`nu_by_chi[c]`). Every obligor that references a chi factor MUST
have `obligor.nu == nu_by_chi[chi_uix]` — otherwise the kernel's
`sqrt(nu_i / W_i)` transform breaks the t-marginal. The canonical
t-copula ingest case is built per-obligor-chi to satisfy this:
obligor i references CHI_i with dof matching its own nu_i.
"""

from __future__ import annotations

from typing import Final

import numpy as np
import numpy.typing as npt

# ==============================================================================
# FACTOR NAMES (UIX = position in this tuple)
# ==============================================================================

CANONICAL_CREDIT_FACTOR_NAMES: Final[tuple[str, ...]] = (
    "GLOBAL_CREDIT",  # 0  — systemic Z (Gaussian, t)
    "IDIO_1",  # 1  — idiosyncratic eps (obligor 1)
    "IDIO_2",  # 2  — idiosyncratic eps (obligor 2)
    "IDIO_3",  # 3  — idiosyncratic eps (obligor 3)
    "CHI_1",  # 4  — chi-square W_1 for obligor 1 (nu=5)
    "CHI_2",  # 5  — chi-square W_2 for obligor 2 (nu=7)
    "CHI_3",  # 6  — chi-square W_3 for obligor 3 (nu=10)
)

CANONICAL_CREDIT_FACTOR_COUNT: Final[int] = len(CANONICAL_CREDIT_FACTOR_NAMES)

CANONICAL_CREDIT_FACTOR_NAME_TO_UIX: Final[dict[str, int]] = {
    name: i for i, name in enumerate(CANONICAL_CREDIT_FACTOR_NAMES)
}

CANONICAL_CREDIT_FACTOR_IDS: Final[npt.NDArray[np.int32]] = np.arange(
    CANONICAL_CREDIT_FACTOR_COUNT, dtype=np.int32
)

# ==============================================================================
# T-COPULA CHI CONFIGURATION
# ==============================================================================
# One chi-square factor per obligor (uids 1, 2, 3). Each obligor's nu_i
# matches the dof of its chi factor — the only configuration that yields
# a true Student-t marginal under `sqrt(nu_i / W_i) * (beta*Z + ... )`.

CANONICAL_T_COPULA_CHI_COLUMNS: Final[npt.NDArray[np.int32]] = np.array(
    [
        CANONICAL_CREDIT_FACTOR_NAME_TO_UIX["CHI_1"],
        CANONICAL_CREDIT_FACTOR_NAME_TO_UIX["CHI_2"],
        CANONICAL_CREDIT_FACTOR_NAME_TO_UIX["CHI_3"],
    ],
    dtype=np.int32,
)


CANONICAL_T_COPULA_CHI_FACTOR_TO_NU_REGISTRY: Final[dict[str, float]] = {
    "CHI_1": 5.0,
    "CHI_2": 7.0,
    "CHI_3": 10.0,
}


# Aligned to CANONICAL_T_COPULA_CHI_COLUMNS by index. The canonical

# src/riskquant/ingest_fixtures/market_data/canonical_market_data_history.py
"""
Canonical historical market data fixture.

Builds a synthetic `(T, N)` historical risk-factor matrix whose columns are
generated dynamically from `CANONICAL_RISK_STATIC_CONFIG_DATA` — guaranteeing
referential integrity with the static metadata.

Math
----
- Equity / FX / commodity spot-like factors follow geometric Brownian motion
  with drift `-0.5 * sigma**2 * dt` so simulated levels remain unbiased in
  log-space.
- Rate factors follow arithmetic Brownian motion (random walk).

Memory layout
-------------
The output matrix is allocated F-contiguous so a downstream caller doing
column-major time-series access reads cache-aligned. `np.cumsum(..., out=...)`
and `np.exp(..., out=...)` mutate the buffer in place; no intermediates.

Output dtype is `float32` to match `build_risk_factor_matrix_from_df`'s
default — generating in float64 would just be downcast immediately.

TODO: split this file into `_parts/{nodes,curves,surfaces}.py` once the
universe count grows past comfort. The dynamic-build pattern makes the split
mechanical.
"""

from __future__ import annotations

import logging
from typing import Final

import numpy as np
import numpy.typing as npt
import pandas as pd

from riskquant.ingest_fixtures.as_of import CANONICAL_AS_OF_DATE
from riskquant.ingest_fixtures.config.risk_static_config.canonical_risk_static_config import (
    CANONICAL_RISK_STATIC_CONFIG_DATA,
)
from riskquant.ingest_fixtures.constants import (
    BUSINESS_DAYS_PER_YEAR,
    CANONICAL_HISTORY_BUSINESS_DAYS,
    EQUITY_DEFAULTS,
    FX_DEFAULTS,
    GAS_COMMODITY_DEFAULTS,
    HISTORY_DTYPE,
    OIL_COMMODITY_DEFAULTS,
    RATE_DEFAULTS,
    VOL_OF_VOL_DEFAULTS,
    _MarketRiskFactorParams,
)
from riskquant.ingest_fixtures.rng import get_canonical_rng

_logger: logging.Logger = logging.getLogger(__name__)


# ==============================================================================
# Time series setup
# ==============================================================================

HISTORICAL_DATES: Final[npt.NDArray[np.datetime64]] = np.sort(
    np.busday_offset(
        CANONICAL_AS_OF_DATE,
        -np.arange(CANONICAL_HISTORY_BUSINESS_DAYS - 1, -1, -1),
    )
)


# ==============================================================================
# Universe build (dynamic — derived from the static config, not parallel-listed)
# ==============================================================================


def _build_canonical_universe() -> list[tuple[str, _MarketRiskFactorParams]]:
    """
    Derive the full historical universe from `CANONICAL_RISK_STATIC_CONFIG_DATA`.
    Returns `[(factor_name, params), ...]` in node->curve_points->surface_points
    order.
    """
    specs: list[tuple[str, _MarketRiskFactorParams]] = []

    # ---- 1. Nodes ----
    node_meta = CANONICAL_RISK_STATIC_CONFIG_DATA["node_meta"]
    for node_name, asset_class in zip(
        node_meta["node_name"], node_meta["asset_class"], strict=True
    ):
        if asset_class == "EQUITY":
            specs.append((str(node_name), EQUITY_DEFAULTS))
        else:
            specs.append((str(node_name), FX_DEFAULTS))

    # ---- 2. Curve points ----
    curve_meta = CANONICAL_RISK_STATIC_CONFIG_DATA["curve_meta"]
    curve_info: dict[object, tuple[str, str]] = {
        cid: (str(name), str(utype))
        for cid, name, utype in zip(
            curve_meta["curve_id"],
            curve_meta["curve_name"],
            curve_meta["unit_type"],
            strict=True,
        )
    }

    curve_point_meta = CANONICAL_RISK_STATIC_CONFIG_DATA["curve_point_meta"]
    for cid, tenor in zip(
        curve_point_meta["curve_id"],
        curve_point_meta["tenor_token"],
        strict=True,
    ):
        curve_name, unit_type = curve_info[cid]
        factor_name = f"{curve_name}{tenor}"

        if unit_type == "RATE":
            specs.append((factor_name, RATE_DEFAULTS))
        elif "GAS" in curve_name:
            specs.append((factor_name, GAS_COMMODITY_DEFAULTS))
        else:
            specs.append((factor_name, OIL_COMMODITY_DEFAULTS))

    # ---- 3. Surface points ----
    surface_meta = CANONICAL_RISK_STATIC_CONFIG_DATA["surface_meta"]
    surface_info: dict[object, tuple[str, str]] = {
        sid: (str(name), str(y_kind))
        for sid, name, y_kind in zip(
            surface_meta["surface_id"],
            surface_meta["surface_name"],
            surface_meta["y_axis_kind"],
            strict=True,
        )
    }

    surface_point_meta = CANONICAL_RISK_STATIC_CONFIG_DATA["surface_point_meta"]
    for sid, x_tenor, y_label in zip(
        surface_point_meta["surface_id"],
        surface_point_meta["x_tenor_token"],
        surface_point_meta["y_label"],
        strict=True,
    ):
        surface_name, y_axis_kind = surface_info[sid]
        factor_name = f"{surface_name}{x_tenor}[{y_axis_kind}={y_label}]"
        specs.append((factor_name, VOL_OF_VOL_DEFAULTS))

    return specs


_FACTOR_SPECS: Final[list[tuple[str, _MarketRiskFactorParams]]] = (
    _build_canonical_universe()
)

CANONICAL_HISTORICAL_FACTOR_NAMES: Final[list[str]] = [
    name for name, _ in _FACTOR_SPECS
]
CANONICAL_HISTORICAL_FACTOR_COUNT: Final[int] = len(CANONICAL_HISTORICAL_FACTOR_NAMES)


# ==============================================================================
# Path generation (in-place, F-contiguous, batched RNG draw)
# ==============================================================================

_rng: np.random.Generator = get_canonical_rng()
_dt: float = 1.0 / BUSINESS_DAYS_PER_YEAR
_sqrt_dt: float = float(np.sqrt(_dt))

# Pre-allocate F-contiguous so per-column traversal is contiguous in memory.
_historical_values: npt.NDArray[np.floating] = np.empty(
    (CANONICAL_HISTORY_BUSINESS_DAYS, CANONICAL_HISTORICAL_FACTOR_COUNT),
    dtype=HISTORY_DTYPE,
    order="F",
)

# Single batched draw — avoids per-column RNG state thrash.
_Z: npt.NDArray[np.floating] = _rng.standard_normal(
    (CANONICAL_HISTORY_BUSINESS_DAYS, CANONICAL_HISTORICAL_FACTOR_COUNT),
    dtype=HISTORY_DTYPE,
)

for _i, (_, _params) in enumerate(_FACTOR_SPECS):
    _col = _historical_values[:, _i]
    _z_col = _Z[:, _i]
    _sigma = _params.annualized_vol

    if _params.is_rate:
        # Arithmetic Brownian motion: level_t = s0 + cumsum(sigma * sqrt(dt) * Z)
        np.cumsum(_sigma * _sqrt_dt * _z_col, out=_col)
        _col += _params.s0
    else:
        # Geometric Brownian motion in log-space, then exponentiate in place.
        _drift = -0.5 * (_sigma * _sigma) * _dt
        _log_returns = _drift + (_sigma * _sqrt_dt * _z_col)
        np.cumsum(_log_returns, out=_col)
        np.exp(_col, out=_col)
        _col *= _params.s0

_logger.debug(
    "Generated %d business-day history for %d canonical factors (dtype=%s)",
    CANONICAL_HISTORY_BUSINESS_DAYS,
    CANONICAL_HISTORICAL_FACTOR_COUNT,
    HISTORY_DTYPE.__name__,
)


# ==============================================================================
# DataFrame view (edge-only — strip before crossing into the engine)
# ==============================================================================

CANONICAL_HISTORICAL_MARKET_DATA_DF: Final[pd.DataFrame] = pd.DataFrame(
    _historical_values,
    index=HISTORICAL_DATES,
    columns=CANONICAL_HISTORICAL_FACTOR_NAMES,
)


# ==============================================================================
# Diagnostics
# ==============================================================================


def _debug_print() -> None:
    from riskquant.utils.printer import (
        print_calling_file,
        print_df,
        print_success_message,
    )

    print_calling_file()

    print("=========================================")
    print("=== HISTORICAL RISK FACTOR MATRIX DF  ===")
    print("=========================================")
    print(
        f"Shape: {CANONICAL_HISTORICAL_MARKET_DATA_DF.shape} "
        f"(T={CANONICAL_HISTORY_BUSINESS_DAYS}, "
        f"N={CANONICAL_HISTORICAL_FACTOR_COUNT}, dtype={HISTORY_DTYPE.__name__})"
    )

    print_df(
        CANONICAL_HISTORICAL_MARKET_DATA_DF.iloc[:, :5].head(5),
        title="Head (First 5 Factors)",
        index=True,
        float_format="{:.4f}",
    )
    print_df(
        CANONICAL_HISTORICAL_MARKET_DATA_DF.iloc[:, :5].tail(5),
        title="Tail (First 5 Factors)",
        index=True,
        float_format="{:.4f}",
    )

    print_success_message(add_space_before=False)


if __name__ == "__main__":
    _debug_print()

def _validate_single_table(data: Mapping[str, Sequence[Any]]) -> None:
    lengths = {len(v) for v in data.values()}

    if len(lengths) > 1:
        raise ValueError(f"Inconsistent column lengths: {lengths}")


def validate_column_lengths(data: Any) -> None:
    """
    Validates:
    - flat columnar dicts: Dict[str, Sequence]
    - nested config: Dict[str, Dict[str, Sequence]]
    """

    # Case 1: Nested (RiskStaticConfigData)
    if all(isinstance(v, Mapping) for v in data.values()):
        for k, sub_table in data.items():
            if not isinstance(sub_table, Mapping):
                raise TypeError(f"{k} is not a valid table")

            _validate_single_table(cast(Mapping[str, Sequence[Any]], sub_table))
        return

    # Case 2: Flat table
    _validate_single_table(cast(Mapping[str, Sequence[Any]], data))


def generate_netting_set_ids(
    *,
    n: int,
    none_probability: float = 0.2,
) -> list[int | None]:

    import numpy as np

    from riskquant.ingest_fixtures.netting_set.minimal_netting_set_master import (
        CANONICAL_NETTING_SET_UIDS,
    )
    from riskquant.ingest_fixtures.rng import get_canonical_rng

    _rng: np.random.Generator = get_canonical_rng()

    # 1. Generate the remaining IDs from valid options
    ns_ids = _rng.choice(
        a=CANONICAL_NETTING_SET_UIDS,
        size=n,
        replace=True,
    ).tolist()

    # 2. Randomly assign None based on the specified probability
    none_mask = _rng.random(size=n) < none_probability
    for i in range(n):
        if none_mask[i]:
            ns_ids[i] = None

    # 3. Combine and shuffle (shuffle works in-place)
    _rng.shuffle(ns_ids)

    return ns_ids
