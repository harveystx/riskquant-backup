```python
# src/riskquant/portfolio/blocks/base_block.py

"""
Base class for all Struct-of-Arrays (SoA) trade blocks.

`Block` carries only what is universal to every compiled block: the flat int64
`trade_uids` array and the `INSTRUMENT_ID` dispatch tag. All economic SoA arrays
live on the concrete subclasses.

`INSTRUMENT_ID` is the single integer the valuation engine dispatches on: it
indexes directly into `BLOCK_EVALUATORS`, so no `type()` calls, no isinstance
ladders, no string keys. The base value is the `_UNSET` sentinel; every
concrete block MUST override it with its `Instrument` enum id. The override is
enforced at class-creation time by `__init_subclass__`, which fails loudly the
moment a new block type is added without wiring its id — a compile-time guard
rather than a runtime surprise.
"""

from __future__ import annotations

from dataclasses import dataclass, fields
from typing import Any, ClassVar

import numpy as np
import numpy.typing as npt

# Sentinel for "concrete subclass forgot to set INSTRUMENT_ID". Negative so it
# can never collide with a real Instrument enum id or index a dispatch table.
_UNSET_INSTRUMENT_ID: int = -999


@dataclass(frozen=True, slots=True, kw_only=True)
class Block:
    """Universal base for compiled SoA trade blocks."""

    # Dispatch tag. ClassVar => not a dataclass field, zero per-instance cost.
    INSTRUMENT_ID: ClassVar[int] = _UNSET_INSTRUMENT_ID

    trade_uids: npt.NDArray[np.int64]

    def __init_subclass__(cls, **kwargs: object) -> None:
        """
        Enforce the dispatch contract for *concrete* block subclasses.

        Intermediate abstract bases (names starting with '_') are exempt: they
        legitimately leave `INSTRUMENT_ID` unset for their own subclasses to
        define. Any concrete block that ships without an id is a wiring bug and
        is rejected here, at import time.
        """
        # super().__init_subclass__(**kwargs)
        is_abstract_base = cls.__name__.startswith("_")
        if not is_abstract_base and cls.INSTRUMENT_ID == _UNSET_INSTRUMENT_ID:
            raise TypeError(
                f"Block subclass '{cls.__name__}' must set INSTRUMENT_ID "
                f"(its riskquant.portfolio.instruments id) for engine dispatch."
            )

    @property
    def num_trades(self) -> int:
        """Number of trades packed into this block."""
        return self.trade_uids.shape[0]

    def print_block(
        self,
        *,
        max_items: int | None = 10,
        float_precision: int = 4,
        sep: str = ", ",
        pad: int = 15,  # Increased default pad slightly to account for the new '_' characters
        thousand_sep: str = "_",
    ) -> None:
        """Print every SoA field with shape, dtype, and a value preview."""

        print(f"{type(self).__name__}:")
        for f in fields(self):
            v = getattr(self, f.name)
            if isinstance(v, np.ndarray):
                # Replace lambdas with typed inner functions to satisfy Pyright/MyPy
                def fmt_float(x: Any) -> str:
                    return f"{x:>{pad}{thousand_sep}.{float_precision}f}"

                def fmt_int(x: Any) -> str:
                    # Use 'd' (decimal integer) instead of a hacked float format
                    return f"{x:>{pad}{thousand_sep}d}"

                # Format the array preview cleanly
                val_preview = np.array2string(
                    v[:max_items],
                    separator=sep,
                    max_line_width=100_000,
                    formatter={
                        "float_kind": fmt_float,
                        "int_kind": fmt_int,
                    },
                )

                print(
                    f"  {f.name:<33} | shape: {v.shape!s:>10} | "
                    f"dtype: {v.dtype!s:>10} | values: {val_preview}"
                )
            else:
                print(f"  {f.name:<33} {v}")
        print()
```

```python
# src/riskquant/portfolio/blocks/spot_block.py
"""
Compiled, memory-efficient container for spot positions.

`SpotBlock` is the struct-of-arrays representation of a set of spot trades.
The block is intentionally compact (NumPy arrays, primitive numeric dtypes)
so it can be held efficiently in memory and passed between processes in
high-throughput valuation workflows.

Architectural contract
----------------------
* Row `i` of every array corresponds to the trade with UID `trade_uids[i]`.
* Every `_uix` field is a zero-based, contiguous index produced by the
  portfolio engine's `UixMapper` — NOT a global static-config ID. The block
  carries execution coordinates only.
* `fx_spots_effective_uix` indexes into `MarketDataMatrix.fx_spots_effective`,
  the deduplicated effective FX-transform matrix. No invert flag is carried;
  the inversion (if any) has been folded into the effective matrix at the
  builder edge. Pricer kernels do one unconditional gather + multiply.
* `INVALID_UIX (< 0)` in any `_uix` field signals "no factor required for
  this trade" and is the only branch the pricer kernel performs (hoisted
  outside the scenario loop).
"""

from __future__ import annotations

from dataclasses import dataclass
from typing import ClassVar

import numpy as np
import numpy.typing as npt

from riskquant.portfolio.blocks.base_block import Block
from riskquant.portfolio.instruments import SPOT_ID


@dataclass(frozen=True, slots=True, kw_only=True)
class SpotBlock(Block):
    """Struct-of-arrays for compiled spot trades."""

    INSTRUMENT_ID: ClassVar[int] = int(SPOT_ID)

    # ========== TRADE IDENTITY ==========
    # Unique trade identifiers, one per row.
    trade_uids: npt.NDArray[np.int64]

    # ========== ECONOMIC TERMS ==========
    # Signed quantities / notionals for the spot positions.
    qty: npt.NDArray[np.float64]

    # ========== MARKET LINKAGES (EXECUTION UIX) ==========
    # Index into `MarketDataMatrix.prices` for this row.
    price_uix: npt.NDArray[np.int32]

    # Index into `MarketDataMatrix.fx_spots_effective`, or INVALID_UIX
    # (< 0) when no FX conversion is required for this trade.
    fx_spots_effective_uix: npt.NDArray[np.int32]
```

```python
Compiled blocks for vanilla European option trades.

Architectural contract
----------------------
* Row `i` of every array corresponds to the trade with UID `trade_uids[i]`.

  block layer.
* `cp_mult` packs `is_call` into `+1` / `-1` for branchless Black-76 payoff.
* `INVALID_UIX (< 0)` in any optional `_uix` field signals "no factor
  required for this trade" and is the only branch the pricer kernel
  performs (hoisted outside the scenario loop).

from typing import ClassVar

import numpy.typing as npt

from riskquant.portfolio.instruments import (
    VANILLA_EUROPEAN_OPTION_CURVE_ID,
    VANILLA_EUROPEAN_OPTION_NODE_ID,
)


@dataclass(frozen=True, slots=True, kw_only=True)
class _VanillaEuropeanOptionBaseBlock(Block):
    """Common SoA fields shared by all vanilla European option blocks."""

    # ========== TRADE IDENTITY ==========
    trade_uids: npt.NDArray[np.int64]

    # ========== ECONOMIC TERMS ==========
    qty: npt.NDArray[np.float64]
    strike: npt.NDArray[np.float64]
    # +1 for call, -1 for put, precomputed for branchless payoff sign.
    cp_mult: npt.NDArray[np.int8]

    # ========== DATES (DATE UIX) ==========
    # Indices into `unique_dates_ord32`.
    expiry_date_uix: npt.NDArray[np.int32]
    settlement_date_uix: npt.NDArray[np.int32]

    # ========== FX LINKAGES (EFFECTIVE UIX) ==========
    # Indices into `MarketDataMatrix.fx_spots_effective`. INVALID_UIX (< 0)

    fx_under_to_settle_effective_uix: npt.NDArray[np.int32]
    fx_strike_to_settle_effective_uix: npt.NDArray[np.int32]

    # ========== DISCOUNT LINKAGES (CURVE UIX) ==========
    # Indices into `MarketDataMatrix.discount_factor_grid` axis 0.
    # INVALID_UIX (< 0) when no discount applies on that leg.
    under_discount_curve_uix: npt.NDArray[np.int32]
    strike_discount_curve_uix: npt.NDArray[np.int32]
```

```python
# src/riskquant/portfolio/compiled.py

"""
Compiled portfolio container.

`CompiledPortfolio` is a pure portfolio-layer artifact: the frozen, valuation-
ready output of block compilation. It holds compiled SoA blocks plus the
metadata the engine needs to walk them — and nothing from the valuation layer.

Layering
--------
This module imports only `portfolio.blocks`. It deliberately does NOT import
pricers, risk evaluators, or the `Evaluator` protocol. Evaluator resolution is
the valuation engine's responsibility: it dispatches on each block's
`INSTRUMENT_ID` class constant via its own `BLOCK_EVALUATORS` table.

The dependency therefore points one way only:

    valuation  ──▶  portfolio        (allowed)
    portfolio  ──▶  valuation        (never)

The previous design bound pricer callables into `_evaluators` inside
`__post_init__`, which made a data container reach up into the valuation layer
and created the `compiled <-> valuation.engine` import cycle. Removing those
references eliminates the cycle by construction — there is no edge left to
break, deferred-import trick or otherwise.
"""

from __future__ import annotations

from collections.abc import Iterator
from dataclasses import dataclass, field
from typing import TypeAlias

import numpy as np
import numpy.typing as npt

from riskquant.portfolio.blocks.spot_block import SpotBlock
from riskquant.portfolio.blocks.vanilla_european_option_block import (
    VanillaEuropeanOptionCurveBlock,
    VanillaEuropeanOptionNodeBlock,
)
from riskquant.portfolio.blocks.vanilla_forward_block import (
    VanillaForwardCurveBlock,
    VanillaForwardNodeBlock,
)

BlockT: TypeAlias = (
    # --- Market Risk ---
    SpotBlock
    | VanillaForwardNodeBlock
    | VanillaForwardCurveBlock
    | VanillaEuropeanOptionNodeBlock
    | VanillaEuropeanOptionCurveBlock
)


@dataclass(frozen=True, slots=True, kw_only=True)
class CompiledPortfolio:
    """
    Valuation-ready portfolio: an ordered set of compiled SoA blocks.

    Each populated block contributes one contiguous row-band to the eventual
    (n_trades x n_scenarios) valuation matrix; `block_offsets[i]` is the first
    row of block `i`. `trade_uids` aggregates every block's uids into one flat
    int64 array in the same order, so a portfolio-row index maps straight to a
    trade uid without per-block bookkeeping.
    """

    # Unique dates across the portfolio, as ordinals truncated to int32 for
    # engine-friendly storage and fast comparisons. Sorted ascending, unique.
    unique_dates_ord32: npt.NDArray[np.int32] = field(
        default_factory=lambda: np.empty(0, dtype=np.int32)
    )

    # Instrument blocks (SoA arrays); None blocks are compacted in __post_init__.
    spot: SpotBlock | None = None
    vanilla_forward_node: VanillaForwardNodeBlock | None = None
    vanilla_forward_curve: VanillaForwardCurveBlock | None = None
    vanilla_european_option_node: VanillaEuropeanOptionNodeBlock | None = None
    vanilla_european_option_curve: VanillaEuropeanOptionCurveBlock | None = None

    # Derived, set in __post_init__.
    _blocks: tuple[BlockT, ...] = field(init=False, repr=False)
    _block_offsets: npt.NDArray[np.int64] = field(init=False, repr=False)
    _total_blocks: int = field(init=False, repr=False)
    _total_trades: int = field(init=False, repr=False)
    _trade_uids: npt.NDArray[np.int64] = field(init=False, repr=False)

    def __post_init__(self) -> None:
        # Construction-ordered slots; None slots are compacted out below. This
        # tuple order defines block ordering and therefore output-matrix bands.
        ordered: tuple[BlockT | None, ...] = (
            self.spot,
            self.vanilla_forward_node,
            self.vanilla_forward_curve,
            self.vanilla_european_option_node,
            self.vanilla_european_option_curve,
        )
        blocks: tuple[BlockT, ...] = tuple(b for b in ordered if b is not None)
        total_blocks: int = len(blocks)

        # Offsets via exclusive prefix-sum of per-block trade counts.
        block_offsets = np.zeros(total_blocks, dtype=np.int64)
        trade_counts = np.fromiter(
            (b.num_trades for b in blocks), dtype=np.int64, count=total_blocks
        )
        total_trades = int(trade_counts.sum())
        if total_blocks > 1:
            np.cumsum(trade_counts[:-1], out=block_offsets[1:])

        # Aggregate every block's uids into one flat array in portfolio order.
        trade_uids = np.empty(total_trades, dtype=np.int64)
        offset = 0
        for block in blocks:
            n = block.num_trades
            trade_uids[offset : offset + n] = block.trade_uids
            offset += n

        object.__setattr__(self, "_blocks", blocks)
        object.__setattr__(self, "_block_offsets", block_offsets)
        object.__setattr__(self, "_total_blocks", total_blocks)
        object.__setattr__(self, "_total_trades", total_trades)
        object.__setattr__(self, "_trade_uids", trade_uids)

    # ------------------------------------------------------------------ #
    # Read-only accessors
    # ------------------------------------------------------------------ #

    @property
    def blocks(self) -> tuple[BlockT, ...]:
        return self._blocks

    @property
    def block_offsets(self) -> npt.NDArray[np.int64]:
        return self._block_offsets

    @property
    def total_blocks(self) -> int:
```

```python
# src/riskquant/portfolio/engine.py
"""
Top-level portfolio orchestrator and compiler.

The engine sits exactly on the boundary between Global State (sparse
`IngestBatch`es indexed by `static_config` Risk Factor IDs) and Local
Execution State (`CompiledPortfolio` blocks indexed by zero-based
contiguous UIX coordinates aligned with the `MarketDataMatrix`).

Pipeline
--------
1. `build_uix_mapper(ingest_batches)`
      Discovers the global universe touched by the run (market IDs,
      discount-curve IDs, dates) AND deduplicates every distinct
      `(fx_rf_uix, invert)` pair across all FX legs of every trade. The
      result, a `UixMapper`, is the single source of truth for ID->UIX
      translation downstream.

2. (Caller fetches market data using the discovered universes.)

3. `compile_portfolio_from_ingest_batch(ingest_batch, uix_mapper)`
      Translates every global ID and ordinal date in the batch into UIX
      execution coordinates, then partitions into per-instrument compiled
      blocks via the registered compiler table. Returns a frozen
      `CompiledPortfolio`.

Design invariants
-----------------
- `UixMapper` is immutable. Its `_universe` arrays are sorted, unique, and
  validated once at construction; runtime `map_*` calls trust the contract
  and use `np.searchsorted` for O(log N) vectorized lookup.
- `fx_effective_pairs` is built in the same pass as the market universe so
  the FX-effective UIX space is part of the global contract. Pricers will
  index a `(K, S)` `fx_spots_effective` matrix built by the market-data
  builder; `INVALID_UIX` (< 0) signals "no FX conversion required" and is
  the only branch in the pricer hot path.
- Compilers take `(batch, start, end, mapper)`. They are the ONLY place
  global IDs are translated to UIXs; blocks never see global IDs.
"""

from __future__ import annotations

from collections.abc import Sequence
from dataclasses import dataclass

import numpy as np
import numpy.typing as npt

from riskquant.constants import INVALID_UIX, NO_FX_RF_REQUIRED
from riskquant.portfolio.compiled import CompiledPortfolio
from riskquant.portfolio.compilers.registry import COMPILERS, FIELD_NAMES
from riskquant.portfolio.ingest.batch import IngestBatch

# ==============================================================================
# UIX TRANSLATION CONTEXT
# ==============================================================================


@dataclass(frozen=True, slots=True, kw_only=True)
class UixMapper:
    """
    Immutable execution-space translator.

    Holds the four global universes touched by the run plus the deduplicated
    FX-effective pair table. All universe arrays are sorted ascending and
    unique; the `__post_init__` validation runs once so the runtime `map_*`
    methods can stay allocation-light and branch-light.

    Field layout
    ------------
    market_id_universe          : sorted-unique int32 IDs used by
                                  prices / forwards / vols / fx-spots.
                                  Indexed by `price_uix`, `vol_uix`,
                                  `underlying_price_curve_uix`, etc.
    discount_curve_id_universe  : sorted-unique int32 IDs used by discount
                                  curves. Kept distinct from `market_id_universe`
                                  because the discount-curve ID range is
                                  disjoint and `MarketDataMatrix.discount_factor_grid`
                                  has its own axis-0 universe.
    date_ord32_universe         : sorted-unique int32 day-ordinals
                                  (days-since-epoch) touched by the run.
                                  Indexed by `*_date_uix` on blocks.
    fx_effective_pairs          : (K, 2) int32 deduplicated `(fx_rf_uix, invert)`
                                  table. `fx_rf_uix` is itself an index into
                                  `market_id_universe`; `invert` is 0 or 1.
                                  The market-data builder turns this into a
                                  `(K, S)` `fx_spots_effective` matrix.
    """

    market_id_universe: npt.NDArray[np.int32]
    discount_curve_id_universe: npt.NDArray[np.int32]
    date_ord32_universe: npt.NDArray[np.int32]
    fx_effective_pairs: npt.NDArray[np.int32]

    # --- INVARIANTS (CHECKED ONCE) -----------------------------------------

    def __post_init__(self) -> None:
        self._validate_universe(self.market_id_universe, "market_id_universe")
        self._validate_universe(
            self.discount_curve_id_universe, "discount_curve_id_universe"
        )
        self._validate_universe(self.date_ord32_universe, "date_ord32_universe")
        self._validate_fx_effective_pairs(self.fx_effective_pairs)

    @staticmethod
    def _validate_universe(arr: npt.NDArray[np.int32], name: str) -> None:
        if arr.dtype != np.int32:
            raise TypeError(f"{name} must be int32; got {arr.dtype}.")
        if arr.ndim != 1:
            raise ValueError(f"{name} must be 1-D; got ndim={arr.ndim}.")
        if not arr.flags.c_contiguous:
            raise ValueError(f"{name} must be C-contiguous.")
        if arr.size > 1 and np.any(arr[1:] <= arr[:-1]):
            raise ValueError(f"{name} must be strictly increasing and unique.")

    @staticmethod
    def _validate_fx_effective_pairs(pairs: npt.NDArray[np.int32]) -> None:
        if pairs.dtype != np.int32:
            raise TypeError(f"fx_effective_pairs must be int32; got {pairs.dtype}.")
        if pairs.ndim != 2 or (pairs.size and pairs.shape[1] != 2):
            raise ValueError(
                f"fx_effective_pairs must have shape (K, 2); got {pairs.shape}"
            )
        if not pairs.flags.c_contiguous:
            raise ValueError("fx_effective_pairs must be C-contiguous.")

    # --- TRANSLATION API ---------------------------------------------------

    def map_market_ids_to_uix(
        self,
        raw_ids: npt.NDArray[np.int32],
    ) -> npt.NDArray[np.int32]:
        """
        Translate global market IDs into `market_id_universe` UIXs.

        Sentinel rows (`raw_ids < 0`) map to `INVALID_UIX`.
        """
        return _searchsorted_with_sentinel(
            universe=self.market_id_universe,
            raw=raw_ids,
        )

    def map_discount_ids_to_uix(
        self,
        raw_ids: npt.NDArray[np.int32],
    ) -> npt.NDArray[np.int32]:
        """
        Translate global discount-curve IDs into
        `discount_curve_id_universe` UIXs.

        Sentinel rows (`raw_ids < 0`) map to `INVALID_UIX`.
        """
        return _searchsorted_with_sentinel(
            universe=self.discount_curve_id_universe,
            raw=raw_ids,
        )

    def map_dates_to_uix(
        self,
        ordinals: npt.NDArray[np.int32],
    ) -> npt.NDArray[np.int32]:
        """
        Translate global day-ordinals into `date_ord32_universe` UIXs.

        Sentinel rows (`ordinals < 0`) map to `INVALID_UIX`.
        """
        return _searchsorted_with_sentinel(
            universe=self.date_ord32_universe,
            raw=ordinals,
        )

    def map_fx_legs_to_effective_uix(
        self,
        *,
        raw_fx_ids: npt.NDArray[np.int32],
        raw_fx_inv: npt.NDArray[np.int8],
    ) -> npt.NDArray[np.int32]:
        """
        Translate per-trade `(raw_fx_id, raw_fx_inv)` into per-trade
        `fx_spots_effective_uix` indices.

        Steps (all vectorized, no Python loop):
          1. Translate `raw_fx_ids` into `market_id_universe` UIXs.
          2. Look the resulting `(fx_uix, invert)` pair up in
             `fx_effective_pairs` to obtain the effective UIX.

        Sentinel rows (`raw_fx_ids == NO_FX_RF_REQUIRED`) map to `INVALID_UIX`.
        """
        if raw_fx_ids.shape != raw_fx_inv.shape:
            raise ValueError(
                f"fx legs shape mismatch: ids={raw_fx_ids.shape}, "
                f"inv={raw_fx_inv.shape}"
            )

        # Stage 1: market-id universe lookup. Sentinels collapse to INVALID_UIX.
        fx_market_uix = _searchsorted_with_sentinel(
            universe=self.market_id_universe,
            raw=raw_fx_ids,
        )

        # Stage 2: dedup-pair lookup. We pack each (uix, inv) pair into a
        # single int64 key for a vectorized one-shot searchsorted into the
        # equivalently-packed effective-pairs table.
        n = raw_fx_ids.shape[0]
        out = np.full(n, INVALID_UIX, dtype=np.int32)

        if self.fx_effective_pairs.size == 0:
            return out

        needs_fx = fx_market_uix != INVALID_UIX
        if not needs_fx.any():
            return out

        needed_uix = fx_market_uix[needs_fx].astype(np.int64, copy=False)
        needed_inv = raw_fx_inv[needs_fx].astype(np.int64, copy=False)
        needed_key = (needed_uix << 1) | needed_inv

        pair_uix = self.fx_effective_pairs[:, 0].astype(np.int64, copy=False)
        pair_inv = self.fx_effective_pairs[:, 1].astype(np.int64, copy=False)
        pair_key = (pair_uix << 1) | pair_inv

        idx = np.searchsorted(pair_key, needed_key)
        # Every needed_key MUST exist in pair_key (it was deduplicated from
        # exactly this input). Validate to fail loud on contract drift.
        if idx.size and (
            idx.max() >= pair_key.size or np.any(pair_key[idx] != needed_key)
        ):
            raise RuntimeError(
                "FX-effective dedup contract violated: per-trade pair not "
                "found in fx_effective_pairs. Mapper was built from a "
                "different batch than this one."
            )

        out[needs_fx] = idx.astype(np.int32, copy=False)
        return out


# ==============================================================================
# INTERNAL HELPERS
# ==============================================================================


def _searchsorted_with_sentinel(
    *,
    universe: npt.NDArray[np.int32],
    raw: npt.NDArray[np.int32],
) -> npt.NDArray[np.int32]:
    """
    Vectorized O(log N) ID translation with sentinel preservation.

    `raw < 0` is treated as a sentinel and mapped to `INVALID_UIX` in the
    output. All other IDs MUST exist in `universe` (the universe is built
    from the same IDs upstream); a violation indicates a wiring bug and is
    caught by `validate_ingest_batches_against_uix_universe`.

    The universe itself MUST be sorted ascending and unique — guaranteed by
    `UixMapper.__post_init__`.
    """
    uix = np.searchsorted(universe, raw).astype(np.int32, copy=False)
    uix[raw < 0] = INVALID_UIX
    return uix


def _dedup_fx_effective_pairs(
    *,
    market_id_universe: npt.NDArray[np.int32],
    raw_fx_id_arrays: tuple[npt.NDArray[np.int32], ...],
    raw_fx_inv_arrays: tuple[npt.NDArray[np.int8], ...],
) -> npt.NDArray[np.int32]:
    """
    Build the deduplicated `(fx_uix, invert)` pair table covering every FX
    leg in the run.

    Inputs are the raw `(fx_id, inv)` arrays from every IngestBatch leg
    (under->settle, strike->settle, ...). The function:
      1. Concatenates them into one stream.
      2. Drops sentinel rows (`fx_id == NO_FX_RF_REQUIRED`).
      3. Translates `fx_id` -> `market_id_universe` UIX.
      4. `np.unique(axis=0)` to get the sorted-unique pair table.

    Returned shape is `(K, 2) int32`. The two columns are
    `(fx_uix_in_market_universe, invert_flag)`. The table is sorted
    lexicographically by `(fx_uix, invert)`, which is critical for the
    `np.searchsorted` lookup that `map_fx_legs_to_effective_uix` performs
    via the packed `key = (uix << 1) | inv` trick.
    """
    if len(raw_fx_id_arrays) != len(raw_fx_inv_arrays):
        raise ValueError(
            f"FX leg array count mismatch: ids={len(raw_fx_id_arrays)}, "
            f"inv={len(raw_fx_inv_arrays)}"
        )

    if not raw_fx_id_arrays:
        return np.empty((0, 2), dtype=np.int32)

    all_ids = np.concatenate(raw_fx_id_arrays)
    all_inv = np.concatenate(raw_fx_inv_arrays)

    mask = all_ids != int(NO_FX_RF_REQUIRED)
    if not mask.any():
        return np.empty((0, 2), dtype=np.int32)

    real_ids = all_ids[mask]
    real_inv = all_inv[mask]

    # Translate raw IDs -> market-universe UIXs.
    real_uix = np.searchsorted(market_id_universe, real_ids).astype(
        np.int32, copy=False
    )

    pairs = np.stack(
        (real_uix, real_inv.astype(np.int32, copy=False)),
        axis=1,
    )
    unique_pairs = np.unique(pairs, axis=0)
    # np.unique already returns sorted lex order; ensure C-contiguous int32.
    return np.ascontiguousarray(unique_pairs.astype(np.int32, copy=False))


# ==============================================================================
# ORCHESTRATOR PHASE (UNIVERSE DISCOVERY)
# ==============================================================================


def build_uix_mapper(ingest_batches: Sequence[IngestBatch]) -> UixMapper:
    """
    Global universe discovery — the "orchestrator pause".

    Walks every `IngestBatch` once to collect:
      - the union of every market ID (underlyings, vols, FX spots)
      - the union of every discount curve ID
      - the union of every trade-relevant date ordinal
      - the deduplicated `(fx_uix, invert)` pair table

    Returns a frozen `UixMapper` that downstream consumers (block compilers,
    market-data builder, validation) use as the single source of truth for
    UIX <-> global-ID translation.
    """
    if not ingest_batches:
        raise ValueError(
            "No ingest batches provided to build_uix_mapper. At least one "
            "IngestBatch is required."
        )

    # ---- MARKET IDs ----
    market_id_arrays = [
        arr
        for b in ingest_batches
        for arr in (b.underlying_ids, b.volatility_ids, b.fx_u2s_ids, b.fx_k2s_ids)
    ]
    raw_market = np.concatenate(market_id_arrays)
    market_id_universe = np.unique(raw_market[raw_market >= 0]).astype(
        np.int32, copy=False
    )

    # ---- DISCOUNT CURVE IDs ----
    discount_id_arrays = [
        arr
        for b in ingest_batches
        for arr in (b.under_discount_curve_ids, b.strike_discount_curve_ids)
    ]
    raw_discount = np.concatenate(discount_id_arrays)
    discount_curve_id_universe = np.unique(raw_discount[raw_discount >= 0]).astype(
        np.int32, copy=False
    )

    # ---- DATES ----
    date_arrays = [
        arr
        for b in ingest_batches
        for arr in (b.settle_ord, b.delivery_ord, b.expiry_ord)
    ]
    raw_dates = np.concatenate(date_arrays)
    date_ord32_universe = np.unique(raw_dates[raw_dates >= 0]).astype(
        np.int32, copy=False
    )

    # ---- FX EFFECTIVE PAIRS ----
    raw_fx_id_arrays = tuple(
        arr for b in ingest_batches for arr in (b.fx_u2s_ids, b.fx_k2s_ids)
    )
    raw_fx_inv_arrays = tuple(
        arr for b in ingest_batches for arr in (b.fx_u2s_inv, b.fx_k2s_inv)
    )
    fx_effective_pairs = _dedup_fx_effective_pairs(
        market_id_universe=market_id_universe,
        raw_fx_id_arrays=raw_fx_id_arrays,
        raw_fx_inv_arrays=raw_fx_inv_arrays,
    )

    return UixMapper(
        market_id_universe=market_id_universe,
        discount_curve_id_universe=discount_curve_id_universe,
        date_ord32_universe=date_ord32_universe,
        fx_effective_pairs=fx_effective_pairs,
    )


# ==============================================================================
# COMPILER PHASE (UIX INJECTION)
# ==============================================================================


def compile_portfolio_from_ingest_batch(
    *,
    ingest_batch: IngestBatch,
    uix_mapper: UixMapper,
) -> CompiledPortfolio:
    """
    Sort an `IngestBatch` by `instrument_id` and compile each contiguous run
    into an execution-ready block using the provided `UixMapper`.

    The function performs the only global-ID -> UIX translation in the
    portfolio layer: it hands each instrument-specific compiler the
    sub-batch and the mapper, and the compiler does the actual gather.
    """
    instrument_ids = ingest_batch.instrument_ids
    n = instrument_ids.size

    if n == 0:
        raise ValueError("Cannot compile an empty batch into a portfolio.")

    # ---- 1. Sort once by instrument_id (stable so original ordering wins
    #         within each instrument group). ----
    order = np.argsort(instrument_ids, kind="stable")
    ingest_batch = ingest_batch.reorder(order)
    sorted_ids = ingest_batch.instrument_ids

    # ---- 2. Find group boundaries via one diff + one concat. ----
    change_idx = np.flatnonzero(sorted_ids[1:] != sorted_ids[:-1]) + 1
    boundaries = np.concatenate(([0], change_idx, [n]))

    # ---- 3. Compile per-instrument blocks. ----
    compiled_fields: dict[str, object] = {}
    compilers = COMPILERS
    field_names = FIELD_NAMES

    n_groups = len(boundaries) - 1
    for i in range(n_groups):
        start = int(boundaries[i])
        end = int(boundaries[i + 1])
        instrument_id = int(sorted_ids[start])

        compiler = compilers[instrument_id]
        if compiler is None:
            continue

        field_name = field_names[instrument_id]
        if field_name is None:
            raise RuntimeError(
                f"Compiler registered for instrument_id={instrument_id} but "
                "no FIELD_NAME mapping. Wire it in compilers.registry."
            )

        block = compiler(ingest_batch, start, end, uix_mapper)
        if block is not None:
            compiled_fields[field_name] = block

    # ---- 4. Forward portfolio-level metadata to the compiled artifact. ----
    compiled_fields["unique_dates_ord32"] = uix_mapper.date_ord32_universe

    return CompiledPortfolio(**compiled_fields)
```

```python
from __future__ import annotations

import re
from typing import Iterable

_ccy_re = re.compile(r"^[A-Z]{3}$")


def validate_ccy(ccy: str) -> str:
    if not isinstance(ccy, str):
        raise TypeError(f"Currency must be str; got {type(ccy).__name__}")
    if len(ccy) != 3:
        raise ValueError(
            f"Currency must be exactly 3 letters; got '{ccy}' ({len(ccy)})"
        )
    if _ccy_re.match(ccy) is None:
        raise ValueError(f"Currency must be 3 uppercase letters A–Z; got '{ccy}'")
    return ccy


def validate_ccys_unique(ccys: Iterable[str]) -> None:
    """Validate each distinct CCY once; fail fast on first invalid."""
    for c in set(ccys):
        validate_ccy(c)


def get_fx_pair_for_under_to_settle_conversion(
    *,
    underlying_ccy: str,
    settlement_ccy: str,
) -> str:
    # Validate scalars only (no allocations)
    u = validate_ccy(underlying_ccy)
    s = validate_ccy(settlement_ccy)
    if u == s:
        return ""
    return f"{u}{s}"
```

```python
# src/riskquant/portfolio/ingest/batch.py
"""
Struct-of-Arrays (SoA) container for validated trade ingestion.

Design goals:
- contiguous homogeneous arrays
- strict dtype enforcement
- cache-friendly SoA layout
- immutable batch semantics
- zero object arrays in engine paths
- explicit debug-only reconstruction utilities

Dtype change in this revision
-----------------------------
`fx_u2s_inv` and `fx_k2s_inv` are now int8 (was int32). Flags are 1 byte per row,
matching the project flag convention. The block compilers convert int8 -> bool
via .view() (same itemsize, no allocation) for the block-layer signatures.

Architecture update
-------------------
- Purged all UIX coordinates. The batch acts purely as a global ID ledger.
- Semantic Collapsing: Sparse curve/node arrays fused into dense `underlying_ids`
  and `volatility_ids`.
- `vol_surface_y_axis_type_ids` is now resolved by `compile_ingest_soa_payload_
  from_df` (sourced from `static_config.surface_meta_dict[id].y_axis_kind`).
  `from_compiled_ingest_soa` is a pure copy with no synthetic sentinels.
"""

from __future__ import annotations

from dataclasses import dataclass
from types import MappingProxyType
from typing import TYPE_CHECKING, Final

import numpy as np
import numpy.typing as npt

from riskquant.constants import (
    NO_DATE_ORD32,
    NO_FX_RF_REQUIRED,
    NO_ID,
    NOT_OPTION_MULT,
)
from riskquant.portfolio.ingest.compile import CompiledIngestSoA
from riskquant.utils.dates import ordinal32_to_datetime64D

if TYPE_CHECKING:
    from pandas import DataFrame

# ==============================================================================
# MODULE-LEVEL DTYPE / LENGTH CONTRACT
# ==============================================================================
# Built once at module load. Each IngestBatch construction reads from these
# rather than rebuilding the spec dict inline.

_EXPECTED_DTYPES: Final = MappingProxyType(
    {
        # CORE
        "instrument_ids": np.dtype(np.int32),
        "uids": np.dtype(np.int64),
        # MARKET RISK
        "qty": np.dtype(np.float64),
        "strike_price": np.dtype(np.float64),
        "cp_mult": np.dtype(np.int8),
        # DATES (Ordinals Only)
        "settle_ord": np.dtype(np.int32),
        "delivery_ord": np.dtype(np.int32),
        "expiry_ord": np.dtype(np.int32),
        # MARKET LINKAGES (Dense)
        "underlying_ids": np.dtype(np.int32),
        "volatility_ids": np.dtype(np.int32),
        # FX
        "fx_u2s_ids": np.dtype(np.int32),
        "fx_u2s_inv": np.dtype(np.int8),
        "fx_k2s_ids": np.dtype(np.int32),
        "fx_k2s_inv": np.dtype(np.int8),
        # CURVES & SURFACES
        "under_discount_curve_ids": np.dtype(np.int32),
        "strike_discount_curve_ids": np.dtype(np.int32),
        "vol_surface_y_axis_type_ids": np.dtype(np.int8),
        # NETTING SET
        "netting_set_id": np.dtype(np.int64),
    }
)

_VARIABLE_LENGTH_FIELDS: Final[tuple[str, ...]] = tuple(_EXPECTED_DTYPES.keys())


# ==============================================================================
# INTERNAL HELPERS
# ==============================================================================


def _assert_c_contiguous(arr: npt.NDArray, *, name: str) -> None:
    if not arr.flags.c_contiguous:
        raise ValueError(f"IngestBatch field '{name}' must be C-contiguous.")


def _assert_dtype(
    arr: npt.NDArray,
    expected_dtype: np.dtype,
    *,
    name: str,
) -> None:
    if arr.dtype != expected_dtype:
        raise TypeError(
            f"IngestBatch field '{name}' has dtype={arr.dtype}, "
            f"expected={expected_dtype}."
        )


def _masked_float_copy(
    arr: npt.NDArray,
    *,
    mask_value: float | int,
    replace_with: float = np.nan,
    copy: bool,
) -> npt.NDArray[np.float64]:
    """Create float view/copy and replace sentinel-like values with NaN."""
    out = arr.copy() if copy else arr.astype(arr.dtype, copy=False)

    mask = out == mask_value

    if np.any(mask):
        out = out.copy()
        out[mask] = replace_with

    return out


def _decode_dates(
    ord_array: npt.NDArray[np.int32],
    *,
    sentinel: int,
    copy: bool,
) -> npt.NDArray[np.datetime64]:
    """Decode ordinal32 dates to datetime64[D] with NaT masking."""
    dts = ordinal32_to_datetime64D(ord_array)

    mask = ord_array == sentinel
    if copy or np.any(mask):
        dts = dts.copy()
    if np.any(mask):
        dts[mask] = np.datetime64("NaT")

    return dts


def _debug_mask_int_sentinel(
    arr: npt.NDArray,
    *,
    sentinel: int,
    replacement: str,
) -> npt.NDArray[np.object_]:
    """
    Debug-only helper.

    Converts integer arrays into object arrays for human-readable auditing.
    NEVER use in pricing/risk kernels.
    """
    out = arr.astype(object, copy=True)
    mask = arr == sentinel
    if np.any(mask):
        out[mask] = replacement
    return out


# ==============================================================================
# INGEST BATCH
# ==============================================================================


@dataclass(frozen=True, slots=True, kw_only=True)
class IngestBatch:
    """
    Immutable Struct-of-Arrays ingestion container.

    All arrays:
    - must be contiguous
    - must be strongly typed (see _EXPECTED_DTYPES)
    - must have length == size (see _VARIABLE_LENGTH_FIELDS)
    """

    size: int

    # ROUTING / DISPATCH
    instrument_ids: npt.NDArray[np.int32]

    # ECONOMICS
    uids: npt.NDArray[np.int64]
    qty: npt.NDArray[np.float64]
    strike_price: npt.NDArray[np.float64]
    cp_mult: npt.NDArray[np.int8]

    # DATES (ORDINAL32)
    settle_ord: npt.NDArray[np.int32]
    delivery_ord: npt.NDArray[np.int32]
    expiry_ord: npt.NDArray[np.int32]

    # MARKET LINKAGES
    underlying_ids: npt.NDArray[np.int32]
    volatility_ids: npt.NDArray[np.int32]

    # FX
    fx_u2s_ids: npt.NDArray[np.int32]
    fx_u2s_inv: npt.NDArray[np.int8]
    fx_k2s_ids: npt.NDArray[np.int32]
    fx_k2s_inv: npt.NDArray[np.int8]

    # DISCOUNT CURVES & SURFACES
    under_discount_curve_ids: npt.NDArray[np.int32]
    strike_discount_curve_ids: npt.NDArray[np.int32]
    vol_surface_y_axis_type_ids: npt.NDArray[np.int8]

    # NETTING SET LINKAGE
    netting_set_id: npt.NDArray[np.int64]

    # ==========================================================================
    # POST INIT VALIDATION
    # ==========================================================================

    def __post_init__(self) -> None:
        """Enforce dtype, contiguity, and length consistency."""
        for field_name, expected_dtype in _EXPECTED_DTYPES.items():
            arr = getattr(self, field_name)
            if not isinstance(arr, np.ndarray):
                raise TypeError(
                    f"IngestBatch field '{field_name}' is not a numpy ndarray."
                )
            _assert_dtype(arr, expected_dtype, name=field_name)
            _assert_c_contiguous(arr, name=field_name)

        for field_name in _VARIABLE_LENGTH_FIELDS:
            arr = getattr(self, field_name)
            if arr.shape[0] != self.size:
                raise ValueError(
                    f"IngestBatch field '{field_name}' has "
                    f"length={arr.shape[0]}, expected={self.size}."
                )

    # ==========================================================================
    # DEBUG RECONSTRUCTION
    # ==========================================================================

    def to_df(self, *, copy: bool = True) -> DataFrame:
        """
        DEBUG / AUDIT ONLY.

        Reconstruct a DataFrame from the SoA container. Alloc-heavy,
        produces object arrays. Not engine-safe.
        """
        from pandas import DataFrame

        strike_price = _masked_float_copy(
            self.strike_price,
            mask_value=0.0,
            copy=copy,
        )
        cp_mult = _masked_float_copy(
            self.cp_mult.astype(np.float64, copy=False),
            mask_value=int(NOT_OPTION_MULT),
            copy=copy,
        )

        return DataFrame(

                # ECONOMICS

                "cp_mult": cp_mult,
                # MARKET LINKAGES
                "und_ids": _debug_mask_int_sentinel(

                    sentinel=int(NO_ID),
                    replacement="NONE",
                ),
                "vol_ids": _debug_mask_int_sentinel(
                    self.volatility_ids,
                    sentinel=int(NO_ID),
                    replacement="NONE",
                ),
                # FX IDs
                "fx_u2s_ids": _debug_mask_int_sentinel(
                    self.fx_u2s_ids,
                    sentinel=int(NO_FX_RF_REQUIRED),
                    replacement="NONE",
                ),
                "fx_u2s_inv": self.fx_u2s_inv,
                "fx_k2s_ids": _debug_mask_int_sentinel(
                    self.fx_k2s_ids,
                    sentinel=int(NO_FX_RF_REQUIRED),
                    replacement="NONE",
                ),
                "fx_k2s_inv": self.fx_k2s_inv,
                # CURVES
                "disc_und_id": _debug_mask_int_sentinel(
                    self.under_discount_curve_ids,
                    sentinel=int(NO_ID),

                    sentinel=int(NO_ID),

    # ==========================================================================
    # REORDERING
    # ==========================================================================

    def reorder(self, order: np.ndarray, *, copy: bool = False) -> IngestBatch:
        """Reorder batch arrays consistently."""
        if order.dtype.kind not in ("i", "u"):
            raise TypeError("order must contain integer indices.")
        if order.ndim != 1:
            raise ValueError("order must be 1-dimensional.")
        if order.shape[0] != self.size:
            raise ValueError(
                f"order length={order.shape[0]} does not match batch size={self.size}."

        def _reorder(arr: np.ndarray) -> np.ndarray:
```

```python
# src/riskquant/portfolio/ingest/canonicalize.py

"""
Canonicalization layer for ingest dataframes.

Purpose
-------
Normalize raw human-facing ingest data into RiskQuant canonical form.

This layer performs ONLY deterministic normalization in-place:

- canonical column names
- canonical column ordering
- string trimming
- string uppercasing
- empty-string normalization

This module intentionally does NOT perform:

- semantic validation
- dtype coercion
- business logic
- instrument validation
- RF mapping
- parsing

Those belong downstream in:
- materialize.py
- validator.py
- batch_builder.py

Design Philosophy
-----------------
Human-readable at the edge.
Canonicalized deterministic frames internally.
"""

from __future__ import annotations

import pandas as pd
from pandas.api.types import (
    is_object_dtype,
    is_string_dtype,
)

from riskquant.exceptions import PrettyKeyError
from riskquant.portfolio.ingest.schema import (
    ALL_ALLOWED_COLS_SET,
    CANONICAL_COLUMN_ORDER,
    NON_NULLABLE_COLS,
    STRING_COLS,
    UNIVERSAL_REQUIRED_COLS,
)

# ==============================================================================
# INTERNAL HELPERS
# ==============================================================================


def _validate_schema(df: pd.DataFrame) -> None:
    """
    Validate ingest dataframe structural schema.

    Validates ONLY:
    - dataframe type
    - unique columns
    - required universal columns
    - allowed column universe

    Does NOT validate:
    - semantics
    - values
    - instruments
    - quantities
    - option logic
    """

    # --------------------------------------------------------------------------
    # EMPTY DATAFRAME CHECK
    # --------------------------------------------------------------------------
    if df.empty:
        raise ValueError("Cannot ingest empty dataframe.")

    # --------------------------------------------------------------------------
    # DUPLICATE COLUMNS
    # --------------------------------------------------------------------------

    if not df.columns.is_unique:
        duplicated_ix = df.columns.duplicated()
        duplicated_cols = [
            str(df.columns[i]) for i, is_dup in enumerate(duplicated_ix) if is_dup
        ]

        raise PrettyKeyError(f"Duplicate ingest columns detected:\n{duplicated_cols}")

    # --------------------------------------------------------------------------
    # REQUIRED UNIVERSAL COLS
    # --------------------------------------------------------------------------

    missing = [col for col in UNIVERSAL_REQUIRED_COLS if col not in df.columns]

    if missing:
        raise PrettyKeyError(f"Missing required ingest columns:\n{missing}")

    # --------------------------------------------------------------------------
    # REQUIRED DOMAIN COLS
    # --------------------------------------------------------------------------

    # We do NOT enforce domain-specific columns at the schema level, as not all
    # instruments will require them. Instead, we allow them to be optional at
    # the schema level and defer domain-specific presence/absence logic to the
    # semantic validation layer (validator.py). This keeps the structural schema
    # flexible while still enforcing the core universal contract.

    # --------------------------------------------------------------------------
    # UNKNOWN / ILLEGAL COLS
    # --------------------------------------------------------------------------

    unknown = sorted(set(df.columns) - ALL_ALLOWED_COLS_SET)

    if unknown:
        raise PrettyKeyError(f"Unknown ingest columns detected:\n{unknown}")


def _canonicalize_column_names(df: pd.DataFrame) -> None:
    """
    Normalize dataframe column names in-place.

    Operations:
    - strip whitespace
    - lowercase
    """
    df.columns = df.columns.str.strip().str.lower()


def _canonicalize_string_series(ser: pd.Series) -> pd.Series:
    """
    Helper to process a single string series.
    """
    if not (is_string_dtype(ser.dtype) or is_object_dtype(ser.dtype)):
        raise TypeError(f"String column '{ser.name}' has invalid dtype={ser.dtype}.")

    non_na = ser.dropna()
    vals = non_na.to_numpy(copy=False)

    if not all(isinstance(x, str) for x in vals):
        raise TypeError(f"String column '{ser.name}' contains non-string values.")

    out = ser.astype("string")
    out = out.str.strip()
    out = out.str.replace(r"\s+", " ", regex=True)
    out = out.mask(out == "", pd.NA)
    out = out.str.upper()

    return out


def _canonicalize_string_columns(df: pd.DataFrame) -> None:
    """
    Canonicalize all configured string columns in-place.
    """
    for col in STRING_COLS:
        if col not in df.columns:
            continue
        # Reassigning the series modifies the DataFrame in-place
        df[col] = _canonicalize_string_series(df[col])


def _reorder_columns(df: pd.DataFrame) -> None:
    """
    Reorder dataframe columns into canonical RiskQuant order in-place.

    Uses pop and insert to modify the internal column order without
    allocating a new DataFrame object.
    """
    ordered_cols = [col for col in CANONICAL_COLUMN_ORDER if col in df.columns]

    for i, col in enumerate(ordered_cols):
        col_data = df.pop(col)
        df.insert(i, col, col_data)


# ==============================================================================
# MAIN ENTRYPOINT
# ==============================================================================


def canonicalize_raw_ingest_df(df: pd.DataFrame) -> None:
    """
    Canonicalize ingest dataframe into deterministic RiskQuant form in-place.

    Pipeline:
    1. canonicalize column names
    2. validate structural schema
    3. canonicalize string columns
    4. reorder columns
    5. validate non-nullable columns

    Returns
    -------
    None
        The dataframe is modified in-place.
    """

    # --------------------------------------------------------------------------
    # COLUMN NAME NORMALIZATION
    # --------------------------------------------------------------------------
    _canonicalize_column_names(df)

    # --------------------------------------------------------------------------
    # STRUCTURAL VALIDATION
    # --------------------------------------------------------------------------
    _validate_schema(df)

    # --------------------------------------------------------------------------
    # STRING NORMALIZATION
    # --------------------------------------------------------------------------
    _canonicalize_string_columns(df)

    # --------------------------------------------------------------------------
    # NON-NULLABLE VALIDATION
    # --------------------------------------------------------------------------
    for col in NON_NULLABLE_COLS:
        if df[col].isna().any():
            bad_idx = df.index[df[col].isna()].tolist()[:10]
            raise ValueError(
                f"Non-nullable column '{col}' "
                f"contains missing values at rows: {bad_idx}"
            )

    # --------------------------------------------------------------------------
    # COLUMN ORDERING
    # --------------------------------------------------------------------------
    _reorder_columns(df)

    # --------------------------------------------------------------------------
    # RESET INDEX
    # --------------------------------------------------------------------------
    df.reset_index(drop=True, inplace=True)
```

```python
# src/riskquant/portfolio/ingest/compile.py
"""
DataFrame -> Struct-of-Arrays compiler.

This is the final hop of the ingest edge: a validated, materialized DataFrame
is collapsed into `CompiledIngestArrays`, a flat homogeneous SoA payload with
no strings and no Python objects. `IngestBatch.from_compiled` wraps the result
for the engine.

Refactor history
----------------
- Semantic Collapsing: `price_rf` and `price_curve` fused into dense `underlying_ids`.
- Semantic Collapsing: `vol_rf` and `vol_surface` fused into dense `volatility_ids`.
- Coordinate Purge: All `_uix` logic removed. Ingest acts purely as a global ID ledger.
- FX Linkages renamed to explicitly end in `_ids` to enforce boundary semantics.
- Underlying currency resolution optimized to occur concurrently with ID mapping.
- Real volatility-container resolution: the `volatility_container_name` schema
  column is mapped to int32 container ids and to the per-trade int8 y-axis-type
  id. Both fields land on `CompiledIngestSoA`; `IngestBatch.from_compiled` is
  a pure copy with no synthetic sentinels.
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
import numpy.typing as npt
import pandas as pd

from riskquant.config.static_config.static_config import RiskStaticConfig
from riskquant.constants import (
    CALL_OPTION_MULT,
    NO_DATE_ORD32,
    NO_FX_RF_REQUIRED,
    NO_ID,
    NOT_IMPLEMENTED_SENTINEL,
    NOT_OPTION_MULT,
    PUT_OPTION_MULT,
)
from riskquant.exceptions import PrettyKeyError
from riskquant.market_data.surfaces.y_axes_types import SurfaceYAxisType
from riskquant.portfolio.ingest.fx_resolver import resolve_fx_rf_ids_inplace
from riskquant.portfolio.ingest.parser import (
    parse_optional_date_series_to_ord32,
)
from riskquant.portfolio.ingest.rf_resolver import resolve_optional_ids

# ==============================================================================
# OUTPUT STRUCTURE
# ==============================================================================


@dataclass(slots=True, frozen=True, kw_only=True)
class CompiledIngestSoA:
    """
    Flat SoA payload emitted by the compiler. All arrays length == size.
    Contains ONLY pure economics, date ordinals, and global IDs.
    Strictly zero UIXs and fully dense semantic arrays.
    """

    size: int

    # --- ROUTING & IDENTITY ---
    instrument_ids: npt.NDArray[np.int32]
    uids: npt.NDArray[np.int64]  # external trade ids stay 64-bit

    # --- ECONOMICS ---
    qty: npt.NDArray[np.float64]
    strike: npt.NDArray[np.float64]
    cp_mult: npt.NDArray[np.int8]

    # --- DATES (Ordinals Only) ---
    settle_ord: npt.NDArray[np.int32]
    delivery_ord: npt.NDArray[np.int32]
    expiry_ord: npt.NDArray[np.int32]

    # --- MARKET DATA LINKAGES (Dense Global IDs) ---
    underlying_ids: npt.NDArray[np.int32]
    volatility_ids: npt.NDArray[np.int32]

    # --- FX LINKAGES (Global IDs) ---
    fx_u2s_ids: npt.NDArray[np.int32]
    fx_u2s_inv: npt.NDArray[np.int8]

    fx_k2s_ids: npt.NDArray[np.int32]
    fx_k2s_inv: npt.NDArray[np.int8]

    # --- DISCOUNT CURVE LINKAGES (Global IDs) ---
    under_discount_curve_ids: npt.NDArray[np.int32]
    strike_discount_curve_ids: npt.NDArray[np.int32]

    # --- VOL-SURFACE Y-AXIS TYPE (int8 per trade) ---
    # Resolved at compile time from `surface_meta_dict[vol_container_id]
    # .y_axis_kind`. Non-option rows and option rows whose volatility
    # container is not a surface (e.g. a future vol-node container) get
    # `NOT_IMPLEMENTED_SENTINEL`. The option-curve pricer's pre-flight
    # check refuses to invoke its kernel with sentinel ids.
    vol_surface_y_axis_type_ids: npt.NDArray[np.int8]

    # --- NETTING SET LINKAGE ---
    netting_set_id: npt.NDArray[np.int64]

    def print_soa(
        self,
        *,
        max_rows: int = 50,
        int_fmt: str = ">10_.0f",
        float_fmt: str = ">16_.5f",
    ) -> None:
        """Print the SoA as a dense matrix."""
        n = min(self.size, max_rows)

        headers = [
            "uid",
            "inst_id",
            "qty",
            "k",
            "cp_mult",
            "set_ord",
            "del_ord",
            "exp_ord",
            "und_id",
            "vol_id",
            "vol_yax",
            "fx_u2s_id",
            "fx_u2s_inv",
            "fx_k2s_id",
            "fx_k2s_inv",
            "disc_und_id",
            "disc_k_id",
            "net_set_id",
        ]

        data = np.column_stack(
            [
                self.uids[:n],
                self.instrument_ids[:n],
                self.qty[:n],
                self.strike[:n],
                self.cp_mult[:n],
                self.settle_ord[:n],
                self.delivery_ord[:n],
                self.expiry_ord[:n],
                self.underlying_ids[:n],
                self.volatility_ids[:n],
                self.vol_surface_y_axis_type_ids[:n],
                self.fx_u2s_ids[:n],
                self.fx_u2s_inv[:n],
                self.fx_k2s_ids[:n],
                self.fx_k2s_inv[:n],
                self.under_discount_curve_ids[:n],
                self.strike_discount_curve_ids[:n],
                self.netting_set_id[:n],
            ]
        )

        fmts = [int_fmt] * len(headers)
        fmts[2] = fmts[3] = float_fmt  # qty, strike

        # Parse field width from format spec (e.g. ">16_.5f" -> 16)
        import re

        def _fmt_width(fmt: str) -> int:
            m = re.match(r"[<>^]?(\d+)", fmt)
            return int(m.group(1)) if m else len(fmt)

        widths = [_fmt_width(f) for f in fmts]

        header = " ".join(f"{h:>{w}}" for h, w in zip(headers, widths, strict=True))
        print(header)
        print("-" * len(header))

        for i in range(n):
            print(" ".join(f"{data[i, j]:{fmts[j]}}" for j in range(len(headers))))

        if self.size > max_rows:
            print(f"... ({self.size - max_rows} more rows)")

def _zero_series(index: pd.Index) -> pd.Series:
    """Float64 zero-fill series for absent numeric columns."""
    return pd.Series(0.0, index=index, dtype=np.float64)


def _blank_series(index: pd.Index) -> pd.Series:
    """NA-filled string series for absent string columns."""
    return pd.Series(pd.NA, index=index, dtype="string")


def _to_numpy_str(ser: pd.Series) -> npt.NDArray[np.str_]:
    """Extract a C-contiguous str array; NA collapses to the empty string."""
    return np.ascontiguousarray(ser.fillna("").to_numpy(dtype=str, copy=False))


def _to_f64(ser: pd.Series) -> npt.NDArray[np.float64]:
    """Extract a C-contiguous float64 array; NA collapses to 0.0."""
    return np.ascontiguousarray(ser.fillna(0.0).to_numpy(dtype=np.float64))


def _compile_option_multiplier(
    df: pd.DataFrame,
    n: int,
) -> npt.NDArray[np.int8]:
    """
    Pack `is_call` into the branchless int8 multiplier.

    +1 call, -1 put, 0 not-an-option. The encoding lets the pricer form the
    payoff sign by multiplication instead of branching.
    """
    cp_mult = np.empty(n, dtype=np.int8)
    cp_mult.fill(NOT_OPTION_MULT)

    if "is_call" not in df.columns:
        return cp_mult

    # is_call is materialized to nullable float64 (1.0 / 0.0 / NaN).
    raw = df["is_call"].to_numpy(copy=False)
    present = pd.notna(raw)
    flags = np.where(present, raw, 0.0).astype(np.int8)

    cp_mult[present & (flags == 1)] = CALL_OPTION_MULT
    cp_mult[present & (flags == 0)] = PUT_OPTION_MULT
    return cp_mult


def _compile_netting_set_id(
    df: pd.DataFrame,
    n: int,
) -> npt.NDArray[np.int64]:
    """
    Compile the effective-netting-set linkage into a flat int64 array.
    Absent column or NA row -> `NO_NETTING_SET_ID`.
    """
    out = np.empty(n, dtype=np.int64)
    if "netting_set_id" not in df.columns:
        out.fill(np.int64(NO_ID))
        return out

    filled = df["netting_set_id"].fillna(np.int64(NO_ID))
    return np.ascontiguousarray(filled.to_numpy(dtype=np.int64))


def _resolve_underlying(
    underlying_names: npt.NDArray[np.str_],
    static_config: RiskStaticConfig,
    n: int,
) -> tuple[npt.NDArray[np.int32], npt.NDArray[np.int8]]:
    """
    Classify each underlying name as a risk factor or a container and return
    a single collapsed semantic underlying_ids array, alongside the ccy_id.

    Vectorized: one pd.Series.map per universe, boolean-mask scatter.
    """
    name_to_rf = static_config.rel_name_to_rf_id
    name_to_container = static_config.container_name_to_id

    und_ser = pd.Series(underlying_names)
    rf_hits = und_ser.map(name_to_rf)
    container_hits = und_ser.map(name_to_container)

    is_rf = rf_hits.notna().to_numpy(copy=False)
    is_container = container_hits.notna().to_numpy(copy=False)

    both = is_rf & is_container
    if both.any():
        bad = sorted({str(v) for v in underlying_names[both]})[:10]
        raise PrettyKeyError(
            "Underlying name resolves to BOTH a risk factor AND a container - "
            f"the static config has a namespace collision. Offending: {bad}"
        )

    neither = ~(is_rf | is_container)
    if neither.any():
        bad = sorted({str(v) for v in underlying_names[neither] if str(v)})[:10]
        blanks = int((underlying_names[neither] == "").sum())
        raise PrettyKeyError(
            f"Underlying name resolves to NEITHER a risk factor NOR a container.\n"
            f"  unknown names ({len(bad)} of {int(neither.sum())} unique): {bad}\n"
            f"  blank/empty values: {blanks}"
        )

    # Dense semantic ID array
    underlying_ids = np.full(n, NO_ID, dtype=np.int32)
    # Currency ID array
    ccy_ids = np.full(n, -1, dtype=np.int8)

    if is_rf.any():
        rf_int32 = rf_hits[is_rf].to_numpy(dtype=np.int32, copy=False)
        underlying_ids[is_rf] = rf_int32
        ccy_ids[is_rf] = static_config.rf_id_to_ccy_id[rf_int32]

    if is_container.any():
        cont_int32 = container_hits[is_container].to_numpy(dtype=np.int32, copy=False)
        underlying_ids[is_container] = cont_int32
        ccy_ids[is_container] = static_config.container_id_to_ccy_id[cont_int32]

    return underlying_ids, ccy_ids


def _resolve_underlying_ccy_strings(
    ccy_ids: npt.NDArray[np.int8],
    static_config: RiskStaticConfig,
    fallback_ccy: npt.NDArray[np.str_],
) -> npt.NDArray[np.str_]:
    """
    Derive the underlying's quote currency NAME per row.
    Maps the ccy_ids derived during underlying resolution back to strings.
    """
    n = ccy_ids.shape[0]
    ccy_id_to_name = static_config.ccy_id_to_name  # list[str], indexed by ccy_id
    out = np.empty(n, dtype=object)

    for i in range(n):
        cid = int(ccy_ids[i])
        if cid < 0:
            out[i] = fallback_ccy[i]
        else:
            out[i] = ccy_id_to_name[cid]

    return np.ascontiguousarray(out.astype(str))

def _resolve_volatility_ids(
    names: npt.NDArray[np.str_],
    static_config: RiskStaticConfig,
) -> npt.NDArray[np.int32]:
    """
    Resolve the `volatility_container_name` column to int32 container ids.

    Blank rows (non-option trades that leave the column None) collapse to
    `NO_ID`. Populated-but-unmapped names are a hard error — the helper
    detects them explicitly because the pre-existing `resolve_optional_ids`
    silently turns NaN into garbage int32 on the typo path.

    The returned array flows into `IngestBatch.volatility_ids` and from
    there into `market_id_universe` (see `validate_ingest_batches_against_
    uix_universe`), which the option block compilers then `searchsorted`
    into per-block `vol_uix` / `vol_surface_uix` slots.
    """
    container_name_to_container_id = static_config.container_name_to_id

    # Reuse the standard optional-id path for the dense vectorized hash
    # join. The result is correct on blank and on successfully-mapped
    # rows; the only failure mode it silently swallows is populated-but-
    # unmapped (a typo) — which we detect explicitly below.
    out = resolve_optional_ids(
        names,
        container_name_to_container_id,
        sentinel=NO_ID,
        ctx="volatility_container_name",
    )

    # Explicit typo detection. `_BLANK` is "" (per canonicalize.py NA -> "").
    blank_mask = names == ""
    mapped = pd.Series(names).map(container_name_to_container_id)
    unmapped = mapped.isna().to_numpy(copy=False) & ~blank_mask
    if unmapped.any():
        bad = sorted({str(v) for v in names[unmapped]})[:10]
        raise PrettyKeyError(
            f"volatility_container_name: {int(unmapped.sum())} name(s) could "
            f"not be resolved to a container id.\n"
            f"Offending names (up to 10): {bad}\n"
            f"Allowed names are surface / vol-node container names from "
            f"`static_config.container_name_to_id`."
        )

    return out


def _resolve_vol_surface_y_axis_type_ids(
    volatility_ids: npt.NDArray[np.int32],
    static_config: RiskStaticConfig,
) -> npt.NDArray[np.int8]:
    """
    Map each non-sentinel `volatility_id` to its surface y-axis-type id (int8).

    The kernel's per-trade y-axis dispatch (STRIKE / LOG_STRIKE / MONEYNESS
    / LOG_MONEYNESS / DELTA_CALL) reads from this column. Resolving here
    keeps the option-curve block compiler a pure slicer — no static-config
    coupling past the ingest edge.

    Sentinel rows (`vol_id == NO_ID`) and rows whose `vol_id` is NOT a
    surface (e.g. a vol-node container, when that universe lands) collapse
    to `NOT_IMPLEMENTED_SENTINEL`. The option-curve pricer's pre-flight
    check refuses the sentinel — that's the contract.

    Performance
    -----------
    O(n) Python loop. Vol-bearing rows are a small minority of the
    portfolio; the surface-meta dict lookup is O(1). If the option count
    ever dominates, swap the loop for a `np.take` on a dense surface-id
    -> y-axis-type lookup table built once from `surface_meta_dict`.
    """
    n = volatility_ids.shape[0]
    out = np.empty(n, dtype=np.int8)
    out.fill(NOT_IMPLEMENTED_SENTINEL)

    surface_meta_dict = static_config.surface_meta_dict
    # Build a small cache so identical surface ids (the common case)
    # only hit the IntEnum lookup once.
    sid_to_y_axis_id: dict[int, np.int8] = {}

    for i in range(n):
        vid = int(volatility_ids[i])
        if vid < 0:
            continue

        cached = sid_to_y_axis_id.get(vid)
        if cached is not None:
            out[i] = cached
            continue

        sm = surface_meta_dict.get(vid)
        if sm is None:
            # Resolved container is not a surface (e.g. future vol-node
            # container). Leave as NOT_IMPLEMENTED_SENTINEL; the option-
            # curve pricer's pre-flight check refuses it. Cache None
            # via the sentinel so we don't re-lookup on repeats.
            sid_to_y_axis_id[vid] = np.int8(NOT_IMPLEMENTED_SENTINEL)
            continue

        # `SurfaceYAxisType` is the runtime enum the option-curve kernel
        # branches on (STRIKE=1, ..., DELTA_CALL=7). Use this not
        # `YAxisKind` which has a different ordering (STRIKE=0, ...).
        try:
            y_id = np.int8(SurfaceYAxisType[sm.y_axis_kind].value)
        except KeyError as exc:
            raise PrettyKeyError(
                f"surface_meta_dict[{vid}].y_axis_kind={sm.y_axis_kind!r} "
                f"is not a SurfaceYAxisType member. The static config and "
                f"the runtime enum have drifted."
            ) from exc

        sid_to_y_axis_id[vid] = y_id
        out[i] = y_id

    return out

# MAIN COMPILER
# ==============================================================================


def compile_ingest_soa_payload_from_df(
    *,
    df: pd.DataFrame,
    instrument_ids: npt.NDArray[np.int32],
    static_config: RiskStaticConfig,
) -> CompiledIngestSoA:
    """
    Compile a validated ingest frame into a flat SoA payload.

    Parameters
    ----------
    df
        Canonicalized, materialized, validated ingest frame (index 0..n-1).
    instrument_ids
        Per-row int32 instrument ids from `dispatch_ingest_validation`.
    static_config
        Compiled RiskStaticConfig. Single source of truth for every name
        lookup, ccy translation, and FX-pair resolution.

    Returns
    -------
    CompiledIngestSoA
        Flat SoA payload.
    """
    n = len(df)
    if n == 0:
        raise ValueError("Empty ingest dataframe.")

    index = df.index

    # --------------------------------------------------------------------------
    # CORE ECONOMICS
    # --------------------------------------------------------------------------
    uids = np.ascontiguousarray(df["uid"].to_numpy(dtype=np.int64, copy=False))
    qty = _to_f64(df.get("quantity", _zero_series(index)))
    strike = _to_f64(df.get("strike_price", _zero_series(index)))
    cp_mult = _compile_option_multiplier(df, n)

    # --------------------------------------------------------------------------
    # DATES -> ORDINAL32 (NO UIX LOGIC)
    # --------------------------------------------------------------------------
    settle_ord = parse_optional_date_series_to_ord32(
        df.get("settlement_date", _blank_series(index)),
        col_name="settlement_date",
        sentinel=NO_DATE_ORD32,
    )
    delivery_ord = parse_optional_date_series_to_ord32(
        df.get("delivery_date", _blank_series(index)),
        col_name="delivery_date",
        sentinel=NO_DATE_ORD32,
    )
    expiry_ord = parse_optional_date_series_to_ord32(
        df.get("expiry_date", _blank_series(index)),
        col_name="expiry_date",
        sentinel=NO_DATE_ORD32,
    )

    # --------------------------------------------------------------------------
    # MARKET LINKAGES (DENSE SEMANTIC IDS)
    # --------------------------------------------------------------------------

    # --- RESOLVE UNDERLYING ---
    underlying_names = _to_numpy_str(df.get("underlying", _blank_series(index)))
    underlying_ids, under_ccy_ids = _resolve_underlying(
        underlying_names, static_config, n
    )

    # --- RESOLVE DISCOUNT CURVE CONTAINERS ---
    container_name_to_container_id = static_config.container_name_to_id

    under_df_curve_ids = resolve_optional_ids(
        _to_numpy_str(df.get("under_discount_curve", _blank_series(index))),
        container_name_to_container_id,
        sentinel=NO_ID,
        ctx="under_discount_curve",
    )
    strike_df_curve_ids = resolve_optional_ids(
        _to_numpy_str(df.get("strike_discount_curve", _blank_series(index))),
        container_name_to_container_id,
        sentinel=NO_ID,
        ctx="strike_discount_curve",
    )

    # --- RESOLVE VOL CONTAINERS + SURFACE Y-AXIS TYPE ---
    volatility_names = _to_numpy_str(
        df.get("volatility_container_name", _blank_series(index))
    )
    volatility_ids = _resolve_volatility_ids(volatility_names, static_config)
    vol_surface_y_axis_type_ids = _resolve_vol_surface_y_axis_type_ids(
        volatility_ids, static_config
    )

    # --------------------------------------------------------------------------
    # FX RESOLUTION (underlying / strike -> settlement)
    # --------------------------------------------------------------------------
    settle_ccy = _to_numpy_str(df.get("settlement_ccy", _blank_series(index)))

    # Derive the underlying CCY string
    under_ccy = _resolve_underlying_ccy_strings(
        ccy_ids=under_ccy_ids,
        static_config=static_config,
        fallback_ccy=settle_ccy,
    )

    # Strike ccy defaults to underlying ccy when blank.
    if "strike_ccy" in df.columns:
        strike_ccy = np.ascontiguousarray(
            df["strike_ccy"]
            .fillna(pd.Series(under_ccy, index=index))
            .to_numpy(dtype=str)
        )
    else:
        strike_ccy = under_ccy

    # ---- FX: UNDERLYING -> SETTLEMENT ----
    fx_u2s_ids = np.full(n, np.int32(NO_FX_RF_REQUIRED), dtype=np.int32)
    fx_u2s_inv = np.zeros(n, dtype=np.int8)
    resolve_fx_rf_ids_inplace(
        base_ccy=under_ccy,
        settlement_ccy=settle_ccy,
        out_fx_rf_ids=fx_u2s_ids,
        out_fx_invert=fx_u2s_inv,
        static_config=static_config,
    )

    # ---- FX: STRIKE -> SETTLEMENT ----
    fx_k2s_ids = np.full(n, np.int32(NO_FX_RF_REQUIRED), dtype=np.int32)
    fx_k2s_inv = np.zeros(n, dtype=np.int8)
    resolve_fx_rf_ids_inplace(
        base_ccy=strike_ccy,
        settlement_ccy=settle_ccy,
        out_fx_rf_ids=fx_k2s_ids,
        out_fx_invert=fx_k2s_inv,
        static_config=static_config,
    )

    # --------------------------------------------------------------------------
    # NETTING SET LINKAGE
    # --------------------------------------------------------------------------
    netting_set_id = _compile_netting_set_id(df, n)

    # --------------------------------------------------------------------------
    # ASSEMBLE
    # --------------------------------------------------------------------------
    return CompiledIngestSoA(
        size=n,
        instrument_ids=np.ascontiguousarray(instrument_ids, dtype=np.int32),
        uids=uids,
        qty=qty,
        strike=strike,
        cp_mult=cp_mult,
        settle_ord=settle_ord,
        delivery_ord=delivery_ord,
        expiry_ord=expiry_ord,
        underlying_ids=underlying_ids,
        volatility_ids=volatility_ids,
        fx_u2s_ids=fx_u2s_ids,
        fx_u2s_inv=fx_u2s_inv,
        fx_k2s_ids=fx_k2s_ids,
        fx_k2s_inv=fx_k2s_inv,
        under_discount_curve_ids=under_df_curve_ids,
        strike_discount_curve_ids=strike_df_curve_ids,
        vol_surface_y_axis_type_ids=vol_surface_y_axis_type_ids,
        netting_set_id=netting_set_id,
    )
```

```python
# src/riskquant/portfolio/ingest/dispatcher.py

"""
Instrument routing for the ingest edge.

This module is the non-OOP dispatch layer that routes materialized ingest rows
to the correct per-instrument validation logic. Routing is performed with dense
lookup tables keyed by integer ids — no class hierarchy, no method overriding,
no string-keyed dictionary lookups in the dispatch path.

Two routing axes mirror the engine's own dual-dispatch model:

  - FAMILY  (primary)   : SPOT / FORWARD / OPTION / CREDIT  -> coarse validators
  - INSTRUMENT (secondary): the concrete Instrument id      -> refinement validators

The block-compiler routing (instrument id -> SoA block compiler) lives
downstream in `portfolio.compilers.registry`; this module deliberately handles
only the validation hop, the single point in the ingest phase that genuinely
needs per-instrument-type branching. It supersedes the hand-wired sequence
previously inlined in `validate.validate_materialized_ingest_df`.
"""

from __future__ import annotations

from collections.abc import Callable
from typing import Final, TypeAlias, cast

import numpy as np
import numpy.typing as npt
import pandas as pd

from riskquant.portfolio.ingest.validate import (
    validate_forward_family,
    validate_option_family,
    validate_spot_family,
    validate_unique_uids,
)
from riskquant.portfolio.instruments import (
    INSTRUMENT_NAME_TO_ID_REGISTRY,
    INSTRUMENT_TO_FAMILY_ARRAY,
    NUM_INSTRUMENTS,
    InstrumentFamily,
)

# A masked validator consumes the frame plus a boolean row selector.
ValidatorFn: TypeAlias = Callable[[pd.DataFrame, npt.NDArray[np.bool_]], None]

# ==============================================================================
# DISPATCH TABLES (DENSE, INTEGER-KEYED)
# ==============================================================================

_NUM_FAMILIES: Final[int] = len(InstrumentFamily)

# Family id -> coarse validator. Dense list => O(1) branchless indexing.
FAMILY_VALIDATORS: Final[list[ValidatorFn | None]] = [None] * _NUM_FAMILIES
FAMILY_VALIDATORS[InstrumentFamily.SPOT] = validate_spot_family
FAMILY_VALIDATORS[InstrumentFamily.FORWARD] = validate_forward_family
FAMILY_VALIDATORS[InstrumentFamily.OPTION] = validate_option_family

# Instrument id -> refinement validator. Most slots stay None: only instruments
# carrying constraints beyond their family contract register here.
INSTRUMENT_VALIDATORS: Final[list[ValidatorFn | None]] = [None] * NUM_INSTRUMENTS

# Instrument id -> family id, narrowed to int8 and frozen as an ndarray so the
# id->family hop is a single advanced-indexing op rather than a Python lookup.
_FAMILY_LUT: Final[npt.NDArray[np.int8]] = np.asarray(
    INSTRUMENT_TO_FAMILY_ARRAY, dtype=np.int8
)


# ==============================================================================
# INSTRUMENT ID RESOLUTION
# ==============================================================================


def resolve_instrument_ids(df: pd.DataFrame) -> npt.NDArray[np.int32]:
    """
    Map the `instrument` name column to int32 instrument ids.

    The strings die here: every downstream stage routes on the integer id.
    """
    mapped = df["instrument"].map(INSTRUMENT_NAME_TO_ID_REGISTRY)

    invalid = mapped.isna().to_numpy(copy=False)
    if invalid.any():
        names = df["instrument"]
        rows = names.index[invalid].tolist()[:10]
        bad = sorted(names[invalid].dropna().unique().tolist())
        raise ValueError(f"Unknown instrument(s):\n{bad}\nRows: {rows}")

    # Generate the contiguous array
    arr = np.ascontiguousarray(mapped.to_numpy(dtype=np.int32))

    # Cast the return value to satisfy the type checker
    return cast(npt.NDArray[np.int32], arr)


# ==============================================================================
# MAIN ENTRYPOINT
# ==============================================================================


def dispatch_ingest_validation(df: pd.DataFrame) -> npt.NDArray[np.int32]:
    """
    Validate a materialized ingest frame via table-driven routing.

    Runs global checks, then routes every row through its family validator and
    any instrument-level refinement validator. The resolved instrument-id array
    is returned so the downstream compiler can reuse it without re-mapping the
    `instrument` strings.

    Parameters
    ----------
    df
        Canonicalized + materialized ingest frame.

    Returns
    -------
    npt.NDArray[np.int32]
        Per-row instrument ids, aligned to `df`.
    """
    # ---- global invariants -------------------------------------------------
    validate_unique_uids(df)

    # ---- id + family resolution (vectorized, no row loops) -----------------
    instrument_ids = resolve_instrument_ids(df)
    family_ids = _FAMILY_LUT[instrument_ids]

    # ---- primary dispatch: family-level validation -------------------------
    for family_id, validator in enumerate(FAMILY_VALIDATORS):
        if validator is None:
            continue
        mask = family_ids == family_id
        if mask.any():
            validator(df, mask)

    # ---- secondary dispatch: instrument-level refinement -------------------
    for instrument_id, validator in enumerate(INSTRUMENT_VALIDATORS):
        if validator is None:
            continue
        mask = instrument_ids == instrument_id
        if mask.any():
            validator(df, mask)

    return instrument_ids
```

```python
# src/riskquant/portfolio/ingest/fx_resolver.py
"""
Vectorized resolution of FX-spot risk-factor ids for currency conversion.

Strategy: two gathers into static_config.fx_spot_matrix + a np.where mask.
For every row whose base currency differs from its settlement currency:

    fx_matrix[base_id, quote_id]   -> direct hit
    fx_matrix[quote_id, base_id]   -> inverse hit (record invert=1)

No string concatenation, no name fallback (e.g. "[ON]"), no priority chain.
The static_config validator guarantees `fx_spot_matrix` is the single
source of truth for FX-pair resolution.

The output `fx_invert` array is int8 (project convention for flags), not
int32. The flag carries a per-row "1 / fx_rate vs fx_rate" decision that
the pricer can apply branchlessly via `r * (1 - invert) + (1/r) * invert`.
"""

from __future__ import annotations

import numpy as np
import numpy.typing as npt
import pandas as pd

from riskquant.config.static_config.static_config import RiskStaticConfig
from riskquant.constants import NO_FX_RF_REQUIRED
from riskquant.exceptions import PrettyKeyError

# Per-row sentinel matching static_config._NO_FX_RF_REQUIRED.
# Re-declared here so the resolver doesn't import from a private symbol.
_NO_FX_INT: int = int(NO_FX_RF_REQUIRED)


def resolve_fx_rf_ids_inplace(
    *,
    base_ccy: npt.NDArray[np.str_],
    settlement_ccy: npt.NDArray[np.str_],
    out_fx_rf_ids: npt.NDArray[np.int32],
    out_fx_invert: npt.NDArray[np.int8],
    static_config: RiskStaticConfig,
) -> None:
    """
    Resolve FX-spot risk-factor ids and invert flags in place.

    Parameters
    ----------
    base_ccy, settlement_ccy
        Per-row currency arrays (ISO 3-letter strings, uppercased upstream
        by canonicalize). Rows where they match are tagged NO_FX_RF_REQUIRED.
    out_fx_rf_ids
        int32 output buffer. Receives the FX-pair rf_id or NO_FX_RF_REQUIRED.
    out_fx_invert
        int8 output buffer. 1 where the inverse pair was used, else 0.
    static_config
        Compiled static config. Supplies `ccy_name_to_id` for the per-row
        currency -> id translation and `fx_spot_matrix` for the principled
        (base_id, quote_id) -> rf_id lookup.

    Raises
    ------
    PrettyKeyError
        If any row carries a currency name not in ccy_name_to_id, or if
        neither the direct nor the inverse pair has been configured.
    """
    if base_ccy.shape != settlement_ccy.shape:
        raise ValueError(
            f"base_ccy {base_ccy.shape} and settlement_ccy {settlement_ccy.shape} "
            "must have identical shapes."
        )
    if out_fx_rf_ids.shape != base_ccy.shape:
        raise ValueError(
            f"out_fx_rf_ids {out_fx_rf_ids.shape} must match input {base_ccy.shape}."
        )
    if out_fx_invert.shape != base_ccy.shape:
        raise ValueError(
            f"out_fx_invert {out_fx_invert.shape} must match input {base_ccy.shape}."
        )
    if out_fx_rf_ids.dtype != np.int32:
        raise TypeError(f"out_fx_rf_ids must be int32; got {out_fx_rf_ids.dtype}.")
    if out_fx_invert.dtype != np.int8:
        raise TypeError(f"out_fx_invert must be int8; got {out_fx_invert.dtype}.")

    out_fx_rf_ids.fill(np.int32(_NO_FX_INT))
    out_fx_invert.fill(np.int8(0))

    # Only rows where base != settle need resolution.
    needs_fx = base_ccy != settlement_ccy
    if not needs_fx.any():
        return

    # ---- Currency NAME -> ID ----
    # One vectorized hash join per side; no per-row Python iteration.
    ccy_name_to_id = static_config.ccy_name_to_id

    base_sub = base_ccy[needs_fx]
    settle_sub = settlement_ccy[needs_fx]

    base_mapped = pd.Series(base_sub).map(ccy_name_to_id)
    settle_mapped = pd.Series(settle_sub).map(ccy_name_to_id)

    _detect_unknown_ccys(base_sub, base_mapped, ctx="base_ccy")
    _detect_unknown_ccys(settle_sub, settle_mapped, ctx="settlement_ccy")

    # Narrow to int (small currency universe; the static_config validator
    # caps max_ccy_id well within int16 range, so int32 is plenty).
    base_ids = base_mapped.to_numpy(dtype=np.int32, copy=False)
    quote_ids = settle_mapped.to_numpy(dtype=np.int32, copy=False)

    # ---- Two-gather lookup into the dense FX matrix ----
    fx_matrix = static_config.fx_spot_matrix
    std_hits = fx_matrix[base_ids, quote_ids]
    inv_hits = fx_matrix[quote_ids, base_ids]

    std_valid = std_hits != _NO_FX_INT
    inv_valid = inv_hits != _NO_FX_INT

    # ---- Unresolved rows are hard errors ----
    missing = ~(std_valid | inv_valid)
    if missing.any():
        i = int(np.flatnonzero(missing)[0])
        raise PrettyKeyError(
            "No FX spot rf configured for currency conversion:\n"
            f"  base_ccy={base_sub[i]!r} (id={int(base_ids[i])})\n"
            f"  quote_ccy={settle_sub[i]!r} (id={int(quote_ids[i])})\n"
            f"  matrix[base, quote] = {int(std_hits[i])}\n"
            f"  matrix[quote, base] = {int(inv_hits[i])}"
        )

    # ---- Scatter resolved ids + invert flags ----
    # Direct hit wins over inverse (canonical orientation). Where direct
    # missed but inverse is present, record invert=1.
    final_ids = np.where(std_valid, std_hits, inv_hits).astype(np.int32, copy=False)
    final_inv = (~std_valid).astype(np.int8, copy=False)

    out_fx_rf_ids[needs_fx] = final_ids
    out_fx_invert[needs_fx] = final_inv


def _detect_unknown_ccys(
    raw: npt.NDArray[np.str_],
    mapped: pd.Series,
    *,
    ctx: str,
) -> None:
    """Raise if any currency string failed to map to a ccy_id."""
    unknown = mapped.isna().to_numpy(copy=False)
    if not unknown.any():
        return
    bad = sorted({str(v) for v in raw[unknown]})[:10]
    raise PrettyKeyError(
        f"{ctx}: {int(unknown.sum())} currency name(s) not in static_config "
        f"ccy_name_to_id.\nUnresolved currencies:\n{bad}"
    )
```

```python
# src/riskquant/portfolio/ingest/materialize.py

"""
Materialization layer for ingest dataframes.

Purpose
-------
Convert canonicalized ingest dataframes into deterministic,
machine-typed dataframes suitable for downstream validation
and SoA extraction.

This layer performs ONLY deterministic dtype normalization in-place:

- numeric coercion
- integer coercion
- flag coercion
- string dtype normalization
- object dtype elimination

This module intentionally does NOT perform:

- semantic validation
- business logic
- RF mapping
- parsing into engine arrays
- instrument validation

Those belong downstream in:
- validator.py
- batch_builder.py

Design Philosophy
-----------------
Canonicalized human-readable dataframe
    ↓
Deterministic machine-typed dataframe (mutated in-place)
"""

from __future__ import annotations

import numpy as np
import pandas as pd

from riskquant.portfolio.ingest.schema import (
    DATE_COLS,
    DATETIME_COLS,
    FLAG_COLS,
    FLOAT_COLS,
    INT_COLS,
    NULLABLE_INT_COLS,
    STRING_COLS,
)

# ==============================================================================
# INTERNAL HELPERS
# ==============================================================================


def _materialize_string_col(
    df: pd.DataFrame,
    col: str,
) -> None:
    """
    Materialize string columns into deterministic pandas StringDtype in-place.

    Notes
    -----
    String columns intentionally remain human-readable at this stage.

    Strings are eliminated later during:
    - RF mapping
    - batch building
    """
    df[col] = df[col].astype("string", copy=False)


def _materialize_float_col(
    df: pd.DataFrame,
    col: str,
) -> None:
    """
    Materialize float columns into deterministic float64 dtype in-place.
    """
    try:
        numeric_ser = pd.to_numeric(
            df[col],
            errors="raise",
        )
    except Exception as e:
        raise TypeError(f"Failed to materialize float column '{col}': {e}") from e

    df[col] = numeric_ser.astype(np.float64, copy=False)


def _materialize_int_col(
    df: pd.DataFrame,
    col: str,
) -> None:
    """
    Materialize integer columns into deterministic int64 dtype in-place.

    Integer columns are:
    - non-nullable
    - exact integers only
    """
    ser = df[col]

    # --------------------------------------------------------------------------
    # NULL VALIDATION
    # --------------------------------------------------------------------------
    if ser.isna().any():
        bad_idx = ser.index[ser.isna()].tolist()[:10]
        raise ValueError(
            f"Integer column '{col}' contains null values at rows: {bad_idx}"
        )

    # --------------------------------------------------------------------------
    # NUMERIC COERCION
    # --------------------------------------------------------------------------
    try:
        numeric_ser = pd.to_numeric(
            ser,
            errors="raise",
        )
    except Exception as e:
        raise TypeError(f"Failed to materialize integer column '{col}': {e}") from e

    # --------------------------------------------------------------------------
    # INTEGER EXACTNESS VALIDATION
    # --------------------------------------------------------------------------
    frac, _ = np.modf(numeric_ser.to_numpy(copy=False))
    valid = frac == 0

    if not np.all(valid):
        bad = ser.iloc[np.flatnonzero(~valid)[:10]].tolist()
        raise TypeError(f"Integer column '{col}' contains non-integer values:\n{bad}")

    df[col] = numeric_ser.astype(np.int64, copy=False)


def _materialize_nullable_int_col(
    df: pd.DataFrame,
    col: str,
) -> None:
    """
    Materialize a nullable integer column into pandas Int64 in-place.

    Sibling of `_materialize_int_col` for columns where NA is a legitimate
    "absent linkage" rather than an error. NA is preserved; the downstream
    compiler substitutes the typed sentinel.
    """
    ser = df[col]

    # Boolean leakage rejection (mirrors the flag-column contract).
    non_na = ser.dropna()
    is_bool = non_na.map(lambda x: isinstance(x, bool))
    if is_bool.any():
        bad = non_na[is_bool].iloc[:10].tolist()
        raise TypeError(f"Nullable-int column '{col}' contains boolean values:\n{bad}")

    try:
        numeric_ser = pd.to_numeric(ser, errors="raise")
    except Exception as e:
        raise TypeError(
            f"Failed to materialize nullable-int column '{col}': {e}"
        ) from e

    # Integer exactness on populated rows; NA rows pass through.
    frac, _ = np.modf(numeric_ser.to_numpy(dtype=np.float64, copy=False))
    valid = numeric_ser.isna().to_numpy(copy=False) | (frac == 0)
    if not np.all(valid):
        bad = ser.iloc[np.flatnonzero(~valid)[:10]].tolist()
        raise TypeError(
            f"Nullable-int column '{col}' contains non-integer values:\n{bad}"
        )

    df[col] = numeric_ser.astype("Int64", copy=False)


def _materialize_flag_col(
    df: pd.DataFrame,
    col: str,
) -> None:
    """
    Materialize nullable flag columns in-place.

    Allowed:
        0
        1
        <NA>

    Rejected:
        True
        False
        arbitrary numerics

    Stored as:
        float64

    Rationale
    ---------
    Downstream ingest logic uses NaN semantics.
    """
    ser = df[col]

    # --------------------------------------------------------------------------
    # BOOLEAN LEAKAGE REJECTION
    # --------------------------------------------------------------------------
    non_na = ser.dropna()
    is_bool = non_na.map(lambda x: isinstance(x, bool))

    if is_bool.any():
        bad = non_na[is_bool].iloc[:10].tolist()
        raise TypeError(
            f"Flag column '{col}' contains {is_bool.sum()} boolean values:\n{bad}"
        )

    # --------------------------------------------------------------------------
    # NUMERIC COERCION
    # --------------------------------------------------------------------------
    try:
        numeric_ser = pd.to_numeric(
            ser,
            errors="raise",
        )
    except Exception as e:
        raise TypeError(f"Failed to materialize flag column '{col}': {e}") from e

    # --------------------------------------------------------------------------
    # DOMAIN VALIDATION
    # --------------------------------------------------------------------------
    valid = numeric_ser.isna() | numeric_ser.isin((0, 1))

    if not valid.all():
        bad = ser[~valid].iloc[:10].tolist()
        raise ValueError(f"Flag column '{col}' contains invalid values:\n{bad}")

    df[col] = numeric_ser.astype(np.float64, copy=False)


def _materialize_date_col(
    df: pd.DataFrame,
    col: str,
) -> None:
    """
    Materialize date columns into deterministic datetime64[ns] in-place.

    Dates are strictly validated against the ISO 8601 'YYYY-MM-DD' format.
    Providing an explicit format string ensures O(N) C-level parsing speed
    and avoids the heavy overhead of generic date inference.
    """
    try:
        df[col] = pd.to_datetime(
            df[col],
            format="%Y-%m-%d",
            exact=True,  # Strict matching, no extra characters allowed
            errors="raise",
        )
    except Exception as e:
        raise ValueError(
            f"Failed to materialize date column '{col}'. "
            f"Dates must be exactly 'YYYY-MM-DD'. "
            f"Original error: {e}"
        ) from e


def _materialize_datetime_col(
    df: pd.DataFrame,
    col: str,
) -> None:
    """
    Materialize datetime columns into deterministic datetime64[ns] in-place.

    Datetimes are strictly validated against the format 'YYYY-MM-DD HH:MM:SS.FFFFFF'.
    Providing an explicit format string ensures O(N) C-level parsing speed
    and avoids the heavy overhead of generic datetime inference.
    """
    try:
        df[col] = pd.to_datetime(
            df[col],
            format="%Y-%m-%d %H:%M:%S.%f",
            exact=True,  # Strict matching, no extra characters allowed
            errors="raise",
        )
    except Exception as e:
        raise ValueError(
            f"Failed to materialize datetime column '{col}'. "
            f"Datetimes must be exactly 'YYYY-MM-DD HH:MM:SS.FFFFFF'. "
            f"Original error: {e}"
        ) from e


# ==============================================================================
# MAIN ENTRYPOINT
# ==============================================================================


def materialize_canonicalized_ingest_df(
    df: pd.DataFrame,
) -> None:
    """
    Materialize canonicalized ingest dataframe into deterministic dtypes in-place.

    Pipeline
    --------
    - string normalization
    - float normalization
    - integer normalization
    - nullable integer normalization
    - flag normalization
    - date normalization
    - object dtype elimination

    Returns
    -------
    None
        The dataframe is mutated in-place.
    """

    # --------------------------------------------------------------------------
    # STRING COLS
    # --------------------------------------------------------------------------
    for col in STRING_COLS:
        if col in df.columns:
            _materialize_string_col(df, col)

    # --------------------------------------------------------------------------
    # FLOAT COLS
    # --------------------------------------------------------------------------
    for col in FLOAT_COLS:
        if col in df.columns:
            _materialize_float_col(df, col)

    # --------------------------------------------------------------------------
    # INT COLS
    # --------------------------------------------------------------------------
    for col in INT_COLS:
        if col in df.columns:
            _materialize_int_col(df, col)

    # --------------------------------------------------------------------------
    # NULLABLE INT COLS
    # --------------------------------------------------------------------------
    for col in NULLABLE_INT_COLS:
        if col in df.columns:
            _materialize_nullable_int_col(df, col)

    # --------------------------------------------------------------------------
    # FLAG COLS
    # --------------------------------------------------------------------------
    for col in FLAG_COLS:
        if col in df.columns:
            _materialize_flag_col(df, col)

    # --------------------------------------------------------------------------
    # DATE COLS
    # --------------------------------------------------------------------------
    for col in DATE_COLS:
        if col in df.columns:
            _materialize_date_col(df, col)

    # --------------------------------------------------------------------------
    # DATETIME COLS
    # --------------------------------------------------------------------------
    for col in DATETIME_COLS:
        if col in df.columns:
            _materialize_datetime_col(df, col)

    # --------------------------------------------------------------------------
    # OBJECT DTYPE ELIMINATION
    # --------------------------------------------------------------------------
    obj_cols = df.select_dtypes(include=["object"]).columns

    if len(obj_cols):
        raise RuntimeError(
            f"Object dtype columns remain after materialization:\n{obj_cols.tolist()}"
        )
```

```python
from __future__ import annotations

from datetime import date, datetime

import numpy as np
import pandas as pd
from pandas.api.types import (
    is_datetime64_any_dtype,
)

from riskquant.portfolio.ingest.utils import is_na_helper
from riskquant.utils.dates import (
    datetime64D_to_ord32,
    is_date_str_valid,
)


def parse_optional_date_series_to_ord32(
    ser: pd.Series,
    *,
    col_name: str,
    sentinel: np.int32,
) -> np.ndarray:
    """
    Convert a date-like column to ordinal32 days since DATE_EPOCH using the canonical
    riskquant date encoding (datetime64[D] -> ordinal32).

    Accepted inputs:
      - NaN / None / ""        -> sentinel
      - datetime64 / Timestamp -> encoded via datetime64D_to_ord32
      - str "YYYY-MM-DD"       -> validated via is_date_str_valid then encoded
    """
    # Number of rows in the input series
    n = int(len(ser))

    # Initialize the output array with the sentinel value
    out = np.empty(n, dtype=np.int32)
    out.fill(np.int32(sentinel))

    # Early exit if the series is empty, since the rest of the code assumes at least one row
    if n == 0:
        return out

    # Convert the series to a numpy array for efficient access, without copying if possible
    date_vals = ser.to_numpy(copy=False)

    # Create a boolean mask for non-NA values.
    nn_mask = pd.notna(date_vals)

    # If the dtype is object, we also want to treat empty strings as NA.
    if date_vals.dtype == object:
        date_vals_bind = date_vals  # local binding for efficiency
        nn_mask_bind = nn_mask  # local binding for efficiency
        for i in range(n):
            if nn_mask_bind[i]:
                v = date_vals_bind[i]
                if isinstance(v, str) and (v == "" or v.isspace()):
                    nn_mask_bind[i] = False

    # If there are no non-NA values, we can return early with the all-sentinel output.
    if not np.any(nn_mask):
        return out

    # If the dtype of the series is timezone-aware datetime, we should raise an error since we don't support timezones in our date encoding.
    if isinstance(ser.dtype, pd.DatetimeTZDtype):
        raise TypeError(
            f"Invalid {col_name}: timezone-aware datetimes are not supported."
        )

    # If the dtype of the non-NA values is already datetime64, we can convert them directly to ordinal32 without having to iterate over them.
    if is_datetime64_any_dtype(ser.dtype):
        # Convert the series to datetime64[D] without copying if possible.
        d64 = date_vals.astype("datetime64[D]", copy=False)
        out[nn_mask] = datetime64D_to_ord32(d64[nn_mask])
        return out

    # Get the indices of the non-NA values
    idx = np.nonzero(nn_mask)[0]

    # Pre-allocate a datetime64[D] array.
    d64 = np.empty(idx.shape[0], dtype="datetime64[D]")

    # Local bindings for efficiency in the loop
    _is_na = is_na_helper
    _is_date_str_valid = is_date_str_valid

    # Iterate over the non-NA values and parse each one according to its type.
    for k, i in enumerate(idx):
        # Get the original value from the series for this index.
        x = date_vals[i]

        # Smoke test! If the value is NA (None, NaN, or empty string), we should have already marked it as NA in the mask, so we can skip it.
        if _is_na(x):
            raise ValueError(
                f"Internal error: encountered NA value at row {int(i)} that was not marked as NA in the mask. Value: {x}"
            )

        # Now we know x is not NA, so we can parse it according to the accepted input types.
        if isinstance(x, (pd.Timestamp, np.datetime64, datetime, date)):
            d64[k] = np.datetime64(x, "D")
            continue

        # If the dtype of the series is object, we may have string values that need to be validated and parsed.
        if isinstance(x, str):
            xs = x.strip()
            if not _is_date_str_valid(xs):
                raise ValueError(
                    f"Invalid {col_name}='{xs}' at row {int(i)}!"
                    f"Got a string that is not a valid date in YYYY-MM-DD format: '{xs}'. "
                    "Expected strict YYYY-MM-DD."
                )
            d64[k] = np.datetime64(xs, "D")
            continue

        # If we get here, it means x is not NA and not one of the accepted types, so we will raise an error with a helpful message.
        raise TypeError(
            f"Invalid {col_name} type at row {int(i)}: value = {x}, type = {type(x).__name__}. "
            "Expected YYYY-MM-DD string, pandas Timestamp, datetime, or date."
        )

    # At this point, we have successfully parsed all the non-NA values into the d64 array as datetime64[D]. We can now convert them to ordinal32 and store them in the output array at the correct indices.
    out[idx] = datetime64D_to_ord32(d64)

    # Return.
    return out
```

```python
# src/riskquant/portfolio/ingest/pipeline.py

"""
Top-level ingest orchestrator.This wires the full
edge pipeline so a raw, human-facing frame is normalized and validated before
any SoA payload is produced.
"""

from __future__ import annotations

import pandas as pd

from riskquant.config.static_config.static_config import RiskStaticConfig
from riskquant.portfolio.ingest.batch import IngestBatch
from riskquant.portfolio.ingest.canonicalize import canonicalize_raw_ingest_df
from riskquant.portfolio.ingest.compile import (
    compile_ingest_soa_payload_from_df,
)
from riskquant.portfolio.ingest.dispatcher import dispatch_ingest_validation
from riskquant.portfolio.ingest.materialize import (
    materialize_canonicalized_ingest_df,
)


def compile_raw_df_into_ingest_batch(
    *,
    raw_df: pd.DataFrame,
    static_config: RiskStaticConfig,
    copy: bool = False,
) -> IngestBatch:
    """
    Run the full ingest pipeline: raw frame -> validated SoA `IngestBatch`.

    Stages
    ------
    1. canonicalize : column names, ordering, string normalization (in place)
    2. materialize  : strict dtype coercion (in place)
    3. dispatch     : table-driven per-instrument validation; returns int32 ids
    4. compile      : DataFrame -> flat SoA `CompiledIngestSoA`
    5. pack         : wrap into the immutable `IngestBatch` contract

    Parameters
    ----------
    raw_df
        Human-facing positions frame (market-risk and/or JTD credit rows).
    static_config
        The full `RiskStaticConfig` is required to validate the raw frame and
        compile the SoA arrays.
    copy
        When True (default) the caller's frame is left untouched: stages 1-2
        mutate a private copy. Set False only when the caller already owns a
        throwaway frame and wants to avoid the edge allocation.
    """
    # Canonicalize / materialize mutate in place; copy once at the edge so the
    # caller's frame is never clobbered.
    df = raw_df.copy() if copy else raw_df

    canonicalize_raw_ingest_df(df)
    materialize_canonicalized_ingest_df(df)

    # Validation routing also yields the int32 instrument ids, so the
    # `instrument` strings are hash-mapped exactly once for the whole pipeline.
    instrument_ids = dispatch_ingest_validation(df)

    # Compile the canonicalized, validated frame into the flat SoA payloads expected
    # by the portfolio engine. This is the last step before packing into the immutable
    # `IngestBatch` contract.
    compiled_ingest_soa = compile_ingest_soa_payload_from_df(
        df=df,
        instrument_ids=instrument_ids,
        static_config=static_config,
    )
    # compiled_ingest_soa.print_soa()

    return IngestBatch.from_compiled_ingest_soa(compiled_ingest_soa)
```

```python
# src/riskquant/portfolio/ingest/rf_resolver.py

"""
Vectorized resolution of human-facing risk-factor name strings into the int32
Universe Indices (UIX) consumed by the compiled blocks and JIT kernels.

Sibling of `fx_resolver.py`: that module resolves FX-pair strings; this one
resolves single-name risk factors (underlyings, discount curves, price/vol
factors). Together they are the layer where "strings die early" — every name
column is collapsed to a flat C-contiguous np.int32 array before any data
leaves the ingest edge.

All resolution is performed through pandas `Series.map` (a C-level hash join).
There are no Python row loops, no `.apply`, and no regex matching.
"""

from __future__ import annotations

from collections.abc import Mapping
from typing import cast

import numpy as np
import numpy.typing as npt
import pandas as pd

from riskquant.exceptions import PrettyKeyError

# A canonicalized string column uses the empty string as its missing sentinel
# (canonicalize.py maps NA -> "" on extraction). "" is never a valid RF name.
_BLANK: str = ""


def _detect_unmapped(
    names: npt.NDArray[np.str_],
    mapped: pd.Series,
    blank_mask: npt.NDArray[np.bool_],
    *,
    ctx: str,
) -> None:
    """Raise on names that are populated but absent from the lookup table."""
    unmapped = mapped.isna().to_numpy(copy=False) & ~blank_mask
    if not unmapped.any():
        return

    bad = sorted({str(v) for v in names[unmapped]})[:10]
    raise PrettyKeyError(
        f"{ctx}: {int(unmapped.sum())} name(s) could not be resolved to a "
        f"risk-factor universe index.\nUnresolved names:\n{bad}"
    )


def resolve_required_ids(
    names: npt.NDArray[np.str_],
    mapping: dict[str, int],
    *,
    ctx: str,
) -> npt.NDArray[np.int32]:
    """
    Resolve a name column where every row MUST carry a mappable name.

    Parameters
    ----------
    names
        Flat string array (blanks are treated as errors here).
    mapping
        Name -> universe-index lookup table.
    ctx
        Human-readable column label for diagnostics.
    """
    blank_mask = names == _BLANK
    mapped = pd.Series(names).map(mapping)
    _detect_unmapped(names, mapped, blank_mask, ctx=ctx)

    if blank_mask.any():
        bad = np.flatnonzero(blank_mask)[:10].tolist()
        raise PrettyKeyError(f"{ctx}: required name missing at rows {bad}.")
    # No NaNs survive past the guards above -> direct int32 narrowing is safe.
    arr = np.ascontiguousarray(mapped.to_numpy(dtype=np.int32))
    return cast(npt.NDArray[np.int32], arr)


def resolve_optional_ids(
    names: npt.NDArray[np.str_],
    mapping: Mapping[str, int | np.int32],
    *,
    sentinel: int | np.integer,
    ctx: str,
) -> npt.NDArray[np.int32]:
    """
    Resolve a name column where blank rows are legitimately absent.

    Blank rows collapse to `sentinel`. A populated-but-unmapped name is still a
    hard error: it signals a typo or a stale universe rather than absence.

    Parameters
    ----------
    names
        Flat string array; blank rows ("") map to `sentinel`.
    mapping
        Name -> universe-index lookup table.
    sentinel
        Typed sentinel (from riskquant.constants) for "no linkage".
    ctx
        Human-readable column label for diagnostics.
    """
    n = names.shape[0]
    blank_mask = names == _BLANK
    mapped = pd.Series(names).map(mapping)

    # Edge allocation; downstream kernels never touch this array directly.
    out = np.empty(n, dtype=np.int32)
    out.fill(np.int32(sentinel))

    populated = ~blank_mask
    if populated.any():
        out[populated] = mapped.to_numpy(dtype=np.float64)[populated].astype(np.int32)

    return out  # np.empty is already C-contiguous
```

```python
import numpy as np
import pandas as pd


def is_na_helper(x: object) -> bool:
    """Check if a scalar value is missing or invalid (None, NaN, pd.NA, ±inf)."""
    if x is None:
        # None is considered missing.
        return True

    if x is pd.NA:
        # pd.NA is considered missing.
        return True

    if isinstance(x, (float, np.floating)):
        # Check for NaN or infinite values.
        # np.isfinite returns False for NaN and ±inf, so we return True for those cases.
        # NaNs and infinite values are considered missing/invalid.
        return not np.isfinite(x)

    # For all other types (e.g., int, str), we consider them valid and not missing.
    return False
```

```python
def _require_cols(df: pd.DataFrame, cols: tuple[str, ...]) -> None:
    missing = [c for c in cols if c not in df.columns]
    if missing:
        raise ValueError(f"Missing required columns:\n{missing}")


def _validate_non_null_masked(
    df: pd.DataFrame,
    mask: npt.NDArray[np.bool_],
    cols: tuple[str, ...],
    ctx: str,
) -> None:
    for col in cols:
        arr = df[col].isna().to_numpy(copy=False)
        bad = mask & arr
        if np.any(bad):
            idx = df.index[bad].tolist()[:10]
            raise ValueError(f"{ctx} missing '{col}' at rows: {idx}")


def _validate_ccy_series(
    df: pd.DataFrame,
    col: str,
    mask: npt.NDArray[np.bool_],
    ctx: str,
) -> None:
    ser = df[col]
    valid_idx = np.flatnonzero(mask & ser.notna().to_numpy(copy=False))

    if len(valid_idx) == 0:
        return

    subset = ser.iloc[valid_idx]

    alpha = subset.str.isalpha()
    if not alpha.all():
        idx = subset.index[~alpha].tolist()[:10]
        raise ValueError(f"{ctx} invalid currency (non-alpha) at rows: {idx}")

    length = subset.str.len().eq(3)
    if not length.all():
        idx = subset.index[~length].tolist()[:10]
        raise ValueError(f"{ctx} invalid currency (len!=3) at rows: {idx}")


def _validate_non_zero_quantity(
    df: pd.DataFrame,
    mask: npt.NDArray[np.bool_],
    ctx: str,
) -> None:
    qty = df["quantity"].to_numpy(copy=False)

    with np.errstate(invalid="ignore"):
        dust = np.abs(qty) < POSITION_DUST_TOLERANCE

    bad = mask & dust
    if np.any(bad):
        idx = df.index[bad].tolist()[:25]
        raise ValueError(
            f"{ctx} contain zero / dust quantities.\n"
            f"Tolerance={POSITION_DUST_TOLERANCE}\nRows: {idx}"
        )

    id_arr = ser.map(INSTRUMENT_NAME_TO_ID_REGISTRY).to_numpy()

    if np.any(pd.isna(id_arr)):
        bad_mask = pd.isna(id_arr)
        idx = ser.index[bad_mask].tolist()[:10]
        vals = sorted(ser[bad_mask].dropna().unique().tolist())

        raise ValueError(f"Invalid instruments:\n{vals}\nRows: {idx}")

    return id_arr.astype(np.int32, copy=False)


def _derive_family_array(
    inst_id_arr: npt.NDArray[np.int32],

    val: int,
```

```python
"""
Pre-compile validation of `IngestBatch` global-ID references against the
`UixMapper` universes.

Runs ONCE between universe discovery (`build_uix_mapper`) and block
compilation. Catches any global ID or ordinal that the upstream resolver
emitted but the universe-builder didn't see — i.e. a referential-integrity
bug somewhere in the ingest pipeline. By doing this before compile, the
hot-path `map_*_to_uix` calls can stay branch-light and trust their input.
"""

from __future__ import annotations

from collections.abc import Iterable, Sequence

import numpy as np
import numpy.typing as npt

from riskquant.portfolio.engine import UixMapper
from riskquant.portfolio.ingest.batch import IngestBatch


def validate_ingest_batches_against_uix_universe(
    *,
    ingest_batches: Sequence[IngestBatch],
    uix_mapper: UixMapper,
) -> None:
    """
    Validate that every non-sentinel ID and ordinal referenced by the
    ingest batches is present in the corresponding `UixMapper` universe.

    Raises `ValueError` with the first batch of offending values it finds,
    grouped by universe.
    """
    # --- MARKET IDs -------------------------------------------------------
    _check_arrays_against_universe(
        arrays=(
            arr
            for b in ingest_batches
            for arr in (
                b.underlying_ids,
                b.volatility_ids,
                b.fx_u2s_ids,
                b.fx_k2s_ids,
            )
        ),
        universe=uix_mapper.market_id_universe,
        label="market IDs",
    )

    # --- DISCOUNT CURVE IDs -----------------------------------------------
    _check_arrays_against_universe(
        arrays=(
            arr
            for b in ingest_batches
            for arr in (
                b.under_discount_curve_ids,
                b.strike_discount_curve_ids,
            )
        ),
        universe=uix_mapper.discount_curve_id_universe,
        label="discount-curve IDs",
    )

    # --- DATES ------------------------------------------------------------
    _check_arrays_against_universe(
        arrays=(
            arr
            for b in ingest_batches
            for arr in (b.settle_ord, b.delivery_ord, b.expiry_ord)
        ),
        universe=uix_mapper.date_ord32_universe,
        label="ordinal dates",
    )


def _check_arrays_against_universe(
    *,
    arrays: Iterable[npt.NDArray[np.int32]],
    universe: npt.NDArray[np.int32],
    label: str,
) -> None:
    """
    For every input array, assert that every value >= 0 also appears in
    `universe`. Vectorized: one `searchsorted` per input array, no Python
    loop over rows.
    """
    for arr in arrays:
        populated = arr[arr >= 0]
        if populated.size == 0:
            continue

        idx = np.searchsorted(universe, populated)
        in_bounds = idx < universe.size

        matches = np.zeros(populated.shape[0], dtype=bool)
        matches[in_bounds] = universe[idx[in_bounds]] == populated[in_bounds]

        if not matches.all():
            missing = np.unique(populated[~matches])
            raise ValueError(
                f"Unknown {label} encountered during UIX validation: {missing}"
            )
```

```python
"""
Scenario-generator protocol, batch container, and the one canonical entry
point for resolving a required-factor universe into a ShockUniverse +
ShockComputePlan.

Design
------
`ScenarioGenerator` is a streaming protocol: generators yield
`ScenarioSetBatch` chunks of fixed `batch_size` instead of materialising
the full (T, K) shock matrix. The runner holds one pre-allocated
shocked-levels buffer and consumes each batch in place.

`resolve_shock_universe` replaces the old dict-based
`build_factor_id_to_col_index` + `build_shock_key_plan` helpers. It
takes `RiskFactorMatrix.factor_ids` (sorted ascending int32) directly
and uses `np.searchsorted` for O(K log N) vectorised resolution.
"""

from __future__ import annotations

from collections.abc import Iterator
from dataclasses import dataclass
from typing import Protocol

import numpy as np
import numpy.typing as npt

from riskquant.simlab.builder.shock_universe import (
    ShockUniverse,
    build_shock_universe,
)

# ============================================================
# Streaming-batch payload
# ============================================================


@dataclass(frozen=True, slots=True)
class ScenarioSetBatch:
    """
    One streaming chunk of scenarios in shock-space.

    `shocks` is F-contiguous `(B, K_req)` so the apply-shocks kernel
    reads down the scenario axis with unit stride. `required_to_shock_uix`
    maps required-exposure rows back to columns of `shocks`.
    """

    shocks: np.ndarray
    required_to_shock_uix: np.ndarray
    shock_scale: np.ndarray
    required_shift_type_id: np.ndarray
    scenario_offset: int
    scenario_count_total: int | None = None


# ============================================================
# Generator protocol
# ============================================================


class ScenarioGenerator(Protocol):
    """Streaming-first scenario generation."""

    total_scenarios: int
    available_factor_ids: np.ndarray

    def iter_batches(
        self,
        *,
        required_shift_type_id: np.ndarray,  # (Nreq,) int8
        source_factor_ids: np.ndarray,  # (Nreq,) int32
        shock_scale: np.ndarray,  # (Nreq,) float32
        batch_size: int,
    ) -> Iterator[ScenarioSetBatch]: ...


# ============================================================
# Vectorised universe resolution
# ============================================================


def resolve_shock_universe(
    *,
    required_shift_type_id: npt.NDArray[np.int8],
    source_factor_ids: npt.NDArray[np.int32],
    available_factor_ids: npt.NDArray[np.int32],
) -> ShockUniverse:
    """
    Build a `ShockUniverse` from the run's resolved-source universe and
    the generator's available-factor index.

    `available_factor_ids` MUST be sorted ascending and unique (the
    `RiskFactorMatrix` invariant). This lets us replace the dict-based
    lookup with one vectorised `np.searchsorted` inside
    `build_shock_universe`.

    The previous two-stage helper pair (`build_factor_id_to_col_index`
    + `build_shock_key_plan`) is gone — both were Python-loop with
    dict-lookup anti-patterns. This is the single entry point all
    generators use.
    """
    return build_shock_universe(
        required_shift_type_id=required_shift_type_id,
        source_factor_ids=source_factor_ids,
        available_factor_ids=available_factor_ids,
    )
```

```python
"""
Portfolio valuation engine.

Owns the instrument-id -> evaluator dispatch table and the scenario valuation
loop. Evaluator resolution lives here, in the valuation layer: `run_valuation`

from typing import Any, Final, Protocol, TypeVar, cast, runtime_checkable

import numpy as np
import numpy.typing as npt

from riskquant.market_data.state import MarketDataMatrix
from riskquant.portfolio.blocks.base_block import Block
from riskquant.portfolio.compiled import CompiledPortfolio
from riskquant.portfolio.instruments import (

from riskquant.valuation.pricers.spot import price_spot_block
from riskquant.valuation.pricers.vanilla_european_option_curve import (

from riskquant.valuation.pricers.vanilla_european_option_node import (

)
from riskquant.valuation.pricers.vanilla_forward_curve import (
    price_vanilla_forward_curve_block,
)
from riskquant.valuation.pricers.vanilla_forward_node import (
    price_vanilla_forward_node_block,
)
from riskquant.valuation.result import ValuationResult

# ==============================================================================
# EVALUATOR PROTOCOL
# ==============================================================================

# 1. Define a contravariant TypeVar bound to Block
B_contra = TypeVar("B_contra", bound=Block, contravariant=True)


@runtime_checkable
class Evaluator(Protocol[B_contra]):
    """
    Structural contract for all block evaluators (pricers + risk evaluators).
    """

    def __call__(
        self,
        *,
        block: B_contra,  # 2. Use the TypeVar instead of base Block
        market_data_mtx: MarketDataMatrix,
        out: npt.NDArray[np.float64],
    ) -> None: ...


# ==============================================================================
# DISPATCH TABLE
# ==============================================================================

# 3. Explicitly type the list elements as Evaluator[Any] | None
_BLOCK_EVALUATORS_MUT: list[Evaluator[Any] | None] = [None] * NUM_INSTRUMENTS


def _register(instrument_id: int, evaluator: Evaluator[Any]) -> None:
    # runtime_checkable ignores TypeVars, so this check still works natively
    if not isinstance(evaluator, Evaluator):
        raise TypeError(
            f"Evaluator for instrument_id={instrument_id} does not satisfy the "
            f"Evaluator protocol: {evaluator!r}"
        )
    _BLOCK_EVALUATORS_MUT[instrument_id] = evaluator


_register(
    int(SPOT_ID),
    price_spot_block,
)
_register(
    int(VANILLA_FORWARD_NODE_ID),
    price_vanilla_forward_node_block,
)
_register(
    int(VANILLA_FORWARD_CURVE_ID),
    price_vanilla_forward_curve_block,
)
_register(
    int(VANILLA_EUROPEAN_OPTION_NODE_ID),
    price_vanilla_european_option_node_block,
)
_register(
    int(VANILLA_EUROPEAN_OPTION_CURVE_ID),
    price_vanilla_european_option_curve_block,
)


# Freeze. No mutations after module load.
# This tells the type checker: "Trust me, the runtime guard above handled the Nones."
BLOCK_EVALUATORS: Final[tuple[Evaluator[Any], ...]] = cast(
    tuple[Evaluator[Any], ...], tuple(_BLOCK_EVALUATORS_MUT)
)

# Hard fail at import if any instrument has no evaluator: catches the case where
# a new Instrument variant is added but not wired up here.
_unregistered = [
    Instrument(i).name for i, ev in enumerate(BLOCK_EVALUATORS) if ev is None
]
if _unregistered:
    raise RuntimeError(
        f"No evaluator registered for instruments: {_unregistered}. "
        f"Register them in engine.py."
    )


def resolve_evaluator(block: Block) -> Evaluator:
    """
    Resolve the evaluator for a compiled block via its `INSTRUMENT_ID`.

    Single-array index — no `type()`, no isinstance ladder, no string keys.
    """
    evaluator = BLOCK_EVALUATORS[block.INSTRUMENT_ID]
    if evaluator is None:  # unreachable given the import-time guard above
        raise RuntimeError(
            f"No evaluator for INSTRUMENT_ID={block.INSTRUMENT_ID} "
            f"({type(block).__name__})."
        )
    return evaluator


# ==============================================================================
# VALUATION ENGINE
# ==============================================================================


def run_valuation(
    *,
    portfolio: CompiledPortfolio,
    market_data_mtx: MarketDataMatrix,
) -> ValuationResult:
    """
    Execute full portfolio valuation across all scenarios.

    Iterates pre-compiled blocks in portfolio order, resolves each block's
    evaluator from its `INSTRUMENT_ID`, and dispatches it. All writes target
    contiguous slices of a single pre-allocated (n_trades x n_scenarios) C-order
    output array. Zero intermediate allocations; the GIL is bypassed inside the
    Numba kernels.

    Returns
    -------
    npt.NDArray[np.float64]
        Shape (n_trades, n_scenarios), C-contiguous. Row i corresponds to the
        trade at portfolio offset i.
    """
    n_trades: int = portfolio.total_trades
    n_scenarios: int = market_data_mtx.num_scenarios

    out: npt.NDArray[np.float64] = np.empty(
        (n_trades, n_scenarios),
        dtype=np.float64,
        order="C",
    )

    # Hoist repeated attribute lookups out of the loop.
    blocks = portfolio.blocks
    offsets = portfolio.block_offsets
    evaluators_table = BLOCK_EVALUATORS
    md = market_data_mtx

    for i in range(portfolio.total_blocks):
        block = blocks[i]
        start = int(offsets[i])
        end = start + block.num_trades

        # id -> evaluator: one O(1) table index, resolved here in the engine.
        evaluator = evaluators_table[block.INSTRUMENT_ID]

        evaluator(
            block=block,
            market_data_mtx=md,
            out=out[start:end, :],
        )

    return ValuationResult(
        trade_uids=portfolio.trade_uids,
        values=out,
    )
```

```python
"""
Shared bare-metal math helpers for vectorized pricer kernels.

Tiny hot-path functions: `@nb.njit(inline='always')` collapses them into
the surrounding kernel at compile time, so the calling kernel sees fused
code with no call-frame overhead.

Anything that needs `prange` or more than a handful of ops should not
live here.
"""

from __future__ import annotations

import math

from riskquant._njit import inline_njit

# 1 / sqrt(2). Lifted so the kernel never recomputes it.
_INV_SQRT_2: float = 0.7071067811865476


@inline_njit
def numba_norm_cdf(x: float) -> float:
    """Standard normal CDF: Phi(x) = 0.5 * (1 + erf(x / sqrt(2)))."""
    return 0.5 * (1.0 + math.erf(x * _INV_SQRT_2))


@inline_njit
def numba_norm_cdf_abrom_steg(x: float) -> float:

    ax = abs(x)

    t = 1.0 / (1.0 + 0.2316419 * ax)

    poly = t * (
        0.319381530
        + t * (-0.356563782 + t * (1.781477937 + t * (-1.821255978 + t * 1.330274429)))
    )

    pdf = 0.39894228040143267794 * math.exp(-0.5 * ax * ax)

    cdf = 1.0 - pdf * poly

    if x < 0.0:
        return 1.0 - cdf

    return cdf


@inline_njit
def linear_interp_scalar(x: float, xs, ys) -> float:
    """
    Branchless 1D linear interpolation with flat extrapolation.

    Drops the `np.interp` call frame and Numba's bounds-check overhead in
    favour of an inlinable binary search + linear blend. LLVM fuses this
    into the caller's scenario loop.

    Both `xs` and `ys` must be 1-D C-contiguous; `xs` strictly ascending.
    For `x` outside `[xs[0], xs[-1]]` returns the corresponding edge value
    (flat extrapolation), matching `np.interp`'s default semantics and
    avoiding accidental log/division blow-ups on wide-strike moneyness.
    """
    n = xs.shape[0]
    if n == 1:
        return ys[0]

    # Edge clamps — flat extrapolation. Branches are predictable
    # (rare) and the compiler can lift them out of vectorised loops.
    if x <= xs[0]:
        return ys[0]
    if x >= xs[n - 1]:
        return ys[n - 1]

    # Binary search for the right-hand pillar. Loop trip count is
    # log2(n) and the compiler unrolls it for small n.
    lo = 0
    hi = n - 1
    while hi - lo > 1:
        mid = (lo + hi) >> 1
        if xs[mid] <= x:
            lo = mid
        else:
            hi = mid

    x0 = xs[lo]
    x1 = xs[hi]
    y0 = ys[lo]
    y1 = ys[hi]
    # x1 > x0 by strict-monotone invariant.
    w = (x - x0) / (x1 - x0)
    return y0 + w * (y1 - y0)
```

```python
"""
Spot pricing kernel.

PV = qty * spot_price * fx_effective

`fx_spots_effective` rows are already the resolved transform (direct or
pre-inverted) — pricer kernels do no inversion math. Per-trade FX is one
gather + one multiply per scenario. The `fx_effective_uix < 0` sentinel
selects the "no FX" branch which is hoisted outside the scenario loop.

from __future__ import annotations

import numba as nb  # type: ignore[import-untyped]

from riskquant._njit import parallel_njit

# ==============================================================================
# Numba layout-pinned aliases — force C-contiguous strides at compile time so
# the kernel emits straight-line SIMD loops with no dynamic stride checks.
# ==============================================================================

_F64_2D = nb.float64[:, ::1]
_F64_1D = nb.float64[::1]
_I32_1D = nb.int32[::1]


@parallel_njit(
    nb.void(
        _F64_2D,  # out
        _F64_2D,  # prices
        _I32_1D,  # price_uix
        _F64_1D,  # qty
        _F64_2D,  # fx_spots_effective
        _I32_1D,  # fx_spots_effective_uix
    ),
)
def _kernel_price_spot(
    out,
    prices,
    price_uix,
    qty,
    fx_spots_effective,
    fx_spots_effective_uix,
) -> None:
    """
    Fused spot pricing kernel: out[i, s] = prices[p, s] * qty[i] * fx[f, s].

    Zero allocations. Each row is a contiguous gather + scale + (gated)
    multiply streaming through C-contiguous memory.
    """
    m_trades, s_scenarios = out.shape

    for i in nb.prange(m_trades):
        p_idx = price_uix[i]
        f_idx = fx_spots_effective_uix[i]
        q = qty[i]

        p_row = prices[p_idx]
        out_row = out[i]

        if f_idx < 0:
            # No FX leg: straight price * qty.
            for s in range(s_scenarios):
                out_row[s] = p_row[s] * q
        else:
            # FX leg: three-way fused multiply.
            f_row = fx_spots_effective[f_idx]
            for s in range(s_scenarios):
                out_row[s] = p_row[s] * q * f_row[s]
```

```python
# src/riskquant/valuation/kernels/vanilla_european_option_curve.py
"""
Vanilla European option pricing kernel, curve variant.

Identical CIP-forward + Black-76 math to the node variant, with two
structural differences:

  - The underlying scalar comes from `price_curve_grid[curve_uix, delivery_uix, s]`.
  - Volatility is sourced from a 4D vol surface and requires per-scenario
    strike-axis interpolation along the pillar coordinate dictated by the
    surface's y-axis type.

FX rows are already EFFECTIVE — no inversion math; the strike-FX rate is
loaded once per scenario, used as a divisor for CIP forward synthesis and
as a multiplier for the PV translation back to settle ccy.

Layout contract (CRITICAL)
--------------------------
`vol_surfaces` MUST be `[surface, expiry, scenario, pillar]` (C-contig).
This guarantees that the pillars for a single trade-scenario tuple form
one contiguous run of memory: `vol_surfaces[v, exp, s]` is a 1-D view
suitable for the binary-search interp. The builder is responsible for
emitting this layout; the kernel relies on it for SIMD-friendly reads.

Branch hoisting
---------------
All sentinel checks (`fx_*_uix < 0`, `df_*_uix < 0`) and y-axis-type
selection are resolved once per trade, outside the scenario loop. Within
the scenario loop only the math executes; CPU branch predictors never
see a `y_type` ladder or a sentinel test per scenario.

Vol-surface-uix sentinel handling
---------------------------------
`v_surf_id = vol_surface_uix[i]` may be `INVALID_UIX (< 0)` when the
ingest compiler resolved no vol container for trade i. With
`boundscheck=False`, `vol_surface_strikes[v_surf_id]` and
`vol_surfaces[v_surf_id, exp_id]` on a negative id are undefined
behaviour — out-of-bounds memory.

This kernel hoists `has_vol = v_surf_id >= 0` once per trade, outside
the scenario loop. The sentinel path writes `out_row[:] = 0.0` and skips
the surface gather + Black-76 math entirely.
"""

from __future__ import annotations

import math

import numba as nb  # type: ignore[import-untyped]

from riskquant._njit import parallel_njit
from riskquant.market_data.surfaces.y_axes_types import (
    SURFACE_Y_AXIS_TYPE_LOG_MONEYNESS_FWD,
    SURFACE_Y_AXIS_TYPE_LOG_MONEYNESS_SPOT,
    SURFACE_Y_AXIS_TYPE_LOG_STRIKE,
    SURFACE_Y_AXIS_TYPE_MONEYNESS_FWD,
    SURFACE_Y_AXIS_TYPE_MONEYNESS_SPOT,
    SURFACE_Y_AXIS_TYPE_STRIKE,
)
from riskquant.valuation.kernels._math import (
    linear_interp_scalar,
    numba_norm_cdf_abrom_steg,
)

_F64_2D = nb.float64[:, ::1]
_F64_3D = nb.float64[:, :, ::1]
_F64_4D = nb.float64[:, :, :, ::1]
_F64_1D = nb.float64[::1]
_I32_1D = nb.int32[::1]
_I8_1D = nb.int8[::1]

_TTE_FLOOR: float = 1e-9
_VOL_STD_DEV_FLOOR: float = 1e-12

# Y-axis kind groupings as int8 keys. Resolved once per trade and packed
# into a 0/1/2/3 dispatch tag so the inner loop sees no branch ladder.
_Y_KIND_STRIKE: int = 0  # raw strike: y = k
_Y_KIND_LOG_STRIKE: int = 1  # y = log(k)
_Y_KIND_MONEYNESS: int = 2  # y = k / fwd
_Y_KIND_LOG_MONEYNESS: int = 3  # y = log(k / fwd)


@parallel_njit(
    nb.void(
        _F64_2D,  # out
        _F64_3D,  # price_curve_grid
        _F64_2D,  # vol_surface_strikes
        _F64_4D,  # vol_surfaces  [surface, expiry, scenario, pillar]
        _F64_2D,  # fx_spots_effective
        _F64_3D,  # df_grid
        _F64_1D,  # tte_grid
        _F64_1D,  # qty
        _F64_1D,  # strike
        _I8_1D,  # cp_mult
        _I32_1D,  # price_curve_uix
        _I32_1D,  # vol_surface_uix
        _I8_1D,  # y_axis_type_id
        _I32_1D,  # fx_u_eff_uix
        _I32_1D,  # df_u_uix
        _I32_1D,  # fx_k_eff_uix
        _I32_1D,  # df_k_uix
        _I32_1D,  # delivery_date_uix
        _I32_1D,  # expiry_date_uix
        _I32_1D,  # settle_date_uix
    ),
)
def _kernel_price_vanilla_european_option_curve(
    out,
    price_curve_grid,
    vol_surface_strikes,
    vol_surfaces,
    fx_spots_effective,
    df_grid,
    tte_grid,
    qty,
    strike,
    cp_mult,
    price_curve_uix,
    vol_surface_uix,
    y_axis_type_id,
    fx_u_eff_uix,
    df_u_uix,
    fx_k_eff_uix,
    df_k_uix,
    delivery_date_uix,
    expiry_date_uix,
    settle_date_uix,
) -> None:
    m_trades, s_scenarios = out.shape

    for i in nb.prange(m_trades):
        q = qty[i]
        k = strike[i]
        cp = cp_mult[i]

        p_curve_id = price_curve_uix[i]
        v_surf_id = vol_surface_uix[i]
        y_type = y_axis_type_id[i]

        fx_u_id = fx_u_eff_uix[i]
        df_u_id = df_u_uix[i]
        fx_k_id = fx_k_eff_uix[i]
        df_k_id = df_k_uix[i]

        deliv_id = delivery_date_uix[i]
        exp_id = expiry_date_uix[i]
        stl_id = settle_date_uix[i]

        out_row = out[i]

        # ---- Hoist vol-surface-uix sentinel: no surface -> zero PV ----
        # Resolving this BEFORE the surface gather is critical:
        # vol_surface_strikes[<0] and vol_surfaces[<0, ...] are UB under
        # boundscheck=False. Sentinel-trade rows write zeros and skip.
        has_vol = v_surf_id >= 0

        if not has_vol:
            for s in range(s_scenarios):
                out_row[s] = 0.0
            continue

        # ---- Trade-level scalars (hoist out of scenario loop) ----
        tte = tte_grid[exp_id]
        safe_tte = tte if tte > _TTE_FLOOR else _TTE_FLOOR
        sqrt_tte = math.sqrt(safe_tte)
        log_k = math.log(k)

        # ---- Pre-compute the 2D plane handles ----
        #   pillars      : 1-D, length=K_pillars  (shared across all scenarios)
        #   v_plane      : 2-D, [scenario, pillar] — contiguous run per scenario
        #   p_row        : 1-D, length=S          (price-curve row for delivery)
        pillars = vol_surface_strikes[v_surf_id]
        v_plane = vol_surfaces[v_surf_id, exp_id]
        p_row = price_curve_grid[p_curve_id, deliv_id]

        # ---- Hoist sentinel branches into per-trade booleans ----
        has_fx_u = fx_u_id >= 0
        has_df_u = df_u_id >= 0
        has_fx_k = fx_k_id >= 0
        has_df_k = df_k_id >= 0

        # ---- Hoist y-axis-type ladder into a 0/1/2/3 dispatch tag ----
        if y_type == SURFACE_Y_AXIS_TYPE_STRIKE:
            y_kind = _Y_KIND_STRIKE
        elif y_type == SURFACE_Y_AXIS_TYPE_LOG_STRIKE:
            y_kind = _Y_KIND_LOG_STRIKE
        elif y_type in (
            SURFACE_Y_AXIS_TYPE_MONEYNESS_FWD,
            SURFACE_Y_AXIS_TYPE_MONEYNESS_SPOT,
        ):
            y_kind = _Y_KIND_MONEYNESS
        elif y_type in (
            SURFACE_Y_AXIS_TYPE_LOG_MONEYNESS_FWD,
            SURFACE_Y_AXIS_TYPE_LOG_MONEYNESS_SPOT,
        ):
            y_kind = _Y_KIND_LOG_MONEYNESS
        else:
            # Unknown y-axis: degrade to strike. Producer validation
            # rejects this case before it reaches the kernel.
            y_kind = _Y_KIND_STRIKE

        for s in range(s_scenarios):
            # ---- CIP forward in strike currency ----
            fwd = p_row[s]
            if has_fx_u:
                fwd *= fx_spots_effective[fx_u_id, s]
            if has_df_u:
                fwd *= df_grid[df_u_id, exp_id, s]

            rate_k = 1.0
            if has_fx_k:
                rate_k = fx_spots_effective[fx_k_id, s]
                fwd /= rate_k
            if has_df_k:
                fwd /= df_grid[df_k_id, exp_id, s]

            # ---- Y-axis target for vol surface lookup ----
            # The branch tag is constant per trade; LLVM should hoist
            # this select out of the SIMD body via loop-unswitching.
            if y_kind == _Y_KIND_STRIKE:
                y_target = k
            elif y_kind == _Y_KIND_LOG_STRIKE:
                y_target = log_k
            elif y_kind == _Y_KIND_MONEYNESS:
                y_target = k / fwd
            else:  # _Y_KIND_LOG_MONEYNESS
                y_target = log_k - math.log(fwd)

            # ---- Strike-axis interpolation ----
            # TODO: QUESTION - DO WE NEED TO DO OTHER TYPES OF INTERPOLATION FOR OTHER
            # Y-AXIS KINDS? E.g. log-log for log-strike?
            # v_plane[s] is a C-contig 1-D run of pillars for THIS scenario.
            sigma = linear_interp_scalar(y_target, pillars, v_plane[s])

            # ---- Black-76 future value ----
            std_dev = sigma * sqrt_tte
            if std_dev < _VOL_STD_DEV_FLOOR:
                std_dev = _VOL_STD_DEV_FLOOR

            d1 = (math.log(fwd) - log_k + 0.5 * sigma * sigma * safe_tte) / std_dev
            d2 = d1 - std_dev

            cdf1 = numba_norm_cdf_abrom_steg(cp * d1)
            cdf2 = numba_norm_cdf_abrom_steg(cp * d2)

            fv = cp * (fwd * cdf1 - k * cdf2)

            # ---- PV discount in strike ccy, FX-translate to settle ccy ----
            if has_df_k:
                fv *= df_grid[df_k_id, stl_id, s]
            fv *= rate_k

            out_row[s] = q * fv
```

```python
"""
Vanilla European option pricing kernel, node variant.

Synthesises the forward in strike currency via Covered Interest Parity,
prices via Black-76, then discounts and FX-translates back to settle ccy.

Math (per trade i, scenario s)
------------------------------
    rate_k = fx_spots_effective[fx_k_eff_uix(i), s]   # k_ccy -> settle_ccy

    # CIP forward in strike currency
    fwd_strike_ccy
        = forward_node[p, s]
          * fx_spots_effective[fx_u_eff_uix(i), s]    # u_ccy -> settle_ccy
          * df_grid[df_u_uix(i), expiry_uix(i), s]    # under-DF at expiry
          / rate_k                                    # settle_ccy -> k_ccy
          / df_grid[df_k_uix(i), expiry_uix(i), s]    # strike-DF at expiry

    # Black-76 future value (cp_mult = +1 call, -1 put)
    sigma = vol_nodes[v, s]
    std_dev = max(sigma * sqrt(tte), eps)
    d1 = (log(fwd / k) + 0.5 * sigma**2 * tte) / std_dev
    d2 = d1 - std_dev
    fv = cp * (fwd * N(cp * d1) - k * N(cp * d2))

    # PV discount in strike ccy, then translate back to settle ccy
    pv_strike_ccy = fv * df_grid[df_k_uix(i), settle_uix(i), s]
    pv_settle_ccy = pv_strike_ccy * rate_k

    out[i, s] = qty[i] * pv_settle_ccy

FX rows are already EFFECTIVE in `MarketDataMatrix.fx_spots_effective`; the
kernel does NO inversion math. The strike-FX rate `rate_k` is loaded once
per scenario and used as a divisor in the forward synthesis and as a
multiplier in the PV translation — equivalent counts to the legacy code
but with the `(1.0 / rate_k) if inv_k else rate_k` gymnastics gone.

Vol-uix sentinel handling
-------------------------
`v_id = vol_uix[i]` may be `INVALID_UIX (< 0)` when the ingest compiler
resolved no vol container for trade i (e.g. a downstream trade that
slipped past validation, or a future-class instrument whose vol path is
not yet wired). With `boundscheck=False`, `vol_nodes[v_id]` on a negative
id is undefined behaviour — out-of-bounds memory.

scenario loop. The sentinel path writes `out_row[:] = 0.0` and skips
the vol gather + Black-76 math entirely. Two scenario loops — one for
the priced path, one for the zero-fill path — keep the SIMD body free
of per-scenario branches.

import numba as nb  # type: ignore[import-untyped]

_F64_1D = nb.float64[::1]

_I8_1D = nb.int8[::1]

_VOL_STD_DEV_FLOOR: float = 1e-12

        _F64_2D,  # out
        _F64_2D,  # forward_nodes
        _F64_2D,  # vol_nodes
        _F64_2D,  # fx_spots_effective
        _F64_3D,  # df_grid

        _F64_1D,  # qty
        _F64_1D,  # strike
        _I8_1D,  # cp_mult
        _I32_1D,  # price_uix
        _I32_1D,  # vol_uix
        _I32_1D,  # fx_u_eff_uix

def _kernel_price_vanilla_european_option_node(

    forward_nodes,
    vol_nodes,
    fx_spots_effective,
    df_grid,
    sqrt_tte_grid,
    half_tte_grid,

    cp_mult,
    price_uix,
    vol_uix,

    df_u_uix,
    fx_k_eff_uix,
    df_k_uix,
    expiry_date_uix,
    settle_date_uix,

    m_trades, s_scenarios = out.shape

    for i in nb.prange(m_trades):
        # ------------------------------------------------------------------
        # Trade constants
        # ------------------------------------------------------------------

        q = qty[i]
        k = strike[i]
        cp = cp_mult[i]

        p_id = price_uix[i]
        v_id = vol_uix[i]

                fwd *= df_u_exp_row[s]  # pyright: ignore[reportPossiblyUnboundVariable]

                rate_k = fx_k_row[s]  # pyright: ignore[reportPossiblyUnboundVariable]

                fv *= df_k_stl_row[s]  # pyright: ignore[reportPossiblyUnboundVariable]

            out_row[s] = q * fv
```

```python
    fx_spots_effective,
    df_grid,
    qty,
    strike,
    price_curve_uix,
    delivery_date_uix,
    fx_u_eff_uix,
    df_u_uix,
    fx_k_eff_uix,
    df_k_uix,
    settle_date_uix,
) -> None:
    m_trades, s_scenarios = out.shape

    for i in nb.prange(m_trades):
        q = qty[i]
        k = strike[i]
        p_curve_id = price_curve_uix[i]
        deliv_id = delivery_date_uix[i]
        fx_u_id = fx_u_eff_uix[i]
        df_u_id = df_u_uix[i]
        fx_k_id = fx_k_eff_uix[i]
        df_k_id = df_k_uix[i]
        stl_id = settle_date_uix[i]

        # Slice the (date, scenario) plane for this trade's underlying curve.
        p_row = price_curve_grid[p_curve_id, deliv_id]
        out_row = out[i]

        for s in range(s_scenarios):
            # UNDERLYING LEG
            leg_u = p_row[s]
            if fx_u_id >= 0:
                leg_u *= fx_spots_effective[fx_u_id, s]
            if df_u_id >= 0:
                leg_u *= df_grid[df_u_id, stl_id, s]

            # STRIKE LEG
            leg_k = k
            if fx_k_id >= 0:
                leg_k *= fx_spots_effective[fx_k_id, s]
            if df_k_id >= 0:
                leg_k *= df_grid[df_k_id, stl_id, s]

            out_row[s] = q * (leg_u - leg_k)
```

```python
"""
Vanilla forward pricing kernel, node variant.

    leg_u = forward_node * fx_u_effective * df_u(stl)
    leg_k = strike       * fx_k_effective * df_k(stl)
    PV    = qty * (leg_u - leg_k)

FX rows are already EFFECTIVE (direct or pre-inverted) inside
`MarketDataMatrix.fx_spots_effective` — pricer kernels do NO inversion math.
Each FX gather is one indexed read + one multiply per scenario.

Optional legs (FX / DF) are signalled by `INVALID_UIX (< 0)` in the
corresponding `_uix` field. These predicates are trade-level invariants;
Numba's LICM hoists them outside the inner scenario loop, so the inner loop
streams straight-line arithmetic over C-contiguous memory.
"""

from __future__ import annotations

import numba as nb  # type: ignore[import-untyped]

from riskquant._njit import parallel_njit

_F64_2D = nb.float64[:, ::1]
_F64_3D = nb.float64[:, :, ::1]
_F64_1D = nb.float64[::1]
_I32_1D = nb.int32[::1]


@parallel_njit(
    nb.void(
        _F64_2D,  # out
        _F64_2D,  # forward_nodes
        _F64_2D,  # fx_spots_effective
        _F64_3D,  # df_grid
        _F64_1D,  # qty
        _F64_1D,  # strike
        _I32_1D,  # price_uix
        _I32_1D,  # fx_u_eff_uix
        _I32_1D,  # df_u_uix
        _I32_1D,  # fx_k_eff_uix
        _I32_1D,  # df_k_uix
        _I32_1D,  # settle_date_uix
    ),
)
def _kernel_price_vanilla_forward_node(
    out,
    forward_nodes,
    fx_spots_effective,
    df_grid,
    qty,
    strike,
    price_uix,
    fx_u_eff_uix,
    df_u_uix,
    fx_k_eff_uix,
    df_k_uix,
    settle_date_uix,
) -> None:
    m_trades, s_scenarios = out.shape

    for i in nb.prange(m_trades):
        # Trade-level scalars (loop-invariant; Numba hoists the branches below)
        q = qty[i]
        k = strike[i]
        p_id = price_uix[i]
        fx_u_id = fx_u_eff_uix[i]
        df_u_id = df_u_uix[i]
        fx_k_id = fx_k_eff_uix[i]
        df_k_id = df_k_uix[i]
        stl_id = settle_date_uix[i]

        p_row = forward_nodes[p_id]
        out_row = out[i]

        for s in range(s_scenarios):
            # UNDERLYING LEG
            leg_u = p_row[s]
            if fx_u_id >= 0:
                leg_u *= fx_spots_effective[fx_u_id, s]
            if df_u_id >= 0:
                leg_u *= df_grid[df_u_id, stl_id, s]

            # STRIKE LEG
            leg_k = k
            if fx_k_id >= 0:
                leg_k *= fx_spots_effective[fx_k_id, s]
            if df_k_id >= 0:
                leg_k *= df_grid[df_k_id, stl_id, s]

            out_row[s] = q * (leg_u - leg_k)
```

```python
"""
Vanilla European option pricer, node variant.

Synthesises the forward in strike currency via Covered Interest Parity,
prices via Black-76, then discounts and FX-translates back to settle ccy.

Math (per trade i, scenario s)
------------------------------
    rate_k = fx_spots_effective[fx_k_eff_uix(i), s]   # k_ccy -> settle_ccy

    # CIP forward in strike currency
    fwd_strike_ccy
        = forward_node[p, s]
          * fx_spots_effective[fx_u_eff_uix(i), s]    # u_ccy -> settle_ccy
          * df_grid[df_u_uix(i), expiry_uix(i), s]    # under-DF at expiry
          / rate_k                                    # settle_ccy -> k_ccy
          / df_grid[df_k_uix(i), expiry_uix(i), s]    # strike-DF at expiry

    # Black-76 future value (cp_mult = +1 call, -1 put)
    sigma = vol_nodes[v, s]
    std_dev = max(sigma * sqrt(tte), eps)
    d1 = (log(fwd / k) + 0.5 * sigma**2 * tte) / std_dev
    d2 = d1 - std_dev
    fv = cp * (fwd * N(cp * d1) - k * N(cp * d2))

    # PV discount in strike ccy, then translate back to settle ccy
    pv_strike_ccy = fv * df_grid[df_k_uix(i), settle_uix(i), s]
    pv_settle_ccy = pv_strike_ccy * rate_k

    out[i, s] = qty[i] * pv_settle_ccy

FX rows are already EFFECTIVE in `MarketDataMatrix.fx_spots_effective`; the
kernel does NO inversion math. The strike-FX rate `rate_k` is loaded once
per scenario and used as a divisor in the forward synthesis and as a
multiplier in the PV translation — equivalent counts to the legacy code
but with the `(1.0 / rate_k) if inv_k else rate_k` gymnastics gone.
"""

from __future__ import annotations

from typing import TYPE_CHECKING

import numpy as np
import numpy.typing as npt

from riskquant.valuation.kernels.vanilla_european_option_node import (
```

```python
# src/riskquant/valuation/result.py
"""
Valuation results container.
"""

from dataclasses import dataclass

import numpy as np
import numpy.typing as npt


@dataclass(frozen=True, slots=True, kw_only=True)
class ValuationResult:
    trade_uids: npt.NDArray[np.int64]  # (n_trades,)
    values: npt.NDArray[np.float64]  # (n_trades, n_scenarios)

    def __post_init__(self) -> None:
        if self.values.shape[0] != self.trade_uids.shape[0]:
            raise ValueError("UIDs and values row mismatch")

    @property
    def num_trades(self) -> int:
        return self.values.shape[0]

    @property
    def num_scenarios(self) -> int:
        return self.values.shape[1]

    def get_trade_values(self, uid: int) -> npt.NDArray[np.float64]:
        idx = np.searchsorted(self.trade_uids, uid)
        if self.trade_uids[idx] != uid:
            raise KeyError(uid)
        return self.values[idx]

    def select(self, uids: npt.NDArray[np.int64]) -> "ValuationResult":
        idx = np.isin(self.trade_uids, uids)
        return ValuationResult(
            trade_uids=self.trade_uids[idx],
            values=self.values[idx],
        )
```
