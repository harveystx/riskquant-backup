# src/riskquant/_arrays.py
"""
Cross-cutting NumPy array primitives shared across packages.

Top-level private module (mirrors `_njit.py`) so leaf consumers in
`proxy_resolver`, `simlab`, `portfolio`, `market_data`, and `credit` can import
these without a layering inversion.

No domain imports — pure array algebra; domain sentinels (e.g., `INVALID_UIX`
or `NO_UID_MATCH_IDX_I32`) are passed in by callers rather than imported here.

Contents
--------
unique_first_seen                        : dtype-preserving first-seen dedup + inverse.
searchsorted_membership                  : bool found-mask over a sorted-unique universe.
searchsorted_or_raise                    : STRICT INNER JOIN. Raises on any miss.
searchsorted_soft_join                   : SAFE LEFT JOIN. Probes existence, maps misses to a sentinel.
searchsorted_or_propagate_sentinel_int32 : UNSAFE LEFT JOIN. Bypasses probe, blindly maps `< 0` to a sentinel.
"""

from __future__ import annotations

from typing import Final

import numpy as np
import numpy.typing as npt
from numpy.typing import NDArray

# Centralized preview limit for error-message diagnostics to prevent
# massive stdout dumps on join failures.
_PREVIEW: Final[int] = 10


def _as_c_contig_f64(arr: NDArray) -> NDArray[np.float64]:
    return np.ascontiguousarray(arr, dtype=np.float64)


def _as_c_contig_f32(arr: NDArray) -> NDArray[np.float32]:
    return np.ascontiguousarray(arr, dtype=np.float32)


def _as_c_contig_i64(arr: NDArray) -> NDArray[np.int64]:
    return np.ascontiguousarray(arr, dtype=np.int64)


def _as_c_contig_i32(arr: NDArray) -> NDArray[np.int32]:
    return np.ascontiguousarray(arr, dtype=np.int32)


def _as_c_contig_i16(arr: NDArray) -> NDArray[np.int16]:
    return np.ascontiguousarray(arr, dtype=np.int16)


def _as_c_contig_i8(arr: NDArray) -> NDArray[np.int8]:
    return np.ascontiguousarray(arr, dtype=np.int8)


def _as_c_contig_bool(arr: NDArray) -> NDArray[np.bool_]:
    return np.ascontiguousarray(arr, dtype=np.bool_)


def unique_first_seen(
    arr: npt.NDArray,
) -> tuple[npt.NDArray, npt.NDArray[np.int32]]:
    """
    First-seen dedup: dtype-preserving in the values, int32 in the inverse.

    `np.unique` sorts; we need first-seen order so shock-compute /
    proxy-resolution plans are deterministic across reruns of identical inputs.

    Returns
    -------
    unique_first_seen
        (K,) — unique values in first-seen order, same dtype as ``arr``.

    inverse_first_seen
        (N,) int32 — for each input position, its index in ``unique_first_seen``.
    """
    if arr.size == 0:
        return arr[:0].copy(), np.empty(0, dtype=np.int32)

    _, first_idx, inverse_sorted = np.unique(
        arr, return_index=True, return_inverse=True
    )
    # Reorder sorted-unique into first-seen order.
    order = np.argsort(first_idx, kind="mergesort")
    out_unique = arr[first_idx[order]]

    # Build sorted_idx -> first_seen_idx remap, then apply to the inverse.
    sorted_to_first_seen = np.empty(order.size, dtype=np.int32)
    sorted_to_first_seen[order] = np.arange(order.size, dtype=np.int32)
    inverse_first_seen = sorted_to_first_seen[inverse_sorted].astype(
        np.int32, copy=False
    )
    return out_unique, inverse_first_seen


# =========================================================================
# VECTORIZED JOIN PRIMITIVES
# =========================================================================
#
# `universe` MUST be sorted ascending and unique (caller's invariant).
# These do NOT validate that invariant; they trust it, exactly as the
# hand-rolled sites they replace did.


def searchsorted_membership(
    *,
    universe: npt.NDArray,
    raw: npt.NDArray,
) -> npt.NDArray[np.bool_]:
    """
    EXISTENCE CHECK: Per-element membership of `raw` in sorted-unique `universe`.

    Returns a `(raw.size,)` bool mask, ``True`` where the value is present.
    No sentinel handling — callers that treat ``raw < 0`` as "not required"
    should mask those out before calling, or ignore the corresponding mask entries.
    """
    if raw.size == 0:
        return np.empty(0, dtype=np.bool_)
    if universe.size == 0:
        return np.zeros(raw.shape[0], dtype=np.bool_)

    idx = np.searchsorted(universe, raw)
    in_bounds = idx < universe.size
    found = np.zeros(raw.shape[0], dtype=np.bool_)
    found[in_bounds] = universe[idx[in_bounds]] == raw[in_bounds]
    return found


def searchsorted_or_raise(
    *,
    universe: npt.NDArray,
    raw: npt.NDArray,
    ctx: str = "ids",
) -> npt.NDArray[np.int64]:
    """
    STRICT INNER JOIN: Locate every `raw` value in sorted-unique `universe`.
    Raises ValueError on any missing target.

    Returns the `(raw.size,)` int64 index array (the natural `np.searchsorted`
    dtype — callers gather with it, no downcast needed).
    """
    if raw.size == 0:
        return np.empty(0, dtype=np.int64)

    n = universe.shape[0]
    if n == 0:
        raise ValueError(
            f"{ctx}: Universe is empty, {raw.shape[0]} target ID(s) cannot be resolved."
        )

    pos = np.searchsorted(universe, raw)
    np.clip(pos, 0, n - 1, out=pos)
    matched = universe[pos] == raw

    if not matched.all():
        bad = np.unique(raw[~matched])[:_PREVIEW].tolist()
        n_bad = int((~matched).sum())
        raise ValueError(
            f"{ctx}: {n_bad} ID(s) absent from the universe. First {len(bad)}: {bad}"
        )

    return pos


def searchsorted_soft_join(
    *,
    universe: npt.NDArray,
    raw: npt.NDArray,
    sentinel: np.int32,
) -> npt.NDArray[np.int32]:
    """
    SAFE LEFT JOIN: Probes existence. Maps ANY missing `raw` value to `sentinel`.

    Use when absent target IDs are a valid, expected runtime case (e.g.,
    a netting set whose obligor is not modelled in the target portfolio).

    Returns `(raw.size,)` int32 indices. Caller filters via ``keep = pos >= 0``
    (assuming sentinel is negative).
    """
    n = universe.shape[0]
    if n == 0:
        return np.full(raw.shape[0], sentinel, dtype=np.int32)

    pos = np.searchsorted(universe, raw)
    np.clip(pos, 0, n - 1, out=pos)
    hit = universe[pos] == raw
    return np.where(hit, pos, sentinel).astype(np.int32, copy=False)


def searchsorted_or_propagate_sentinel_int32(
    *,
    universe: npt.NDArray,
    raw: npt.NDArray,
    sentinel: np.int32,
) -> npt.NDArray[np.int32]:
    """
    UNSAFE LEFT JOIN: High-performance ID translation with sentinel preservation.

    `raw < 0` rows are explicitly treated as sentinels and mapped to `sentinel`.

    CRITICAL: All other IDs (>= 0) are BLINDLY ASSUMED present in `universe`.
    This primitive does NOT probe membership (no equality check, no bounds clip).
    If a missing positive ID is passed, it will silently map to the next-largest
    element's index. Only use this when upstream referential integrity is 100%
    guaranteed by prior validation phases.
    """
    uix = np.searchsorted(universe, raw).astype(np.int32, copy=False)
    uix[raw < 0] = sentinel
    return uix

"""

from datetime import date

# src/riskquant/numba.py

"""
Centralized Numba decorator definitions used throughout RiskQuant.

Rationale
---------
RiskQuant standardizes Numba compilation settings to ensure consistent
performance characteristics, numerical behavior, and cache utilization
across the codebase.

This module provides three decorators:

    @njit
        Default choice for most compiled functions.

    @parallel_njit
        For kernels that parallelize outer loops using ``numba.prange``.

    @inline_njit
        For tiny helper functions that should be inlined into callers to
        eliminate function-call overhead.

By centralizing these definitions, compilation settings can be adjusted in
one location without touching hundreds of kernels throughout the system.
"""

from __future__ import annotations

from functools import partial

import numba as nb

# Type aliases for Numba-compiled functions — enforce C-contiguous layout for all
# arrays to ensure the compiler emits straight-line SIMD loops with no dynamic
# stride checks. These are used in type signatures for compiled functions below.
_F64_1D = nb.float64[::1]
_F64_2D = nb.float64[:, ::1]
_F64_3D = nb.float64[:, :, ::1]
_F64_4D = nb.float64[:, :, :, ::1]

_F32_1D = nb.float32[::1]
_F32_2D = nb.float32[:, ::1]
_F32_3D = nb.float32[:, :, ::1]
_F32_4D = nb.float32[:, :, :, ::1]

_I64_1D = nb.int64[::1]
_I32_1D = nb.int32[::1]
_I8_1D = nb.int8[::1]

_BOOL_1D = nb.bool_[::1]

#: Default Numba compiler configuration used throughout RiskQuant.
#:
#: Use for the vast majority of compiled functions.
#:
#: Features:
#:     - Explicit fast-math optimizations
#:     - GIL-free execution
#:     - Bounds-check elimination
#:     - On-disk compilation caching
#:     - NumPy-style floating-point error behavior
#:
#: Example:
#:
#:     @njit
#:     def black_scholes(...):
#:         ...
njit = partial(
    nb.njit,
    fastmath={"contract", "reassoc", "nsz"},
    nogil=True,
    boundscheck=False,
    error_model="numpy",
    cache=True,
)


#: Parallel Numba compiler configuration.
#:
#: Use for computational kernels that distribute work across threads using
#: ``numba.prange``.
#:
#: Typical usage:
#:
#:     @parallel_njit
#:     def price_kernel(...):
#:         for i in nb.prange(num_trades):
#:             ...
#:
#: Avoid for:
#:     - Small loops
#:     - Helper functions
#:     - Functions that do not use ``prange``
parallel_njit = partial(
    nb.njit,
    parallel=True,
    fastmath={"contract", "reassoc", "nsz"},
    nogil=True,
    boundscheck=False,
    error_model="numpy",
    cache=True,
)


#: Inline Numba compiler configuration.
#:
#: Intended for extremely small helper functions that are called repeatedly
#: inside hot loops.
#:
#: Examples:
#:     - Normal CDF approximations
#:     - Small mathematical transforms
#:     - Utility functions used inside pricing kernels
#:
#: The ``inline="always"`` hint requests that Numba inline the function body
#: into callers whenever possible.
#:
#: ``error_model="numpy"`` is set to match ``njit``/``parallel_njit``: division
#: by zero yields inf/nan rather than raising inside a nopython (and possibly
#: parallel) caller. Without it, an unguarded division in an inlined helper can
#: turn into a hard ``ZeroDivisionError`` mid-kernel.
inline_njit = partial(
    nb.njit,
    fastmath={"contract", "reassoc", "nsz"},
    nogil=True,
    boundscheck=False,
    error_model="numpy",
    cache=True,
    inline="always",
)

def validate_string_cols(df: DataFrame, cols: tuple[str, ...] | list[str]) -> None:
    for col in cols:
        if col not in df.columns:
            continue

        s = df[col]

        if not is_string_dtype(s):
            raise ValueError(f"Column '{col}' must be of string dtype.")

        non_na = s.dropna()
        if non_na.empty:
            continue

        # Length
        max_len = non_na.str.len().max()
        if max_len > 255:
            raise ValueError(
                f"Column '{col}' contains strings longer than 255 characters "
                f"(max={max_len})."
            )

        # Canonical format
        pattern = r"^[A-Z0-9]+(?:_[A-Z0-9]+)*(?:\[(?:[A-Z]+[+-]\d+|(?:ON|TN|SN|\d+[DWMY]))\])?$"
        if not non_na.str.match(pattern).all():
            raise ValueError(
                f"Column '{col}' contains non-canonical strings. "
                f"Expected format: UPPERCASE tokens separated by underscores."
            )

def coerce_as_of_date_to_pd_timestamp(
    as_of_date: pd.Timestamp | dt.date | pd.Timestamp | str,
) -> pd.Timestamp:
    """
    Accepts: pd.Timestamp | datetime.datetime | datetime.date | str
    Returns: pd.Timestamp (tz-naive)
    """
    # If str, validate format.
    if isinstance(as_of_date, str):
        assert_date_str_is_iso_format(as_of_date)

    # As pd.Timestamp
    ts = pd.Timestamp(as_of_date)

    # If caller passed a date (no time), treat it as end-of-day inclusive
    # so 2026-01-02 includes all times on 2026-01-02.
    if isinstance(as_of_date, dt.date) and not isinstance(as_of_date, dt.datetime):
        ts = ts + pd.Timedelta(days=1) - pd.Timedelta(nanoseconds=1)  # type: ignore[operator]  # mypy doesn't know Timestamp supports this math

    # Normalize tz handling (if any). Adjust policy as you like.
    if ts.tzinfo is not None:
        ts = ts.tz_convert(None)

    return ts


def is_date_str_valid(date_str: str) -> bool:
    """Checks if date_str which must have form %Y-%m-%d is a valid calendar date."""
    if not isinstance(date_str, str):
        return False
    if len(date_str) != 10 or date_str[4] != "-" or date_str[7] != "-":
        return False
    if not date_str.replace("-", "").isdigit():
        return False

    try:
        pd.to_datetime(date_str, format="%Y-%m-%d", errors="raise")
    except ValueError:
        return False

    return True


def assert_date_str_is_iso_format(date_str: str) -> None:
    """Assert that date_str is in ISO format 'YYYY-MM-DD'."""
    if not is_date_str_valid(date_str):
        raise ValueError(f"Date string '{date_str}' is not in 'YYYY-MM-DD' format.")


def date_str_to_date(date_str: str) -> dt.date:
    """Convert a date string in 'YYYY-MM-DD' format to a datetime.date object."""
    assert_date_str_is_iso_format(date_str)
    return pd.to_datetime(date_str, format="%Y-%m-%d", errors="raise").date()


def date_to_ord32(d: dt.date) -> np.int32:
    return np.int32((d - DATE_EPOCH).days)


def ordinal32_to_date(days: int) -> dt.date:
    return DATE_EPOCH + pd.Timedelta(days=int(days))  # or datetime.timedelta


def datetime64D_to_ord32(dts: np.ndarray) -> np.ndarray:
    """
    Convert numpy datetime64[D] array -> int32 days since DATE_EPOCH.
    Returns a view/copy depending on input; no pandas involvement.
    """
    dts = np.asarray(dts, dtype="datetime64[D]")
    return (dts - DATE_EPOCH_AS_DATETIME64D).astype(np.int32, copy=False)


def ordinal32_to_datetime64D(days: np.ndarray) -> np.ndarray:
    """
    Convert int32 days since DATE_EPOCH -> numpy datetime64[D] array.
    """
    days = np.asarray(days, dtype=np.int32)
    # Cast to timedelta64[D] then add epoch
    return DATE_EPOCH_AS_DATETIME64D + days.astype("timedelta64[D]", copy=False)


def ym_str_to_int_tuple(ym: str) -> tuple[int, int]:
    """Convert 'YYYYMM' -> (YYYY, MM)."""
    if len(ym) != 6 or not ym.isdigit():
        raise ValueError(f"Invalid YM string: {ym}")
    y, m = int(ym[:4]), int(ym[4:6])
    if not 1 <= m <= 12:
        raise ValueError(f"Invalid YM string: {ym}")
    return y, m


def ymd_str_to_date(ymd: str) -> dt.date:
    """Convert 'YYYYMMDD' -> datetime.date."""
    if len(ymd) != 8 or not ymd.isdigit():
        raise ValueError(f"Invalid YMD string: {ymd}")
    return dt.date(int(ymd[:4]), int(ymd[4:6]), int(ymd[6:8]))


def month_diff(
    *,
    anchor_ym: tuple[int, int],
    target_ym: tuple[int, int],
) -> int:
    """Difference in months: target - anchor."""
    ay, am = anchor_ym
    ty, tm = target_ym
    return (ty - ay) * 12 + (tm - am)


def next_half_year_boundary(
    t0: dt.date,
    ym_boundaries: Iterable[str] = ("0630", "1231"),
) -> dt.date:
    """
    Return the first boundary date >= t0 among {YYYY0630, YYYY1231} (or configured boundaries).
    If none remain in the year, return the first boundary in the next year.
    """
    ym_bounds = list(ym_boundaries)
    if not ym_bounds:
        raise ValueError("No half-year boundaries provided.")

    year = t0.year
    candidates = []
    for b in ym_bounds:
        if len(b) != 4 or not b.isdigit():
            raise ValueError(f"Invalid year-month boundary 'MMDD': {b}")
        m = int(b[:2])
        d = int(b[2:])
        candidates.append(dt.date(year, m, d))

    for d in sorted(candidates):
        if d >= t0:
            return d

    # none left this year -> first boundary next year (sorted)
    year += 1
    b0 = sorted(ym_bounds)[0]
    return dt.date(year, int(b0[:2]), int(b0[2:]))


def half_year_index(d: dt.date) -> tuple[int, int]:
    """
    Map boundary date to a half-year index:
      YYYY-06-30 -> (YYYY, 0)
      YYYY-12-31 -> (YYYY, 1)
    """
    if (d.month, d.day) == (6, 30):
        return d.year, 0
    if (d.month, d.day) == (12, 31):
        return d.year, 1
    raise ValueError(f"Not a recognized half-year boundary: {d.isoformat()}")


def half_year_diff(
    *,
    anchor: dt.date,
    target: dt.date,
) -> int:
    """Difference in half-years: target - anchor, where each boundary step counts as 1."""
    ay, ah = half_year_index(anchor)
    ty, th = half_year_index(target)
    return (ty - ay) * 2 + (th - ah)


def get_previous_business_date_as_iso_str(
    *,
    lag_in_bus_days: int = 0,
    holiday_dates: list[str] | tuple[str, ...] | None = None,
    time_str_to_append_to_date: str | None = None,
) -> str:
    # Ensure holidays is a list of strings.
    if holiday_dates is None:
        holiday_dates = []
    if isinstance(holiday_dates, tuple):
        holiday_dates = list(holiday_dates)
    if not isinstance(holiday_dates, list):
        raise TypeError("holiday_dates must be a list!")

    # Create a custom business day with the holidays.
    custom_business_day = pd.offsets.CustomBusinessDay(holidays=holiday_dates)

    # Start from today and apply the lag and one additional business day.
    start_date = dt.datetime.today().date()
    adjusted_date = start_date - custom_business_day * (lag_in_bus_days + 1)
    date_as_str = adjusted_date.strftime("%Y-%m-%d")

    # Append the time string if provided, and return.
    if time_str_to_append_to_date is not None:
        return date_as_str + f" {time_str_to_append_to_date}"
    return date_as_str


def get_previous_business_date_as_date(
    *,
    lag_in_bus_days: int = 0,
    holiday_dates: list[str] | tuple[str, ...] | None = None,
    time_str_to_append_to_date: str | None = None,
) -> dt.date:
    # Get as string.
    date_string = get_previous_business_date_as_iso_str(
        lag_in_bus_days=lag_in_bus_days,
        holiday_dates=holiday_dates,
        time_str_to_append_to_date=time_str_to_append_to_date,
    )

    # To date.
    return dt.date.fromisoformat(date_string)

        # Explicitly delete frame references to avoid reference cycles
        del frame

    dt_str = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    text = f"{dt_str} | RUNNING FILE: {filename}"

    # Handle text wrapping
    lines = textwrap.wrap(text, width=width)
    if not lines:
        lines = [""]
    max_len = max(len(line) for line in lines)
    inner_width = max_len + padding * 2

    c = color or ""
    reset = ANSI_RESET if color else ""
    bold = ANSI_BOLD if color else ""

    if add_space_before:
        print()

    # Top border
    print(f"{c}{bold}╭{'─' * inner_width}╮{reset}")

    # Body (Color applied to the entire line)
    for line in lines:
        print(
            f"{c}{bold}│"
            + " " * padding
            + line.ljust(max_len)
            + " " * padding
            + f"│{reset}"
        )

    # Bottom border
    print(f"{c}{bold}╰{'─' * inner_width}╯{reset}")

    if add_space_after:
        print()


def print_dict(
    d: dict[Any, Any | np.integer | np.floating]
    | Mapping[Any, Any | np.integer | np.floating],

    limit: int | None = None,
    add_space_before: bool = True,
    add_space_after: bool = True,
    title_color: str | None = None,
) -> None:
    """Pretty-print a dictionary with aligned keys and nested support."""
    items = sorted(d.items()) if sort_keys else list(d.items())
    if limit is not None:
        items = items[:limit]

    try:
        max_key_len = max(len(str(k)) for k in d)
    except ValueError:
        max_key_len = 0

    pad = " " * indent

    if add_space_before:
        print()

    if title:
        print_header(
            title=f"Dictionary: {title}",
            color=title_color,
            add_space_before=False,
            add_space_after=False,
        )

    print(pad + "{")
    for key, value in items:
        key_str = str(key)
        key_pad = key_str.ljust(max_key_len)

        if isinstance(value, (Mapping, dict)):
            print(pad + " " * indent_step + f"{key_pad}:")

                indent=indent + indent_step * 2,
                indent_step=indent_step,
                sort_keys=sort_keys,
                limit=None,
                add_space_before=False,
                add_space_after=False,
            )
        else:
            print(pad + " " * indent_step + f"{key_pad}: {value}")

    print(pad + "}")

    if add_space_after:
        print()


class _FloatFormatter:
    __slots__ = ("_fmt", "_zero_tol")

    def __init__(self, spec: str, zero_tol: float = 0.0) -> None:
        if "_" in spec:
            clean_spec = spec.replace("_", ",")
            self._fmt = lambda x, s=clean_spec: f"{x:{s}}".replace(",", "_")
        else:
            self._fmt = lambda x, s=spec: f"{x:{s}}"
        self._zero_tol = float(zero_tol)

    def __call__(self, x: float) -> str:
        if x is None or (isinstance(x, float) and math.isnan(x)):
            # Handle NaN alignment for Numpy formats
            try:
                width = int(
                    self._fmt.__defaults__[0]
                    .split(".")[0]
                    .replace(">", "")
                    .replace("<", "")
                )
                return "nan".rjust(width)
            except (ValueError, TypeError, IndexError):
                return "nan"
        if self._zero_tol and abs(x) < self._zero_tol:
            x = 0.0
        try:
            return self._fmt(x)
        except (ValueError, TypeError):
            return str(x)

from __future__ import annotations

from datetime import date
from typing import Optional, Tuple


def parse_bracket_token(tenor_token: str) -> Tuple[str, Optional[int]]:
    """
    "[Y-2]" -> ("Y", -2)
    "[M+0]" -> ("M", 0)
    "[D+24]" -> ("D", 24)
    "[HY+1]" -> ("HY", 1)
    "[ON]" -> ("ON", None)   (LABEL)
    """
    # Error.
    if tenor_token is None:
        raise ValueError("tenor_token cannot be None!")

    # Strip white space.
    s = tenor_token.strip()

    # Must start with '[' and end with ']'.
    if s.startswith("[") and s.endswith("]"):
        s = s[1:-1].strip()

    # Find the plus or minus sign.
    plus = s.rfind("+")
    minus = s.rfind("-")
    sign_pos = max(plus, minus)

    # If no sign.
    if sign_pos <= 0:
        return s, None

    # Unit (Y, M, D, HY, etc.)
    unit = s[:sign_pos]

    # Signed offset.
    offset = int(s[sign_pos:])

    # Return.
    return unit, offset


def get_spot_year(
    *,
    as_of_date: date,
    roll_month: int,
    roll_day: int,
    post_roll_shift: int,
) -> int:
    """
    Defines which delivery year corresponds to Y+0 under ROLL_ON_MMDD.
    """

    # Get the roll date,
    roll_date = date(as_of_date.year, roll_month, roll_day)

    # Compare to as-of-date.
    # As-of-date is on or after the roll date.
    if as_of_date >= roll_date:
        return (
            as_of_date.year + post_roll_shift
        )  # Current year plus the post_roll_shift amount (in years).

    # As-of-date is before the roll date.
    # Previous year plus the post_roll_shift amount (in years).
    return (as_of_date.year - 1) + post_roll_shift


def add_months_to_date_and_return_yyyymm01(
    *,
    d: date,
    months_to_add: int,
) -> date:
    """Add months to a month-start date (day=1)."""
    y = d.year + (d.month - 1 + months_to_add) // 12
    m = (d.month - 1 + months_to_add) % 12 + 1
    return date(y, m, 1)


def fmt_date_as_str_yyyy(d: date) -> str:
    return f"{d.year:04d}"

# src/riskquant/portfolio/blocks/base_block.py

"""
Base class for all Struct-of-Arrays (SoA) trade blocks.

`Block` carries only what is universal to every compiled block: the flat int64
`trade_uids` array and the `BLOCK_TYPE_ID` dispatch tag. All economic SoA arrays
live on the concrete subclasses.

`BLOCK_TYPE_ID` is the single integer the valuation engine dispatches on: it
indexes directly into `BLOCK_EVALUATORS`, so no `type()` calls, no isinstance
ladders, no string keys. The base value is the `UNSET_BLOCK_TYPE_ID_SENTINEL`
sentinel; every concrete block MUST override it with its `BlockType` enum id.
The override is enforced at class-creation time by `__init_subclass__`, which
fails loudly the moment a new block type is added without wiring its id — a
compile-time guard rather than a runtime surprise.
"""

from __future__ import annotations

from dataclasses import dataclass, fields
from typing import Any, ClassVar

import numpy as np
import numpy.typing as npt

from riskquant.valuation.blocks.block_type import (
    UNSET_BLOCK_TYPE_ID_SENTINEL,
    VALID_BLOCK_TYPE_IDS,
)


@dataclass(frozen=True, slots=True, kw_only=True)
class Block:
    """
    Universal base for compiled SoA valuation blocks.

    Every array element of every field corresponds to the same row index.
    Concrete subclasses define the economic SoA fields required by their
    valuation kernel.

    `trade_uids[i]` identifies the originating trade that produced row `i` of the block.
    """

    # Block type identifier for dispatching to the right pricer kernel. Every concrete
    # subclass MUST override this with its `BlockType` enum id, which is
    # guaranteed to be unique.
    BLOCK_TYPE_ID: ClassVar[int] = UNSET_BLOCK_TYPE_ID_SENTINEL

    trade_uids: npt.NDArray[np.int64]

    def __init_subclass__(cls) -> None:
        super().__init_subclass__()

        if cls.__name__.startswith("_"):
            return

        if cls.BLOCK_TYPE_ID == UNSET_BLOCK_TYPE_ID_SENTINEL:
            raise TypeError(f"{cls.__name__} must define BLOCK_TYPE_ID")

        if cls.BLOCK_TYPE_ID not in VALID_BLOCK_TYPE_IDS:
            raise ValueError(
                f"{cls.__name__} has invalid BLOCK_TYPE_ID {cls.BLOCK_TYPE_ID}"
            )

    @property
    def size(self) -> int:
        """
        Number of elements in a block, i.e. number of cashflows, legs, or
        trades it represents.
        """
        return self.trade_uids.shape[0]

    def print_block(
        self,
        *,
        max_items: int | None = 10,
        float_precision: int = 4,
        sep: str = ", ",
        pad: int = 15,  # Increased default pad slightly to account for the new '_' characters
        thousand_sep: str = "_",
    ) -> None:
        """Print every SoA field with shape, dtype, and a value preview."""

        print(f"{type(self).__name__}:")
        for f in fields(self):
            v = getattr(self, f.name)
            if isinstance(v, np.ndarray):
                # Replace lambdas with typed inner functions to satisfy Pyright/MyPy
                def fmt_float(x: Any) -> str:
                    return f"{x:>{pad}{thousand_sep}.{float_precision}f}"

                def fmt_int(x: Any) -> str:
                    # Use 'd' (decimal integer) instead of a hacked float format
                    return f"{x:>{pad}{thousand_sep}d}"

                # Format the array preview cleanly
                val_preview = np.array2string(
                    v[:max_items],
                    separator=sep,
                    max_line_width=100_000,
                    formatter={
                        "float_kind": fmt_float,
                        "int_kind": fmt_int,
                    },
                )

                print(
                    f"  {f.name:<33} | shape: {v.shape!s:>10} | "
                    f"dtype: {v.dtype!s:>10} | values: {val_preview}"
                )
            else:
                print(f"  {f.name:<33} {v}")
        print()

"""
Compiled valuation primitives (Block Types).

These are the executable, structure-of-arrays (SoA) blocks consumed
by the valuation engine.  They are **not** economic instruments.

Examples
--------
VANILLA_IRS
    ├── FIXED_CASHFLOW
    └── FLOATING_CASHFLOW

FIXED_RATE_BOND
    ├── FIXED_CASHFLOW
    └── PRINCIPAL_CASHFLOW

FRN
    ├── FLOATING_CASHFLOW
    └── PRINCIPAL_CASHFLOW
"""

from enum import IntEnum
from types import MappingProxyType
from typing import Final

# ---------------------------------------------------------------------------
# Sentinel value for an explicitly unset / missing block type ID.
# ---------------------------------------------------------------------------
UNSET_BLOCK_TYPE_ID_SENTINEL: Final[int] = -999


class BlockType(IntEnum):
    """
    Executable SoA block primitives.

    Each member carries a stable integer ID used for serialisation,
    dispatch tables, and AoS ↔ SoA conversion.
    """

    # ======================================================================
    # MARKET RISK PRIMITIVES
    # ======================================================================
    SPOT = 0

    VANILLA_FORWARD = 1

    VANILLA_EUROPEAN_OPTION = 2

    # ======================================================================
    # INTEREST-RATE CASHFLOW PRIMITIVES
    # ======================================================================
    FIXED_CASHFLOW = 3
    FLOATING_CASHFLOW = 4
    PRINCIPAL_CASHFLOW = 5

    # ------------------------------------------------------------------
    # Convenience constructors
    # ------------------------------------------------------------------
    @classmethod
    def from_id(cls, id_: int) -> "BlockType":
        """Return the member whose ``value`` equals *id_*."""
        if id_ not in BLOCK_TYPE_ID_TO_NAME:  # pragma: no cover
            raise ValueError(f"Invalid BlockType id: {id_}")
        # IntEnum allows instantiation directly from the integer value.
        return cls(id_)

    @classmethod
    def from_name(cls, name: str) -> "BlockType":
        """Return the member whose ``name`` equals *name* (case-sensitive)."""
        if name not in BLOCK_TYPE_NAME_TO_ID:  # pragma: no cover
            raise ValueError(f"Invalid BlockType name: {name!r}")
        return cls[name]


# ==========================================================================
# CONVENIENCE ID CONSTANTS FOR DISPATCH / LOOKUP
# ==========================================================================

SPOT_BLOCK_TYPE_ID: int = BlockType.SPOT.value
VANILLA_FORWARD_BLOCK_TYPE_ID: int = BlockType.VANILLA_FORWARD.value
VANILLA_EUROPEAN_OPTION_BLOCK_TYPE_ID: int = BlockType.VANILLA_EUROPEAN_OPTION.value
FIXED_CASHFLOW_BLOCK_TYPE_ID: int = BlockType.FIXED_CASHFLOW.value
FLOATING_CASHFLOW_BLOCK_TYPE_ID: int = BlockType.FLOATING_CASHFLOW.value
PRINCIPAL_CASHFLOW_BLOCK_TYPE_ID: int = BlockType.PRINCIPAL_CASHFLOW.value

# ---------------------------------------------------------------------------
# Immutable lookup tables (single source of truth).
# ---------------------------------------------------------------------------

NUM_BLOCK_TYPES: Final[int] = len(BlockType)

BLOCK_TYPE_NAME_TO_ID: Final[MappingProxyType[str, int]] = MappingProxyType(
    {member.name: member.value for member in BlockType}
)

BLOCK_TYPE_ID_TO_NAME: Final[MappingProxyType[int, str]] = MappingProxyType(
    {member.value: member.name for member in BlockType}
)

# Frozen sets for fast membership tests / validation.
VALID_BLOCK_TYPE_IDS: Final[frozenset[int]] = frozenset(BLOCK_TYPE_ID_TO_NAME.keys())
VALID_BLOCK_TYPE_NAMES: Final[frozenset[str]] = frozenset(BLOCK_TYPE_NAME_TO_ID.keys())

# Guard: the sentinel must never collide with a real block type.
assert UNSET_BLOCK_TYPE_ID_SENTINEL not in VALID_BLOCK_TYPE_IDS, (
    "Sentinel collides with a valid block-type ID"
)


# ---------------------------------------------------------------------------
# Light-weight demo / introspection (kept intentionally simple).
# ---------------------------------------------------------------------------
if __name__ == "__main__":
    print("\n--- BlockType Enumeration ---\n")
    for block_type in BlockType:
        print(f"{block_type.name:40s}  id = {block_type.value:>2d}")

    print("\nID → Name mapping:")
    print(BLOCK_TYPE_ID_TO_NAME)

    print("\nName → ID mapping:")
    print(BLOCK_TYPE_NAME_TO_ID)

    print("\nValid BlockType IDs:")
    print(VALID_BLOCK_TYPE_IDS)

    print("\nValid BlockType Names:")
    print(VALID_BLOCK_TYPE_NAMES)

    print("\nSentinel value for unset block type ID:")
    print(UNSET_BLOCK_TYPE_ID_SENTINEL)

    assert UNSET_BLOCK_TYPE_ID_SENTINEL not in VALID_BLOCK_TYPE_IDS, (
        "Sentinel collides with a valid block-type ID"
    )

    print("\nAll checks passed: sentinel does not collide with valid block type IDs.")
    print("\n----------------------------\n")

from typing import ClassVar

    Compiled deterministic principal cashflows.

    CashFlow(T) = Amount

    PV(0) = CashFlow(T) * DiscountFactor(0, T) * FxSpot(0)

    Examples
    --------
    - Bond redemption
    - Deposit repayment
    - Cross-currency swap notional exchange
    - Bullet loan principal repayment
    - Final notional return on an FRN

    BLOCK_TYPE_ID: ClassVar[int] = int(BlockType.PRINCIPAL_CASHFLOW)

    # --- TRADE SPECIFICS ---

class FixedCashflowBlock(_DiscountedCashflowBlock):
    """
    Compiled fixed-rate cashflows.

    CashFlow(T) = Notional * AccrualFactor * FixedRate * PayReceiveMult

    PV(0) = CashFlow(T) * DiscountFactor(0, T) * FxSpot(0)
    """

    BLOCK_TYPE_ID: ClassVar[int] = int(BlockType.FIXED_CASHFLOW)

    # --- TRADE SPECIFICS ---

    accrual_factor: npt.NDArray[np.float64]
    fixed_rate: npt.NDArray[np.float64]

class FloatingCashflowBlock(_DiscountedCashflowBlock):
    """
    Compiled floating-rate cashflows.

    FloatingRate(0, s, T) = ForwardRate(0, s, T) + Spread

    CashFlow(T) = Notional * FloatingRate(0, s, T) * AccrualFactor * PayReceiveMult

    PV(0) = CashFlow(T) * DiscountFactor(0, T) * FxSpot(0)
    """

    BLOCK_TYPE_ID: ClassVar[int] = int(BlockType.FLOATING_CASHFLOW)

"""
Portfolio valuation engine.

Architectural role
------------------
The valuation engine is responsible for dispatching compiled portfolio
blocks to their registered block pricers.

The engine performs no pricing logic itself.

Responsibilities:

    * iterate compiled blocks
    * resolve block pricers from BLOCK_TYPE_ID
    * route block output into the correct slice of the portfolio output matrix

Non-responsibilities:

    * market-data interpretation
    * pricing-model mathematics
    * output allocation
    * result aggregation
    * reporting

All pricing logic resides in block pricers and pricing kernels.
"""

from __future__ import annotations

from typing import Any

import numpy as np
import numpy.typing as npt

from riskquant.market_data.state import ValuationState
from riskquant.portfolio.compiled import CompiledPortfolio
from riskquant.valuation.pricer_registry import BLOCK_PRICERS, Pricer


def run_valuation(
    *,
    portfolio: CompiledPortfolio,
    valuation_state: ValuationState,
    out: npt.NDArray[np.float64],
) -> None:
    """
    Execute full portfolio valuation across all scenarios in valuation batch.

    Parameters
    ----------
    portfolio
        Compiled portfolio.

    valuation_state
        Fully-resolved valuation factors for the current scenario batch.

    out
        Preallocated output matrix.

        Shape:

            (portfolio.total_trades, valuation_state.batch_size)

        Row ordering matches portfolio.trade_uids.
    """

    blocks = portfolio.blocks
    offsets = portfolio.block_offsets
    block_pricers: tuple[Pricer[Any], ...] = BLOCK_PRICERS

    for block_idx in range(portfolio.total_blocks):
        block = blocks[block_idx]

        start = offsets[block_idx]
        end = start + block.size

        block_pricers[block.BLOCK_TYPE_ID](
            block=block,
            valuation_state=valuation_state,
            out=out[start:end],
        )

# src/riskquant/valuation/kernels/_math.py
"""
Shared bare-metal math helpers for vectorized pricer kernels.

Tiny hot-path functions: `@nb.njit(inline='always')` collapses them into
the surrounding kernel at compile time, so the calling kernel sees fused
code with no call-frame overhead.

Anything that needs `prange` or more than a handful of ops should not
live here.
"""

from __future__ import annotations

import math

from riskquant.numba import inline_njit

# 1 / sqrt(2). Lifted so the kernel never recomputes it.
_INV_SQRT_2: float = 0.7071067811865476


@inline_njit
def numba_norm_cdf(x: float) -> float:
    """Standard normal CDF: Phi(x) = 0.5 * (1 + erf(x / sqrt(2)))."""
    return 0.5 * (1.0 + math.erf(x * _INV_SQRT_2))


@inline_njit
def numba_norm_cdf_abrom_steg(x: float) -> float:

    ax = abs(x)

    t = 1.0 / (1.0 + 0.2316419 * ax)

    poly = t * (
        0.319381530
        + t * (-0.356563782 + t * (1.781477937 + t * (-1.821255978 + t * 1.330274429)))
    )

    pdf = 0.39894228040143267794 * math.exp(-0.5 * ax * ax)

    cdf = 1.0 - pdf * poly

    if x < 0.0:
        return 1.0 - cdf

    return cdf


@inline_njit
def linear_interp_scalar(x: float, xs, ys) -> float:
    """
    Branchless 1D linear interpolation with flat extrapolation.

    Drops the `np.interp` call frame and Numba's bounds-check overhead in
    favour of an inlinable binary search + linear blend. LLVM fuses this
    into the caller's scenario loop.

    Both `xs` and `ys` must be 1-D C-contiguous; `xs` strictly ascending.
    For `x` outside `[xs[0], xs[-1]]` returns the corresponding edge value
    (flat extrapolation), matching `np.interp`'s default semantics and
    avoiding accidental log/division blow-ups on wide-strike moneyness.
    """
    n = xs.shape[0]
    if n == 1:
        return ys[0]

    # Edge clamps — flat extrapolation. Branches are predictable
    # (rare) and the compiler can lift them out of vectorised loops.
    if x <= xs[0]:
        return ys[0]
    if x >= xs[n - 1]:
        return ys[n - 1]

    # Binary search for the right-hand pillar. Loop trip count is
    # log2(n) and the compiler unrolls it for small n.
    lo = 0
    hi = n - 1
    while hi - lo > 1:
        mid = (lo + hi) >> 1
        if xs[mid] <= x:
            lo = mid
        else:
            hi = mid

    x0 = xs[lo]
    x1 = xs[hi]
    y0 = ys[lo]
    y1 = ys[hi]
    # x1 > x0 by strict-monotone invariant.
    w = (x - x0) / (x1 - x0)
    return y0 + w * (y1 - y0)

"""
Vanilla European option pricing kernel, curve variant.

Identical CIP-forward + Black-76 math to the node variant, with two
structural differences:

  - The underlying scalar comes from `price_curve_grid[curve_uix, delivery_uix, s]`.
  - Volatility is sourced from a 4D vol surface and requires per-scenario
    strike-axis interpolation along the pillar coordinate dictated by the
    surface's y-axis type.

FX rows are already EFFECTIVE — no inversion math; the strike-FX rate is
loaded once per scenario, used as a divisor for CIP forward synthesis and
as a multiplier for the PV translation back to settle ccy.

Layout contract (CRITICAL)
--------------------------
`vol_surfaces` MUST be `[surface, expiry, scenario, pillar]` (C-contig).
This guarantees that the pillars for a single trade-scenario tuple form
one contiguous run of memory: `vol_surfaces[v, exp, s]` is a 1-D view
suitable for the binary-search interp. The builder is responsible for
emitting this layout; the kernel relies on it for SIMD-friendly reads.

Branch hoisting
---------------
All sentinel checks (`fx_*_uix < 0`, `df_*_uix < 0`) and y-axis-type
selection are resolved once per trade, outside the scenario loop. Within
the scenario loop only the math executes; CPU branch predictors never
see a `y_type` ladder or a sentinel test per scenario.

Vol-surface-uix sentinel handling
---------------------------------
`v_surf_id = vol_surface_uix[i]` may be `INVALID_UIX (< 0)` when the
ingest compiler resolved no vol container for trade i. With
`boundscheck=False`, `vol_surface_strikes[v_surf_id]` and
`vol_surfaces[v_surf_id, exp_id]` on a negative id are undefined
behaviour — out-of-bounds memory.

This kernel hoists `has_vol = v_surf_id >= 0` once per trade, outside
the scenario loop. The sentinel path writes `out_row[:] = 0.0` and skips
the surface gather + Black-76 math entirely.
"""

from __future__ import annotations

import math

import numba as nb  # type: ignore[import-untyped]

from riskquant.market_data.surfaces.y_axes_types import (
    SURFACE_Y_AXIS_TYPE_LOG_MONEYNESS_FWD,
    SURFACE_Y_AXIS_TYPE_LOG_MONEYNESS_SPOT,
    SURFACE_Y_AXIS_TYPE_LOG_STRIKE,
    SURFACE_Y_AXIS_TYPE_MONEYNESS_FWD,
    SURFACE_Y_AXIS_TYPE_MONEYNESS_SPOT,
    SURFACE_Y_AXIS_TYPE_STRIKE,
)
from riskquant.numba import parallel_njit
from riskquant.valuation.pricer_kernels._deprecated._math_v1 import (
    linear_interp_scalar,
    numba_norm_cdf_abrom_steg,
)

_F64_2D = nb.float64[:, ::1]
_F64_3D = nb.float64[:, :, ::1]
_F64_4D = nb.float64[:, :, :, ::1]
_F64_1D = nb.float64[::1]
_I32_1D = nb.int32[::1]
_I8_1D = nb.int8[::1]

_TTE_FLOOR: float = 1e-9
_VOL_STD_DEV_FLOOR: float = 1e-12

# Y-axis kind groupings as int8 keys. Resolved once per trade and packed
# into a 0/1/2/3 dispatch tag so the inner loop sees no branch ladder.
_Y_KIND_STRIKE: int = 0  # raw strike: y = k
_Y_KIND_LOG_STRIKE: int = 1  # y = log(k)
_Y_KIND_MONEYNESS: int = 2  # y = k / fwd
_Y_KIND_LOG_MONEYNESS: int = 3  # y = log(k / fwd)


@parallel_njit(
    nb.void(
        _F64_2D,  # out
        _F64_3D,  # price_curve_grid
        _F64_2D,  # vol_surface_strikes
        _F64_4D,  # vol_surfaces  [surface, expiry, scenario, pillar]
        _F64_2D,  # fx_spots_effective
        _F64_3D,  # df_grid
        _F64_1D,  # tte_grid
        _F64_1D,  # qty
        _F64_1D,  # strike
        _I8_1D,  # cp_mult
        _I32_1D,  # price_curve_uix
        _I32_1D,  # vol_surface_uix
        _I8_1D,  # y_axis_type_id
        _I32_1D,  # fx_u_eff_uix
        _I32_1D,  # df_u_uix
        _I32_1D,  # fx_k_eff_uix
        _I32_1D,  # df_k_uix
        _I32_1D,  # delivery_date_uix
        _I32_1D,  # expiry_date_uix
        _I32_1D,  # settle_date_uix
    ),
)
def _kernel_price_vanilla_european_option_curve(
    out,
    price_curve_grid,
    vol_surface_strikes,
    vol_surfaces,
    fx_spots_effective,
    df_grid,
    tte_grid,
    qty,
    strike,
    cp_mult,
    price_curve_uix,
    vol_surface_uix,
    y_axis_type_id,
    fx_u_eff_uix,
    df_u_uix,
    fx_k_eff_uix,
    df_k_uix,
    delivery_date_uix,
    expiry_date_uix,
    settle_date_uix,
) -> None:
    m_trades, s_scenarios = out.shape

    for i in nb.prange(m_trades):
        q = qty[i]
        k = strike[i]
        cp = cp_mult[i]

        p_curve_id = price_curve_uix[i]
        v_surf_id = vol_surface_uix[i]
        y_type = y_axis_type_id[i]

        fx_u_id = fx_u_eff_uix[i]
        df_u_id = df_u_uix[i]
        fx_k_id = fx_k_eff_uix[i]
        df_k_id = df_k_uix[i]

        deliv_id = delivery_date_uix[i]
        exp_id = expiry_date_uix[i]
        stl_id = settle_date_uix[i]

        out_row = out[i]

        # ---- Hoist vol-surface-uix sentinel: no surface -> zero PV ----
        # Resolving this BEFORE the surface gather is critical:
        # vol_surface_strikes[<0] and vol_surfaces[<0, ...] are UB under
        # boundscheck=False. Sentinel-trade rows write zeros and skip.
        has_vol = v_surf_id >= 0

        if not has_vol:
            for s in range(s_scenarios):
                out_row[s] = 0.0
            continue

        # ---- Trade-level scalars (hoist out of scenario loop) ----
        tte = tte_grid[exp_id]
        safe_tte = tte if tte > _TTE_FLOOR else _TTE_FLOOR
        sqrt_tte = math.sqrt(safe_tte)
        log_k = math.log(k)

        # ---- Pre-compute the 2D plane handles ----
        #   pillars      : 1-D, length=K_pillars  (shared across all scenarios)
        #   v_plane      : 2-D, [scenario, pillar] — contiguous run per scenario
        #   p_row        : 1-D, length=S          (price-curve row for delivery)
        pillars = vol_surface_strikes[v_surf_id]
        v_plane = vol_surfaces[v_surf_id, exp_id]
        p_row = price_curve_grid[p_curve_id, deliv_id]

        # ---- Hoist sentinel branches into per-trade booleans ----
        has_fx_u = fx_u_id >= 0
        has_df_u = df_u_id >= 0
        has_fx_k = fx_k_id >= 0
        has_df_k = df_k_id >= 0

        # ---- Hoist y-axis-type ladder into a 0/1/2/3 dispatch tag ----
        if y_type == SURFACE_Y_AXIS_TYPE_STRIKE:
            y_kind = _Y_KIND_STRIKE
        elif y_type == SURFACE_Y_AXIS_TYPE_LOG_STRIKE:
            y_kind = _Y_KIND_LOG_STRIKE
        elif y_type in (
            SURFACE_Y_AXIS_TYPE_MONEYNESS_FWD,
            SURFACE_Y_AXIS_TYPE_MONEYNESS_SPOT,
        ):
            y_kind = _Y_KIND_MONEYNESS
        elif y_type in (
            SURFACE_Y_AXIS_TYPE_LOG_MONEYNESS_FWD,
            SURFACE_Y_AXIS_TYPE_LOG_MONEYNESS_SPOT,
        ):
            y_kind = _Y_KIND_LOG_MONEYNESS
        else:
            # Unknown y-axis: degrade to strike. Producer validation
            # rejects this case before it reaches the kernel.
            y_kind = _Y_KIND_STRIKE

        for s in range(s_scenarios):
            # ---- CIP forward in strike currency ----
            fwd = p_row[s]
            if has_fx_u:
                fwd *= fx_spots_effective[fx_u_id, s]
            if has_df_u:
                fwd *= df_grid[df_u_id, exp_id, s]

            rate_k = 1.0
            if has_fx_k:
                rate_k = fx_spots_effective[fx_k_id, s]
                fwd /= rate_k
            if has_df_k:
                fwd /= df_grid[df_k_id, exp_id, s]

            # ---- Y-axis target for vol surface lookup ----
            # The branch tag is constant per trade; LLVM should hoist
            # this select out of the SIMD body via loop-unswitching.
            if y_kind == _Y_KIND_STRIKE:
                y_target = k
            elif y_kind == _Y_KIND_LOG_STRIKE:
                y_target = log_k
            elif y_kind == _Y_KIND_MONEYNESS:
                y_target = k / fwd
            else:  # _Y_KIND_LOG_MONEYNESS
                y_target = log_k - math.log(fwd)

            # ---- Strike-axis interpolation ----
            # TODO: QUESTION - DO WE NEED TO DO OTHER TYPES OF INTERPOLATION FOR OTHER
            # Y-AXIS KINDS? E.g. log-log for log-strike?
            # v_plane[s] is a C-contig 1-D run of pillars for THIS scenario.
            sigma = linear_interp_scalar(y_target, pillars, v_plane[s])

            # ---- Black-76 future value ----
            std_dev = sigma * sqrt_tte
            if std_dev < _VOL_STD_DEV_FLOOR:
                std_dev = _VOL_STD_DEV_FLOOR

            d1 = (math.log(fwd) - log_k + 0.5 * sigma * sigma * safe_tte) / std_dev
            d2 = d1 - std_dev

            cdf1 = numba_norm_cdf_abrom_steg(cp * d1)
            cdf2 = numba_norm_cdf_abrom_steg(cp * d2)

            fv = cp * (fwd * cdf1 - k * cdf2)

            # ---- PV discount in strike ccy, FX-translate to settle ccy ----
            if has_df_k:
                fv *= df_grid[df_k_id, stl_id, s]
            fv *= rate_k

            out_row[s] = q * fv

"""
Vanilla European option pricing kernel, node variant.

Synthesises the forward in strike currency via Covered Interest Parity,
prices via Black-76, then discounts and FX-translates back to settle ccy.

Math (per trade i, scenario s)
------------------------------
    rate_k = fx_spots_effective[fx_k_eff_uix(i), s]   # k_ccy -> settle_ccy

    # CIP forward in strike currency
    fwd_strike_ccy
        = forward_node[p, s]
          * fx_spots_effective[fx_u_eff_uix(i), s]    # u_ccy -> settle_ccy
          * df_grid[df_u_uix(i), expiry_uix(i), s]    # under-DF at expiry
          / rate_k                                    # settle_ccy -> k_ccy
          / df_grid[df_k_uix(i), expiry_uix(i), s]    # strike-DF at expiry

    # Black-76 future value (cp_mult = +1 call, -1 put)
    sigma = vol_nodes[v, s]
    std_dev = max(sigma * sqrt(tte), eps)
    d1 = (log(fwd / k) + 0.5 * sigma**2 * tte) / std_dev
    d2 = d1 - std_dev
    fv = cp * (fwd * N(cp * d1) - k * N(cp * d2))

    # PV discount in strike ccy, then translate back to settle ccy
    pv_strike_ccy = fv * df_grid[df_k_uix(i), settle_uix(i), s]
    pv_settle_ccy = pv_strike_ccy * rate_k

    out[i, s] = qty[i] * pv_settle_ccy

FX rows are already EFFECTIVE in `MarketDataMatrix.fx_spots_effective`; the
kernel does NO inversion math. The strike-FX rate `rate_k` is loaded once
per scenario and used as a divisor in the forward synthesis and as a
multiplier in the PV translation — equivalent counts to the legacy code
but with the `(1.0 / rate_k) if inv_k else rate_k` gymnastics gone.

Vol-uix sentinel handling
-------------------------
`v_id = vol_uix[i]` may be `INVALID_UIX (< 0)` when the ingest compiler
resolved no vol container for trade i (e.g. a downstream trade that
slipped past validation, or a future-class instrument whose vol path is
not yet wired). With `boundscheck=False`, `vol_nodes[v_id]` on a negative
id is undefined behaviour — out-of-bounds memory.

This kernel hoists `has_vol = v_id >= 0` once per trade, outside the
scenario loop. The sentinel path writes `out_row[:] = 0.0` and skips
the vol gather + Black-76 math entirely. Two scenario loops — one for
the priced path, one for the zero-fill path — keep the SIMD body free
of per-scenario branches.
"""

from __future__ import annotations

import math

import numba as nb  # type: ignore[import-untyped]

from riskquant.numba import parallel_njit
from riskquant.valuation.pricer_kernels._deprecated._math_v1 import (
    numba_norm_cdf_abrom_steg,
)

_F64_2D = nb.float64[:, ::1]
_F64_3D = nb.float64[:, :, ::1]
_F64_1D = nb.float64[::1]
_I32_1D = nb.int32[::1]
_I8_1D = nb.int8[::1]

_TTE_FLOOR: float = 1e-9
_VOL_STD_DEV_FLOOR: float = 1e-12


@parallel_njit(
    nb.void(
        _F64_2D,  # out
        _F64_2D,  # forward_nodes
        _F64_2D,  # vol_nodes
        _F64_2D,  # fx_spots_effective
        _F64_3D,  # df_grid
        _F64_1D,  # sqrt_tte_grid
        _F64_1D,  # half_tte_grid
        _F64_1D,  # qty
        _F64_1D,  # strike
        _I8_1D,  # cp_mult
        _I32_1D,  # price_uix
        _I32_1D,  # vol_uix
        _I32_1D,  # fx_u_eff_uix
        _I32_1D,  # df_u_uix
        _I32_1D,  # fx_k_eff_uix
        _I32_1D,  # df_k_uix
        _I32_1D,  # expiry_date_uix
        _I32_1D,  # settle_date_uix
    ),
)
def _kernel_price_vanilla_european_option_node(
    out,
    forward_nodes,
    vol_nodes,
    fx_spots_effective,
    df_grid,
    sqrt_tte_grid,
    half_tte_grid,
    qty,
    strike,
    cp_mult,
    price_uix,
    vol_uix,
    fx_u_eff_uix,
    df_u_uix,
    fx_k_eff_uix,
    df_k_uix,
    expiry_date_uix,
    settle_date_uix,
) -> None:

    m_trades, s_scenarios = out.shape

    for i in nb.prange(m_trades):
        # ------------------------------------------------------------------
        # Trade constants
        # ------------------------------------------------------------------

        q = qty[i]
        k = strike[i]
        cp = cp_mult[i]

        p_id = price_uix[i]
        v_id = vol_uix[i]

        fx_u_id = fx_u_eff_uix[i]
        df_u_id = df_u_uix[i]

        fx_k_id = fx_k_eff_uix[i]
        df_k_id = df_k_uix[i]

        exp_id = expiry_date_uix[i]
        stl_id = settle_date_uix[i]

        out_row = out[i]

        # ---- Hoist vol-uix sentinel: no vol container -> zero PV ----
        has_vol = v_id >= 0

        if not has_vol:
            for s in range(s_scenarios):
                out_row[s] = 0.0
            continue

        sqrt_tte = sqrt_tte_grid[exp_id]
        half_tte = half_tte_grid[exp_id]
        log_k = math.log(k)

        # ------------------------------------------------------------------
        # Scenario rows
        # ------------------------------------------------------------------

        p_row = forward_nodes[p_id]
        v_row = vol_nodes[v_id]

        has_fx_u = fx_u_id >= 0
        has_df_u = df_u_id >= 0
        has_fx_k = fx_k_id >= 0
        has_df_k = df_k_id >= 0

        if has_fx_u:
            fx_u_row = fx_spots_effective[fx_u_id]

        if has_fx_k:
            fx_k_row = fx_spots_effective[fx_k_id]

        if has_df_u:
            df_u_exp_row = df_grid[df_u_id, exp_id]

        if has_df_k:
            df_k_exp_row = df_grid[df_k_id, exp_id]
            df_k_stl_row = df_grid[df_k_id, stl_id]

        # ------------------------------------------------------------------
        # Scenario loop
        # ------------------------------------------------------------------

        for s in range(s_scenarios):
            fwd = p_row[s]

            if has_fx_u:
                fwd *= fx_u_row[s]  # pyright: ignore[reportPossiblyUnboundVariable]

            if has_df_u:
                fwd *= df_u_exp_row[s]  # pyright: ignore[reportPossiblyUnboundVariable]

            rate_k = 1.0

            if has_fx_k:
                rate_k = fx_k_row[s]  # pyright: ignore[reportPossiblyUnboundVariable]
                fwd *= 1.0 / rate_k

            if has_df_k:
                fwd *= 1.0 / df_k_exp_row[s]  # pyright: ignore[reportPossiblyUnboundVariable]

            sigma = v_row[s]
            sigma_sq = sigma * sigma

            std_dev = sigma * sqrt_tte
            if std_dev < _VOL_STD_DEV_FLOOR:
                std_dev = _VOL_STD_DEV_FLOOR

            log_fwd = math.log(fwd)

            d1 = (log_fwd - log_k + sigma_sq * half_tte) / std_dev

            d2 = d1 - std_dev

            cdf1 = numba_norm_cdf_abrom_steg(cp * d1)
            cdf2 = numba_norm_cdf_abrom_steg(cp * d2)

            fv = cp * (fwd * cdf1 - k * cdf2)

            if has_df_k:
                fv *= df_k_stl_row[s]  # pyright: ignore[reportPossiblyUnboundVariable]

            fv *= rate_k

            out_row[s] = q * fv

            out_row[s] = accrued_notional_i * floating_rate_s * df_row[s] * fx_row[s]

"""
Spot pricing kernel.

PV = qty * spot_price * fx_effective

`fx_spots_effective` rows are already the resolved transform (direct or
pre-inverted) — pricer kernels do no inversion math. Per-trade FX is one
gather + one multiply per scenario. The `fx_effective_uix < 0` sentinel
selects the "no FX" branch which is hoisted outside the scenario loop.
"""

from __future__ import annotations

import numba as nb  # type: ignore[import-untyped]

from riskquant.numba import parallel_njit

# ==============================================================================
# Numba layout-pinned aliases — force C-contiguous strides at compile time so
# the kernel emits straight-line SIMD loops with no dynamic stride checks.
# ==============================================================================

_F64_2D = nb.float64[:, ::1]
_F64_1D = nb.float64[::1]
_I32_1D = nb.int32[::1]


@parallel_njit(
    nb.void(
        _F64_2D,  # out
        _F64_2D,  # spot_prices
        _F64_2D,  # fx_spots_effective
        _F64_1D,  # qty
        _I32_1D,  # spot_prices_uix
        _I32_1D,  # fx_spots_effective_uix
    ),
)
def _kernel_price_spot(
    # OUTPUT
    out,  # (N, S)
    # MARKET DATA
    spot_prices,  # (P, S)
    fx_spots_effective,  # (F, S)
    # BLOCK FIELDS
    qty,  # (N,)
    spot_price_uix,  # (N,)
    fx_spot_effective_uix,  # (N,)
) -> None:
    """
    Fused spot pricing kernel: out[i, s] = spot_prices[p, s] * qty[i] * fx[f, s].

    Zero allocations. Each row is a contiguous gather + scale + (gated)
    multiply streaming through C-contiguous memory.
    """
    m_trades, s_scenarios = out.shape

    for i in nb.prange(m_trades):
        qty_i = qty[i]

        spot_price_id = spot_price_uix[i]
        fx_id = fx_spot_effective_uix[i]

        spot_prices_row = spot_prices[spot_price_id]
        fx_row = fx_spots_effective[fx_id]

        out_row = out[i]

        for s in range(s_scenarios):
            out_row[s] = qty_i * spot_prices_row[s] * fx_row[s]

import numba as nb  # type: ignore[import-untyped]

        # OUTPUT
        _F64_2D,  # out
        # MARKET DATA
        _F64_2D,  # forward_prices
        _F64_2D,  # vols
        _F64_2D,  # discount_factors
        _F64_2D,  # fx_spots_effective
        # BLOCK FIELDS (TRADE SPECIFICS)
        _F64_1D,  # qty
        _F64_1D,  # strike
        _I8_1D,  # cp_mult
        _F64_1D,  # sqrt_tte
        _F64_1D,  # half_tte
        # BLOCK FIELDS (LINKAGES / UIX)
        _I32_1D,  # forward_price_uix
        _I32_1D,  # vol_uix
        _I32_1D,  # fx_spot_effective_u2s_uix
        _I32_1D,  # fx_spot_effective_k2s_uix
        _I32_1D,  # df_u_exp_uix
        _I32_1D,  # df_k_exp_uix
        _I32_1D,  # df_k_settle_uix
    ),
)
def _kernel_price_vanilla_european_option(
    # OUTPUT
    out,  # (N, S)
    # MARKET DATA
    forward_prices,  # (P, S)
    vols,  # (V, S)
    discount_factors,  # (D, S)
    fx_spots_effective,  # (F, S)
    # BLOCK FIELDS (TRADE SPECIFICS)
    qty,  # (N,)
    strike,  # (N,)
    cp_mult,  # (N,)
    sqrt_tte,  # (N,)
    half_tte,  # (N,)
    # BLOCK FIELDS (LINKAGES / UIX)
    forward_price_uix,  # (N,)
    vol_uix,  # (N,)
    discount_factor_k_uix,  # (N,)
    fx_spot_effective_k2s_uix,  # (N,)
) -> None:
    m_trades, s_scenarios = out.shape

    for i in nb.prange(m_trades):
        # --- TRADE SPECIFICS (FOR ROW) ---
        qty_i = qty[i]
        strike_i = strike[i]
        sqrt_tte_i = sqrt_tte[i]
        half_tte_i = half_tte[i]
        cp_mult_i = cp_mult[i]

        # --- UIX (FOR ROW) ---
        forward_price_id = forward_price_uix[i]
        vol_id = vol_uix[i]
        df_k_id = discount_factor_k_uix[i]
        fx_k2s_id = fx_spot_effective_k2s_uix[i]

        # --- MARKET DATA (FOR ROW) ---
        forwards_row = forward_prices[forward_price_id]
        vols_row = vols[vol_id]
        df_k_row = discount_factors[df_k_id]
        fx_k2s_row = fx_spots_effective[fx_k2s_id]

        # --- HOISTED TRADE-SPECIFIC SCALARS (OUTSIDE SCENARIO LOOP) ---
        log_strike_i = math.log(strike_i)

        # --- PRICING ACROSS SCENARIOS (ACROSS ROW) ---
        for s in range(s_scenarios):
            fwd_s = forwards_row[s]
            vol_s = vols_row[s]

            fv_s = _black76_fv_with_kernel_hoists(
                fwd_s,
                strike_i,
                log_strike_i,
                sqrt_tte_i,
                half_tte_i,
                vol_s,
                cp_mult_i,
            )

            # PV = FX RATE * DISCOUNT FACTOR * QTY * FV
            pv_s = fx_k2s_row[s] * df_k_row[s] * qty_i * fv_s

            out[i, s] = pv_s

# PRICER PROTOCOL
# ==============================================================================

# 1. Define a contravariant TypeVar bound to Block
B_contra = TypeVar("B_contra", bound=Block, contravariant=True)


@runtime_checkable
class Pricer(Protocol[B_contra]):
    """
    Structural contract for all block pricers.
    """

    def __call__(
        self,
        *,
        block: B_contra,  # 2. Use the TypeVar instead of base Block
        valuation_state: ValuationState,
        out: npt.NDArray[np.float64],
    ) -> None: ...

# DISPATCH TABLE
# ==============================================================================

# 3. Explicitly type the list elements as Pricer[Any] | None
_BLOCK_PRICERS_MUT: list[Pricer[Any] | None] = [None] * NUM_BLOCK_TYPES


def _register_block_pricer(block_type_id: int, pricer: Pricer[Any]) -> None:
    # runtime_checkable ignores TypeVars, so this check still works natively
    if not isinstance(pricer, Pricer):
        raise TypeError(
            f"Pricer for block_type_id={block_type_id} does not satisfy the "
            f"Pricer protocol: {pricer!r}"
        )
    _BLOCK_PRICERS_MUT[block_type_id] = pricer

    int(SPOT_BLOCK_TYPE_ID),
    price_spot_block,

_register_block_pricer(

)
_register_block_pricer(
    int(FLOATING_CASHFLOW_BLOCK_TYPE_ID),
    price_floating_cashflow_block,
)

# Freeze. No mutations after module load.
# This tells the type checker: "Trust me, the runtime guard above handled the Nones."
BLOCK_PRICERS: Final[tuple[Pricer[Any], ...]] = cast(
    tuple[Pricer[Any], ...], tuple(_BLOCK_PRICERS_MUT)
)

# Hard fail at import if any block has no pricer: catches the case where
# a new Block variant is added but not wired up here.
_unregistered_blocks = [
    BlockType(i).name for i, ev in enumerate(BLOCK_PRICERS) if ev is None
]
if _unregistered_blocks:
    raise RuntimeError(
        f"No pricer registered for blocks: {_unregistered_blocks}. "
        f"Register them in registry.py."
    )


def resolve_pricer(block: Block) -> Pricer:
    """
    Resolve the pricer for a compiled block via its `BLOCK_TYPE_ID`.

    Single-array index — no `type()`, no isinstance ladder, no string keys.
    """
    pricer = BLOCK_PRICERS[block.BLOCK_TYPE_ID]
    if pricer is None:  # unreachable given the import-time guard above
        raise RuntimeError(
            f"No pricer for BLOCK_TYPE_ID={block.BLOCK_TYPE_ID} "
            f"({type(block).__name__})."
        )
    return pricer

"""
Vanilla European option pricer, node variant.

Synthesises the forward in strike currency via Covered Interest Parity,
prices via Black-76, then discounts and FX-translates back to settle ccy.

Math (per trade i, scenario s)
------------------------------
    rate_k = fx_spots_effective[fx_k_eff_uix(i), s]   # k_ccy -> settle_ccy

    # CIP forward in strike currency
    fwd_strike_ccy
        = forward_node[p, s]
          * fx_spots_effective[fx_u_eff_uix(i), s]    # u_ccy -> settle_ccy
          * df_grid[df_u_uix(i), expiry_uix(i), s]    # under-DF at expiry
          / rate_k                                    # settle_ccy -> k_ccy
          / df_grid[df_k_uix(i), expiry_uix(i), s]    # strike-DF at expiry

    # Black-76 future value (cp_mult = +1 call, -1 put)
    sigma = vol_nodes[v, s]
    std_dev = max(sigma * sqrt(tte), eps)
    d1 = (log(fwd / k) + 0.5 * sigma**2 * tte) / std_dev
    d2 = d1 - std_dev
    fv = cp * (fwd * N(cp * d1) - k * N(cp * d2))

    # PV discount in strike ccy, then translate back to settle ccy
    pv_strike_ccy = fv * df_grid[df_k_uix(i), settle_uix(i), s]
    pv_settle_ccy = pv_strike_ccy * rate_k

    out[i, s] = qty[i] * pv_settle_ccy

FX rows are already EFFECTIVE in `MarketDataMatrix.fx_spots_effective`; the
kernel does NO inversion math. The strike-FX rate `rate_k` is loaded once
per scenario and used as a divisor in the forward synthesis and as a
multiplier in the PV translation — equivalent counts to the legacy code

# src/riskquant/valuation/pricing_models/black76/api.py
"""
Public API for the Black-76 pricing model.

This module provides user-facing entry points for evaluating the
Black-76 model.

Architectural role
------------------
The API layer sits above the model-evaluation kernels and provides a
stable, convenient interface for callers that wish to evaluate the
Black-76 model directly.

Responsibilities include:

    * allocating output arrays
    * exposing user-friendly function signatures
    * input validation
    * dispatching to optimized batch kernels

The API intentionally hides implementation details such as:

    * Numba kernels
    * memory-layout requirements
    * internal helper functions

Relationship to the valuation framework
---------------------------------------
The functions in this module evaluate the Black-76 model directly and
are independent of RiskQuant's portfolio valuation architecture.

They do not require:

    * Blocks
    * ValuationState
    * Compiled portfolios

Typical consumers include:

    * calibration routines
    * market-data tooling
    * research notebooks
    * standalone pricing workflows
    * testing and validation

Portfolio valuation should generally occur through product-specific
valuation kernels rather than through this API.
"""

from __future__ import annotations

import numpy as np
from numpy.typing import NDArray

from riskquant._arrays import _as_c_contig_bool, _as_c_contig_f64
from riskquant.valuation.pricing_models.black76.kernels import _kernel_black76_fv_batch


def price_black76_fv_batch(
    *,
    forwards: NDArray[np.float64],
    vols: NDArray[np.float64],
    strike: NDArray[np.float64],
    tte: NDArray[np.float64],
    is_call: NDArray[np.bool_],
) -> NDArray[np.float64]:
    """
    Price a batch of options under the Black 76 model. Returns FUTURE VALUE (FV) of the option, not PV.
    The caller is responsible for multiplying by the discount factor to get PV.

    Preconditions
    -------------
    The caller guarantees:

        forward > 0
        strike > 0
        volatility > 0
        tte > 0
    """

    # --- CONTIGUITY ENFORCEMENT ---

    forwards = _as_c_contig_f64(forwards)
    vols = _as_c_contig_f64(vols)
    strike = _as_c_contig_f64(strike)
    tte = _as_c_contig_f64(tte)
    is_call = _as_c_contig_bool(is_call)

    # --- SHAPE VALIDATION ---

    # Market data arrays are 2D (num_options, batch_size).

    if forwards.ndim != 2:
        raise ValueError("forwards must be a 2D array")

    if vols.ndim != 2:
        raise ValueError("vols must be a 2D array")

    # Trade-specific arrays are 1D (num_options,).

    if strike.ndim != 1:
        raise ValueError("strike must be a 1D array")

    if tte.ndim != 1:
        raise ValueError("tte must be a 1D array")

    if is_call.ndim != 1:
        raise ValueError("is_call must be a 1D array")

    m_options, s_scenarios = forwards.shape

    if vols.shape != (m_options, s_scenarios):
        raise ValueError("vols must have the same shape as forwards")

    if strike.shape != (m_options,):
        raise ValueError("strike must have the same number of rows as forwards")

    if tte.shape != (m_options,):
        raise ValueError("tte must have the same number of rows as forwards")

    if is_call.shape != (m_options,):
        raise ValueError("is_call must have the same number of rows as forwards")

    # --- VALUES VALIDATION ---

    if np.any(forwards <= 0.0):
        raise ValueError("forwards must be strictly positive")

    if np.any(vols <= 0.0):
        raise ValueError("vols must be strictly positive")

    if np.any(strike <= 0.0):
        raise ValueError("strike must be strictly positive")

    if np.any(tte <= 0.0):
        raise ValueError("tte must be strictly positive")

    # --- FINITE CHECKS ---

    if not np.all(np.isfinite(forwards)):
        raise ValueError("forwards must be finite")

    if not np.all(np.isfinite(vols)):
        raise ValueError("vols must be finite")

    if not np.all(np.isfinite(strike)):
        raise ValueError("strike must be finite")

    if not np.all(np.isfinite(tte)):
        raise ValueError("tte must be finite")

    # --- KERNEL DISPATCH ---

    cp_mult = np.where(
        is_call,
        np.int8(1),
        np.int8(-1),
    )
    sqrt_tte = np.sqrt(tte)
    half_tte = 0.5 * tte

    out = np.empty((m_options, s_scenarios), dtype=np.float64)

    _kernel_black76_fv_batch(
        # OUTPUT
        out,
        # MARKET DATA
        forwards,
        vols,
        # TRADE SPECIFICS
        strike,
        sqrt_tte,
        half_tte,
        cp_mult,
    )

    return out

# src/riskquant/valuation/pricing_models/black76/kernels.py
"""
High-throughput Black-76 model-evaluation kernels.

This module contains batch-oriented Numba kernels that evaluate the
Black-76 model across many instruments and/or scenarios.

Architectural role
------------------
These kernels are model-evaluation kernels, not product-valuation
kernels.

They operate directly on arrays of Black-76 inputs:

    * forwards
    * strikes
    * volatilities
    * times to expiry

and produce arrays of Black-76 outputs.

They do not know about:

    * portfolios
    * blocks
    * valuation states
    * settlement conventions
    * FX conversion
    * discount-factor construction
    * market-data compilation

Typical consumers include:

    * pricing-model APIs
    * volatility-surface calibration
    * implied-volatility stripping
    * research tooling
    * benchmarking

Product valuation kernels should generally call the scalar functions in
`model.py` directly rather than routing through these batch kernels.

These kernels exist to provide efficient bulk evaluation of the Black-76
model outside the portfolio valuation framework.
"""

from __future__ import annotations

import math

import numba as nb  # type: ignore[import-untyped]

from riskquant.numba import (
    _F64_1D,
    _F64_2D,
    _I8_1D,
    parallel_njit,
)
from riskquant.valuation.pricing_models.black76.model import (
    _black76_fv_with_kernel_hoists,
)


@parallel_njit(
    nb.void(
        # OUTPUT
        _F64_2D,  # out
        # MARKET DATA
        _F64_2D,  # forwards
        _F64_2D,  # vols
        # TRADE SPECIFICS
        _F64_1D,  # strike
        _F64_1D,  # sqrt_tte
        _F64_1D,  # half_tte
        _I8_1D,  # cp_mult
    ),
)
def _kernel_black76_fv_batch(
    # OUTPUT
    out,  # (m, S)
    # MARKET DATA
    forwards,  # (m, S)
    vols,  # (m, S)
    # TRADE SPECIFICS
    strike,  # (m, )
    sqrt_tte,  # (m, )
    half_tte,  # (m, )
    cp_mult,  # (m,)
):
    """
    Black 76 pricing kernel. Returns FUTURE VALUE (FV) of the option, not PV.
    The caller is responsible for multiplying by the discount factor to get PV.

    Preconditions
    -------------
    The caller guarantees:

        forward > 0
        strike > 0
        volatility > 0
        sqrt_tte > 0

    Violating these invariants results in undefined behaviour.

    Validation belongs in the compiler layer, not the pricing kernel.

    Arguments
    ---------
    out : np.ndarray, shape (m, S)
        Output array to write the FV of the option for each trade and scenario.
    forwards : np.ndarray, shape (m, S)
        Forward prices for each trade and scenario.
    vols : np.ndarray, shape (m, S)
        Volatilities for each trade and scenario.
    strike : np.ndarray, shape (m,)
        Strike prices for each trade.
    sqrt_tte : np.ndarray, shape (m,)
        Square root of time to expiry for each trade.
    half_tte : np.ndarray, shape (m,)
        Half of the time to expiry for each trade.
    cp_mult : np.ndarray, shape (m,)
        Call/put multiplier for each trade.

    """

    m_trades, s_scenarios = out.shape

    for i in nb.prange(m_trades):
        # --- TRADE SPECIFICS (FOR ROW) ---
        strike_i = strike[i]
        sqrt_tte_i = sqrt_tte[i]
        half_tte_i = half_tte[i]
        cp_mult_i = cp_mult[i]

        # --- MARKET DATA (FOR ROW) ---
        forwards_row = forwards[i]
        vols_row = vols[i]

        # --- HOISTED TRADE-SPECIFIC SCALARS (OUTSIDE SCENARIO LOOP) ---
        log_strike_i = math.log(strike_i)

        # --- PRICING ACROSS SCENARIOS (ACROSS ROW) ---
        for s in range(s_scenarios):
            fwd_s = forwards_row[s]
            vol_s = vols_row[s]

            fv_s = _black76_fv_with_kernel_hoists(
                fwd_s,
                strike_i,
                log_strike_i,
                sqrt_tte_i,
                half_tte_i,
                vol_s,
                cp_mult_i,
            )

            out[i, s] = fv_s

"""
Black 76 pricing model primitives.

Scalar Black-76 mathematical primitives.

This module contains the core Black-76 pricing model expressed as
small, allocation-free functions operating on scalar inputs.

Architectural role
------------------
Functions in this module represent Black-76 mathematics only.

They do not know about:

    * portfolios
    * valuation states
    * blocks
    * market-data containers
    * discounting conventions
    * FX conversion
    * settlement mechanics
    * batch execution

They are intended to be:

    * reusable
    * inline-able
    * allocation-free
    * independent of the wider RiskQuant valuation framework

Typical consumers include:

    * product valuation kernels
    * model-evaluation kernels
    * calibration routines
    * implied-volatility solvers
    * research tooling

The functions in this module form the single source of truth for
Black-76 mathematics throughout RiskQuant.
"""

from __future__ import annotations

import math

from riskquant.numba import inline_njit
from riskquant.valuation.pricing_models.math import (
    numba_norm_cdf_abrom_steg,
    numba_norm_pdf,
)

# =======================
# PRICING
# =======================


@inline_njit
def _black76_fv_with_kernel_hoists(
    fwd: float,
    strike: float,
    log_strike: float,
    sqrt_tte: float,
    half_tte: float,
    vol: float,
    cp_mult: int,
) -> float:
    """Undiscounted Black 76 option price. Returns FV, not PV. Caller is responsible
    for multiplying by the discount factor to get PV."""

    log_fk = math.log(fwd) - log_strike

    std_dev = vol * sqrt_tte

    d1 = (log_fk + vol * vol * half_tte) / std_dev
    d2 = d1 - std_dev

    cdf_d1 = numba_norm_cdf_abrom_steg(cp_mult * d1)
    cdf_d2 = numba_norm_cdf_abrom_steg(cp_mult * d2)

    fv = cp_mult * (fwd * cdf_d1 - strike * cdf_d2)

    return fv


@inline_njit
def _black76_fv(
    fwd: float,
    strike: float,
    sqrt_tte: float,
    half_tte: float,
    vol: float,
    cp_mult: int,
) -> float:
    """Undiscounted Black 76 option price. Returns FV, not PV. Caller is responsible
    for multiplying by the discount factor to get PV."""
    return _black76_fv_with_kernel_hoists(
        fwd,
        strike,
        math.log(strike),
        sqrt_tte,
        half_tte,
        vol,
        cp_mult,
    )


@inline_njit
def _black76_fv_call(
    fwd: float,
    strike: float,
    sqrt_tte: float,
    half_tte: float,
    vol: float,
) -> float:
    """Undiscounted Black 76 call price. Returns FV, not PV. Caller is
    responsible for multiplying by the discount factor to get PV."""
    return _black76_fv(fwd, strike, sqrt_tte, half_tte, vol, 1)


@inline_njit
def _black76_fv_put(
    fwd: float,
    strike: float,
    sqrt_tte: float,
    half_tte: float,
    vol: float,
) -> float:
    """Undiscounted Black 76 put price. Returns FV, not PV. Caller is
    responsible for multiplying by the discount factor to get PV."""
    return _black76_fv(fwd, strike, sqrt_tte, half_tte, vol, -1)

@inline_njit
def _black76_fv_vega(
    fwd: float,
    strike: float,
    sqrt_tte: float,
    half_tte: float,
    vol: float,
) -> float:
    """
    Undiscounted Black-76 vega: dC/dvol = F * phi(d1) * sqrt(T).

    Returns 0.0 for non-positive vol or tte. This makes vega the natural
    quality weight on a calibrated IV: it is 0 for failed solves (sentinel
    vol < 0) and for at-intrinsic zero-vol solves, tiny in flat-vega regions
    (deep ITM/OTM, ultra-short tenors) where the IV is under-determined, and
    large where the inversion is well-conditioned.
    """
    if vol <= 0.0 or sqrt_tte <= 0.0:
        return 0.0

    log_fk = math.log(fwd) - math.log(strike)

    std_dev = vol * sqrt_tte

    d1 = (log_fk + vol * vol * half_tte) / std_dev

    pdf_d1 = numba_norm_pdf(d1)

    return fwd * pdf_d1 * sqrt_tte

# src/riskquant/valuation/pricing_models/math.py
"""
Shared bare-metal math helpers for vectorized pricer kernels.

Tiny hot-path functions: `@nb.njit(inline='always')` collapses them into
the surrounding kernel at compile time, so the calling kernel sees fused
code with no call-frame overhead.

Anything that needs `prange` or more than a handful of ops should not
live here.
"""

from __future__ import annotations

import math

from riskquant.numba import inline_njit

# 1 / sqrt(2). Lifted so the kernel never recomputes it.
_INV_SQRT_2: float = 0.7071067811865476

# 1 / sqrt(2 * pi). Lifted so the kernel never recomputes it.
_INV_SQRT_2PI: float = 0.39894228040143267794


@inline_njit
def numba_norm_cdf(x: float) -> float:
    """Standard normal CDF: Phi(x) = 0.5 * (1 + erf(x / sqrt(2)))."""
    return 0.5 * (1.0 + math.erf(x * _INV_SQRT_2))


@inline_njit
def numba_norm_pdf(x: float) -> float:
    """Standard normal PDF: phi(x) = exp(-0.5 * x^2) / sqrt(2 * pi)."""
    return _INV_SQRT_2PI * math.exp(-0.5 * x * x)


@inline_njit
def numba_norm_cdf_abrom_steg(x: float) -> float:

    ax = abs(x)

    t = 1.0 / (1.0 + 0.2316419 * ax)

    poly = t * (
        0.319381530
        + t * (-0.356563782 + t * (1.781477937 + t * (-1.821255978 + t * 1.330274429)))
    )

    pdf = 0.39894228040143267794 * math.exp(-0.5 * ax * ax)

    cdf = 1.0 - pdf * poly

    if x < 0.0:
        return 1.0 - cdf

    return cdf

# src/riskquant/valuation/result.py
"""
Valuation results container.
"""

from dataclasses import dataclass

import numpy as np
import numpy.typing as npt


@dataclass(frozen=True, slots=True, kw_only=True)
class ValuationResult:
    trade_uids: npt.NDArray[np.int64]  # (n_trades,)
    values: npt.NDArray[np.float64]  # (n_trades, n_scenarios)

    def __post_init__(self) -> None:
        if self.values.shape[0] != self.trade_uids.shape[0]:
            raise ValueError("UIDs and values row mismatch")

    @property
    def num_trades(self) -> int:
        return self.values.shape[0]

    @property
    def num_scenarios(self) -> int:
        return self.values.shape[1]

    def get_trade_values(self, uid: int) -> npt.NDArray[np.float64]:
        idx = np.searchsorted(self.trade_uids, uid)
        if self.trade_uids[idx] != uid:
            raise KeyError(uid)
        return self.values[idx]

    def select(self, uids: npt.NDArray[np.int64]) -> "ValuationResult":
        idx = np.isin(self.trade_uids, uids)
        return ValuationResult(
            trade_uids=self.trade_uids[idx],
            values=self.values[idx],
        )
