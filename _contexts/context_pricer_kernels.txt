```python
# scripts/demos/valuation/norm_cdf_comparison.py

from __future__ import annotations

import matplotlib.pyplot as plt
import numba as nb
import numpy as np

from riskquant.valuation.kernels._math import (
    numba_norm_cdf,
    numba_norm_cdf_abrom_steg,
)


@nb.njit(cache=True, parallel=True)
def eval_exact(xs: np.ndarray) -> np.ndarray:
    n = xs.shape[0]
    out = np.empty(n, dtype=np.float64)

    for i in nb.prange(n):
        out[i] = numba_norm_cdf(xs[i])

    return out


@nb.njit(cache=True, parallel=True)
def eval_abrom_steg(xs: np.ndarray) -> np.ndarray:
    n = xs.shape[0]
    out = np.empty(n, dtype=np.float64)

    for i in nb.prange(n):
        out[i] = numba_norm_cdf_abrom_steg(xs[i])

    return out


def main() -> None:

    xs = np.linspace(
        -8.0,
        8.0,
        1_000_000,
        dtype=np.float64,
    )

    print("Compiling Numba kernels...")
    exact = eval_exact(xs)
    approx = eval_abrom_steg(xs)

    abs_error = np.abs(exact - approx)

    rel_error = abs_error / np.maximum(
        np.abs(exact),
        1e-16,
    )

    signed_error = exact - approx

    idx_max = np.argmax(abs_error)

    print()
    print("Accuracy Statistics")
    print("-" * 60)
    print(f"Max Abs Error : {abs_error.max():.12e}")
    print(f"Mean Abs Error: {abs_error.mean():.12e}")
    print(f"RMSE          : {np.sqrt(np.mean(abs_error**2)):.12e}")
    print(f"Max Rel Error : {rel_error.max():.12e}")
    print()
    print(f"x @ max error : {xs[idx_max]:.8f}")
    print(f"Exact         : {exact[idx_max]:.12e}")
    print(f"Approx        : {approx[idx_max]:.12e}")

    # ==========================================================
    # Combined Figure
    # ==========================================================

    fig, axes = plt.subplots(
        nrows=2,
        ncols=2,
        figsize=(16, 12),
    )

    # ----------------------------------------------------------
    # Panel 1: CDF Comparison
    # ----------------------------------------------------------

    axes[0, 0].plot(
        xs,
        exact,
        label="Exact (erf)",
    )

    axes[0, 0].plot(
        xs,
        approx,
        "--",
        label="Abramowitz-Stegun",
    )

    axes[0, 0].set_title("Normal CDF")
    axes[0, 0].set_xlabel("x")
    axes[0, 0].set_ylabel("CDF")
    axes[0, 0].grid(True)
    axes[0, 0].legend()

    # ----------------------------------------------------------
    # Panel 2: Absolute Error (Log Scale)
    # ----------------------------------------------------------

    axes[0, 1].semilogy(
        xs,
        abs_error,
    )

    axes[0, 1].set_title("Absolute Error")
    axes[0, 1].set_xlabel("x")
    axes[0, 1].set_ylabel("|Exact - Approx|")
    axes[0, 1].grid(True)

    # ----------------------------------------------------------
    # Panel 3: Signed Error
    # ----------------------------------------------------------

    axes[1, 0].plot(
        xs,
        signed_error,
    )

    axes[1, 0].set_title("Signed Error")
    axes[1, 0].set_xlabel("x")
    axes[1, 0].set_ylabel("Exact - Approx")
    axes[1, 0].grid(True)

    # ----------------------------------------------------------
    # Panel 4: Relative Error (Log Scale)
    # ----------------------------------------------------------

    axes[1, 1].semilogy(
        xs,
        rel_error,
    )

    axes[1, 1].set_title("Relative Error")
    axes[1, 1].set_xlabel("x")
    axes[1, 1].set_ylabel("Relative Error")
    axes[1, 1].grid(True)

    plt.suptitle(
        "Normal CDF Approximation Validation",
        fontsize=16,
    )

    plt.tight_layout()
    plt.show()


if __name__ == "__main__":
    main()
```

```python
# scripts/demos/valuation/norm_cdf_throughput.py

from __future__ import annotations

import time

import numba as nb

from riskquant.valuation.kernels._math import (
    numba_norm_cdf,
    numba_norm_cdf_abrom_steg,
)


@nb.njit(cache=True)
def bench_exact(n: int) -> float:

    acc = 0.0

    for i in range(n):
        x = ((i % 10000) - 5000) * 0.001
        acc += numba_norm_cdf(x)

    return acc


@nb.njit(cache=True)
def bench_abrom(n: int) -> float:

    acc = 0.0

    for i in range(n):
        x = ((i % 10000) - 5000) * 0.001
        acc += numba_norm_cdf_abrom_steg(x)

    return acc


def benchmark(name, fn, n):

    print(f"\n{name}")
    print("-" * 60)

    fn(1)

    t0 = time.perf_counter()
    result = fn(n)
    t1 = time.perf_counter()

    elapsed = t1 - t0

    print(f"Time:        {elapsed:.4f} sec")
    print(f"Evaluations: {n:,}")
    print(f"Throughput:  {n / elapsed / 1e9:.3f} Billion evals/sec")
    print(f"Checksum:    {result:.6f}")

    return elapsed


if __name__ == "__main__":
    n = 1_000_000_000

    exact = benchmark(
        "Exact ERF CDF",
        bench_exact,
        n,
    )

    approx = benchmark(
        "Abramowitz-Stegun",
        bench_abrom,
        n,
    )

    print("\nSummary")
    print("-" * 60)
    print(f"Speedup: {exact / approx:.3f}x")
    print("-" * 60)
    print("-" * 60)
```

```python
# scripts/demos/valuation/spot_kernel_throughput.py

import time

import numpy as np

from riskquant.constants import RNG_SEED
from riskquant.utils.printer import print_calling_file, print_success_message

# ============================================================
# IMPORT PRODUCTION KERNEL
# ============================================================
from riskquant.valuation.kernels.spot import _kernel_price_spot

# ============================================================
# BENCHMARK RUNNER
# ============================================================


def _rng():
    # Fixed seed for reproducibility
    return np.random.default_rng(seed=RNG_SEED)


def run_benchmark():
    print_calling_file()

    # --- 1. Scale Parameters ---
    m_trades = 100_000  # 100k Trades
    s_scenarios = 10_000  # 10k Scenarios
    p_prices = 1_000  # 1k Unique price factors
    f_fx_spots = 50  # 50 Unique FX pairs

    print("--- S-TIER SPOT KERNEL BENCHMARK ---")
    print(f"Trades:    {m_trades:,}")
    print(f"Scenarios: {s_scenarios:,}")
    print(f"Total PVs: {m_trades * s_scenarios:,} (1 Billion valuations)")
    print("-" * 36)

    # --- 2. Allocate C-Contiguous Memory ---
    print("Allocating engine memory...")
    out = np.zeros((m_trades, s_scenarios), dtype=np.float64)

    rng = _rng()

    # Generate random market data
    prices = rng.uniform(10.0, 100.0, (p_prices, s_scenarios)).astype(np.float64)
    fx_rates = rng.uniform(0.5, 1.5, (f_fx_spots, s_scenarios)).astype(np.float64)

    # Generate random portfolio semantics
    qty = rng.uniform(-1000, 1000, m_trades).astype(np.float64)
    price_uix = rng.integers(0, p_prices, m_trades).astype(np.int32)

    # Mix base currency (-1) and foreign currency trades
    fx_uix = rng.integers(-1, f_fx_spots, m_trades).astype(np.int32)

    # Validate memory layout
    assert out.flags.c_contiguous
    assert prices.flags.c_contiguous

    # Outer loop.
    for _ in range(3):
        # --- 3. JIT Warmup ---
        print("Warming up JIT compiler...")
        start_compile = time.perf_counter()

        # Call the imported kernel
        _kernel_price_spot(
            out[:10, :],  # Sliced outer dimension only to preserve C-contiguity
            prices,
            price_uix[:10],
            qty[:10],
            fx_rates,
            fx_uix[:10],
        )

        print(
            f"Compilation/Cache-load finished in {time.perf_counter() - start_compile:.4f}s"
        )
        print("-" * 36)

        # --- 4. The Hot Run ---
        print("Executing 1 Billion valuations...")
        runs = 10
        total_time = 0.0

        for r in range(runs):
            start_time = time.perf_counter()

            # Call the imported kernel
            _kernel_price_spot(out, prices, price_uix, qty, fx_rates, fx_uix)

            run_time = time.perf_counter() - start_time
            total_time += run_time
            print(f"  Run {r + 1}: {run_time:.4f} seconds")

        # --- 5. Throughput Math ---
        avg_time = total_time / runs
        total_valuations = m_trades * s_scenarios
        throughput = total_valuations / avg_time

        # Memory Bandwidth Calc (Writing 8 bytes per PV)
        bytes_written = total_valuations * 8
        bandwidth_gb_s = (bytes_written / avg_time) / (1024**3)

        print("-" * 36)
        print(f"Average Time: {avg_time:.4f} seconds")
        print(f"Throughput:   {throughput / 1_000_000_000:.2f} Billion PVs/sec")
        print(f"Write B/W:    {bandwidth_gb_s:.2f} GB/s")
        print("-" * 36)

    print_success_message()


if __name__ == "__main__":
    run_benchmark()
```

```python
# scripts/demos/valuation/vanilla_european_option_node_greeks.py

import matplotlib.pyplot as plt
import numpy as np

from riskquant.valuation.kernels.vanilla_european_option_node import (
    _kernel_price_vanilla_european_option_node,
)


def plot_option_greeks():
    print("--- Generating Option Greeks Visualization ---")

    # ------------------------------------------------------------------
    # Engine Setup (Vectorizing over Scenarios for X-Axis Grid)
    # ------------------------------------------------------------------
    m_trades = 2  # Trade 0: Call, Trade 1: Put
    s_scenarios = 1_000  # 1000 points on the X-axis (Forward Price)

    # Pre-allocate SoA output buffers for Base, Delta (Up/Down), and Vega (Up)
    out_base = np.zeros((m_trades, s_scenarios), dtype=np.float64)
    out_up = np.zeros((m_trades, s_scenarios), dtype=np.float64)
    out_dn = np.zeros((m_trades, s_scenarios), dtype=np.float64)
    out_vol_up = np.zeros((m_trades, s_scenarios), dtype=np.float64)

    # X-Axis Grid: Forward nodes from 50.0 to 150.0
    fwd_grid = np.linspace(50.0, 150.0, s_scenarios)

    # Finite Difference Bumps
    delta_underlying = 0.01  # 1 bp bump for Delta/Gamma
    delta_bump_vols = 0.0001  # 1 bp bump for Vega (0.01%)

    # Market Data Matrices
    fwd_nodes_base = fwd_grid.reshape(1, s_scenarios).astype(np.float64)
    fwd_nodes_up = (
        (fwd_grid + delta_underlying).reshape(1, s_scenarios).astype(np.float64)
    )
    fwd_nodes_dn = (
        (fwd_grid - delta_underlying).reshape(1, s_scenarios).astype(np.float64)
    )

    vol_nodes_base = np.full((1, s_scenarios), 0.20, dtype=np.float64)  # 20% flat vol
    vol_nodes_up = vol_nodes_base + delta_bump_vols

    # Dummy grids for optional legs (-1 bypasses these in the kernel)
    fx_rates = np.ones((1, 1), dtype=np.float64)  # No FX adjustment for simplicity
    df_grid = np.ones((1, 1, 1), dtype=np.float64)  # No discounting for simplicity

    # Time to Expiry: 1 Year
    sqrt_tte_grid = np.array([1.0], dtype=np.float64)
    half_tte_grid = np.array([0.5], dtype=np.float64)

    # ------------------------------------------------------------------
    # Trade Semantics
    # ------------------------------------------------------------------
    qty = np.array([1.0, 1.0], dtype=np.float64)
    strike = np.array([100.0, 100.0], dtype=np.float64)
    cp_mult = np.array([1, -1], dtype=np.int8)  # +1 Call, -1 Put

    uix_zeros = np.zeros(m_trades, dtype=np.int32)
    uix_neg1 = np.full(m_trades, -1, dtype=np.int32)

    # ------------------------------------------------------------------
    # Kernel Execution (Zero Allocations, C-Contiguous)
    # ------------------------------------------------------------------
    def run_pricer(out_buffer, fwd_data, vol_data):
        _kernel_price_vanilla_european_option_node(
            out_buffer,
            fwd_data,
            vol_data,
            fx_rates,
            df_grid,
            sqrt_tte_grid,
            half_tte_grid,
            qty,
            strike,
            cp_mult,
            uix_zeros,
            uix_zeros,
            uix_neg1,
            uix_neg1,
            uix_neg1,
            uix_neg1,
            uix_zeros,
            uix_zeros,
        )

    run_pricer(out_base, fwd_nodes_base, vol_nodes_base)
    run_pricer(out_up, fwd_nodes_up, vol_nodes_base)
    run_pricer(out_dn, fwd_nodes_dn, vol_nodes_base)
    run_pricer(out_vol_up, fwd_nodes_base, vol_nodes_up)

    # ------------------------------------------------------------------
    # Finite Difference Math
    # ------------------------------------------------------------------
    delta = (out_up - out_dn) / (2 * delta_underlying)
    gamma = (out_up - 2 * out_base + out_dn) / (delta_underlying**2)
    vega = (out_vol_up - out_base) / delta_bump_vols

    # ------------------------------------------------------------------
    # Plotting
    # ------------------------------------------------------------------
    fig, axes = plt.subplots(2, 2, figsize=(14, 10))
    fig.suptitle(
        "Vanilla European Option Node - Mathematical Stability Verification",
        fontsize=16,
    )

    labels = ["Call (K=100)", "Put (K=100)"]
    colors = ["#1f77b4", "#ff7f0e"]

    for i in range(m_trades):
        # Present Value
        axes[0, 0].plot(fwd_grid, out_base[i], label=labels[i], color=colors[i], lw=2)
        axes[0, 0].set_title("Present Value (PV)")
        axes[0, 0].set_ylabel("Price")
        axes[0, 0].grid(True, alpha=0.3)
        axes[0, 0].axvline(100.0, color="k", linestyle="--", alpha=0.5)

        # Delta
        axes[0, 1].plot(fwd_grid, delta[i], label=labels[i], color=colors[i], lw=2)
        axes[0, 1].set_title("Delta (∂V/∂S)")
        axes[0, 1].set_ylabel("Delta")
        axes[0, 1].grid(True, alpha=0.3)
        axes[0, 1].axvline(100.0, color="k", linestyle="--", alpha=0.5)

        # Gamma
        axes[1, 0].plot(fwd_grid, gamma[i], label=labels[i], color=colors[i], lw=2)
        axes[1, 0].set_title("Gamma (∂²V/∂S²)")
        axes[1, 0].set_xlabel("Underlying Forward Price")
        axes[1, 0].set_ylabel("Gamma")
        axes[1, 0].grid(True, alpha=0.3)
        axes[1, 0].axvline(100.0, color="k", linestyle="--", alpha=0.5)

        # Vega
        axes[1, 1].plot(fwd_grid, vega[i], label=labels[i], color=colors[i], lw=2)
        axes[1, 1].set_title("Vega (∂V/∂σ)")
        axes[1, 1].set_xlabel("Underlying Forward Price")
        axes[1, 1].set_ylabel("Vega (per 1bp Vol)")
        axes[1, 1].grid(True, alpha=0.3)
        axes[1, 1].axvline(100.0, color="k", linestyle="--", alpha=0.5)

    for ax in axes.flat:
        ax.legend()

    plt.tight_layout()
    plt.show()


if __name__ == "__main__":
    plot_option_greeks()
```

```python
from riskquant.constants import RNG_SEED
from riskquant.utils.printer import print_calling_file, print_success_message

# ============================================================
# IMPORT PRODUCTION KERNEL
# ============================================================
from riskquant.valuation.kernels.vanilla_european_option_node import (
    _kernel_price_vanilla_european_option_node,
)

# ============================================================
# BENCHMARK RUNNER
# ============================================================


def _rng():
    # Fixed seed for reproducibility
    return np.random.default_rng(seed=RNG_SEED)


def run_benchmark():
    print_calling_file()

    # --- 1. Scale Parameters ---
    m_trades = 100_000  # 100k Trades
    s_scenarios = 10_000  # 10k Scenarios

    # Market Data Dimensions
    p_prices = 1_000  # 1k Unique underlying forward nodes
    v_vols = 500  # 500 Unique volatility nodes
    f_fx_spots = 50  # 50 Unique FX pairs
    c_df_curves = 20  # 20 Unique discount curves
    t_tenors = 60  # 60 Standardized dates (for expiry, settle, and TTE)

    print("--- S-TIER VANILLA EUROPEAN OPTION BENCHMARK ---")
    print(f"Trades:    {m_trades:,}")
    print(f"Scenarios: {s_scenarios:,}")
    print(f"Total PVs: {m_trades * s_scenarios:,} (1 Billion valuations)")
    print("-" * 52)

    # --- 2. Allocate C-Contiguous Memory ---
    print("Allocating engine memory...")
    out = np.zeros((m_trades, s_scenarios), dtype=np.float64)

    rng = _rng()

    # Generate random market data
    forward_nodes = rng.uniform(10.0, 100.0, (p_prices, s_scenarios)).astype(np.float64)
    vol_nodes = rng.uniform(0.1, 0.5, (v_vols, s_scenarios)).astype(
        np.float64
    )  # 10% to 50% vol
    fx_rates = rng.uniform(0.5, 1.5, (f_fx_spots, s_scenarios)).astype(np.float64)
    df_grid = rng.uniform(0.8, 1.0, (c_df_curves, t_tenors, s_scenarios)).astype(
        np.float64
    )

    # Time to expiry grid (e.g., 0.1 to 10.0 years)
    tte_grid = np.linspace(0.1, 10.0, t_tenors).astype(np.float64)
    sqrt_tte_grid = np.sqrt(tte_grid)
    half_tte_grid = 0.5 * tte_grid

    # Generate random portfolio semantics
    qty = rng.uniform(-1000, 1000, m_trades).astype(np.float64)
    strike = rng.uniform(10.0, 100.0, m_trades).astype(np.float64)
    cp_mult = rng.choice([-1, 1], size=m_trades).astype(np.int8)  # Calls and Puts

    # Indices for lookups
    price_uix = rng.integers(0, p_prices, m_trades).astype(np.int32)
    vol_uix = rng.integers(0, v_vols, m_trades).astype(np.int32)
    expiry_date_uix = rng.integers(0, t_tenors, m_trades).astype(np.int32)
    settle_date_uix = rng.integers(0, t_tenors, m_trades).astype(np.int32)

    # Mix base currency (-1) and foreign currency trades
    fx_u_eff_uix = rng.integers(-1, f_fx_spots, m_trades).astype(np.int32)
    fx_k_eff_uix = rng.integers(-1, f_fx_spots, m_trades).astype(np.int32)

    # Mix un-discounted (-1) and discounted trades
    df_u_uix = rng.integers(-1, c_df_curves, m_trades).astype(np.int32)
    df_k_uix = rng.integers(-1, c_df_curves, m_trades).astype(np.int32)

    # Validate memory layout
    assert out.flags.c_contiguous
    assert forward_nodes.flags.c_contiguous
    assert vol_nodes.flags.c_contiguous
    assert fx_rates.flags.c_contiguous
    assert df_grid.flags.c_contiguous

    # Outer loop.
    for _ in range(3):
        # --- 3. JIT Warmup ---
        print("Warming up JIT compiler...")
        start_compile = time.perf_counter()

        # Call the imported kernel with sliced outer dimensions for compilation
        _kernel_price_vanilla_european_option_node(
            out[:10, :],
            forward_nodes,
            vol_nodes,
            fx_rates,
            df_grid,
            sqrt_tte_grid,
            half_tte_grid,
            qty[:10],
            strike[:10],
            cp_mult[:10],
            price_uix[:10],
            vol_uix[:10],
            fx_u_eff_uix[:10],
            df_u_uix[:10],
            fx_k_eff_uix[:10],
            df_k_uix[:10],
            expiry_date_uix[:10],
            settle_date_uix[:10],
        )

        print(
            f"Compilation/Cache-load finished in {time.perf_counter() - start_compile:.4f}s"
        )
        print("-" * 52)

        # --- 4. The Hot Run ---
        print("Executing 1 Billion valuations...")
        runs = 10
        total_time = 0.0

        for r in range(runs):
            start_time = time.perf_counter()

            # Call the imported kernel
            _kernel_price_vanilla_european_option_node(
                out,
                forward_nodes,
                vol_nodes,
                fx_rates,
                df_grid,
                sqrt_tte_grid,
                half_tte_grid,
                qty,
                strike,
                cp_mult,
                price_uix,
                vol_uix,
                fx_u_eff_uix,
                df_u_uix,
                fx_k_eff_uix,
                df_k_uix,
                expiry_date_uix,
                settle_date_uix,
            )

            run_time = time.perf_counter() - start_time
            total_time += run_time
            print(f"  Run {r + 1}: {run_time:.4f} seconds")

        # --- 5. Throughput Math ---
        avg_time = total_time / runs
        total_valuations = m_trades * s_scenarios
        throughput = total_valuations / avg_time

        # Memory Bandwidth Calc (Writing 8 bytes per PV)
        bytes_written = total_valuations * 8
        bandwidth_gb_s = (bytes_written / avg_time) / (1024**3)

        print("-" * 52)
        print(f"Average Time: {avg_time:.4f} seconds")
        print(f"Throughput:   {throughput / 1_000_000_000:.2f} Billion PVs/sec")
        print(f"Write B/W:    {bandwidth_gb_s:.2f} GB/s")
        print("-" * 52)

    print_success_message()


if __name__ == "__main__":
    run_benchmark()
```

```python
SURFACE_Y_AXIS_TYPE_STRIKE = 0


def test_spot_kernel():
    print("--- Testing Spot Kernel ---")
    m_trades, s_scenarios = 7, 1
    out = np.zeros((m_trades, s_scenarios), dtype=np.float64)

    # 1D Market Data
    prices = np.array([[100.0]], dtype=np.float64)  # p_idx = 0
    fx_rates = np.array([[0.5]], dtype=np.float64)  # f_idx = 0

    # Trade Semantics
    # Trade 0: Qty  0, Px 100, No FX  -> Expected    0.0
    # Trade 1: Qty  1, Px 100, No FX  -> Expected  100.0
    # Trade 2: Qty  1, Px 100, FX 0.5 -> Expected   50.0
    # Trade 3: Qty  2, Px 100, FX 0.5 -> Expected  100.0
    # Trade 4: Qty -1, Px 100, No FX  -> Expected -100.0
    # Trade 5: Qty -1, Px 100, FX 0.5 -> Expected  -50.0
    # Trade 6: Qty -2, Px 100, FX 0.5 -> Expected -100.0
    qty = np.array([0.0, 1.0, 1.0, 2.0, -1.0, -1.0, -2.0], dtype=np.float64)
    price_uix = np.array([0, 0, 0, 0, 0, 0, 0], dtype=np.int32)
    fx_uix = np.array([-1, -1, 0, 0, -1, 0, 0], dtype=np.int32)  # -1 skips FX branch

    _kernel_price_spot(out, prices, price_uix, qty, fx_rates, fx_uix)

    np.testing.assert_allclose(
        out[:, 0], [0.0, 100.0, 50.0, 100.0, -100.0, -50.0, -100.0]
    )
    print("Spot Kernel: PASS\n")
```

```python
# src/riskquant/valuation/kernels/_math.py
"""
Shared bare-metal math helpers for vectorized pricer kernels.

Tiny hot-path functions: `@nb.njit(inline='always')` collapses them into
the surrounding kernel at compile time, so the calling kernel sees fused
code with no call-frame overhead.

Anything that needs `prange` or more than a handful of ops should not
live here.
"""

from __future__ import annotations

import math

from riskquant._njit import inline_njit

# 1 / sqrt(2). Lifted so the kernel never recomputes it.
_INV_SQRT_2: float = 0.7071067811865476


@inline_njit
def numba_norm_cdf(x: float) -> float:
    """Standard normal CDF: Phi(x) = 0.5 * (1 + erf(x / sqrt(2)))."""
    return 0.5 * (1.0 + math.erf(x * _INV_SQRT_2))


@inline_njit
def numba_norm_cdf_abrom_steg(x: float) -> float:

    ax = abs(x)

    t = 1.0 / (1.0 + 0.2316419 * ax)

    poly = t * (
        0.319381530
        + t * (-0.356563782 + t * (1.781477937 + t * (-1.821255978 + t * 1.330274429)))
    )

    pdf = 0.39894228040143267794 * math.exp(-0.5 * ax * ax)

    cdf = 1.0 - pdf * poly

    if x < 0.0:
        return 1.0 - cdf

    return cdf


@inline_njit
def linear_interp_scalar(x: float, xs, ys) -> float:
    """
    Branchless 1D linear interpolation with flat extrapolation.

    Drops the `np.interp` call frame and Numba's bounds-check overhead in
    favour of an inlinable binary search + linear blend. LLVM fuses this
    into the caller's scenario loop.

    Both `xs` and `ys` must be 1-D C-contiguous; `xs` strictly ascending.
    For `x` outside `[xs[0], xs[-1]]` returns the corresponding edge value
    (flat extrapolation), matching `np.interp`'s default semantics and
    avoiding accidental log/division blow-ups on wide-strike moneyness.
    """
    n = xs.shape[0]
    if n == 1:
        return ys[0]

    # Edge clamps — flat extrapolation. Branches are predictable
    # (rare) and the compiler can lift them out of vectorised loops.
    if x <= xs[0]:
        return ys[0]
    if x >= xs[n - 1]:
        return ys[n - 1]

    # Binary search for the right-hand pillar. Loop trip count is
    # log2(n) and the compiler unrolls it for small n.
    lo = 0
    hi = n - 1
    while hi - lo > 1:
        mid = (lo + hi) >> 1
        if xs[mid] <= x:
            lo = mid
        else:
            hi = mid

    x0 = xs[lo]
    x1 = xs[hi]
    y0 = ys[lo]
    y1 = ys[hi]
    # x1 > x0 by strict-monotone invariant.
    w = (x - x0) / (x1 - x0)
    return y0 + w * (y1 - y0)
```

```python
# src/riskquant/valuation/kernels/spot.py
"""
Spot pricing kernel.

PV = qty * spot_price * fx_effective

`fx_spots_effective` rows are already the resolved transform (direct or
pre-inverted) — pricer kernels do no inversion math. Per-trade FX is one
gather + one multiply per scenario. The `fx_effective_uix < 0` sentinel
selects the "no FX" branch which is hoisted outside the scenario loop.
"""

from __future__ import annotations

import numba as nb  # type: ignore[import-untyped]

from riskquant._njit import parallel_njit

# ==============================================================================
# Numba layout-pinned aliases — force C-contiguous strides at compile time so
# the kernel emits straight-line SIMD loops with no dynamic stride checks.
# ==============================================================================

_F64_2D = nb.float64[:, ::1]
_F64_1D = nb.float64[::1]
_I32_1D = nb.int32[::1]


@parallel_njit(
    nb.void(
        _F64_2D,  # out
        _F64_2D,  # prices
        _I32_1D,  # price_uix
        _F64_1D,  # qty
        _F64_2D,  # fx_spots_effective
        _I32_1D,  # fx_spots_effective_uix
    ),
)
def _kernel_price_spot(
    out,
    prices,
    price_uix,
    qty,
    fx_spots_effective,
    fx_spots_effective_uix,
) -> None:
    """
    Fused spot pricing kernel: out[i, s] = prices[p, s] * qty[i] * fx[f, s].

    Zero allocations. Each row is a contiguous gather + scale + (gated)
    multiply streaming through C-contiguous memory.
    """
    m_trades, s_scenarios = out.shape

    for i in nb.prange(m_trades):
        p_idx = price_uix[i]
        f_idx = fx_spots_effective_uix[i]
        q = qty[i]

        p_row = prices[p_idx]
        out_row = out[i]

        if f_idx < 0:
            # No FX leg: straight price * qty.
            for s in range(s_scenarios):
                out_row[s] = p_row[s] * q
        else:
            # FX leg: three-way fused multiply.
            f_row = fx_spots_effective[f_idx]
            for s in range(s_scenarios):
                out_row[s] = p_row[s] * q * f_row[s]
```

```python
# src/riskquant/valuation/kernels/vanilla_european_option_curve.py
"""
Vanilla European option pricing kernel, curve variant.

Identical CIP-forward + Black-76 math to the node variant, with two
structural differences:

  - The underlying scalar comes from `price_curve_grid[curve_uix, delivery_uix, s]`.
  - Volatility is sourced from a 4D vol surface and requires per-scenario
    strike-axis interpolation along the pillar coordinate dictated by the
    surface's y-axis type.

FX rows are already EFFECTIVE — no inversion math; the strike-FX rate is
loaded once per scenario, used as a divisor for CIP forward synthesis and
as a multiplier for the PV translation back to settle ccy.

Layout contract (CRITICAL)
--------------------------
`vol_surfaces` MUST be `[surface, expiry, scenario, pillar]` (C-contig).
This guarantees that the pillars for a single trade-scenario tuple form
one contiguous run of memory: `vol_surfaces[v, exp, s]` is a 1-D view
suitable for the binary-search interp. The builder is responsible for
emitting this layout; the kernel relies on it for SIMD-friendly reads.

Branch hoisting
---------------
All sentinel checks (`fx_*_uix < 0`, `df_*_uix < 0`) and y-axis-type
selection are resolved once per trade, outside the scenario loop. Within
the scenario loop only the math executes; CPU branch predictors never
see a `y_type` ladder or a sentinel test per scenario.

Vol-surface-uix sentinel handling
---------------------------------
`v_surf_id = vol_surface_uix[i]` may be `INVALID_UIX (< 0)` when the
ingest compiler resolved no vol container for trade i. With
`boundscheck=False`, `vol_surface_strikes[v_surf_id]` and
`vol_surfaces[v_surf_id, exp_id]` on a negative id are undefined
behaviour — out-of-bounds memory.

This kernel hoists `has_vol = v_surf_id >= 0` once per trade, outside
the scenario loop. The sentinel path writes `out_row[:] = 0.0` and skips
the surface gather + Black-76 math entirely.
"""

from __future__ import annotations

import math

import numba as nb  # type: ignore[import-untyped]

from riskquant._njit import parallel_njit
from riskquant.market_data.surfaces.y_axes_types import (
    SURFACE_Y_AXIS_TYPE_LOG_MONEYNESS_FWD,
    SURFACE_Y_AXIS_TYPE_LOG_MONEYNESS_SPOT,
    SURFACE_Y_AXIS_TYPE_LOG_STRIKE,
    SURFACE_Y_AXIS_TYPE_MONEYNESS_FWD,
    SURFACE_Y_AXIS_TYPE_MONEYNESS_SPOT,
    SURFACE_Y_AXIS_TYPE_STRIKE,
)
from riskquant.valuation.kernels._math import (
    linear_interp_scalar,
    numba_norm_cdf_abrom_steg,
)

_F64_2D = nb.float64[:, ::1]
_F64_3D = nb.float64[:, :, ::1]
_F64_4D = nb.float64[:, :, :, ::1]
_F64_1D = nb.float64[::1]
_I32_1D = nb.int32[::1]
_I8_1D = nb.int8[::1]

_TTE_FLOOR: float = 1e-9
_VOL_STD_DEV_FLOOR: float = 1e-12

# Y-axis kind groupings as int8 keys. Resolved once per trade and packed
# into a 0/1/2/3 dispatch tag so the inner loop sees no branch ladder.
_Y_KIND_STRIKE: int = 0  # raw strike: y = k
_Y_KIND_LOG_STRIKE: int = 1  # y = log(k)
_Y_KIND_MONEYNESS: int = 2  # y = k / fwd
_Y_KIND_LOG_MONEYNESS: int = 3  # y = log(k / fwd)


@parallel_njit(
    nb.void(
        _F64_2D,  # out
        _F64_3D,  # price_curve_grid
        _F64_2D,  # vol_surface_strikes
        _F64_4D,  # vol_surfaces  [surface, expiry, scenario, pillar]
        _F64_2D,  # fx_spots_effective
        _F64_3D,  # df_grid
        _F64_1D,  # tte_grid
        _F64_1D,  # qty
        _F64_1D,  # strike
        _I8_1D,  # cp_mult
        _I32_1D,  # price_curve_uix
        _I32_1D,  # vol_surface_uix
        _I8_1D,  # y_axis_type_id
        _I32_1D,  # fx_u_eff_uix
        _I32_1D,  # df_u_uix
        _I32_1D,  # fx_k_eff_uix
        _I32_1D,  # df_k_uix
        _I32_1D,  # delivery_date_uix
        _I32_1D,  # expiry_date_uix
        _I32_1D,  # settle_date_uix
    ),
)
def _kernel_price_vanilla_european_option_curve(
    out,
    price_curve_grid,
    vol_surface_strikes,
    vol_surfaces,
    fx_spots_effective,
    df_grid,
    tte_grid,
    qty,
    strike,
    cp_mult,
    price_curve_uix,
    vol_surface_uix,
    y_axis_type_id,
    fx_u_eff_uix,
    df_u_uix,
    fx_k_eff_uix,
    df_k_uix,
    delivery_date_uix,
    expiry_date_uix,
    settle_date_uix,
) -> None:
    m_trades, s_scenarios = out.shape

    for i in nb.prange(m_trades):
        q = qty[i]
        k = strike[i]
        cp = cp_mult[i]

        p_curve_id = price_curve_uix[i]
        v_surf_id = vol_surface_uix[i]
        y_type = y_axis_type_id[i]

        fx_u_id = fx_u_eff_uix[i]
        df_u_id = df_u_uix[i]
        fx_k_id = fx_k_eff_uix[i]
        df_k_id = df_k_uix[i]

        deliv_id = delivery_date_uix[i]
        exp_id = expiry_date_uix[i]
        stl_id = settle_date_uix[i]

        out_row = out[i]

        # ---- Hoist vol-surface-uix sentinel: no surface -> zero PV ----
        # Resolving this BEFORE the surface gather is critical:
        # vol_surface_strikes[<0] and vol_surfaces[<0, ...] are UB under
        # boundscheck=False. Sentinel-trade rows write zeros and skip.
        has_vol = v_surf_id >= 0

        if not has_vol:
            for s in range(s_scenarios):
                out_row[s] = 0.0
            continue

        # ---- Trade-level scalars (hoist out of scenario loop) ----
        tte = tte_grid[exp_id]
        safe_tte = tte if tte > _TTE_FLOOR else _TTE_FLOOR
        sqrt_tte = math.sqrt(safe_tte)
        log_k = math.log(k)

        # ---- Pre-compute the 2D plane handles ----
        #   pillars      : 1-D, length=K_pillars  (shared across all scenarios)
        #   v_plane      : 2-D, [scenario, pillar] — contiguous run per scenario
        #   p_row        : 1-D, length=S          (price-curve row for delivery)
        pillars = vol_surface_strikes[v_surf_id]
        v_plane = vol_surfaces[v_surf_id, exp_id]
        p_row = price_curve_grid[p_curve_id, deliv_id]

        # ---- Hoist sentinel branches into per-trade booleans ----
        has_fx_u = fx_u_id >= 0
        has_df_u = df_u_id >= 0
        has_fx_k = fx_k_id >= 0
        has_df_k = df_k_id >= 0

        # ---- Hoist y-axis-type ladder into a 0/1/2/3 dispatch tag ----
        if y_type == SURFACE_Y_AXIS_TYPE_STRIKE:
            y_kind = _Y_KIND_STRIKE
        elif y_type == SURFACE_Y_AXIS_TYPE_LOG_STRIKE:
            y_kind = _Y_KIND_LOG_STRIKE
        elif y_type in (
            SURFACE_Y_AXIS_TYPE_MONEYNESS_FWD,
            SURFACE_Y_AXIS_TYPE_MONEYNESS_SPOT,
        ):
            y_kind = _Y_KIND_MONEYNESS
        elif y_type in (
            SURFACE_Y_AXIS_TYPE_LOG_MONEYNESS_FWD,
            SURFACE_Y_AXIS_TYPE_LOG_MONEYNESS_SPOT,
        ):
            y_kind = _Y_KIND_LOG_MONEYNESS
        else:
            # Unknown y-axis: degrade to strike. Producer validation
            # rejects this case before it reaches the kernel.
            y_kind = _Y_KIND_STRIKE

        for s in range(s_scenarios):
            # ---- CIP forward in strike currency ----
            fwd = p_row[s]
            if has_fx_u:
                fwd *= fx_spots_effective[fx_u_id, s]
            if has_df_u:
                fwd *= df_grid[df_u_id, exp_id, s]

            rate_k = 1.0
            if has_fx_k:
                rate_k = fx_spots_effective[fx_k_id, s]
                fwd /= rate_k
            if has_df_k:
                fwd /= df_grid[df_k_id, exp_id, s]

            # ---- Y-axis target for vol surface lookup ----
            # The branch tag is constant per trade; LLVM should hoist
            # this select out of the SIMD body via loop-unswitching.
            if y_kind == _Y_KIND_STRIKE:
                y_target = k
            elif y_kind == _Y_KIND_LOG_STRIKE:
                y_target = log_k
            elif y_kind == _Y_KIND_MONEYNESS:
                y_target = k / fwd
            else:  # _Y_KIND_LOG_MONEYNESS
                y_target = log_k - math.log(fwd)

            # ---- Strike-axis interpolation ----
            # TODO: QUESTION - DO WE NEED TO DO OTHER TYPES OF INTERPOLATION FOR OTHER
            # Y-AXIS KINDS? E.g. log-log for log-strike?
            # v_plane[s] is a C-contig 1-D run of pillars for THIS scenario.
            sigma = linear_interp_scalar(y_target, pillars, v_plane[s])

            # ---- Black-76 future value ----
            std_dev = sigma * sqrt_tte
            if std_dev < _VOL_STD_DEV_FLOOR:
                std_dev = _VOL_STD_DEV_FLOOR

            d1 = (math.log(fwd) - log_k + 0.5 * sigma * sigma * safe_tte) / std_dev
            d2 = d1 - std_dev

            cdf1 = numba_norm_cdf_abrom_steg(cp * d1)
            cdf2 = numba_norm_cdf_abrom_steg(cp * d2)

            fv = cp * (fwd * cdf1 - k * cdf2)

            # ---- PV discount in strike ccy, FX-translate to settle ccy ----
            if has_df_k:
                fv *= df_grid[df_k_id, stl_id, s]
            fv *= rate_k

            out_row[s] = q * fv
```

```python
    rate_k = fx_spots_effective[fx_k_eff_uix(i), s]   # k_ccy -> settle_ccy

    fwd_strike_ccy
        = forward_node[p, s]

        exp_id = expiry_date_uix[i]
        stl_id = settle_date_uix[i]

        out_row = out[i]

        # ---- Hoist vol-uix sentinel: no vol container -> zero PV ----
        has_vol = v_id >= 0

        if not has_vol:
            for s in range(s_scenarios):
                out_row[s] = 0.0
            continue

        sqrt_tte = sqrt_tte_grid[exp_id]
        half_tte = half_tte_grid[exp_id]
        log_k = math.log(k)

        # ------------------------------------------------------------------
        # Scenario rows
        # ------------------------------------------------------------------

        p_row = forward_nodes[p_id]
        v_row = vol_nodes[v_id]

        has_fx_u = fx_u_id >= 0
        has_df_u = df_u_id >= 0
        has_fx_k = fx_k_id >= 0
        has_df_k = df_k_id >= 0

        if has_fx_u:
            fx_u_row = fx_spots_effective[fx_u_id]

        if has_fx_k:
            fx_k_row = fx_spots_effective[fx_k_id]

        if has_df_u:
            df_u_exp_row = df_grid[df_u_id, exp_id]

        if has_df_k:
            df_k_exp_row = df_grid[df_k_id, exp_id]
            df_k_stl_row = df_grid[df_k_id, stl_id]

        # ------------------------------------------------------------------
        # Scenario loop
        # ------------------------------------------------------------------

        for s in range(s_scenarios):
            fwd = p_row[s]

            if has_fx_u:
                fwd *= fx_u_row[s]  # pyright: ignore[reportPossiblyUnboundVariable]

            if has_df_u:
                fwd *= df_u_exp_row[s]  # pyright: ignore[reportPossiblyUnboundVariable]

            rate_k = 1.0

            if has_fx_k:
                rate_k = fx_k_row[s]  # pyright: ignore[reportPossiblyUnboundVariable]
                fwd *= 1.0 / rate_k

            if has_df_k:
                fwd *= 1.0 / df_k_exp_row[s]  # pyright: ignore[reportPossiblyUnboundVariable]

            sigma = v_row[s]
            sigma_sq = sigma * sigma

            std_dev = sigma * sqrt_tte
            if std_dev < _VOL_STD_DEV_FLOOR:
                std_dev = _VOL_STD_DEV_FLOOR

            log_fwd = math.log(fwd)

            d1 = (log_fwd - log_k + sigma_sq * half_tte) / std_dev

            d2 = d1 - std_dev

            cdf1 = numba_norm_cdf_abrom_steg(cp * d1)
            cdf2 = numba_norm_cdf_abrom_steg(cp * d2)

            fv = cp * (fwd * cdf1 - k * cdf2)

            if has_df_k:
                fv *= df_k_stl_row[s]  # pyright: ignore[reportPossiblyUnboundVariable]

            fv *= rate_k

            out_row[s] = q * fv
```

```python
# src/riskquant/valuation/kernels/vanilla_forward_curve.py
"""
Vanilla forward pricing kernel, curve variant.

Identical math to the node variant; the only structural change is that the
underlying scalar comes from `price_curve_grid[curve_uix, delivery_uix, s]`
instead of `forward_nodes[uix, s]`.

FX rows are already EFFECTIVE in `MarketDataMatrix.fx_spots_effective`;
no inversion math survives into this kernel.
"""

from __future__ import annotations

import numba as nb  # type: ignore[import-untyped]

from riskquant._njit import parallel_njit

_F64_2D = nb.float64[:, ::1]
_F64_3D = nb.float64[:, :, ::1]
_F64_1D = nb.float64[::1]
_I32_1D = nb.int32[::1]


@parallel_njit(
    nb.void(
        _F64_2D,  # out
        _F64_3D,  # price_curve_grid
        _F64_2D,  # fx_spots_effective
        _F64_3D,  # df_grid
        _F64_1D,  # qty
        _F64_1D,  # strike
        _I32_1D,  # price_curve_uix
        _I32_1D,  # delivery_date_uix
        _I32_1D,  # fx_u_eff_uix
        _I32_1D,  # df_u_uix
        _I32_1D,  # fx_k_eff_uix
        _I32_1D,  # df_k_uix
        _I32_1D,  # settle_date_uix
    ),
)
def _kernel_price_vanilla_forward_curve(
    out,
    price_curve_grid,
    fx_spots_effective,
    df_grid,
    qty,
    strike,
    price_curve_uix,
    delivery_date_uix,
    fx_u_eff_uix,
    df_u_uix,
    fx_k_eff_uix,
    df_k_uix,
    settle_date_uix,
) -> None:
    m_trades, s_scenarios = out.shape

    for i in nb.prange(m_trades):
        q = qty[i]
        k = strike[i]
        p_curve_id = price_curve_uix[i]
        deliv_id = delivery_date_uix[i]
        fx_u_id = fx_u_eff_uix[i]
        df_u_id = df_u_uix[i]
        fx_k_id = fx_k_eff_uix[i]
        df_k_id = df_k_uix[i]
        stl_id = settle_date_uix[i]

        # Slice the (date, scenario) plane for this trade's underlying curve.
        p_row = price_curve_grid[p_curve_id, deliv_id]
        out_row = out[i]

        for s in range(s_scenarios):
            # UNDERLYING LEG
            leg_u = p_row[s]
            if fx_u_id >= 0:
                leg_u *= fx_spots_effective[fx_u_id, s]
            if df_u_id >= 0:
                leg_u *= df_grid[df_u_id, stl_id, s]

            # STRIKE LEG
            leg_k = k
            if fx_k_id >= 0:
                leg_k *= fx_spots_effective[fx_k_id, s]
            if df_k_id >= 0:
                leg_k *= df_grid[df_k_id, stl_id, s]

            out_row[s] = q * (leg_u - leg_k)
```

```python
"""
Vanilla forward pricing kernel, node variant.

    leg_u = forward_node * fx_u_effective * df_u(stl)
    leg_k = strike       * fx_k_effective * df_k(stl)
    PV    = qty * (leg_u - leg_k)

FX rows are already EFFECTIVE (direct or pre-inverted) inside
`MarketDataMatrix.fx_spots_effective` — pricer kernels do NO inversion math.
Each FX gather is one indexed read + one multiply per scenario.

Optional legs (FX / DF) are signalled by `INVALID_UIX (< 0)` in the
corresponding `_uix` field. These predicates are trade-level invariants;
Numba's LICM hoists them outside the inner scenario loop, so the inner loop
streams straight-line arithmetic over C-contiguous memory.
"""

    nb.void(
        _F64_2D,  # out
        _F64_2D,  # forward_nodes
        _F64_2D,  # fx_spots_effective
        _F64_3D,  # df_grid
        _F64_1D,  # qty
        _F64_1D,  # strike
        _I32_1D,  # price_uix
        _I32_1D,  # fx_u_eff_uix
        _I32_1D,  # df_u_uix
        _I32_1D,  # fx_k_eff_uix
        _I32_1D,  # df_k_uix
        _I32_1D,  # settle_date_uix
    ),
)

    forward_nodes,
    fx_spots_effective,
    df_grid,
    qty,
    strike,
```
