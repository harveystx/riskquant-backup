# src/riskquant/config/enums/discount_basis.py
"""
Discount basis taxonomy for IR discount curves.

A single currency carries multiple discount curves: one per OIS / IBOR
tenor / CSA collateral choice. The basis enum is the per-(currency,
basis) discriminator the ingest layer uses to resolve a trade's
discount-curve linkage when it specifies a non-default basis.

Sentinel:
  NOT_A_DISCOUNT_CURVE - explicitly marks a curve as non-discount. All
  curve_meta rows carry a basis; non-discount curves carry this value.
  The validator rejects any attempt to use a non-discount curve as a
  discount-curve linkage.

Standard convention: the firm-wide default discount basis is OIS. The
static-config validator enforces exactly one is_default_discount=True
curve per currency that has any discount curve, and that the default's
basis is OIS.
"""

from __future__ import annotations

from enum import IntEnum


class DiscountBasis(IntEnum):
    NOT_A_DISCOUNT_CURVE = 0

    OIS = 1

    IBOR_1M = 11
    IBOR_3M = 13
    IBOR_6M = 16

    CSA_USD = 30
    CSA_EUR = 31
    CSA_GBP = 32
    CSA_CHF = 33
    CSA_JPY = 34

    INTERNAL_FVA = 90


# Firm-wide standard. The default discount curve per currency must use this basis.
FIRM_DEFAULT_DISCOUNT_BASIS: DiscountBasis = DiscountBasis.OIS


VALID_DISCOUNT_BASES: frozenset[str] = frozenset(b.name for b in DiscountBasis)

DISCOUNT_BASES_EXCLUDING_SENTINEL: frozenset[str] = frozenset(
    b.name for b in DiscountBasis if b is not DiscountBasis.NOT_A_DISCOUNT_CURVE
)

"""
Roll method taxonomy for spot roll rules.

Replaces the prior two ad-hoc declarations: the string-keyed
`_VALID_SPOT_ROLL_METHODS` frozenset in validate.py and the
`_SpotRollMethod` constant class in rf_resolution.py.

A roll method specifies how a relative tenor token (e.g. [M+1], [Y+0],
[HY+0]) resolves to an absolute risk-factor name for a given as-of date.
The (XAxisKind, RollMethod) pair determines both the token grammar and
the absolute name format; allowed pairs are enumerated explicitly in
ALLOWED_X_AXIS_KIND_ROLL_METHOD_PAIRS so the validator and the
resolution dispatch cannot drift.

Absolute name formats by combination:
  (OFFSET_M,  CALENDAR_PERIOD)     -> "<curve>_YYYYMM"
  (OFFSET_M,  HALF_YEAR_BY_DATE)   -> "<curve>_YYYYMMDD"
  (OFFSET_Y,  CALENDAR_PERIOD)     -> "<curve>_YYYY"
  (OFFSET_Y,  ROLL_ON_MMDD)        -> "<curve>_YYYY"
  (OFFSET_HY, HALF_YEAR_BY_DATE)   -> "<curve>_YYYYMMDD"
  (OFFSET_HY, HALF_YEAR_BY_LABEL)  -> "<curve>_YYYY_FH" or "<curve>_YYYY_BH"
"""

from __future__ import annotations

from enum import IntEnum

from riskquant.config.enums.x_axis_kinds import XAxisKind


class RollMethod(IntEnum):
    CALENDAR_PERIOD = 0
    ROLL_ON_MMDD = 1
    HALF_YEAR_BY_DATE = 2
    HALF_YEAR_BY_LABEL = 3


VALID_ROLL_METHODS: frozenset[str] = frozenset(m.name for m in RollMethod)


ALLOWED_X_AXIS_KIND_ROLL_METHOD_PAIRS: frozenset[tuple[XAxisKind, RollMethod]] = frozenset(
    {
        (XAxisKind.OFFSET_M, RollMethod.CALENDAR_PERIOD),
        (XAxisKind.OFFSET_M, RollMethod.HALF_YEAR_BY_DATE),
        (XAxisKind.OFFSET_Y, RollMethod.CALENDAR_PERIOD),
        (XAxisKind.OFFSET_Y, RollMethod.ROLL_ON_MMDD),
        (XAxisKind.OFFSET_HY, RollMethod.HALF_YEAR_BY_DATE),
        (XAxisKind.OFFSET_HY, RollMethod.HALF_YEAR_BY_LABEL),
    }
)

# src/riskquant/config/enums/x_axis_kinds.py
"""
X-axis kind taxonomy for curves and surfaces.

Replaces the legacy (TenorType, AliasType, AliasKind) triple. A single
XAxisKind tells the resolution layer exactly how a curve / surface point
addresses its position on the time axis:

  LABEL      - fixed, as-of-independent string labels (e.g. [ON], [1W], [1M]).
               Used by IR conventional curves.
  OFFSET_M   - monthly offsets from a roll-determined spot month (e.g. [M+0],
               [M-1], [M+12]). Used by monthly commodity contracts.
  OFFSET_Y   - annual offsets from a roll-determined spot year (e.g. [Y+0],
               [Y-2]). Used by annual commodity contracts.
  OFFSET_HY  - half-year offsets from a roll-determined spot half (e.g.
               [HY+0], [HY+3]). Used by half-year power contracts.

The X-axis kind defines token grammar only. How an offset resolves to an
absolute date label (calendar period, roll-on-MMDD, half-year boundary)
is independent and lives on the spot roll rule via RollMethod. Allowed
(XAxisKind, RollMethod) pairs are enumerated explicitly in
ALLOWED_X_AXIS_KIND_ROLL_METHOD_PAIRS.
"""

from __future__ import annotations

import re
from enum import IntEnum
from re import Pattern


class XAxisKind(IntEnum):
    LABEL = 0
    OFFSET_M = 1
    OFFSET_Y = 2
    OFFSET_HY = 3


# Token grammars by kind. LABEL matches the standard IR conventions plus the
# generic [\d+(D|W|M|Y)] form. OFFSET kinds match a strictly signed integer
# inside brackets prefixed by the unit (M, Y, HY).
X_AXIS_TOKEN_REGEX: dict[XAxisKind, Pattern[str]] = {
    XAxisKind.LABEL: re.compile(r"^\[(?:ON|SN|TN|\d+D|\d+W|\d+M|\d+Y)\]$"),
    XAxisKind.OFFSET_M: re.compile(r"^\[M[+-]\d+\]$"),
    XAxisKind.OFFSET_Y: re.compile(r"^\[Y[+-]\d+\]$"),
    XAxisKind.OFFSET_HY: re.compile(r"^\[HY[+-]\d+\]$"),
}

X_AXIS_TOKEN_EXAMPLE: dict[XAxisKind, str] = {
    XAxisKind.LABEL: "[3M]",
    XAxisKind.OFFSET_M: "[M+1]",
    XAxisKind.OFFSET_Y: "[Y-2]",
    XAxisKind.OFFSET_HY: "[HY+0]",
}

VALID_X_AXIS_KINDS: frozenset[str] = frozenset(k.name for k in XAxisKind)

# OFFSET kinds require a spot roll rule. LABEL does not (and must not).
X_AXIS_KINDS_REQUIRING_ROLL: frozenset[str] = frozenset(
    k.name for k in (XAxisKind.OFFSET_M, XAxisKind.OFFSET_Y, XAxisKind.OFFSET_HY)
)

# src/riskquant/config/static_config/audit.py
"""
Tabular audit of a compiled RiskStaticConfig.

Imports Pandas deliberately — auditing is an edge-only diagnostic, not a
runtime path. Never imported by valuation, simlab, or any kernel.

Layout: one DataFrame per logical table the new 8-table schema exposes,
plus reverse-mapped derived structures (FX spot lookup, discount-curve
lookup, dense per-rf attribute arrays). Each section prints `head(n)`
with a configurable row cap.
"""

from __future__ import annotations

from dataclasses import asdict

import numpy as np
import pandas as pd

from riskquant.config.enums.container_topology import ContainerTopologyType
from riskquant.config.enums.discount_basis import DiscountBasis
from riskquant.config.enums.x_axis_kinds import XAxisKind
from riskquant.config.static_config.compile import (
    compile_risk_static_config_from_xlsx,
)
from riskquant.config.static_config.static_config import RiskStaticConfig


def _print_title(title: str) -> None:
    len_title = len(title) + 4
    ruler = "#" * len_title
    print(f"\n{ruler}")
    print(f"# {title} #")
    print(f"{ruler}")


def _print_subheader(title: str) -> None:
    ruler = "-" * (len(title) + 0)
    print("\n")
    print(f"{title}")
    print(f"{ruler}")


def _df_from_dataclass_values(values) -> pd.DataFrame:
    """Build a DataFrame from an iterable of frozen dataclasses."""
    rows = [asdict(v) for v in values]
    return pd.DataFrame(rows) if rows else pd.DataFrame()


# ============================================================
# Per-table renderers
# ============================================================


def _audit_ccy(rsc: RiskStaticConfig, n: int) -> None:
    _print_subheader("1. Currencies (ccy)")
    df = _df_from_dataclass_values(rsc.ccy_dict.values())
    if df.empty:
        print("(empty)")
        return
    print(df.head(n).to_string(index=False))
    print(f"  Total: {rsc.num_ccys}")


def _audit_holiday_calendars(rsc: RiskStaticConfig, n: int) -> None:
    _print_subheader("2. Holiday calendars (counts per calendar)")
    rows = [
        {
            "holiday_calendar_name": name,
            "n_holidays": int(cal.holiday_dates_ord.size),
            "first_ord": int(cal.holiday_dates_ord[0])
            if cal.holiday_dates_ord.size
            else None,
            "last_ord": int(cal.holiday_dates_ord[-1])
            if cal.holiday_dates_ord.size
            else None,
        }
        for name, cal in rsc.holiday_calendars_dict.items()
    ]
    df = pd.DataFrame(rows)
    if df.empty:
        print("(empty)")
        return
    print(df.head(n).to_string(index=False))
    print(f"  Total calendars: {len(rsc.holiday_calendars_dict)}")


def _audit_spot_roll_rules(rsc: RiskStaticConfig, n: int) -> None:
    _print_subheader("3. Spot roll rules")
    df = _df_from_dataclass_values(rsc.spot_roll_rule_dict.values())
    if df.empty:
        print("(empty)")
        return
    print(df.head(n).to_string(index=False))
    print(f"  Total: {len(rsc.spot_roll_rule_dict)}")


def _audit_node_meta(rsc: RiskStaticConfig, n: int) -> None:
    _print_subheader(
        "4. Node meta (scalar risk factors; FX + equity + credit single names)"
    )
    df = _df_from_dataclass_values(rsc.node_meta_dict.values())
    if df.empty:
        print("(empty)")
        return
    print(df.head(n).to_string(index=False))
    print(
        f"  Total nodes: {rsc.num_nodes}  (FX: {sum(1 for n_ in rsc.node_meta_dict.values() if n_.fx_base_ccy is not None)})"
    )


def _audit_curve_meta(rsc: RiskStaticConfig, n: int) -> None:
    _print_subheader("5. Curve meta (1D functions; discount + non-discount)")
    df = _df_from_dataclass_values(rsc.curve_meta_dict.values())
    if df.empty:
        print("(empty)")
        return
    print(df.head(n).to_string(index=False))
    is_disc = df["discount_basis"] != DiscountBasis.NOT_A_DISCOUNT_CURVE.name
    print(
        f"  Total curves: {rsc.num_curves}  "
        f"(discount: {int(is_disc.sum())}, "
        f"default-flagged: {int(df['is_default_discount_for_ccy'].sum())})"
    )


def _audit_curve_point_meta(rsc: RiskStaticConfig, n: int) -> None:
    _print_subheader("6. Curve point counts (per parent curve)")
    counts: dict[int, int] = {}
    for pt in rsc.curve_point_meta_dict.values():
        counts[pt.curve_id] = counts.get(pt.curve_id, 0) + 1
    rows = []
    for cid, cm in rsc.curve_meta_dict.items():

"""
Single entry-point for building a RiskStaticConfig from source data.

Pipeline:
    dict_of_dfs  ->  validate  ->  compile (typed scalars)  ->  RiskStaticConfig

The 7 (+ ccy + spot_roll_rule + holiday_calendars = 10) required sheets are:
    ccy, holiday_calendars, spot_roll_rule,
    node_meta, curve_meta, curve_point_meta,
    surface_meta, surface_point_meta.

All system-defined taxonomy (asset class, unit, shift, interp/extrap,
x/y axis kind, roll method, discount basis) lives in
riskquant.config.enums / riskquant.market_data and is resolved by the
RiskStaticConfig constructor.

Nothing in this module is imported by the valuation or simlab packages.
"""

from __future__ import annotations

import math
from pathlib import Path

import numpy as np
import pandas as pd

from riskquant.config.static_config.loader import (
    load_risk_static_config_xlsx_into_dict,
)
from riskquant.config.static_config.schema_objects import (
    Ccy,
    CurveMeta,
    CurvePointMeta,
    HolidayCalendar,
    NodeMeta,
    SpotRollRule,
    SurfaceMeta,
    SurfacePointMeta,
)
from riskquant.config.static_config.static_config import RiskStaticConfig
from riskquant.config.static_config.validate import validate_static_config_dfs
from riskquant.constants import DATE_EPOCH

# ---------------------------------------------------------------------------
# Cell extraction primitives
# ---------------------------------------------------------------------------
# The only place raw DataFrame cell values become typed Python scalars.
# Deliberately thin — no business logic, just safe type coercion.


def _iter(df: pd.DataFrame) -> list[dict]:
    """
    Materialize a DataFrame into plain dicts with None for missing values.

    Per-column conversion avoids .where(..., None), which would force
    every column to object dtype.
    """
    if df is None or df.empty:
        return []

    cols: dict[str, list] = {}
    for col in df.columns:
        lst = df[col].tolist()
        cols[col] = [
            None if (v is pd.NA or (isinstance(v, float) and v != v)) else v
            for v in lst
        ]
    keys = list(cols.keys())
    return [
        dict(zip(keys, row, strict=True)) for row in zip(*cols.values(), strict=True)
    ]


def _str(v: object, *, upper: bool = False) -> str:
    if v is None:
        raise ValueError("Expected a non-null string cell, got None.")
    s = str(v).strip()
    return s.upper() if upper else s


def _opt_str(v: object, *, upper: bool = False) -> str | None:
    if v is None:
        return None
    s = str(v).strip()
    if s == "":
        return None
    return s.upper() if upper else s


def _int(v: object) -> int:
    if v is None:
        raise ValueError("Expected a non-null int cell, got None.")
    return int(v)  # type: ignore[arg-type]


def _opt_int(v: object) -> int | None:
    if v is None:
        return None
    if isinstance(v, float) and math.isnan(v):
        return None
    return int(v)  # type: ignore[arg-type]


def _bool(v: object) -> bool:
    if v is None or (isinstance(v, float) and math.isnan(v)):
        return False
    if isinstance(v, bool):
        return v
    if isinstance(v, (int, np.integer)):
        return bool(int(v))
    s = str(v).strip().upper()
    if s in ("TRUE", "1", "Y", "YES"):
        return True
    if s in ("FALSE", "0", "N", "NO", ""):
        return False
    raise ValueError(f"Cannot interpret {v!r} as boolean.")


# ---------------------------------------------------------------------------
# Per-table compilers
# ---------------------------------------------------------------------------


def _compile_ccy(df: pd.DataFrame) -> dict[str, Ccy]:
    out: dict[str, Ccy] = {}
    for r in _iter(df):
        name = _str(r["ccy_name"], upper=True)
        out[name] = Ccy(ccy_name=name, ccy_id=_int(r["ccy_id"]))
    return out


def _compile_holiday_calendars(df: pd.DataFrame) -> dict[str, HolidayCalendar]:
    out: dict[str, HolidayCalendar] = {}
    if df is None or df.empty:
        return out

    epoch_ts = pd.Timestamp(DATE_EPOCH)
    dt_series = pd.to_datetime(df["holiday_date"], format="%Y-%m-%d", errors="raise")
    work = df.copy()
    work["_ord"] = (dt_series - epoch_ts).dt.days.astype(np.int32)

    for cal_name, group in work.groupby("holiday_calendar_name"):
        out[str(cal_name)] = HolidayCalendar(
            holiday_calendar_name=str(cal_name),
            holiday_dates_ord=np.sort(group["_ord"].unique().astype(np.int32)),
        )
    return out


def _compile_spot_roll_rule(df: pd.DataFrame) -> dict[str, SpotRollRule]:
    out: dict[str, SpotRollRule] = {}
    for r in _iter(df):
        name = _str(r["rule_name"], upper=True)
        out[name] = SpotRollRule(
            rule_name=name,
            roll_method=_str(r["roll_method"], upper=True),
            offset_in_tenor_units=_int(r["offset_in_tenor_units"]),
            roll_month=_opt_int(r.get("roll_month")),
            roll_day=_opt_int(r.get("roll_day")),
            fh_bh_mmdd=_opt_str(r.get("fh_bh_mmdd")),
            post_roll_shift=_int(r["post_roll_shift"]),
        )
    return out


def _compile_node_meta(df: pd.DataFrame) -> dict[int, NodeMeta]:
    out: dict[int, NodeMeta] = {}
    for r in _iter(df):
        node_id = _int(r["node_id"])
        out[node_id] = NodeMeta(

            node_name=_str(r["node_name"], upper=True),

            unit_type=_str(r["unit_type"], upper=True),

            shift_type=_str(r["shift_type"], upper=True),
            fx_base_ccy=_opt_str(r.get("fx_base_ccy"), upper=True),
        )
    return out


def _compile_curve_meta(df: pd.DataFrame) -> dict[int, CurveMeta]:
    out: dict[int, CurveMeta] = {}
    for r in _iter(df):
        curve_id = _int(r["curve_id"])
        out[curve_id] = CurveMeta(
            curve_id=curve_id,
            curve_name=_str(r["curve_name"], upper=True),
            quote_ccy=_str(r["quote_ccy"], upper=True),
            asset_class=_str(r["asset_class"], upper=True),
            asset_subclass=_str(r["asset_subclass"], upper=True),
            unit_type=_str(r["unit_type"], upper=True),
            unit_scale=_str(r["unit_scale"], upper=True),

            x_interp_method=_str(r["x_interp_method"], upper=True),
            x_extrap_method=_str(r["x_extrap_method"], upper=True),
            spot_roll_rule_name=_opt_str(r.get("spot_roll_rule_name"), upper=True),

    return out


def _compile_curve_point_meta(

def compile_risk_static_config_from_dict_of_dfs(
    dict_of_dfs: dict[str, pd.DataFrame],

    """
    Validate and compile a dict of DataFrames into a RiskStaticConfig.

    Parameters
    ----------
    dict_of_dfs
        Sheet-name -> DataFrame produced by the loader. All 8 required
        sheets must be present and well-formed; see REQUIRED_SHEETS in
        validate.py.

    Returns
    -------
    RiskStaticConfig
        Fully compiled, Pandas-free runtime config with every derived
        index already built.
    """
    validate_static_config_dfs(dict_of_dfs)

    ccy_dict = _compile_ccy(dict_of_dfs["ccy"])
    holiday_calendars_dict = _compile_holiday_calendars(
        dict_of_dfs["holiday_calendars"]
    )
    spot_roll_rule_dict = _compile_spot_roll_rule(dict_of_dfs["spot_roll_rule"])
    node_meta_dict = _compile_node_meta(dict_of_dfs["node_meta"])
    curve_meta_dict = _compile_curve_meta(dict_of_dfs["curve_meta"])
    curve_point_meta_dict = _compile_curve_point_meta(
        dict_of_dfs["curve_point_meta"], curve_meta_dict
    )
    surface_meta_dict = _compile_surface_meta(dict_of_dfs["surface_meta"])
    surface_point_meta_dict = _compile_surface_point_meta(
        dict_of_dfs["surface_point_meta"], surface_meta_dict
    )

    return RiskStaticConfig(
        ccy_dict=ccy_dict,
        holiday_calendars_dict=holiday_calendars_dict,
        spot_roll_rule_dict=spot_roll_rule_dict,
        node_meta_dict=node_meta_dict,
        curve_meta_dict=curve_meta_dict,
        curve_point_meta_dict=curve_point_meta_dict,
        surface_meta_dict=surface_meta_dict,
        surface_point_meta_dict=surface_point_meta_dict,
    )

    """

    Read every required sheet from an .xlsx workbook into a dict of DataFrames.

    Parameters
    ----------
    path
        Path to the Excel workbook.

    Returns
    -------
    dict[str, pd.DataFrame]
        Sheet-name -> DataFrame for every name in `REQUIRED_SHEETS`.

    Raises
    ------
    FileNotFoundError
        If the workbook does not exist.
    ValueError
        If a required sheet is missing from the workbook.
    """
    p = Path(path)
    if not p.is_file():
        raise FileNotFoundError(f"Workbook not found: {p}")

    excel = pd.ExcelFile(p)
    available: set[str] = {str(name) for name in excel.sheet_names}
    missing = REQUIRED_SHEETS - available
    if missing:
        raise ValueError(
            f"Workbook {p.name} is missing required sheet(s): {sorted(missing)}.\n"
            f"Sheets present: {sorted(available)}."
        )

    return {name: excel.parse(name) for name in REQUIRED_SHEETS}

"""
Date-sensitive resolution of relative risk-factor names to absolute names.

Examples for as-of=2026-05-23:
    BRENT[M+0]          ->   BRENT_202607        (NEXT_MONTH, CALENDAR_PERIOD)
    HYDRO[Y+0]          ->   HYDRO_2026          (ANNUAL_ROLL_APRIL_BGN, ROLL_ON_MMDD)
    ERCOT[HY+0]         ->   ERCOT_2026_BH       (HALF_YEAR_LABEL, HALF_YEAR_BY_LABEL)

The public entry point is `build_risk_factor_name_mapping_for_as_of_date`
which returns a frozen, bidirectionally indexed `RiskFactorAbsToRelNameIdMap`.

Dispatch is a registry keyed by (XAxisKind, RollMethod) — never a
string-equality ladder. The validator guarantees only allowed pairs
ever reach the resolver.

Nodes do not resolve: their absolute name equals their declared name
(node_name). Surface point names follow the same model as curve points
with an additional bracketed y label.
"""

from __future__ import annotations

from bisect import bisect_left
from collections.abc import Callable
from dataclasses import dataclass, field
from datetime import date
from typing import overload

from riskquant.config.enums.roll_methods import RollMethod
from riskquant.config.enums.x_axis_kinds import XAxisKind
from riskquant.config.static_config.schema_objects import (
    SpotRollRule,
)

    add_months_to_date_and_return_yyyymm01,

# Half-year helpers
# ============================================================


def _require_fh_bh_mmdd(ctx: _ResolverContext) -> str:
    """Half-year roll methods require fh_bh_mmdd to be populated."""
    if not ctx.rule.fh_bh_mmdd:
        raise ValueError(
            f"Spot roll rule {ctx.rule.rule_name!r} requires fh_bh_mmdd "
            f"but it is absent.\n"
            f"relative={ctx.relative_name!r}, container={ctx.container_name!r}"
        )
    return ctx.rule.fh_bh_mmdd


def _parse_fh_bh_mmdd(fh_bh_mmdd: str) -> tuple[int, int, int, int]:
    """Parse 'MMDD|MMDD' into (fh_month, fh_day, bh_month, bh_day)."""
    parts = fh_bh_mmdd.split("|")
    if len(parts) != 2 or len(parts[0]) != 4 or len(parts[1]) != 4:
        raise ValueError(f"fh_bh_mmdd must be 'MMDD|MMDD', got {fh_bh_mmdd!r}")
    fh_mm, fh_dd = int(parts[0][:2]), int(parts[0][2:])
    bh_mm, bh_dd = int(parts[1][:2]), int(parts[1][2:])
    return fh_mm, fh_dd, bh_mm, bh_dd


def _half_year_boundary_by_date(
    *,
    as_of_date: date,
    fh_bh_mmdd: str,
    token_offset: int,
    relative_name: str,
    container_name: str,
) -> date:
    """
    Resolve a half-year boundary date.

    Builds a sorted list of FH/BH boundary dates spanning a window around
    as_of_date, then picks the boundary at (bisect_left index + token_offset).
    """
    fh_mm, fh_dd, bh_mm, bh_dd = _parse_fh_bh_mmdd(fh_bh_mmdd)
    as_of_year = as_of_date.year
    look = max(3, abs(token_offset) + 1)

    boundaries: list[date] = []
    for y in range(as_of_year - look, as_of_year + look + 1):
        boundaries.append(date(y, fh_mm, fh_dd))
        boundaries.append(date(y, bh_mm, bh_dd))
    if fh_mm > bh_mm or (fh_mm == bh_mm and fh_dd > bh_dd):
        boundaries.sort()

    idx_target = bisect_left(boundaries, as_of_date) + token_offset
    if idx_target < 0 or idx_target >= len(boundaries):
        raise ValueError(
            f"HALF_YEAR_BY_DATE offset {token_offset} out of range.\n"
            f"relative={relative_name!r}, container={container_name!r}"
        )
    return boundaries[idx_target]


def _half_year_label_suffix(
    *,
    as_of_date: date,
    fh_bh_mmdd: str,
    token_offset: int,
) -> str:
    """Resolve a half-year label suffix (YYYY_FH / YYYY_BH)."""
    fh_mm, fh_dd, _, _ = _parse_fh_bh_mmdd(fh_bh_mmdd)
    current_half_idx = 0 if as_of_date <= date(as_of_date.year, fh_mm, fh_dd) else 1
    total = (as_of_date.year * 2) + current_half_idx + token_offset
    target_year = total // 2
    target_label = "FH" if (total % 2 == 0) else "BH"
    return f"{target_year:04d}_{target_label}"

# ============================================================


@dataclass(frozen=True, slots=True, kw_only=True)
class RiskFactorAbsToRelNameIdMap:
    """
    Bidirectionally indexed mapping between absolute and relative
    risk-factor names for one as-of date.

    Conceptual model
    ----------------
    Relative names (from config):
        BRENT[M+0]                 -- curve point
        VOL_BRENT[M+0][STRIKE=50]  -- surface point
        USDEUR                     -- node (same as absolute)

    Absolute names (as-of-resolved):
        BRENT_202607
        VOL_BRENT_202607[STRIKE=50]
        USDEUR

    Lookup contract
    ---------------
    O(1) lookup for both absolute and relative forms via the unified
    `lookup` dict. Hot-path callers should use `get_rf_id(name)` and
    treat it as the only lookup function they need.
    """

    as_of_date: date
    abs_to_rf_id: dict[str, int]
    abs_to_rel_name: dict[str, str]

    _rel_to_abs: dict[str, str] = field(init=False, repr=False)
    _rel_to_id: dict[str, int] = field(init=False, repr=False)
    _lookup: dict[str, int] = field(init=False, repr=False)

    def __post_init__(self) -> None:
        abs_keys = set(self.abs_to_rf_id)
        rel_keys = set(self.abs_to_rel_name)
        if abs_keys != rel_keys:
            sym_diff = sorted(abs_keys ^ rel_keys)[:25]
            raise ValueError(
                f"abs_to_rf_id and abs_to_rel_name keys disagree: {sym_diff}"
            )

        rel_to_abs: dict[str, str] = {}
        rel_to_id: dict[str, int] = {}
        for abs_name, rel_name in self.abs_to_rel_name.items():
            rf_id = self.abs_to_rf_id[abs_name]
            if rel_name in rel_to_abs and rel_to_abs[rel_name] != abs_name:
                raise ValueError(
                    f"Duplicate relative name {rel_name!r} maps to both "
                    f"{rel_to_abs[rel_name]!r} and {abs_name!r}"
                )
            rel_to_abs[rel_name] = abs_name
            rel_to_id[rel_name] = rf_id

        # Defensive disjointness check.
        # The static_config validator already enforces (a) global name uniqueness
        # across all five namespaces and (b) that the only legitimate overlap is
        # nodes, where abs_name == rel_name == node_name AND abs_id == rel_id.
        # This loop only fires if those invariants are broken by a future change
        # to validate.py or schema_objects.py — at which point the loud error here
        # is exactly what we want.
        for name in abs_keys & set(rel_to_id):
            if self.abs_to_rf_id[name] != rel_to_id[name]:
                raise ValueError(
                    f"Name {name!r} appears as both absolute and relative "
                    f"with different rf_ids "
                    f"(abs={self.abs_to_rf_id[name]}, rel={rel_to_id[name]})"
                )

        lookup = {**self.abs_to_rf_id, **rel_to_id}

        object.__setattr__(self, "_rel_to_abs", rel_to_abs)
        object.__setattr__(self, "_rel_to_id", rel_to_id)
        object.__setattr__(self, "_lookup", lookup)

    # ----- Single hot-path lookup -----

    def get_rf_id(self, name: str) -> int:
        """Resolve either an absolute or relative name -> rf_id."""
        return self._lookup[name]

    # ----- Explicit lookups (clarity / validation paths) -----

    def get_rf_id_for_abs_name(self, abs_name: str) -> int:
        return self.abs_to_rf_id[abs_name]

    @overload
    def get_rf_id_for_rel_name(self, rel_name: str) -> int: ...
    @overload
    def get_rf_id_for_rel_name(self, rel_name: list[str]) -> list[int]: ...

    def get_rf_id_for_rel_name(self, rel_name: str | list[str]) -> int | list[int]:
        m = self._rel_to_id
        if isinstance(rel_name, str):
            return m[rel_name]
        return [m[r] for r in rel_name]

    def get_abs_name_for_rel_name(self, rel_name: str | list[str]) -> str | list[str]:
        m = self._rel_to_abs
        if isinstance(rel_name, str):
            return m[rel_name]
        return [m[r] for r in rel_name]

    def get_rel_name_for_abs_name(self, abs_name: str) -> str:
        return self.abs_to_rel_name[abs_name]

    # ----- Properties -----

    @property
    def num_risk_factors(self) -> int:
        return len(self.abs_to_rf_id)

    @property
    def rel_name_to_rf_id(self) -> dict[str, int]:
        return self._rel_to_id

    @property
    def rel_name_to_abs_name(self) -> dict[str, str]:
        return self._rel_to_abs

    @property
    def lookup(self) -> dict[str, int]:
        return self._lookup


# ============================================================
# Resolution engine
# ============================================================


def build_risk_factor_name_mapping_for_as_of_date(
    *,
    risk_static_config: RiskStaticConfig,
    as_of_date: date | str,
) -> RiskFactorAbsToRelNameIdMap:
    """
    Resolve every relative risk-factor name to its absolute name for one as_of_date.

    Three sources are merged:

      1. Nodes — abs_name == rel_name == node_name. No resolution.
      2. Curve points — abs_name = f"{curve_name}_{resolved_suffix}".
         LABEL curves: resolved_suffix is the bare label (e.g. "ON", "3M");
         OFFSET curves: resolved_suffix is computed via the registry.
      3. Surface points — abs_name = f"{surface_name}_{resolved_suffix}[{y_label}]".
         Same resolution rules as curves for the x axis.

    Parameters
    ----------
    risk_static_config
        Compiled static configuration.
    as_of_date
        Valuation date used to resolve OFFSET tokens. ISO string accepted.

    Returns
    -------
    RiskFactorAbsToRelNameIdMap
        Bidirectional abs<->rel<->rf_id mapping.
    """
    if as_of_date is None:
        raise ValueError("as_of_date is required.")
    if isinstance(as_of_date, str):
        as_of_date = date.fromisoformat(as_of_date)

    abs_to_rf_id: dict[str, int] = {}
    abs_to_rel_name: dict[str, str] = {}

    # ---- Nodes ----
    for node in risk_static_config.node_meta_dict.values():
        abs_to_rf_id[node.node_name] = int(node.node_id)
        abs_to_rel_name[node.node_name] = node.node_name

    # ---- Curve points ----
    for pt in risk_static_config.curve_point_meta_dict.values():
        cm = risk_static_config.curve_meta_dict[pt.curve_id]
        rel_name = f"{cm.curve_name}{pt.tenor_token}"
        suffix = _resolve_curve_or_surface_suffix(
            container_name=cm.curve_name,
            container_kind="curve",
            x_axis_kind_name=cm.x_axis_kind,
            spot_roll_rule_name=cm.spot_roll_rule_name,
            spot_roll_rule_dict=risk_static_config.spot_roll_rule_dict,
            tenor_token=pt.tenor_token,
            relative_name=rel_name,
            as_of_date=as_of_date,
        )
        abs_name = f"{cm.curve_name}_{suffix}"
        abs_to_rf_id[abs_name] = int(pt.rf_id)
        abs_to_rel_name[abs_name] = rel_name

    # ---- Surface points ----
    for pt in risk_static_config.surface_point_meta_dict.values():
        sm = risk_static_config.surface_meta_dict[pt.surface_id]
        rel_name = f"{sm.surface_name}{pt.x_tenor_token}[{pt.y_label}]"
        suffix = _resolve_curve_or_surface_suffix(
            container_name=sm.surface_name,
            container_kind="surface",
            x_axis_kind_name=sm.x_axis_kind,
            spot_roll_rule_name=sm.spot_roll_rule_name,
            spot_roll_rule_dict=risk_static_config.spot_roll_rule_dict,
            tenor_token=pt.x_tenor_token,
            relative_name=rel_name,
            as_of_date=as_of_date,
        )
        abs_name = f"{sm.surface_name}_{suffix}[{pt.y_label}]"
        abs_to_rf_id[abs_name] = int(pt.rf_id)
        abs_to_rel_name[abs_name] = rel_name

    return RiskFactorAbsToRelNameIdMap(
        as_of_date=as_of_date,
        abs_to_rf_id=abs_to_rf_id,
        abs_to_rel_name=abs_to_rel_name,
    )


def _resolve_curve_or_surface_suffix(
    *,
    container_name: str,
    container_kind: str,
    x_axis_kind_name: str,
    spot_roll_rule_name: str | None,
    spot_roll_rule_dict: dict[str, SpotRollRule],
    tenor_token: str,
    relative_name: str,
    as_of_date: date,
) -> str:
    """Resolve a single curve/surface point's absolute-name suffix."""
    x_kind = XAxisKind[x_axis_kind_name]

    if x_kind == XAxisKind.LABEL:
        label, signed_offset = parse_bracket_token(tenor_token)
        if signed_offset is not None:
            raise ValueError(
                f"LABEL {container_kind} {container_name!r} has offset token "
                f"{tenor_token!r}; LABEL containers accept label-only tokens."
            )
        return label

    if spot_roll_rule_name is None:
        raise ValueError(
            f"OFFSET {container_kind} {container_name!r} has no spot_roll_rule_name."
        )
    rule = spot_roll_rule_dict.get(spot_roll_rule_name)
    if rule is None:
        raise ValueError(
            f"Unknown spot_roll_rule_name={spot_roll_rule_name!r} for "
            f"{container_kind} {container_name!r}."
        )

    _, token_offset = parse_bracket_token(tenor_token)
    if token_offset is None:
        raise ValueError(
            f"OFFSET {container_kind} {container_name!r} requires a signed "
            f"integer offset token, got {tenor_token!r}."
        )

    method = RollMethod[rule.roll_method]
    resolver = _OFFSET_RESOLVERS.get((x_kind, method))
    if resolver is None:
        raise ValueError(
            f"Unsupported (x_axis_kind, roll_method) pair: "
            f"({x_kind.name}, {method.name})."
        )

    return resolver(
        _ResolverContext(
            container_name=container_name,
            relative_name=relative_name,
            rule=rule,
            token_offset=token_offset,
            as_of_date=as_of_date,
        )
    )

"""
Immutable value objects for the god-tier static config pipeline.

One dataclass per input table. Every field is non-null in its own table:
no nullability cascades, no "must be null when this is null" rules. The
three first-class object kinds (Node, Curve, Surface) each get their own
metadata table and child point table.

  NodeMeta              - scalar observable. A node IS its own risk factor.
                          FX rate nodes additionally carry fx_base_ccy.
  CurveMeta             - 1D function metadata. Optional discount-basis tag
                          marks IR discount curves.
  CurvePointMeta        - one row per point on a curve. The point's
                          tenor_token is parsed using the parent's
                          x_axis_kind.
  SurfaceMeta           - 2D function metadata (expiry x strike-axis).
  SurfacePointMeta      - one row per point on a surface; x_tenor_token
                          is parsed using the surface's x_axis_kind,
                          y_label using its y_axis_kind.
  SpotRollRule          - one named rule. PK is the rule name.
  HolidayCalendar       - one calendar with sorted int32 holiday ordinals.
  Ccy                   - one currency with its dense int8 id.

All system-defined taxonomies (asset class, unit, shift, interp/extrap,
x/y axis kind, roll method, discount basis) live in riskquant.config.enums
and riskquant.market_data — never inside these dataclasses as derived state.
"""

from dataclasses import dataclass

import numpy.typing as npt

# ============================================================

    PK is `rule_name`. The rule's `roll_method` determines which

    against the method:

      CALENDAR_PERIOD   -> uses offset_in_tenor_units
      ROLL_ON_MMDD      -> uses roll_month, roll_day, post_roll_shift
      HALF_YEAR_BY_DATE -> uses fh_bh_mmdd

    roll_method: str
    offset_in_tenor_units: int
    roll_month: int | None
    roll_day: int | None
    fh_bh_mmdd: str | None
    post_roll_shift: int

# NODE  (scalar; IS its own risk factor)
# ============================================================


@dataclass(frozen=True, slots=True, kw_only=True)

    """
    A scalar observable. Every node is both a container (for ingest
    classification) and a risk factor (for market-data lookup). The
    node_id IS the rf_id; there is no child point table.

    fx_base_ccy contract:
      asset_class == "FX"  <=> fx_base_ccy is not None
      For FX nodes, node_name == fx_base_ccy + quote_ccy is enforced.

    node_id: int
    node_name: str

    quote_ccy: str
    asset_class: str
    asset_subclass: str

"""
Runtime container for the compiled god-tier static config.

Strictly native Python structures plus typed NumPy arrays — no Pandas.
All derived lookups are computed eagerly in `__post_init__`. The object
is fully valid and fully indexed the moment construction completes.

Conceptual model
----------------
Three disjoint container topology types:

  Node     - scalar observable; IS its own risk factor.
             (FX rates, equity spot, single-name credit spread proxy, etc.)
  Curve    - 1D function; addressable risk-factor children.
             (Commodity forward curves, IR discount curves, etc.)
  Surface  - 2D function; addressable risk-factor children.
             (Vol surfaces in (expiry, strike) or equivalent.)

Name namespaces are kept globally unique: every node_name, curve_name,
surface_name, curve-point relative name, and surface-point relative name
is distinct from every other in the same RiskStaticConfig. The validator
enforces this at compile time so the ingest layer can classify a string
deterministically.

Two lookup tables matter at ingest:

  name_to_rf_id        : node_name | curve_point_rel_name | surface_point_rel_name
                         -> rf_id (int32). Resolves directly to a runtime
                         risk factor.
  container_name_to_id : curve_name | surface_name -> container_id (int32).
                         Resolves to a parent container that requires
                         interpolation at runtime.

A trade's `underlying` resolves into exactly one of these two tables.
Both-match and neither-match are validation errors.

FX resolution
-------------
fx_spot_lookup is the typed table the ingest FX resolver consumes:

  (under_ccy_id: int8, settle_ccy_id: int8) -> rf_id: int32

It is built from NodeMeta rows tagged asset_class="FX" using their
declared (fx_base_ccy, quote_ccy). No string heuristics, no "[ON]"
guessing.

Discount resolution
-------------------
Two layers:

  ccy_id_to_default_discount_curve_id  : per-ccy default (OIS).
  discount_curve_lookup                : (ccy_id, basis_id) -> curve_id
                                          for explicit (basis) overrides.

The compiler enforces every active currency has at least one discount
curve and exactly one default.

def _build_currency_state(ccy_dict: dict[str, Ccy]) -> CurrencyState:
    """Stage 1: Resolve the currency universe."""
    if not ccy_dict:
        raise ValueError("ccy_dict is empty.")

    ccy_name_to_id: dict[str, np.int8] = {
        name: np.int8(c.ccy_id) for name, c in ccy_dict.items()
    }
    max_ccy_id = max((int(c.ccy_id) for c in ccy_dict.values()), default=0)

    ccy_id_to_name: list[str] = [""] * (max_ccy_id + 1)
    for c in ccy_dict.values():
        ccy_id_to_name[int(c.ccy_id)] = c.ccy_name

    return CurrencyState(
        ccy_name_to_id=ccy_name_to_id,
        ccy_id_to_name=tuple(ccy_id_to_name),
        max_ccy_id=max_ccy_id,
    )


def _build_name_lookup_state(
    state: CurrencyState,
    node_meta_dict: dict[int, NodeMeta],
    curve_meta_dict: dict[int, CurveMeta],
    curve_point_meta_dict: dict[int, CurvePointMeta],
    surface_meta_dict: dict[int, SurfaceMeta],
    surface_point_meta_dict: dict[int, SurfacePointMeta],
) -> NameLookupState:
    """Stage 2: Build globally-unique name -> id mappings."""
    rel_name_to_rf_id: dict[str, np.int32] = {}
    container_name_to_id: dict[str, np.int32] = {}

    # Nodes: name is the risk-factor name
    for n in node_meta_dict.values():
        rel_name_to_rf_id[n.node_name] = np.int32(n.node_id)

    # Curve points: "{curve_name}{tenor_token}"
    for pt in curve_point_meta_dict.values():
        rel_name = f"{pt.curve_name}{pt.tenor_token}"
        if rel_name in rel_name_to_rf_id:
            raise ValueError(
                f"Duplicate relative risk-factor name {rel_name!r} "
                f"(rf_id {pt.rf_id} collides with {rel_name_to_rf_id[rel_name]})."
            )
        rel_name_to_rf_id[rel_name] = np.int32(pt.rf_id)

    # Surface points: "{surface_name}{x_tenor_token}[{y_label}]"
    for pt in surface_point_meta_dict.values():
        rel_name = f"{pt.surface_name}{pt.x_tenor_token}[{pt.y_label}]"
        if rel_name in rel_name_to_rf_id:
            raise ValueError(
                f"Duplicate relative risk-factor name {rel_name!r} "
                f"(rf_id {pt.rf_id} collides with {rel_name_to_rf_id[rel_name]})."
            )
        rel_name_to_rf_id[rel_name] = np.int32(pt.rf_id)

    # Containers: curves and surfaces
    for cm in curve_meta_dict.values():
        container_name_to_id[cm.curve_name] = np.int32(cm.curve_id)

    for sm in surface_meta_dict.values():
        if sm.surface_name in container_name_to_id:
            raise ValueError(
                f"Container name {sm.surface_name!r} declared as both "
                "curve and surface."
            )
        container_name_to_id[sm.surface_name] = np.int32(sm.surface_id)

    # Cross-namespace disjointness
    overlap = set(rel_name_to_rf_id) & set(container_name_to_id)
    if overlap:
        raise ValueError(
            f"Names appear in BOTH the risk-factor and container "
            f"namespaces: {sorted(overlap)}"
        )

    return NameLookupState(
        ccy_name_to_id=state.ccy_name_to_id,
        ccy_id_to_name=state.ccy_id_to_name,
        max_ccy_id=state.max_ccy_id,
        rel_name_to_rf_id=rel_name_to_rf_id,
        container_name_to_id=container_name_to_id,
    )


def _build_rf_state(
    state: NameLookupState,
    node_meta_dict: dict[int, NodeMeta],
    curve_meta_dict: dict[int, CurveMeta],
    curve_point_meta_dict: dict[int, CurvePointMeta],
    surface_meta_dict: dict[int, SurfaceMeta],
    surface_point_meta_dict: dict[int, SurfacePointMeta],
) -> RfState:
    """Stage 3: Build per-risk-factor dense attribute arrays."""
    rf_ids: list[int] = (
        [n.node_id for n in node_meta_dict.values()]
        + [p.rf_id for p in curve_point_meta_dict.values()]
        + [p.rf_id for p in surface_point_meta_dict.values()]
    )
    max_rf_id = max(rf_ids) if rf_ids else 0
    n = max_rf_id + 1

    parent_topology_type = np.full(n, -1, dtype=np.int8)
    parent_id = np.full(n, -1, dtype=np.int32)
    ccy_id = np.full(n, -1, dtype=np.int8)
    shift_id = np.full(n, -1, dtype=np.int8)
    unit_scale = np.full(n, np.nan, dtype=np.float64)

    ccy_name_to_id = state.ccy_name_to_id

    for node in node_meta_dict.values():
        i = node.node_id
        parent_topology_type[i] = np.int8(ContainerTopologyType.NODE.value)
        parent_id[i] = i
        ccy_id[i] = ccy_name_to_id[node.quote_ccy]
        shift_id[i] = _resolve_shift_id(node.shift_type)
        unit_scale[i] = _resolve_unit_scale_factor(node.unit_scale)

    for pt in curve_point_meta_dict.values():
        cm = curve_meta_dict[pt.curve_id]
        i = pt.rf_id
        parent_topology_type[i] = np.int8(ContainerTopologyType.CURVE.value)
        parent_id[i] = pt.curve_id
        ccy_id[i] = ccy_name_to_id[cm.quote_ccy]
        shift_id[i] = _resolve_shift_id(cm.shift_type)
        unit_scale[i] = _resolve_unit_scale_factor(cm.unit_scale)

    for pt in surface_point_meta_dict.values():
        sm = surface_meta_dict[pt.surface_id]
        i = pt.rf_id
        parent_topology_type[i] = np.int8(ContainerTopologyType.SURFACE.value)
        parent_id[i] = pt.surface_id
        ccy_id[i] = ccy_name_to_id[sm.quote_ccy]
        shift_id[i] = _resolve_shift_id(sm.shift_type)
        unit_scale[i] = _resolve_unit_scale_factor(sm.unit_scale)

    return RfState(
        ccy_name_to_id=state.ccy_name_to_id,
        ccy_id_to_name=state.ccy_id_to_name,
        max_ccy_id=state.max_ccy_id,
        rel_name_to_rf_id=state.rel_name_to_rf_id,
        container_name_to_id=state.container_name_to_id,
        max_rf_id=max_rf_id,
        rf_id_to_parent_topology_type=parent_topology_type,
        rf_id_to_parent_id=parent_id,
        rf_id_to_ccy_id=ccy_id,
        rf_id_to_shift_type_id=shift_id,
        rf_id_to_unit_scale_factor=unit_scale,
    )

def _build_curve_tenor_index(
    curve_meta_dict: dict[int, CurveMeta],
    curve_point_meta_dict: dict[int, CurvePointMeta],
) -> dict[int, CurveTenorIndex]:
    """
    Build one CurveTenorIndex per curve.

    OFFSET curves: sorted offset axis + parallel rf_id array, ready for
    `np.searchsorted(offsets, requested_offset)` at runtime.
    LABEL curves : direct dict label -> rf_id.
    """
    by_curve_offsets: dict[int, list[int]] = {}
    by_curve_ids: dict[int, list[int]] = {}
    by_curve_labels: dict[int, dict[str, int]] = {}

    for pt in curve_point_meta_dict.values():
        cm = curve_meta_dict[pt.curve_id]
        x_kind = XAxisKind[cm.x_axis_kind]

        if x_kind == XAxisKind.LABEL:
            unit, offset = parse_bracket_token(pt.tenor_token)
            if offset is not None:
                raise ValueError(
                    f"LABEL curve {cm.curve_name!r} contains an offset token "
                    f"{pt.tenor_token!r}; LABEL curves accept label-only tokens."
                )
            by_curve_labels.setdefault(pt.curve_id, {})[unit] = int(pt.rf_id)
            continue

        _, offset = parse_bracket_token(pt.tenor_token)
        if offset is None:
            raise ValueError(
                f"OFFSET curve {cm.curve_name!r} contains a label token "
                f"{pt.tenor_token!r}; OFFSET curves require a signed integer offset."
            )
        by_curve_offsets.setdefault(pt.curve_id, []).append(int(offset))
        by_curve_ids.setdefault(pt.curve_id, []).append(int(pt.rf_id))

    out: dict[int, CurveTenorIndex] = {}

    for curve_id, offsets_list in by_curve_offsets.items():
        offsets_arr = np.asarray(offsets_list, dtype=np.int64)
        ids_arr = np.asarray(by_curve_ids[curve_id], dtype=np.int32)
        order = np.argsort(offsets_arr, kind="mergesort")
        out[curve_id] = CurveTenorIndex(
            sorted_offsets=offsets_arr[order],
            rf_ids_by_offset=ids_arr[order],
            label_to_rf_id=by_curve_labels.get(curve_id, {}),
        )

    for curve_id, lbl_map in by_curve_labels.items():
        if curve_id in out:
            continue
        out[curve_id] = CurveTenorIndex(
            sorted_offsets=np.empty(0, dtype=np.int64),
            rf_ids_by_offset=np.empty(0, dtype=np.int32),
            label_to_rf_id=lbl_map,

                sorted_offsets=np.empty(0, dtype=np.int64),
                rf_ids_by_offset=np.empty(0, dtype=np.int32),
                label_to_rf_id={},
            )

    return out

def _build_fx_state(
    state: CurveTenorState,
    node_meta_dict: dict[int, NodeMeta],
) -> FxState:
    """Stage 6: Build FX spot lookup table."""
    out: dict[tuple[np.int8, np.int8], np.int32] = {}
    ccy_name_to_id = state.ccy_name_to_id

    for node in node_meta_dict.values():
        if node.fx_base_ccy is None:
            continue
        base_id = ccy_name_to_id[node.fx_base_ccy]
        quote_id = ccy_name_to_id[node.quote_ccy]
        key = (base_id, quote_id)
        if key in out:
            raise ValueError(
                f"Duplicate FX spot pair "
                f"(base={node.fx_base_ccy}, quote={node.quote_ccy}): "
                f"node {node.node_name} collides with another FX node."
            )
        out[key] = np.int32(node.node_id)

    return FxState(
        ccy_name_to_id=state.ccy_name_to_id,
        ccy_id_to_name=state.ccy_id_to_name,
        max_ccy_id=state.max_ccy_id,
        rel_name_to_rf_id=state.rel_name_to_rf_id,
        container_name_to_id=state.container_name_to_id,

        rf_id_to_ccy_id=state.rf_id_to_ccy_id,
        rf_id_to_shift_type_id=state.rf_id_to_shift_type_id,
        rf_id_to_unit_scale_factor=state.rf_id_to_unit_scale_factor,
        max_container_id=state.max_container_id,
        container_id_to_topology_type=state.container_id_to_topology_type,

        container_id_to_extrap_x_id=state.container_id_to_extrap_x_id,
        container_id_to_interp_y_id=state.container_id_to_interp_y_id,
        container_id_to_extrap_y_id=state.container_id_to_extrap_y_id,
        container_id_to_ccy_id=state.container_id_to_ccy_id,
        container_id_to_spot_roll_rule_name=state.container_id_to_spot_roll_rule_name,
        container_id_to_x_axis_kind=state.container_id_to_x_axis_kind,
        curve_id_to_tenor_index=state.curve_id_to_tenor_index,

    )


def _build_fx_matrix_state(
    state: FxState,
) -> FxMatrixState:
    """Stage 6b: Build dense FX spot matrix."""
    n = state.max_ccy_id + 1

def _build_discount_state(
    state: FxMatrixState,
    curve_meta_dict: dict[int, CurveMeta],
) -> DiscountState:
    """Stage 7: Build discount-curve lookup tables.  Terminal stage."""
    max_ccy_id = state.max_ccy_id
    ccy_name_to_id = state.ccy_name_to_id

    default_per_ccy = np.full(max_ccy_id + 1, _NO_DISCOUNT_CURVE_ID, dtype=np.int32)
    explicit: dict[tuple[np.int8, np.int8], np.int32] = {}

    for cm in curve_meta_dict.values():
        if cm.discount_basis == DiscountBasis.NOT_A_DISCOUNT_CURVE.name:
            continue
        ccy_id = ccy_name_to_id[cm.quote_ccy]
        basis_id = np.int8(int(DiscountBasis[cm.discount_basis]))
        key = (ccy_id, basis_id)
        if key in explicit:
            raise ValueError(
                f"Duplicate discount curve for "
                f"(ccy={cm.quote_ccy}, basis={cm.discount_basis}): "
                f"curve {cm.curve_name}."
            )
        explicit[key] = np.int32(cm.curve_id)

        if cm.is_default_discount_for_ccy:
            if default_per_ccy[int(ccy_id)] != _NO_DISCOUNT_CURVE_ID:
                raise ValueError(
                    f"More than one default discount curve flagged for "
                    f"ccy={cm.quote_ccy}."
                )
            default_per_ccy[int(ccy_id)] = np.int32(cm.curve_id)

    return DiscountState(
        ccy_name_to_id=state.ccy_name_to_id,
        ccy_id_to_name=state.ccy_id_to_name,
        max_ccy_id=state.max_ccy_id,
        rel_name_to_rf_id=state.rel_name_to_rf_id,
        container_name_to_id=state.container_name_to_id,
        max_rf_id=state.max_rf_id,
        rf_id_to_parent_topology_type=state.rf_id_to_parent_topology_type,
        rf_id_to_parent_id=state.rf_id_to_parent_id,
        rf_id_to_ccy_id=state.rf_id_to_ccy_id,
        rf_id_to_shift_type_id=state.rf_id_to_shift_type_id,
        rf_id_to_unit_scale_factor=state.rf_id_to_unit_scale_factor,
        max_container_id=state.max_container_id,
        container_id_to_topology_type=state.container_id_to_topology_type,
        container_id_to_interp_x_id=state.container_id_to_interp_x_id,
        container_id_to_extrap_x_id=state.container_id_to_extrap_x_id,
        container_id_to_interp_y_id=state.container_id_to_interp_y_id,
        container_id_to_extrap_y_id=state.container_id_to_extrap_y_id,
        container_id_to_ccy_id=state.container_id_to_ccy_id,
        container_id_to_spot_roll_rule_name=state.container_id_to_spot_roll_rule_name,
        container_id_to_x_axis_kind=state.container_id_to_x_axis_kind,
        curve_id_to_tenor_index=state.curve_id_to_tenor_index,
        fx_spot_lookup=state.fx_spot_lookup,
        fx_spot_matrix=state.fx_spot_matrix,
        ccy_id_to_default_discount_curve_id=default_per_ccy,
        discount_curve_lookup=explicit,
    )

@dataclass(frozen=True, slots=True)
class RiskStaticConfig:
    """
    Compiled, Pandas-free static configuration.

    Constructed by ``compile_risk_static_config_from_dict_of_dfs``.
    All derived fields are computed eagerly in ``__post_init__`` via a
    type-state pipeline and frozen before the constructor returns.

    Public access is through delegating properties that read from the
    internal frozen ``_state``.
    """

    # ------------------------------------------------------------------ #
    # Init fields — supplied by compile.py                               #
    # ------------------------------------------------------------------ #

    ccy_dict: dict[str, Ccy] = field(repr=False)
    holiday_calendars_dict: dict[str, HolidayCalendar] = field(repr=False)
    spot_roll_rule_dict: dict[str, SpotRollRule] = field(repr=False)

    node_meta_dict: dict[int, NodeMeta] = field(repr=False)
    curve_meta_dict: dict[int, CurveMeta] = field(repr=False)
    curve_point_meta_dict: dict[int, CurvePointMeta] = field(repr=False)
    surface_meta_dict: dict[int, SurfaceMeta] = field(repr=False)
    surface_point_meta_dict: dict[int, SurfacePointMeta] = field(repr=False)

    # ------------------------------------------------------------------ #
    # Internal frozen state (populated in __post_init__)                 #
    # ------------------------------------------------------------------ #

    _state: DiscountState = field(init=False, repr=False)

    # ------------------------------------------------------------------ #
    # Construction                                                       #
    # ------------------------------------------------------------------ #

    def __post_init__(self) -> None:
        """
        Run the type-state pipeline and freeze the result.

        The pipeline order is encoded in the type signatures:
        each stage accepts exactly the state produced by the previous
        stage.  It is structurally impossible for a future maintainer
        to read a field before it has been computed.
        """
        if not self.ccy_dict:
            raise ValueError("ccy_dict is empty.")
        if not (self.node_meta_dict or self.curve_meta_dict or self.surface_meta_dict):
            raise ValueError(
                "RiskStaticConfig has no nodes, no curves, and no surfaces."
            )

        # Pipeline: Currency -> NameLookup -> Rf -> Container -> Tenor -> Fx -> Discount
        s1 = _build_currency_state(self.ccy_dict)
        s2 = _build_name_lookup_state(
            s1,
            self.node_meta_dict,
            self.curve_meta_dict,
            self.curve_point_meta_dict,
            self.surface_meta_dict,
            self.surface_point_meta_dict,
        )
        s3 = _build_rf_state(
            s2,
            self.node_meta_dict,
            self.curve_meta_dict,
            self.curve_point_meta_dict,
            self.surface_meta_dict,
            self.surface_point_meta_dict,
        )
        s4 = _build_container_state(
            s3,
            self.curve_meta_dict,
            self.surface_meta_dict,
        )
        s5 = _build_curve_tenor_state(
            s4,
            self.curve_meta_dict,
            self.curve_point_meta_dict,
        )
        s6 = _build_fx_state(
            s5,
            self.node_meta_dict,
        )
        s7 = _build_fx_matrix_state(
            s6,
        )
        s8 = _build_discount_state(
            s7,
            self.curve_meta_dict,
        )

        frozen = _freeze_state(s8)
        object.__setattr__(self, "_state", frozen)

    # ------------------------------------------------------------------ #
    # Delegating properties — public API                                 #
    # ------------------------------------------------------------------ #

    @property

        return self._state.ccy_id_to_name

             * every active currency has at least one discount curve

def _check_required_sheets(dict_of_dfs: dict[str, pd.DataFrame]) -> None:
    missing = REQUIRED_SHEETS - dict_of_dfs.keys()
    if missing:
        raise ConfigurationError(
            f"Missing required sheet(s): {sorted(missing)}.\n"
            f"Required: {sorted(REQUIRED_SHEETS)}"
        )


def _check_columns(df: pd.DataFrame, cols: frozenset[str], sheet_name: str) -> None:
    missing = cols - set(df.columns)
    if missing:
        raise ConfigurationError(
            f"Sheet {sheet_name!r} missing required columns: {sorted(missing)}.\n"
            f"Found: {sorted(df.columns)}"
        )


def _check_unique(df: pd.DataFrame, col: str, sheet_name: str) -> None:
    if not df[col].is_unique:
        dupes = df[col][df[col].duplicated()].unique().tolist()
        raise ConfigurationError(
            f"Uniqueness violation: {sheet_name}.{col} must be unique. "
            f"Duplicates: {dupes}"
        )


def _check_composite_unique(
    df: pd.DataFrame,
    cols: tuple[str, ...],
    sheet_name: str,
) -> None:
    dupes_mask = df.duplicated(subset=list(cols), keep=False)
    if dupes_mask.any():
        offending = df.loc[dupes_mask, list(cols)]
        raise ConfigurationError(
            f"Composite uniqueness violation: "
            f"{sheet_name}({', '.join(cols)}) must be unique.\n"
            f"Duplicates:\n{offending.to_string(index=False)}"
        )


def _check_enum_col(
    df: pd.DataFrame,
    col: str,
    valid_set: frozenset[str],
    sheet_name: str,
    enum_name: str,
    *,
    nullable: bool,
) -> None:
    if col not in df.columns:
        return
    series = df[col]
    populated = series.dropna()
    if not nullable and len(populated) != len(series):
        n_null = int(series.isna().sum())
        raise ConfigurationError(
            f"Enum column {sheet_name}.{col} ({enum_name}) "
            f"contains {n_null} null value(s) but is non-nullable."
        )
    if populated.empty:
        return
    invalid = set(populated.astype(str)) - valid_set
    if invalid:
        raise ConfigurationError(
            f"Enum violation in {sheet_name}.{col}: values not in {enum_name}: "
            f"{sorted(invalid)}.\nValid: {sorted(valid_set)}"
        )


def _check_fk(
    child_df: pd.DataFrame,
    child_col: str,
    parent_df: pd.DataFrame,
    parent_col: str,
    *,
    child_sheet: str,
    parent_sheet: str,
) -> None:
    child_vals = child_df[child_col].dropna().to_numpy()
    parent_vals = parent_df[parent_col].dropna().to_numpy()
    missing = np.setdiff1d(child_vals, parent_vals, assume_unique=False)
    if missing.size > 0:
        raise ConfigurationError(
            f"FK violation: {child_sheet}.{child_col} references values absent "
            f"from {parent_sheet}.{parent_col}.\nMissing: {missing.tolist()[:25]}"
        )


def _check_no_blank_strings(
    df: pd.DataFrame, col: str, sheet_name: str, *, nullable: bool
) -> None:
    if col not in df.columns:
        return
    series = df[col]
    if nullable:
        series = series.dropna()
    if series.empty:
        return
    is_blank = series.astype(str).str.strip() == ""
    if is_blank.any():
        idx = series.index[is_blank].tolist()[:10]
        raise ConfigurationError(
            f"Blank string violation: {sheet_name}.{col} has empty values at "
            f"rows: {idx}"
        )

def _validate_uniqueness(dict_of_dfs: dict[str, pd.DataFrame]) -> None:
    single_pk: dict[str, tuple[str, ...]] = {
        "ccy": ("ccy_name", "ccy_id"),
        "spot_roll_rule": ("rule_name",),
        "node_meta": ("node_id", "node_name"),

    for sheet_name, cols in single_pk.items():

    _check_composite_unique(
        dict_of_dfs["curve_point_meta"],
        ("curve_id", "tenor_token"),
        "curve_point_meta",
    )
    _check_composite_unique(
        dict_of_dfs["surface_point_meta"],

        "surface_point_meta",
    )

def _validate_fks(dict_of_dfs: dict[str, pd.DataFrame]) -> None:
    ccy_df = dict_of_dfs["ccy"]
    node_df = dict_of_dfs["node_meta"]
    curve_df = dict_of_dfs["curve_meta"]
    curve_pt_df = dict_of_dfs["curve_point_meta"]
    surf_df = dict_of_dfs["surface_meta"]
    surf_pt_df = dict_of_dfs["surface_point_meta"]
    srr_df = dict_of_dfs["spot_roll_rule"]

    # ccy FKs
    _check_fk(
        node_df,
        "quote_ccy",
        ccy_df,
        "ccy_name",
        child_sheet="node_meta",
        parent_sheet="ccy",
    )
    _check_fk(
        curve_df,
        "quote_ccy",
        ccy_df,
        "ccy_name",
        child_sheet="curve_meta",
        parent_sheet="ccy",
    )
    _check_fk(
        surf_df,
        "quote_ccy",
        ccy_df,
        "ccy_name",
        child_sheet="surface_meta",
        parent_sheet="ccy",
    )

    # node_meta.fx_base_ccy is nullable; FK only on populated rows
    populated_fx_base = node_df.loc[node_df["fx_base_ccy"].notna(), ["fx_base_ccy"]]
    if not populated_fx_base.empty:
        _check_fk(
            populated_fx_base,
            "fx_base_ccy",
            ccy_df,
            "ccy_name",
            child_sheet="node_meta",
            parent_sheet="ccy",
        )

    # Curve / surface spot_roll_rule_name FKs (only populated rows)
    cm_pop = curve_df.loc[
        curve_df["spot_roll_rule_name"].notna(), ["spot_roll_rule_name"]
    ]
    if not cm_pop.empty:
        _check_fk(
            cm_pop,
            "spot_roll_rule_name",
            srr_df,
            "rule_name",
            child_sheet="curve_meta",
            parent_sheet="spot_roll_rule",
        )

    sm_pop = surf_df.loc[
        surf_df["spot_roll_rule_name"].notna(), ["spot_roll_rule_name"]
    ]
    if not sm_pop.empty:
        _check_fk(
            sm_pop,
            "spot_roll_rule_name",
            srr_df,
            "rule_name",
            child_sheet="surface_meta",
            parent_sheet="spot_roll_rule",
        )

    # Curve / surface point parent FKs
    _check_fk(
        curve_pt_df,
        "curve_id",
        curve_df,
        "curve_id",
        child_sheet="curve_point_meta",
        parent_sheet="curve_meta",
    )
    _check_fk(
        surf_pt_df,
        "surface_id",
        surf_df,
        "surface_id",
        child_sheet="surface_point_meta",
        parent_sheet="surface_meta",
    )

def _validate_global_name_uniqueness(dict_of_dfs: dict[str, pd.DataFrame]) -> None:
    """No name appears in more than one of {node, curve, surface} namespaces."""
    node_names = set(dict_of_dfs["node_meta"]["node_name"].astype(str))
    curve_names = set(dict_of_dfs["curve_meta"]["curve_name"].astype(str))
    surface_names = set(dict_of_dfs["surface_meta"]["surface_name"].astype(str))

    nc = node_names & curve_names
    ns = node_names & surface_names
    cs = curve_names & surface_names

    overlap = nc | ns | cs
    if overlap:
        raise ConfigurationError(
            f"Name collision across container namespaces: {sorted(overlap)}.\n"
            f"Every node_name / curve_name / surface_name must be globally unique."
        )


def _validate_rf_id_uniqueness(dict_of_dfs: dict[str, pd.DataFrame]) -> None:
    """rf_ids (node_id ∪ curve_point.rf_id ∪ surface_point.rf_id) must be globally unique."""
    node_ids = dict_of_dfs["node_meta"]["node_id"].to_numpy()
    cp_ids = dict_of_dfs["curve_point_meta"]["rf_id"].to_numpy()
    sp_ids = dict_of_dfs["surface_point_meta"]["rf_id"].to_numpy()

    all_rf_ids = np.concatenate([node_ids, cp_ids, sp_ids])

    if np.unique(all_rf_ids).size != all_rf_ids.size:
        unique, counts = np.unique(all_rf_ids, return_counts=True)
        dupes = unique[counts > 1].tolist()
        raise ConfigurationError(
            f"CRITICAL: rf_id collision across atomic namespaces: {dupes[:25]}.\n"
            f"node_id, curve_point.rf_id, and surface_point.rf_id share one ID space "
            f"and cannot overlap."
        )


def _validate_container_id_uniqueness(dict_of_dfs: dict[str, pd.DataFrame]) -> None:
    """Ensure internal primary key uniqueness within container tables."""
    # Cross-container collisions are prevented by the namespace bounds.
    # We only need to check that the tables themselves don't have duplicate IDs.
    for table, col in [("curve_meta", "curve_id"), ("surface_meta", "surface_id")]:
        ids = dict_of_dfs[table][col].to_numpy()
        if np.unique(ids).size != ids.size:
            raise ConfigurationError(
                f"CRITICAL: Duplicate primary keys found in {table}.{col}."
            )


def _validate_global_id_namespace(dict_of_dfs: dict[str, pd.DataFrame]) -> None:
    """IDs that share a namespace must be globally bounded."""
    # 1. Gather ALL atomic Risk Factor IDs (Dimension = 0)
    node_ids = dict_of_dfs["node_meta"][
        "node_id"
    ].to_numpy()  # Node IDs are RF IDs by definition
    cp_ids = dict_of_dfs["curve_point_meta"]["rf_id"].to_numpy()
    sp_ids = dict_of_dfs["surface_point_meta"]["rf_id"].to_numpy()
    all_rf_ids = np.concatenate([node_ids, cp_ids, sp_ids])

    # 2. Gather Containers
    curve_ids = dict_of_dfs["curve_meta"]["curve_id"].to_numpy()
    surface_ids = dict_of_dfs["surface_meta"]["surface_id"].to_numpy()

    # 3. Enforce the mathematical bounds
    enforce_id_namespace(ids=all_rf_ids, bounds=GlobalIdNamespace.NODES)
    enforce_id_namespace(ids=curve_ids, bounds=GlobalIdNamespace.CURVES)
    enforce_id_namespace(ids=surface_ids, bounds=GlobalIdNamespace.SURFACES)


def _validate_asset_class_subclass_consistency(
    dict_of_dfs: dict[str, pd.DataFrame],
) -> None:
    """asset_subclass must map to the declared asset_class for all 3 meta tables."""

    def _expected_class(subclass_val: str) -> str | None:
        try:
            return ASSET_SUBCLASS_TO_CLASS[AssetSubclass(subclass_val)].value
        except ConfigurationError:
            return None

    for sheet in ("node_meta", "curve_meta", "surface_meta"):
        df = dict_of_dfs[sheet]
        if "asset_class" not in df.columns or "asset_subclass" not in df.columns:
            continue
        valid = df[["asset_class", "asset_subclass"]].dropna()
        if valid.empty:
            continue
        expected = valid["asset_subclass"].map(_expected_class)
        bad_mask = expected.notna() & (valid["asset_class"] != expected)
        if bad_mask.any():
            offending = valid.loc[bad_mask].copy()
            offending["expected_class"] = expected[bad_mask]
            raise ConfigurationError(
                f"Domain violation: {sheet} has inconsistent "
                f"asset_class / asset_subclass.\n"
                f"{offending.to_string(index=False)}"
            )


def _validate_fx_node_invariants(dict_of_dfs: dict[str, pd.DataFrame]) -> None:
    """
    FX node contract:
      asset_class == "FX"  <=>  fx_base_ccy is non-null
      For FX nodes: node_name == fx_base_ccy + quote_ccy.
    """
    df = dict_of_dfs["node_meta"]
    is_fx = df["asset_class"] == AssetClass.FX.value
    has_base = df["fx_base_ccy"].notna()

    inconsistent = is_fx != has_base
    if inconsistent.any():
        offending = df.loc[
            inconsistent,
            ["node_name", "asset_class", "fx_base_ccy", "quote_ccy"],
        ]
        raise ConfigurationError(
            "Domain violation: asset_class=='FX' iff fx_base_ccy is non-null.\n"
            f"{offending.to_string(index=False)}"
        )

    fx_rows = df.loc[is_fx]
    if fx_rows.empty:
        return
    expected_names = fx_rows["fx_base_ccy"].astype(str) + fx_rows["quote_ccy"].astype(
        str
    )
    name_mismatch = fx_rows["node_name"] != expected_names
    if name_mismatch.any():
        offending = fx_rows.loc[
            name_mismatch, ["node_name", "fx_base_ccy", "quote_ccy"]
        ]
        raise ConfigurationError(
            "Domain violation: FX node_name must equal fx_base_ccy + quote_ccy.\n"
            f"{offending.to_string(index=False)}"
        )

    # FX pairs are unique: (base, quote) appears at most once.
    composite = list(zip(fx_rows["fx_base_ccy"], fx_rows["quote_ccy"], strict=True))
    if len(set(composite)) != len(composite):
        from collections import Counter

        dupes = [k for k, c in Counter(composite).items() if c > 1]
        raise ConfigurationError(
            f"FX pair duplication: (base, quote) pairs appear more than once: {dupes}"
        )


def _validate_axis_roll_pairs(dict_of_dfs: dict[str, pd.DataFrame]) -> None:
    """
    (XAxisKind, RollMethod) compatibility:

      - LABEL curves must have spot_roll_rule_name == NULL.
      - OFFSET_* curves must have spot_roll_rule_name populated.
      - The resolved (XAxisKind, RollMethod) pair must be in the
        ALLOWED_X_AXIS_KIND_ROLL_METHOD_PAIRS table.

    Vectorized: a single pass over each meta table, no row loops.
    """
    srr_df = dict_of_dfs["spot_roll_rule"]
    rule_to_method: dict[str, str] = dict(
        zip(
            srr_df["rule_name"].astype(str),
            srr_df["roll_method"].astype(str),
            strict=True,
        )
    )

    # Pre-compute the allowed-pair set as plain strings (cheap O(K) build).
    allowed_pair_names: frozenset[tuple[str, str]] = frozenset(
        (xk.name, rm.name) for xk, rm in ALLOWED_X_AXIS_KIND_ROLL_METHOD_PAIRS
    )

    for sheet, name_col in (
        ("curve_meta", "curve_name"),
        ("surface_meta", "surface_name"),
    ):
        df = dict_of_dfs[sheet]
        if df.empty:
            continue

        x_kind = df["x_axis_kind"].astype(str)
        rule = df["spot_roll_rule_name"]
        rule_populated = rule.notna()
        needs_roll = x_kind.isin(X_AXIS_KINDS_REQUIRING_ROLL)

        # Missing rule on OFFSET container.
        missing = needs_roll & ~rule_populated
        if missing.any():
            offending = df.loc[missing, [name_col, "x_axis_kind"]]
            raise ConfigurationError(
                f"{sheet}: OFFSET containers missing spot_roll_rule_name:\n"
                f"{offending.to_string(index=False)}"
            )

        # Populated rule on LABEL container.
        extraneous = ~needs_roll & rule_populated
        if extraneous.any():
            offending = df.loc[
                extraneous, [name_col, "x_axis_kind", "spot_roll_rule_name"]
            ]
            raise ConfigurationError(
                f"{sheet}: LABEL containers with populated spot_roll_rule_name:\n"
                f"{offending.to_string(index=False)}"
            )

        # For OFFSET rows, check the (x_kind, method) pair is allowed.
        offset_rows = df.loc[needs_roll].copy()
        if offset_rows.empty:
            continue
        # method = offset_rows["spot_roll_rule_name"].astype(str).map(rule_to_method)
        # pair_strs = list(
        #     zip(offset_rows["x_axis_kind"].astype(str), method, strict=True)
        # )
        # bad_mask = np.array(
        #     [p not in allowed_pair_names for p in pair_strs], dtype=bool
        # )
        method = offset_rows["spot_roll_rule_name"].astype(str).map(rule_to_method)
        # Vectorized pair check: encode each (x_kind, roll_method) tuple as a
        # single "X|M" string and isin() against the precomputed allowed set.
        # Drops the Python-level list comprehension in the validation hot path.
        joined = offset_rows["x_axis_kind"].astype(str) + "|" + method.astype(str)
        allowed_join = frozenset(f"{xk}|{rm}" for xk, rm in allowed_pair_names)
        bad_mask = ~joined.isin(allowed_join).to_numpy(copy=False)
        if bad_mask.any():
            offending = offset_rows.loc[
                bad_mask, [name_col, "x_axis_kind", "spot_roll_rule_name"]
            ].copy()
            offending["roll_method"] = method[bad_mask].values
            raise ConfigurationError(
                f"{sheet}: disallowed (x_axis_kind, roll_method) combinations:\n"
                f"{offending.to_string(index=False)}"
            )


def _validate_fh_bh_mmdd(dict_of_dfs: dict[str, pd.DataFrame]) -> None:
    fh_bh_mmdd = dict_of_dfs["spot_roll_rule"]["fh_bh_mmdd"]
    if fh_bh_mmdd.empty:
        return
    fh_bh_mmdd = fh_bh_mmdd.dropna().astype(str)

    # Assert correct length of 9 (e.g., 0401|1001)
    bad_length = fh_bh_mmdd.str.len() != 9
    if bad_length.any():
        offending = fh_bh_mmdd[bad_length]
        raise ConfigurationError(
            f"Invalid fh_bh_mmdd length:\n{offending.to_string(index=False)}"
        )

    # Assert correct format of 4 digits, pipe, 4 digits.
    bad_format = ~fh_bh_mmdd.str.match(r"^\d{4}\|\d{4}$")
    if bad_format.any():
        offending = fh_bh_mmdd[bad_format]
        raise ConfigurationError(
            f"Invalid fh_bh_mmdd format (expected 'MMDD|MMDD'):\n"
            f"{offending.to_string(index=False)}"
        )


def _validate_curve_point_token_grammar(
    dict_of_dfs: dict[str, pd.DataFrame],
) -> None:
    """
    Every curve_point.tenor_token matches its parent's x_axis_kind regex.

    Vectorized: group rows by parent x_axis_kind, apply pd.Series.str.match
    with the kind-specific regex to each group in one vectorized call.
    """
    cm_df = dict_of_dfs["curve_meta"]
    cp_df = dict_of_dfs["curve_point_meta"]
    if cp_df.empty:
        return

    # Join parent x_axis_kind onto each point row via map (vectorized).
    parent_kind_by_id = cm_df.set_index("curve_id")["x_axis_kind"].astype(str)
    work = cp_df[["curve_id", "rf_id", "tenor_token"]].copy()
    work["x_axis_kind"] = work["curve_id"].map(parent_kind_by_id)

    bad_pieces: list[pd.DataFrame] = []
    for kind_name, group in work.groupby("x_axis_kind"):
        # Explicitly cast kind_name to a string
        kind_name_str = str(kind_name)
        regex = X_AXIS_TOKEN_REGEX[XAxisKind[kind_name_str]]
        good = group["tenor_token"].astype(str).str.match(regex)
        bad = ~good
        if bad.any():
            bad_pieces.append(group.loc[bad])

    if bad_pieces:
        offending = pd.concat(bad_pieces, ignore_index=True)
        raise ConfigurationError(
            "Curve point token grammar violations:\n"
            f"{offending.head(25).to_string(index=False)}"
        )


def _validate_surface_point_token_grammar(
    dict_of_dfs: dict[str, pd.DataFrame],
) -> None:
    """
    Every surface_point.x_tenor_token matches its parent's x_axis_kind regex.

def _validate_discount_invariants(dict_of_dfs: dict[str, pd.DataFrame]) -> None:
    """
    Discount-curve rules:
      - is_default_discount_for_ccy=True curves must have a real basis
        (not NOT_A_DISCOUNT_CURVE).
      - Non-discount curves (basis=NOT_A_DISCOUNT_CURVE) cannot be default.
      - Discount curves must have asset_class=='IR'.
      - The default's basis must equal FIRM_DEFAULT_DISCOUNT_BASIS.
      - Exactly one default per currency that has any discount curve.
      - (ccy, basis) is unique among discount curves.
      - Every currency in the ccy table that appears as a quote_ccy on
        any other container must have at least one discount curve.
    """
    curve_df = dict_of_dfs["curve_meta"]
    node_df = dict_of_dfs["node_meta"]
    surf_df = dict_of_dfs["surface_meta"]
    # ccy_df = dict_of_dfs["ccy"]

    is_disc = curve_df["discount_basis"] != DiscountBasis.NOT_A_DISCOUNT_CURVE.name
    is_default = curve_df["is_default_discount_for_ccy"].astype(bool)

    # Default => is a real discount curve.
    bad = is_default & ~is_disc
    if bad.any():
        offending = curve_df.loc[bad, ["curve_name", "discount_basis"]]
        raise ConfigurationError(
            "Default discount flag set on non-discount curve(s):\n"
            f"{offending.to_string(index=False)}"
        )

    # Discount curves must be IR.
    bad_class = is_disc & (curve_df["asset_class"] != AssetClass.IR.value)
    if bad_class.any():
        offending = curve_df.loc[bad_class, ["curve_name", "asset_class"]]
        raise ConfigurationError(
            "Discount curves must have asset_class=='IR'.\n"
            f"{offending.to_string(index=False)}"
        )

    # Default basis must equal the firm standard.
    bad_basis = is_default & (
        curve_df["discount_basis"] != FIRM_DEFAULT_DISCOUNT_BASIS.name
    )
    if bad_basis.any():
        offending = curve_df.loc[bad_basis, ["curve_name", "discount_basis"]]
        raise ConfigurationError(
            f"Default discount curves must use the firm-standard basis "
            f"{FIRM_DEFAULT_DISCOUNT_BASIS.name!r}.\n"
            f"{offending.to_string(index=False)}"
        )

    # Exactly one default per currency that has any discount curve.
    disc_curves = curve_df.loc[
        is_disc,
        ["curve_name", "quote_ccy", "discount_basis", "is_default_discount_for_ccy"],
    ]
    default_counts = (
        disc_curves.groupby("quote_ccy")["is_default_discount_for_ccy"]
        .sum()
        .astype(int)
    )
    bad_ccys = default_counts[default_counts != 1]
    if not bad_ccys.empty:
        raise ConfigurationError(
            f"Each currency with a discount curve must flag exactly one as "
            f"default. Currencies with !=1 default:\n"
            f"{bad_ccys.to_string()}"
        )

    # (ccy, basis) is unique among discount curves.
    composite = disc_curves[["quote_ccy", "discount_basis"]]
    dupe_mask = composite.duplicated(keep=False)
    if dupe_mask.any():
        offending = disc_curves.loc[dupe_mask]
        raise ConfigurationError(
            "Duplicate (quote_ccy, discount_basis) among discount curves:\n"
            f"{offending.to_string(index=False)}"
        )

    # Coverage: every currency that's a quote_ccy somewhere must have a
    # discount curve. (Skips currencies that exist in `ccy` but aren't used.)
    quote_ccys_used = pd.Index(
        pd.concat([curve_df["quote_ccy"], surf_df["quote_ccy"], node_df["quote_ccy"]])
        .dropna()
        .unique()
    )
    ccys_with_discount = set(disc_curves["quote_ccy"].astype(str))
    missing = sorted(set(quote_ccys_used.astype(str)) - ccys_with_discount)
    if missing:
        raise ConfigurationError(
            f"Currencies with no discount curve: {missing}. "
            f"Every active quote_ccy must have at least one discount curve."
        )

    # Sanity: discount_basis used by discount curves must be one of the real bases.
    real_bases = set(disc_curves["discount_basis"].astype(str))
    bogus = real_bases - DISCOUNT_BASES_EXCLUDING_SENTINEL
    if bogus:
        raise ConfigurationError(
            f"Unexpected discount_basis on discount curves: {sorted(bogus)}. "
            f"Allowed: {sorted(DISCOUNT_BASES_EXCLUDING_SENTINEL)}"
        )

from enum import IntEnum

import numpy as np


class ExtrapMethod(IntEnum):
    FLAT = 1
    LINEAR = 2


# compiler-friendly int8 constants
EXTRAP_FLAT: np.int8 = np.int8(ExtrapMethod.FLAT)
EXTRAP_LINEAR: np.int8 = np.int8(ExtrapMethod.LINEAR)


# Extrap methods
EXTRAP_METHOD_NAMES: tuple[str, ...] = tuple(
    method.name.upper() for method in ExtrapMethod
)

# Central registry
EXTRAP_METHOD_REGISTRY: dict[str, ExtrapMethod] = {
    "LINEAR": ExtrapMethod.LINEAR,
    "FLAT": ExtrapMethod.FLAT,
}


def validate_extrapolation_methods_dict(
    extrapolation_methods_dict: dict[str, int],
) -> None:
    expected = {
        name: int(member) for name, member in ExtrapMethod.__members__.items()
    }
    got = {str(k): int(v) for k, v in extrapolation_methods_dict.items()}

    if got == expected:
        return

    missing = {k: expected[k] for k in expected.keys() - got.keys()}
    extra = {k: got[k] for k in got.keys() - expected.keys()}
    different = {
        k: {"expected": expected[k], "got": got[k]}
        for k in (expected.keys() & got.keys())
        if expected[k] != got[k]
    }

    raise ValueError(
        "extrapolation_methods sheet does not match riskquant ExtrapMethod enum.\n"
        f"missing={missing}\n"
        f"extra={extra}\n"
        f"different={different}\n"
        "Fix the Excel extrapolation_methods table or update riskquant.market_data.curves.extrap_methods.ExtrapMethod."
    )

# src/riskquant/market_data/curves/interp_defs.py
from __future__ import annotations

import math

import numba as nb
import numpy as np

# ==============================================================================
# --- LINEAR INTERPOLATION WITH FLAT EXTRAPOLATION ---
# ==============================================================================


def interp_linear_flat_extrap_3d(
    tenors: np.ndarray,
    values: np.ndarray,
    targets: np.ndarray,
) -> np.ndarray:
    """
    Vectorized 3D Linear Interpolation with Flat Extrapolation.
    Shapes: tenors (c_curves, m_pillars), values (c_curves, m_pillars, s_scenarios), targets (d_targets,)
    Returns: (c_curves, d_targets, s_scenarios)
    """
    c_curves, m_pillars = tenors.shape
    idx1 = np.clip(
        np.sum(tenors[:, None, :] <= targets[None, :, None], axis=2),
        1,
        m_pillars - 1,
    )
    idx0 = idx1 - 1

    c_idx = np.arange(c_curves)[:, None]
    t0, t1 = tenors[c_idx, idx0], tenors[c_idx, idx1]
    y0, y1 = values[c_idx, idx0, :], values[c_idx, idx1, :]

    dt = np.where((t1 - t0) == 0, 1.0, t1 - t0)
    w = (targets[None, :] - t0) / dt

    # FLAT EXTRAPOLATION: Clamp weights to [0.0, 1.0]
    w = np.clip(w, 0.0, 1.0)

    return y0 + w[:, :, None] * (y1 - y0)


@nb.njit(
    parallel=True,
    fastmath={"contract", "reassoc", "nsz"},
    nogil=True,
    boundscheck=False,
    error_model="numpy",
    cache=True,
)
def interp_linear_flat_extrap_3d_numba_kernel(
    tenors: np.ndarray,
    values: np.ndarray,
    targets: np.ndarray,
    out: np.ndarray,
) -> None:
    """Zero-allocation 3D Linear Interpolator with Flat Extrapolation."""
    c_curves, m_pillars = tenors.shape
    d_target_dates = targets.shape[0]
    s_scenarios = values.shape[2]

    for c in nb.prange(c_curves):
        for d in range(d_target_dates):
            target = targets[d]

            idx = np.searchsorted(tenors[c], target, side="right")
            idx1 = min(max(idx, 1), m_pillars - 1)
            idx0 = idx1 - 1

            t0 = tenors[c, idx0]
            t1 = tenors[c, idx1]

            dt = t1 - t0
            if dt == 0.0:
                dt = 1.0

            w = (target - t0) / dt
            w = min(max(w, 0.0), 1.0)

            for s in range(s_scenarios):
                y0 = values[c, idx0, s]
                y1 = values[c, idx1, s]
                out[c, d, s] = y0 + w * (y1 - y0)

def interp_log_linear_flat_extrap_3d(
    tenors: np.ndarray,
    values: np.ndarray,
    targets: np.ndarray,
) -> np.ndarray:
    """
    Vectorized 3D Log-Linear Interpolation with Constant Forward Extrapolation.
    """
    c_curves, m_pillars = tenors.shape
    mask = tenors[:, None, :] <= targets[None, :, None]
    idx1 = np.sum(mask, axis=2)
    idx1 = np.clip(idx1, 1, m_pillars - 1)
    idx0 = idx1 - 1

    c_idx = np.arange(c_curves)[:, None]
    t0 = tenors[c_idx, idx0]
    t1 = tenors[c_idx, idx1]

    y0 = values[c_idx, idx0, :]
    y1 = values[c_idx, idx1, :]

    dt = t1 - t0
    dt = np.where(dt == 0, 1.0, dt)
    w = (targets[None, :] - t0) / dt

    ln_interp = y0 + w[:, :, None] * (y1 - y0)
    return np.exp(ln_interp)

# ==============================================================================


def interp_strike_linear_flat_extrap(
    pillars: np.ndarray,  # Shape: (N_trades, K_strikes)
    values: np.ndarray,  # Shape: (N_trades, K_strikes, S_scenarios)
    targets: np.ndarray,  # Shape: (N_trades, S_scenarios)
) -> np.ndarray:
    """
    Vectorized 1D linear interpolation along the strike/Y-axis with flat extrapolation.
    Used dynamically inside Option Pricers to resolve exact strike vol across scenarios.
    """
    N, K_dim = pillars.shape
    S = targets.shape[1]

    mask = pillars[:, None, :] <= targets[:, :, None]
    idx1 = np.sum(mask, axis=2)
    idx1 = np.clip(idx1, 1, K_dim - 1)
    idx0 = idx1 - 1

    n_idx = np.arange(N)[:, None]
    s_idx = np.arange(S)[None, :]

    x0 = pillars[n_idx, idx0]
    x1 = pillars[n_idx, idx1]

    y0 = values[n_idx, idx0, s_idx]
    y1 = values[n_idx, idx1, s_idx]

    dx = np.where((x1 - x0) == 0, 1.0, x1 - x0)
    w = (targets - x0) / dx

    w = np.clip(w, 0.0, 1.0)
    return y0 + w * (y1 - y0)

    Zero-allocation 2D Surface Strike Interpolator.
    Pre-allocates an (N_trades, S_scenarios) array and resolves exactly.

    s_scenarios = targets.shape[1]

    for n in nb.prange(n_trades):

            w = min(max(w, 0.0), 1.0)

            y0 = values[n, idx0, s]
            y1 = values[n, idx1, s]

            out[n, s] = y0 + w * (y1 - y0)

    interpolation_methods_dict: Mapping[str, int],
) -> None:
    expected = {name: int(member) for name, member in InterpMethod.__members__.items()}
    got = {str(k): int(v) for k, v in interpolation_methods_dict.items()}

    if got == expected:
        return

    missing = {k: expected[k] for k in expected.keys() - got.keys()}
    extra = {k: got[k] for k in got.keys() - expected.keys()}
    different = {
        k: {"expected": expected[k], "got": got[k]}
        for k in (expected.keys() & got.keys())
        if expected[k] != got[k]
    }

    raise ValueError(
        "interpolation_methods sheet does not match riskquant InterpolationMethod enum.\n"
        f"missing={missing}\n"
        f"extra={extra}\n"
        f"different={different}\n"
        "Fix the Excel interpolation_methods table or update riskquant.market_data.curves.interp_methods.InterpMethod."

# riskquant/utils/tenors.py
from __future__ import annotations

from datetime import date
from typing import Optional, Tuple


def parse_bracket_token(tenor_token: str) -> Tuple[str, Optional[int]]:
    """
    "[Y-2]" -> ("Y", -2)
    "[M+0]" -> ("M", 0)
    "[D+24]" -> ("D", 24)
    "[HY+1]" -> ("HY", 1)
    "[ON]" -> ("ON", None)   (LABEL)
    """
    # Error.
    if tenor_token is None:
        raise ValueError("tenor_token cannot be None!")

    # Strip white space.
    s = tenor_token.strip()

    # Must start with '[' and end with ']'.
    if s.startswith("[") and s.endswith("]"):
        s = s[1:-1].strip()

    # Find the plus or minus sign.
    plus = s.rfind("+")
    minus = s.rfind("-")
    sign_pos = max(plus, minus)

    # If no sign.
    if sign_pos <= 0:
        return s, None

    # Unit (Y, M, D, HY, etc.)
    unit = s[:sign_pos]

    # Signed offset.
    offset = int(s[sign_pos:])

    # Return.
    return unit, offset


def get_spot_year(
    *,
    as_of_date: date,
    roll_month: int,
    roll_day: int,
    post_roll_shift: int,
) -> int:
    """
    Defines which delivery year corresponds to Y+0 under ROLL_ON_MMDD.
    """

    # Get the roll date,
    roll_date = date(as_of_date.year, roll_month, roll_day)

    # Compare to as-of-date.
    # As-of-date is on or after the roll date.
    if as_of_date >= roll_date:
        return (
            as_of_date.year + post_roll_shift
        )  # Current year plus the post_roll_shift amount (in years).

    # As-of-date is before the roll date.
    # Previous year plus the post_roll_shift amount (in years).
    return (as_of_date.year - 1) + post_roll_shift


def add_months_to_date_and_return_yyyymm01(
    *,
    d: date,
    months_to_add: int,
) -> date:
    """Add months to a month-start date (day=1)."""
    y = d.year + (d.month - 1 + months_to_add) // 12
    m = (d.month - 1 + months_to_add) % 12 + 1
    return date(y, m, 1)


def fmt_date_as_str_yyyy(d: date) -> str:
    return f"{d.year:04d}"


def fmt_date_as_str_yyyymm(d: date) -> str:
    return f"{d.year:04d}{d.month:02d}"


def fmt_date_as_str_yyyymmdd(d: date) -> str:
    return f"{d.year:04d}{d.month:02d}{d.day:02d}"
